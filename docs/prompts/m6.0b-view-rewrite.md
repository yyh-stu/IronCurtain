## 三、渲染规格(权威)

### 3.1 材质(URP)
- shader 一律 `Shader.Find("Universal Render Pipeline/Lit")`。
- `ViewAssets.Mat(albedo, metallic, smoothness, emissionColor, emissionIntensity)`:emission 颜色分量可 >1(HDR,喂 Bloom);`EnableKeyword("_EMISSION")`、`SetColor("_EmissionColor", emission * emissionIntensity)`。
- `ViewAssets.TeamMat(side)`:蓝 = albedo(0.15,0.20,0.28)+emission(0.2,0.85,1)×1.2;红 = albedo(0.28,0.15,0.15)+emission(1,0.3,0.3)×1.2;中立 = albedo(0.3,0.3,0.3)+emission(0.85,0.85,0.85)×0.8。
- `ViewAssets.TeamColor(side)` 同上发光色。
- `ViewAssets.GlowMat(side)` = 纯黑 albedo + 强 emission(×2.5)。
- 金属质感:机体 metallic 0.55、smoothness 0.6。

### 3.2 自建网格(MeshBuilder)
`ViewAssets` 内提供网格构建帮助(顶点/三角形直接写,禁止 CreatePrimitive 组合当主体;允许小装饰用极简 mesh):
- `AddBox(verts, tris, center, size)`(24 顶点立方体)
- `AddPrism(verts, tris, center, bottomSize, topSize, height, segments)`(棱台,segments=4 或 8,侧面四边形三角化)
- `AddCylinder(verts, tris, center, radiusTop, radiusBottom, height, segments)`
- 每个构建方法返回 (verts, tris, 子网格归属):**submesh 0 = 主体材质,submesh 1 = 发光材质**。
- `CreateMesh(verts, tris, submeshCount)`:SetVertices/SetTriangles 分 submesh/RecalculateNormals。
- 单位/建筑网格生成后挂 MeshFilter+MeshRenderer(双材质数组:[主体材质, 发光材质])。

```csharp
using IronCurtain.Sim;
using UnityEngine;

namespace IronCurtain.View
{
    public static class ViewAssets
    {
        private static Font _font;

        public static Material Mat(Color albedo, Color emission, float emissionStrength)
        {
            var m = new Material(Shader.Find("Standard"));
            m.color = albedo;
            if (emissionStrength > 0f)
            {
                m.EnableKeyword("_EMISSION");
                m.SetColor("_EmissionColor", emission * emissionStrength);
            }
            return m;
        }

        public static Material TeamMat(Side side)
        {
            Color c = TeamColor(side);
            Color albedo;
            switch (side)
            {
                case Side.Blue: albedo = new Color(0.16f, 0.22f, 0.30f); break;
                case Side.Red: albedo = new Color(0.30f, 0.16f, 0.16f); break;
                default: albedo = new Color(0.30f, 0.30f, 0.30f); break;
            }
            return Mat(albedo, c, 1.1f);
        }

        public static Color TeamColor(Side side)
        {
            switch (side)
            {
                case Side.Blue: return new Color(0.2f, 0.85f, 1f);
                case Side.Red: return new Color(1f, 0.3f, 0.3f);
                default: return new Color(0.85f, 0.85f, 0.85f);
            }
        }

        public static Font UiFont(int size)
        {
            if (_font == null) _font = Resources.GetBuiltinResource<Font>("LegacyRuntime.ttf");
            return _font;
        }

        public static GameObject Box(Material mat, Vector3 localPos, Vector3 localScale, Transform parent, string name = "box")
        {
            var go = GameObject.CreatePrimitive(PrimitiveType.Cube);
            go.name = name;
            go.transform.SetParent(parent, false);
            go.transform.localPosition = localPos;
            go.transform.localScale = localScale;
            if (mat != null) go.GetComponent<Renderer>().sharedMaterial = mat;
            Object.Destroy(go.GetComponent<Collider>());
            return go;
        }

        public static GameObject Sphere(Material mat, Vector3 localPos, float diameter, Transform parent, string name = "sphere")
        {
            var go = GameObject.CreatePrimitive(PrimitiveType.Sphere);
            go.name = name;
            go.transform.SetParent(parent, false);
            go.transform.localPosition = localPos;
            go.transform.localScale = Vector3.one * diameter;
            if (mat != null) go.GetComponent<Renderer>().sharedMaterial = mat;
            Object.Destroy(go.GetComponent<Collider>());
            return go;
        }

        public static GameObject Cylinder(Material mat, Vector3 localPos, float diameter, float height, Transform parent, string name = "cyl")
        {
            var go = GameObject.CreatePrimitive(PrimitiveType.Cylinder);
            go.name = name;
            go.transform.SetParent(parent, false);
            go.transform.localPosition = localPos;
            go.transform.localScale = new Vector3(diameter, height * 0.5f, diameter);
            if (mat != null) go.GetComponent<Renderer>().sharedMaterial = mat;
            Object.Destroy(go.GetComponent<Collider>());
            return go;
        }

        public static GameObject Quad(Material mat, Vector3 localPos, Vector3 localScale, Transform parent, string name = "quad")
        {
            var go = GameObject.CreatePrimitive(PrimitiveType.Quad);
            go.name = name;
            go.transform.SetParent(parent, false);
            go.transform.localPosition = localPos;
            go.transform.localScale = localScale;
            if (mat != null) go.GetComponent<Renderer>().sharedMaterial = mat;
            Object.Destroy(go.GetComponent<Collider>());
            return go;
        }
    }
}
```

### 4.2 EntityView.cs 全文
```csharp
using IronCurtain.Sim;
using UnityEngine;

namespace IronCurtain.View
{
    public sealed class EntityView : MonoBehaviour
    {
        private Entity _entity;
        private Side _side;
        private HeroId? _hero;
        private BuildingId? _building;

        private GameObject _model;
        private Renderer[] _glowRenderers;
        private Material _rubbleMat;
        private Material[] _origGlowMats;

        private Transform _hpRoot, _hpFill, _shieldFill;
        private float _hpBarWidth = 1.5f;
        private float _hpBarHeight = 0.16f;
        private float _deadTimer = -1f;
        private bool _dead;

        private Material _hpGreen, _hpYellow, _hpRed, _hpBack, _shieldMat;
        private Vector3 _baseScale = Vector3.one;

        public void Bind(Entity e, Side side, HeroId? hero, BuildingId? building)
        {
            _entity = e;
            _side = side;
            _hero = hero;
            _building = building;

            InitMaterials();
            if (e.Kind == EntityKind.Unit)
                BuildUnit(hero.HasValue ? hero.Value : HeroId.PanShi);
            else
                BuildBuilding(building.HasValue ? building.Value : BuildingId.Tower);
            BuildHpBar();

            Vector3 p = BoardView.WorldPos(e.Pos, e.Kind == EntityKind.Building);
            transform.position = p;
            if (e.Kind == EntityKind.Unit)
                transform.rotation = Quaternion.Euler(0f, side == Side.Blue ? 90f : -90f, 0f);
            SetHp(e.Hp, e.MaxHp);
        }

        private void InitMaterials()
        {
            if (_hpGreen != null) return;
            _hpBack = ViewAssets.Mat(new Color(0.08f, 0.08f, 0.08f), Color.black, 0f);
            _hpGreen = ViewAssets.Mat(Color.green, Color.green * 0.6f, 1f);
            _hpYellow = ViewAssets.Mat(Color.yellow, Color.yellow * 0.6f, 1f);
            _hpRed = ViewAssets.Mat(Color.red, Color.red * 0.6f, 1f);
            _shieldMat = ViewAssets.Mat(new Color(0.4f, 0.8f, 1f), new Color(0.4f, 0.8f, 1f), 1f);
            _rubbleMat = ViewAssets.Mat(new Color(0.28f, 0.26f, 0.24f), Color.black, 0f);
        }

        public void FlashHit() { }

        public void SetDead()
        {
            if (_dead) return;
            _dead = true;
            _deadTimer = 0f;
        }

        public void SetRubbled(bool on)
        {
            if (_model == null) return;
            if (on)
            {
                _model.transform.localScale = new Vector3(1f, 0.35f, 1f);
                foreach (var r in _model.GetComponentsInChildren<Renderer>())
                    if (System.Array.IndexOf(_glowRenderers, r) < 0)
                        r.sharedMaterial = _rubbleMat;
                foreach (var r in _glowRenderers) r.enabled = false;
            }
            else
            {
                _model.transform.localScale = _baseScale;
                for (int i = 0; i < _glowRenderers.Length; i++)
                {
                    _glowRenderers[i].enabled = true;
                    _glowRenderers[i].sharedMaterial = _origGlowMats[i];
                }
            }
        }

        public void SetHp(float hp, float maxHp)
        {
            if (_hpFill == null) return;
            float t = maxHp > 0f ? Mathf.Clamp01(hp / maxHp) : 0f;
            _hpFill.localScale = new Vector3(_hpBarWidth * t, _hpBarHeight, 1f);
            _hpFill.localPosition = new Vector3(-_hpBarWidth * (1f - t) * 0.5f, 0f, 0f);
            _hpFill.GetComponent<Renderer>().sharedMaterial = t > 0.5f ? _hpGreen : (t > 0.25f ? _hpYellow : _hpRed);
        }

        public void TickBillboard(Camera cam)
        {
            if (_hpRoot != null && cam != null)
            {
                _hpRoot.rotation = Quaternion.LookRotation(_hpRoot.position - cam.transform.position);
            }
        }

        public void Tick(float dt, Camera cam)
        {
            if (_entity == null) return;
            bool isBuilding = _entity.Kind == EntityKind.Building;
            Vector3 target = BoardView.WorldPos(_entity.Pos, isBuilding);
            transform.position = Vector3.Lerp(transform.position, target, 1f - Mathf.Exp(-14f * dt));
            SetHp(_entity.Hp, _entity.MaxHp);
            if (_entity.Rubbled && _model != null && _model.transform.localScale.y > 0.9f) SetRubbled(true);
            if (!_entity.Rubbled && _model != null && _model.transform.localScale.y < 0.9f && _baseScale.y >= 1f) SetRubbled(false);
            UpdateShield();
            TickBillboard(cam);

            if (_dead)
            {
                _deadTimer += dt;
                float s = Mathf.Max(0.01f, 1f - _deadTimer * 4f);
                transform.localScale = Vector3.one * s;
            }
            else transform.localScale = Vector3.one;
        }

        private void UpdateShield()
        {
            if (_shieldFill == null) return;
            bool show = _entity.Shield > 0.01f && _entity.ShieldMax > 0.01f;
            _shieldFill.gameObject.SetActive(show);
            if (show)
            {
                float t = Mathf.Clamp01(_entity.Shield / _entity.ShieldMax);
                _shieldFill.localScale = new Vector3(_hpBarWidth * t, 0.05f, 1f);
                _shieldFill.localPosition = new Vector3(-_hpBarWidth * (1f - t) * 0.5f, 0.15f, 0f);
            }
        }

        private void BuildHpBar()
        {
            var root = new GameObject("HpBar");
            root.transform.SetParent(transform, false);
            float y = _entity.Kind == EntityKind.Building ? 3.7f : 3.0f;
            root.transform.localPosition = new Vector3(0f, y, 0f);
            _hpRoot = root.transform;

            ViewAssets.Quad(_hpBack, Vector3.zero, new Vector3(_hpBarWidth, _hpBarHeight, 1f), root.transform, "Back");
            _hpFill = ViewAssets.Quad(_hpGreen, Vector3.zero, new Vector3(_hpBarWidth, _hpBarHeight, 1f), root.transform, "Fill").transform;
            var sq = ViewAssets.Quad(_shieldMat, new Vector3(0f, 0.15f, 0f), new Vector3(_hpBarWidth, 0.05f, 1f), root.transform, "Shield");
            _shieldFill = sq.transform;
            sq.SetActive(false);
        }

        private void BuildUnit(HeroId hero)
        {
            _model = new GameObject("Model");
            _model.transform.SetParent(transform, false);
            Material team = ViewAssets.TeamMat(_side);
            Color glow = ViewAssets.TeamColor(_side);
            Material glowMat = ViewAssets.Mat(Color.black, glow, 1.6f);
            Material metal = ViewAssets.Mat(new Color(0.35f, 0.38f, 0.42f), Color.black, 0f);

            Transform m = _model.transform;
            var glows = new System.Collections.Generic.List<Renderer>();

            switch (hero)
            {
                case HeroId.PanShi:
                    ViewAssets.Box(team, new Vector3(0f, 0.9f, 0f), new Vector3(0.9f, 0.9f, 0.7f), m);
                    ViewAssets.Box(metal, new Vector3(0.55f, 1.0f, 0f), new Vector3(0.12f, 1.3f, 0.9f), m);
                    glows.Add(ViewAssets.Box(glowMat, new Vector3(-0.48f, 1.3f, 0.3f), new Vector3(0.06f, 0.5f, 0.08f), m).GetComponent<Renderer>());
                    glows.Add(ViewAssets.Box(glowMat, new Vector3(-0.48f, 1.3f, -0.3f), new Vector3(0.06f, 0.5f, 0.08f), m).GetComponent<Renderer>());
                    ViewAssets.Sphere(team, new Vector3(0f, 1.5f, 0f), 0.35f, m, "Head");
                    break;
                case HeroId.ZhenYue:
                    ViewAssets.Box(team, new Vector3(0f, 0.95f, 0f), new Vector3(1.0f, 1.0f, 0.7f), m);
                    ViewAssets.Cylinder(metal, new Vector3(0.6f, 0.9f, 0f), 0.1f, 1.1f, m);
                    ViewAssets.Box(metal, new Vector3(0.6f, 1.5f, 0f), new Vector3(0.55f, 0.4f, 0.4f), m);
                    glows.Add(ViewAssets.Box(glowMat, new Vector3(-0.52f, 1.2f, 0f), new Vector3(0.06f, 0.9f, 0.1f), m).GetComponent<Renderer>());
                    ViewAssets.Sphere(team, new Vector3(0f, 1.6f, 0f), 0.35f, m, "Head");
                    break;
                case HeroId.DuanFeng:
                    ViewAssets.Box(team, new Vector3(0f, 1.0f, 0f), new Vector3(0.55f, 1.3f, 0.5f), m);
                    var spear = ViewAssets.Box(metal, new Vector3(0.5f, 1.4f, 0f), new Vector3(1.7f, 0.07f, 0.07f), m);
                    spear.transform.localRotation = Quaternion.Euler(0f, 0f, 20f);
                    glows.Add(ViewAssets.Box(glowMat, new Vector3(1.25f, 1.68f, 0f), new Vector3(0.2f, 0.09f, 0.09f), spear.transform).GetComponent<Renderer>());
                    ViewAssets.Sphere(team, new Vector3(0f, 1.75f, 0f), 0.32f, m, "Head");
                    break;
                case HeroId.ZhuFeng:
                    ViewAssets.Box(team, new Vector3(0f, 0.8f, 0f), new Vector3(0.5f, 0.8f, 0.45f), m);
                    var lance = ViewAssets.Box(metal, new Vector3(0.8f, 0.9f, 0f), new Vector3(1.9f, 0.06f, 0.06f), m);
                    ViewAssets.Box(team, new Vector3(-0.2f, 0.9f, 0.35f), new Vector3(0.5f, 0.08f, 0.25f), m);
                    ViewAssets.Box(team, new Vector3(-0.2f, 0.9f, -0.35f), new Vector3(0.5f, 0.08f, 0.25f), m);
                    glows.Add(ViewAssets.Box(glowMat, new Vector3(1.75f, 0.9f, 0f), new Vector3(0.15f, 0.1f, 0.1f), lance.transform).GetComponent<Renderer>());
                    ViewAssets.Sphere(team, new Vector3(0f, 1.35f, 0f), 0.3f, m, "Head");
                    break;
```

```csharp
using System.Collections.Generic;
using IronCurtain.Sim;
using UnityEngine;

namespace IronCurtain.View
{
    public sealed class BoardView : MonoBehaviour
    {
        private SimCore _sim;
        private readonly Dictionary<int, EntityView> _views = new Dictionary<int, EntityView>();

        public static Vector3 WorldPos(GridPos p, bool isBuilding)
        {
            return new Vector3(p.Col * 2f, isBuilding ? 0f : 0.9f, (1 - p.Row) * 2f);
        }

        public void Init(SimCore sim, Transform parent)
        {
            _sim = sim;
            BuildBoard();
            foreach (var kv in sim.Entities)
            {
                var e = kv.Value;
                if (e != null) CreateView(e);
            }
        }

        public EntityView GetView(int entityId)
        {
            EntityView v;
            return _views.TryGetValue(entityId, out v) ? v : null;
        }

        public void OnSimEvent(SimEvent ev)
        {
            switch (ev.Type)
            {
                case SimEventType.EntitySpawned:
                    var e = _sim.GetEntity(ev.SubjectId);
                    if (e != null) CreateView(e);
                    break;
                case SimEventType.EntityRemoved:
                    EntityView rv;
                    if (_views.TryGetValue(ev.SubjectId, out rv))
                    {
                        _views.Remove(ev.SubjectId);
                        if (rv != null && rv.gameObject != null) Destroy(rv.gameObject);
                    }
                    break;
                case SimEventType.DamageDealt:
                    var dv = GetView(ev.SubjectId);
                    if (dv != null) dv.FlashHit();
                    break;
                case SimEventType.EntityDied:
                    var dv2 = GetView(ev.SubjectId);
                    if (dv2 != null) dv2.SetDead();
                    break;
            }
        }

        public void TickViews()
        {
            Camera cam = Camera.main;
            float dt = Time.deltaTime;
            foreach (var kv in _views)
            {
                var v = kv.Value;
                if (v == null) continue;
                var e = _sim.GetEntity(kv.Key);
                if (e == null || e.State == EntityState.Dead) v.SetDead();
                v.Tick(dt, cam);
            }
        }

        private void CreateView(Entity e)
        {
            if (_views.ContainsKey(e.Id)) return;
            var go = new GameObject("Entity_" + e.Id);
            go.transform.SetParent(transform, false);
            var view = go.AddComponent<EntityView>();
            view.Bind(e, e.Side, e.Hero, e.Building);
            _views[e.Id] = view;
        }

        private void BuildBoard()
        {
            Material darkA = ViewAssets.Mat(new Color32(0x15, 0x18, 0x20, 0xff), Color.black, 0f);
            Material darkB = ViewAssets.Mat(new Color32(0x2c, 0x31, 0x3d, 0xff), Color.black, 0f);
            Material fence = ViewAssets.Mat(new Color(0.15f, 0.16f, 0.19f), Color.black, 0f);

            // 废土延展地面(棋盘之外的底色,避免画幅上下空洞)
            Material wasteland = ViewAssets.Mat(new Color32(0x0e, 0x10, 0x16, 0xff), Color.black, 0f);
            ViewAssets.Box(wasteland, new Vector3(29f, -0.18f, -2f), new Vector3(78f, 0.15f, 26f), transform, "Wasteland");

            for (int row = 0; row < LaneGrid.Rows; row++)
            {
                for (int col = 0; col < LaneGrid.Cols; col++)
                {
                    Material m = ((row + col) & 1) == 0 ? darkA : darkB;
                    Vector3 pos = new Vector3(col * 2f, -0.05f, (1 - row) * 2f);
                    ViewAssets.Box(m, pos, new Vector3(2f, 0.1f, 2f), transform, "Tile_" + row + "_" + col);
                }
            }

            float cx = (LaneGrid.Cols - 1) * 1f;
            ViewAssets.Box(fence, new Vector3(cx, 0.3f, -3.4f), new Vector3(LaneGrid.Cols * 2f + 1.6f, 0.3f, 0.25f), transform, "Fence_N");

        }
    }
```

```csharp
using IronCurtain.Sim;
using UnityEngine;

namespace IronCurtain.View
{
    public sealed class GameBootstrap : MonoBehaviour
    {
        public SimCore Sim;
        public BoardView Board;
        public Camera Cam;
        public bool Paused;
        public float Speed = 1f;

        private double _accumulator;

        private void Awake()
        {
            var cfg = GameConfigParser.FromJson(Resources.Load<TextAsset>("gameconfig").text);
            Sim = MatchSetup.Build(cfg, null, AiTier.Normal);

            CreateCamera();
            CreateLighting();

            var boardGo = new GameObject("Board");
            boardGo.transform.SetParent(transform, false);
            Board = boardGo.AddComponent<BoardView>();
            Board.Init(Sim, transform);
            Sim.AddEventListener(Board.OnSimEvent);
            gameObject.AddComponent<CaptureRunner>();

            Debug.Log("[GameBootstrap] Match started");
        }

        private void Update()
        {
            if (Paused) return;
            double interval = 1.0 / Sim.Config.TickRate;
            _accumulator += Time.deltaTime * Speed;
            while (_accumulator >= interval)
            {
                _accumulator -= interval;
                Sim.Step();
            }
            Board.TickViews();
        }

        private void OnDestroy()
        {
            if (Sim != null) Sim.ClearEventListeners();
        }

        private void CreateCamera()
        {
            var go = new GameObject("MainCamera");
            go.transform.SetParent(transform, false);
            go.tag = "MainCamera";
            Cam = go.AddComponent<Camera>();
            Cam.clearFlags = CameraClearFlags.SolidColor;
            Cam.backgroundColor = new Color32(0x0a, 0x0e, 0x18, 0xff);
            Cam.nearClipPlane = 0.3f;
            Cam.farClipPlane = 200f;
            Cam.orthographic = true;
            Cam.orthographicSize = 4.5f;
            go.transform.position = new Vector3(12f, 12f, -9.5f);
            go.transform.LookAt(new Vector3(12f, 0f, 0f));
            go.AddComponent<AudioListener>();
        }

        private void CreateLighting()
        {
            var go = new GameObject("Directional Light");
            go.transform.SetParent(transform, false);
            var light = go.AddComponent<Light>();
            light.type = LightType.Directional;
            light.color = Color.white;
            light.intensity = 0.85f;
            go.transform.rotation = Quaternion.Euler(60f, 30f, 0f);
            RenderSettings.ambientMode = UnityEngine.Rendering.AmbientMode.Flat;
            RenderSettings.ambientLight = new Color(0.3f, 0.3f, 0.34f, 1f);
            RenderSettings.fog = true;
            RenderSettings.fogMode = FogMode.ExponentialSquared;
            RenderSettings.fogColor = new Color(0.04f, 0.06f, 0.10f);
            RenderSettings.fogDensity = 0.012f;
        }
    }
}
```
