你是《铁幕:前线》Unity 6(6000.0.30f1,URP 17)表现层主程。本模块把 Sim 层已完成的**椭圆形大地图(30 行 × 286 列包围盒,长半轴 143、短半轴 15,楚河汉界河宽 10,上下两座 5 格桥,河中央岛)**落到画面:椭圆石路+少量草地、蓝色河面、木桥、椭圆外暗色虚空,并把相机抬高到约 1/3 地图视野(约 95 格宽),建筑视觉放大一倍(逻辑仍单格)。风格=英雄联盟式石路+少量草地(拒绝满屏绿),中性化。

## 二、需求与规范(权威数值)

### 2.1 地图与坐标
- 包围盒 Row 0..29 × Col 0..285,椭圆中心 (143,15),长半轴 143,短半轴 15。
- 世界坐标映射(已有,沿用 BoardView.WorldPos):**x = X 米,z = 3 − Y 米**。地图中心世界坐标 =(143, 0, −12)。
- 地形判定**必须调用 `TerrainMap` 静态谓词**,不得自己重新实现:
  - `TerrainMap.IsInside(SimVec2)` / `IsInside(GridPos)`:椭圆内判定
  - `TerrainMap.IsWater(SimVec2)`:河面(河 X∈[138,148),减去桥与岛)
  - `TerrainMap.IsBridge(SimVec2)`:两座桥(Y∈[25,30) 或 Y∈[0,5),且 X∈[138,148))
  - `TerrainMap.IsIsland(SimVec2)`:中央岛(X∈[139,147),Y∈[11,19))
  - 常量:`TerrainMap.Rows=30`、`Cols=286`(也可用 `LaneGrid.Rows/Cols`)

### 2.2 地形渲染规格(颜色/高度/材质,权威)
用 `ViewAssets.Flat(color, smoothness)`(URP/Lit,metallic=0)或 `ViewAssets.Mat`:

| 元素 | 覆盖格 | 颜色(hex) | smoothness | 高度(top y) | 做法 |
|---|---|---|---|---|---|
| 石路(主) | 椭圆内、非水域 | 深 `#787a7e` / 浅 `#86888d` 按 `(row+col)&1` 两色 | 0.35 | 0 | 网格盒 |
| 草地(少量) | 椭圆内、非水域中的草地格 | `#5f8149` | 0.12 | 0 | 网格盒 |
| 河面 | `TerrainMap.IsWater` 格 | `#2f6fb3` | 0.92 | −0.12 | 网格盒 |
| 桥面 | `TerrainMap.IsBridge` 格 | `#9a7a52` | 0.3 | +0.06 | 网格盒 + 栏杆 |
| 中央岛 | `TerrainMap.IsIsland` 格 | 草地色 `#5f8149` | 0.12 | 0 | 并入地面网格 |
| 虚空 | 椭圆外全部(包围盒全域垫底) | `#0a0c12` | 0.05 | −0.35 | 一块大盒 |

- **草地判定(确定性 hash,不用 System.Random)**:`int h = row * 31 + col * 17;` 若 `(h & 0xFF) < 38`(约 15%)则该格为草地;**中央岛所有格强制草地**;河面/桥格不算草地。
- 网格单元:每格一个 1×0.1×1 的盒(用 `ViewAssets.AddBox`,中心 `(col+0.5, topY−0.05, 3−row−0.5)`)。
- **分块硬约束(必须遵守)**:Unity 单 Mesh 上限 65535 顶点。地面按列分块:**每 60 列一块**(col 0-59 / 60-119 / 120-179 / 180-239 / 240-285),每块一个 GameObject(`Ground_0`…`Ground_4`),每块内 submesh[0]=石路深、submesh[1]=石路浅、submesh[2]=草地(草地含中央岛格),材质数组对应三色。每块顶点数 ≤ 60×30×24 = 43200 < 65535,安全。
- 河面:单网格(约 150 格,盒);桥面:单网格(100 格,盒),每座桥两侧各加 1 条栏杆 `ViewAssets.DecoBox`(色 `#6f5638`,长 10 格、截面 0.12×0.5,沿桥两侧 y 0.35)。桥面盒 top +0.06 覆盖在河面上方(河面格在桥处被桥面盖住,但河面网格桥格也照常生成,无妨)。
- 虚空:`ViewAssets.DecoBox`,中心 (143, −0.4, −12),尺寸 (300, 0.6, 34)(top y=−0.1,垫在椭圆外露出的区域下方)。
- 全部地形挂到 `Terrain` 子节点(transform 挂入传入的 parent)。

### 3.1 TerrainMap.cs(Sim 地形权威 API,只读)
```csharp
using System;

namespace IronCurtain.Sim
{
    /// <summary>椭圆大地图地形规则(权威规格,详见 M6.5)。</summary>
    public static class TerrainMap
    {
        public const int Rows = 30;
        public const int Cols = 286;

        // 椭圆参数
        public const float CenterX = 143f;
        public const float CenterY = 15f;
        public const float HalfX = 143f;
        public const float HalfY = 15f;

        // 河(楚河汉界)
        public const float RiverMinX = 138f;
        public const float RiverMaxX = 148f;
        // 桥
        public const float BridgeTopMinY = 25f;
        public const float BridgeBotMaxY = 5f;
        // 中央岛
        public const float IsleMinX = 139f;
        public const float IsleMaxX = 147f;
        public const float IsleMinY = 11f;
        public const float IsleMaxY = 19f;

        public static bool IsInside(SimVec2 v)
        {
            float dx = (v.X - CenterX) / HalfX;
            float dy = (v.Y - CenterY) / HalfY;
            return dx * dx + dy * dy <= 1f;
        }

        public static bool IsInside(GridPos p)
        {
            return IsInside(SimVec2.FromGrid(p));
        }

        public static bool IsBridge(SimVec2 v)
        {
            if (v.X < RiverMinX || v.X >= RiverMaxX) return false;
            return v.Y >= BridgeTopMinY || v.Y < BridgeBotMaxY;
        }

        public static bool IsIsland(SimVec2 v)
        {
            return v.X >= IsleMinX && v.X < IsleMaxX && v.Y >= IsleMinY && v.Y < IsleMaxY;
        }

        public static bool IsWater(SimVec2 v)
        {
            if (v.X < RiverMinX || v.X >= RiverMaxX) return false;
            return !IsBridge(v) && !IsIsland(v);
        }

        public static float MoveFactor(SimVec2 v)
        {
            return IsWater(v) ? 0.5f : 1f;
        }

        /// <summary>径向投影:把 v 推回椭圆(含边界)。</summary>
        public static SimVec2 ClampToEllipse(SimVec2 v)
        {
            if (IsInside(v)) return v;
            float dx = v.X - CenterX;
            float dy = v.Y - CenterY;
            // 防退化:恰在中心时向右推
            if (dx == 0f && dy == 0f) return new SimVec2(CenterX + HalfX, CenterY);
            double a = HalfX, b = HalfY;
            // 参数角:取点方向,然后牛顿迭代求边界到方向射线的交点
            double theta = Math.Atan2(dy / b, dx / a);
            double ct = Math.Cos(theta), st = Math.Sin(theta);
            // 边界点: x = a*ct, y = b*st (椭圆参数化)
            float bx = (float)(CenterX + a * ct);
            float by = (float)(CenterY + b * st);
            // 参数化点保证在椭圆上;再向内收一点点避免浮点误差
            var r = new SimVec2(bx, by);
            if (!IsInside(r))
                r = new SimVec2((bx - CenterX) * 0.999f + CenterX, (by - CenterY) * 0.999f + CenterY);
            return r;
        }
    }
}
```

### 3.2 SimVec2.cs(连续坐标,只读)
```csharp
using System;

namespace IronCurtain.Sim
{
    /// <summary>连续坐标(米制)。1 格 = 1 米;X=列方向,Y=行方向。</summary>
    public struct SimVec2
    {
        public float X;
        public float Y;

        public SimVec2(float x, float y)
        {
            X = x;
            Y = y;
        }

        public static SimVec2 FromGrid(GridPos p) => new SimVec2(p.Col + 0.5f, p.Row + 0.5f);

        public static SimVec2 operator +(SimVec2 a, SimVec2 b) => new SimVec2(a.X + b.X, a.Y + b.Y);
        public static SimVec2 operator -(SimVec2 a, SimVec2 b) => new SimVec2(a.X - b.X, a.Y - b.Y);
        public static SimVec2 operator *(SimVec2 a, float s) => new SimVec2(a.X * s, a.Y * s);

        public float SqrMagnitude => X * X + Y * Y;
        public float Magnitude => (float)Math.Sqrt(X * X + Y * Y);

        public SimVec2 Normalized()
        {
            float m = Magnitude;
            return m > 1e-6f ? new SimVec2(X / m, Y / m) : new SimVec2(0f, 0f);
        }

        public static float Distance(SimVec2 a, SimVec2 b)
        {
            float dx = a.X - b.X, dy = a.Y - b.Y;
            return (float)Math.Sqrt(dx * dx + dy * dy);
        }

        public static float SqrDistance(SimVec2 a, SimVec2 b)
        {
            float dx = a.X - b.X, dy = a.Y - b.Y;
            return dx * dx + dy * dy;
        }

        public static SimVec2 Lerp(SimVec2 a, SimVec2 b, float t) =>
            new SimVec2(a.X + (b.X - a.X) * t, a.Y + (b.Y - a.Y) * t);

        /// <summary>连续坐标 → 所在格(向下取整)。</summary>
        public GridPos ToGrid()
        {
            int col = X < 0 ? 0 : (int)X;
            int row = Y < 0 ? 0 : (int)Y;
            if (col > LaneGrid.Cols - 1) col = LaneGrid.Cols - 1;
            if (row > LaneGrid.Rows - 1) row = LaneGrid.Rows - 1;
            return new GridPos(row, col);
        }

        public override string ToString() => $"({X:F2},{Y:F2})";
    }
}
```

### 3.3 BoardView.cs 全文(现状,将 BuildBoard 替换为 TerrainView 调用)
```csharp
using System.Collections.Generic;
using IronCurtain.Sim;
using UnityEngine;

namespace IronCurtain.View
{
    public sealed class BoardView : MonoBehaviour
    {
        private SimCore _sim;
        private readonly Dictionary<int, EntityView> _views = new Dictionary<int, EntityView>();

        public static Vector3 WorldPos(SimVec2 p) { return new Vector3(p.X, 0f, 3f - p.Y); }
        public static Vector3 WorldPos(GridPos p) { return new Vector3(p.Col, 0f, 3f - p.Row); }

        public void Init(SimCore sim, Transform parent)
        {
            _sim = sim;
            BuildBoard();
            foreach (var kv in sim.Entities)
            {
                var e = kv.Value;
                if (e != null) CreateView(e);
            }
        }

        public EntityView GetView(int entityId)
        {
            EntityView v;
            return _views.TryGetValue(entityId, out v) ? v : null;
        }

        public void OnSimEvent(SimEvent ev)
        {
            switch (ev.Type)
            {
                case SimEventType.EntitySpawned:
                    var e = _sim.GetEntity(ev.SubjectId);
                    if (e != null) CreateView(e);
                    break;
                case SimEventType.EntityRemoved:
                    EntityView rv;
                    if (_views.TryGetValue(ev.SubjectId, out rv))
                    {
                        _views.Remove(ev.SubjectId);
                        if (rv != null && rv.gameObject != null) Destroy(rv.gameObject);
                    }
                    break;
                case SimEventType.DamageDealt:
                    var dv = GetView(ev.SubjectId);
                    if (dv != null) dv.FlashHit();
                    break;
                case SimEventType.EntityDied:
                    var dv2 = GetView(ev.SubjectId);
                    if (dv2 != null) dv2.SetDead();
                    break;
            }
        }

        public void TickViews()
        {
            Camera cam = Camera.main;
            float dt = Time.deltaTime;
            foreach (var kv in _views)
            {
                var v = kv.Value;
                if (v == null) continue;
                var e = _sim.GetEntity(kv.Key);
                if (e == null || e.State == EntityState.Dead) v.SetDead();
                v.Tick(dt, cam);
            }
        }

        private void CreateView(Entity e)
        {
            if (_views.ContainsKey(e.Id)) return;
            var go = new GameObject("Entity_" + e.Id);
            go.transform.SetParent(transform, false);
            var view = go.AddComponent<EntityView>();
            view.Bind(e, e.Side, e.Hero, e.Building);
            _views[e.Id] = view;
        }

        private void BuildBoard()
        {
            // 棋盘:6×255 全格合并为 1 个 Mesh,双 submesh 深浅两色
            var dv = new List<Vector3>(); var dt = new List<int>();
            var lv = new List<Vector3>(); var lt = new List<int>();
            for (int row = 0; row < LaneGrid.Rows; row++)
            {
                for (int col = 0; col < LaneGrid.Cols; col++)
                {
                    var v = ((row + col) & 1) == 0 ? dv : lv;
                    var t = ((row + col) & 1) == 0 ? dt : lt;
                    ViewAssets.AddBox(v, t, new Vector3(col + 0.5f, -0.05f, 2.5f - row), new Vector3(1f, 0.1f, 1f));
                }
            }
            var mesh = ViewAssets.CreateMesh(new List<Vector3>[] { dv, lv }, new List<int>[] { dt, lt });
            ViewAssets.AttachMesh("Board", transform, mesh, new Material[] {
                ViewAssets.Flat(new Color32(0x16, 0x1a, 0x21, 0xff), 0.2f),
                ViewAssets.Flat(new Color32(0x26, 0x2c, 0x36, 0xff), 0.2f) });

            // 废土延展面 + 碎块(固定种子)
            Material waste = ViewAssets.Flat(new Color32(0x0d, 0x10, 0x16, 0xff), 0.15f);
            ViewAssets.DecoBox(waste, new Vector3(127.5f, -0.12f, 0f), new Vector3(274f, 0.1f, 34f), transform, "Wasteland");
            var rng = new System.Random(1234);
            for (int i = 0; i < 8; i++)
            {
                float x = 4f + (float)rng.NextDouble() * 248f;
                float z = (i & 1) == 0 ? 4.5f + (float)rng.NextDouble() * 3f : -4.5f - (float)rng.NextDouble() * 3f;
                ViewAssets.DecoBox(ViewAssets.Flat(new Color32(0x14, 0x16, 0x1c, 0xff)),
                    new Vector3(x, 0.1f, z), new Vector3(0.8f + i * 0.15f, 0.35f, 0.7f + i * 0.1f), transform, "Debris" + i);
            }

            // 边缘围栏:南北发光条,东西短墙
            Material rail = ViewAssets.Mat(new Color(0.02f, 0.02f, 0.02f), 0f, 0.9f, new Color(0.45f, 0.55f, 0.7f), 0.3f);
            ViewAssets.DecoBox(rail, new Vector3(127.5f, 0.06f, 3.25f), new Vector3(255f, 0.06f, 0.14f), transform, "Rail_N");
            ViewAssets.DecoBox(rail, new Vector3(127.5f, 0.06f, -3.25f), new Vector3(255f, 0.06f, 0.14f), transform, "Rail_S");
            Material wall = ViewAssets.Flat(new Color32(0x1a, 0x1e, 0x26, 0xff));
            ViewAssets.DecoBox(wall, new Vector3(-0.3f, 0.3f, 0f), new Vector3(0.25f, 0.6f, 6.6f), transform, "Wall_W");
            ViewAssets.DecoBox(wall, new Vector3(255.3f, 0.3f, 0f), new Vector3(0.25f, 0.6f, 6.6f), transform, "Wall_E");

            // 中央标记:中立资源点列(127)暗金中线
            Material mid = ViewAssets.Mat(new Color(0.02f, 0.02f, 0.02f), 0f, 0.9f, new Color(0.8f, 0.62f, 0.25f), 0.6f);
            ViewAssets.DecoBox(mid, new Vector3(127.5f, 0.06f, 0f), new Vector3(0.15f, 0.05f, 6f), transform, "MidLine");
        }
    }
```

### 3.4 GameBootstrap.cs 全文(现状,只改 CreateCamera 数值)
```csharp
using IronCurtain.Sim;
using UnityEngine;
using UnityEngine.Rendering;
using UnityEngine.Rendering.Universal;

namespace IronCurtain.View
{
    public sealed class GameBootstrap : MonoBehaviour
    {
        public SimCore Sim;
        public BoardView Board;
        public Camera Cam;
        public bool Paused;
        public float Speed = 1f;

        private double _accumulator;

        private void Awake()
        {
            if (Sim != null) return; // 域重载防护:已初始化则跳过
            var cfg = GameConfigParser.FromJson(Resources.Load<TextAsset>("gameconfig").text);
            Sim = MatchSetup.Build(cfg, null, AiTier.Normal);

            CreateCamera();
            CreateLighting();
            CreatePostProcess();

            var boardGo = new GameObject("Board");
            boardGo.transform.SetParent(transform, false);
            Board = boardGo.AddComponent<BoardView>();
            Board.Init(Sim, transform);
            Sim.AddEventListener(Board.OnSimEvent);
            gameObject.AddComponent<CaptureRunner>();

            Debug.Log("[GameBootstrap] Match started");
        }

        private void Update()
        {
            if (Paused || Sim == null) return;
            double interval = 1.0 / Sim.Config.TickRate;
            _accumulator += Time.deltaTime * Speed;
            while (_accumulator >= interval)
            {
                _accumulator -= interval;
                Sim.Step();
            }
            Board.TickViews();
        }
```

```csharp
        private IEnumerator Run()

            if (boot == null || boot.Sim == null)
            {
                Debug.LogError("[capture] GameBootstrap missing");
                yield break;
            }

                if (simSec >= Marks[i])

                if (guard > 200f)
                {
                    Debug.Log("[capture] timeout");
                    break;
                }
            }
```

```csharp
        private static void Quad(List<Vector3> v, List<int> t, Vector3 a, Vector3 b, Vector3 c, Vector3 d)
        {
            int i = v.Count;
            v.Add(a); v.Add(b); v.Add(c); v.Add(d);
            t.Add(i); t.Add(i + 1); t.Add(i + 2);
            t.Add(i); t.Add(i + 2); t.Add(i + 3);
        }

        public static void AddBox(List<Vector3> v, List<int> t, Vector3 c, Vector3 s)
        {
            float x = s.x * 0.5f, y = s.y * 0.5f, z = s.z * 0.5f;
            Quad(v, t, c + new Vector3(-x, -y, z), c + new Vector3(x, -y, z), c + new Vector3(x, y, z), c + new Vector3(-x, y, z));
            Quad(v, t, c + new Vector3(x, -y, -z), c + new Vector3(-x, -y, -z), c + new Vector3(-x, y, -z), c + new Vector3(x, y, -z));
            Quad(v, t, c + new Vector3(x, -y, z), c + new Vector3(x, -y, -z), c + new Vector3(x, y, -z), c + new Vector3(x, y, z));
            Quad(v, t, c + new Vector3(-x, -y, -z), c + new Vector3(-x, -y, z), c + new Vector3(-x, y, z), c + new Vector3(-x, y, -z));
            Quad(v, t, c + new Vector3(-x, y, z), c + new Vector3(x, y, z), c + new Vector3(x, y, -z), c + new Vector3(-x, y, -z));
        }

        // 底面中心 c,向上 h;botHalf/topHalf 为 x/z 半宽;seg 边数;angOff 起始角偏移
        public static void AddPrism(List<Vector3> v, List<int> t, Vector3 c, Vector2 botHalf, Vector2 topHalf, float h, int seg, float angOff = 0f)
        {
            float step = Mathf.PI * 2f / seg;
            for (int i = 0; i < seg; i++)
            {
                float a0 = angOff + step * i, a1 = angOff + step * (i + 1);
                var b0 = new Vector3(c.x + Mathf.Cos(a0) * botHalf.x, c.y, c.z + Mathf.Sin(a0) * botHalf.y);
                var b1 = new Vector3(c.x + Mathf.Cos(a1) * botHalf.x, c.y, c.z + Mathf.Sin(a1) * botHalf.y);
                var t0 = new Vector3(c.x + Mathf.Cos(a0) * topHalf.x, c.y + h, c.z + Mathf.Sin(a0) * topHalf.y);
                var t1 = new Vector3(c.x + Mathf.Cos(a1) * topHalf.x, c.y + h, c.z + Mathf.Sin(a1) * topHalf.y);
                Quad(v, t, b0, b1, t1, t0);
                int tc = v.Count;
                v.Add(new Vector3(c.x, c.y + h, c.z));
                t.Add(tc); t.Add(tc + 1); t.Add(tc + 2);
                v.Add(t0); v.Add(t1);
                t.Add(tc); t.Add(tc + 3); t.Add(tc + 4);
                int bc = v.Count;
                v.Add(new Vector3(c.x, c.y, c.z));
                t.Add(bc); t.Add(bc + 2); t.Add(bc + 1);
                v.Add(b0); v.Add(b1);
                t.Add(bc); t.Add(bc + 4); t.Add(bc + 3);
            }
        }

        public static void AddCylinder(List<Vector3> v, List<int> t, Vector3 c, float rBot, float rTop, float h, int seg, float angOff = 0f)
        {
            AddPrism(v, t, c, new Vector2(rBot, rBot), new Vector2(rTop, rTop), h, seg, angOff);
        }

        public static void AddRing(List<Vector3> v, List<int> t, Vector3 c, float radius, float width, int seg)
        {
            float step = Mathf.PI * 2f / seg, ri = radius - width, ro = radius + width;
            for (int i = 0; i < seg; i++)
            {
                float a0 = step * i, a1 = step * (i + 1);
                Quad(v, t,
                    new Vector3(c.x + Mathf.Cos(a0) * ri, c.y, c.z + Mathf.Sin(a0) * ri),
                    new Vector3(c.x + Mathf.Cos(a1) * ri, c.y, c.z + Mathf.Sin(a1) * ri),
                    new Vector3(c.x + Mathf.Cos(a1) * ro, c.y, c.z + Mathf.Sin(a1) * ro),
                    new Vector3(c.x + Mathf.Cos(a0) * ro, c.y, c.z + Mathf.Sin(a0) * ro));
            }
        }

        public static Mesh CreateMesh(List<Vector3>[] verts, List<int>[] tris)
        {
            var m = new Mesh();
            var all = new List<Vector3>();
            for (int s = 0; s < verts.Length; s++) all.AddRange(verts[s]);
            m.SetVertices(all);
            m.subMeshCount = verts.Length;
            for (int s = 0; s < verts.Length; s++)
            {
                int off = 0;
                for (int p = 0; p < s; p++) off += verts[p].Count;
                var idx = new List<int>(tris[s].Count);
                for (int i = 0; i < tris[s].Count; i++) idx.Add(tris[s][i] + off);
                m.SetTriangles(idx, s);
            }
            m.RecalculateNormals();
            return m;
        }

        public static GameObject AttachMesh(string name, Transform parent, Mesh mesh, Material[] mats)
        {
            var go = new GameObject(name);
            go.transform.SetParent(parent, false);
            go.AddComponent<MeshFilter>().sharedMesh = mesh;
            go.AddComponent<MeshRenderer>().sharedMaterials = mats;
            return go;
        }

        // 极简小装饰(血条/驻军光点等),单 submesh 小网格
        public static GameObject DecoBox(Material mat, Vector3 lpos, Vector3 size, Transform parent, string name = "deco")
        {
            var bv = new List<Vector3>(); var bt = new List<int>();
            AddBox(bv, bt, Vector3.zero, size);
            var mesh = CreateMesh(new List<Vector3>[] { bv }, new List<int>[] { bt });
            var go = AttachMesh(name, parent, mesh, new Material[] { mat });
            go.transform.localPosition = lpos;
            return go;
        }

        public static GameObject DecoQuad(Material mat, Vector3 lpos, Vector2 size, Transform parent, string name = "decoQ")
        {
            if (_quadMesh == null)
            {
                var v = new List<Vector3>(); var t = new List<int>();
                Quad(v, t, new Vector3(-0.5f, -0.5f, 0), new Vector3(0.5f, -0.5f, 0), new Vector3(0.5f, 0.5f, 0), new Vector3(-0.5f, 0.5f, 0));
                _quadMesh = CreateMesh(new List<Vector3>[] { v }, new List<int>[] { t });
            }
            var go = new GameObject(name);
            go.transform.SetParent(parent, false);
            go.transform.localPosition = lpos;
            go.transform.localScale = new Vector3(size.x, size.y, 1f);
            go.AddComponent<MeshFilter>().sharedMesh = _quadMesh;
            go.AddComponent<MeshRenderer>().sharedMaterial = mat;
            return go;
        }
    }
}
```
