### 坐标与渲染惯例
- Sim 连续坐标 `SimVec2{X=列米, Y=行米}`;`Entity.PosF`=单位权威位置;`Entity.Pos`=格(GridPos{Row,Col},建筑)。
- 世界坐标:`BoardView.WorldPos(SimVec2)` / `BoardView.WorldPos(GridPos)` → 世界 x=X、z=3−Y、y=0 地面。
- 现有 IMGUI 面板布局(互不遮挡):左上 BarracksOverviewPanel(static `LastPanelRect`,GUI 坐标 y 从顶起);右下 MinimapPanel(static `LastPanelRect`);底中撤退按钮;顶中布阵提示。**顶栏=屏幕最顶部横条**,与上述无冲突。

public struct FloatItem
{
    public int EntityId;      // 受击者 id(渲染时若实体已移除,用 Origin 定位)
    public SimVec2 Origin;    // 事件时刻受击者 PosF 快照
    public string Text;       // "-12"
    public int ColorKind;     // 0=己方受击(红字) 1=敌方/中立受击(金黄字)
    public float Age;         // 已存活秒
}
public sealed class FloatingTextModel
    public const float Life = 1.0f;
    public const int MaxActive = 40;
    public readonly List<FloatItem> Items = new List<FloatItem>();
    public bool TryAdd(SimCore sim, Side playerSide, int entityId, float amount)
        // 1) e=sim.GetEntity(entityId),null 或 State==Dead → false
        // 2) 可见性:e.Side==playerSide → 可见;单位(非建筑)→ VisionSystem.IsUnitVisible(sim, playerSide, entityId);建筑(Kind==Building)→ VisionSystem.IsBuildingKnown(sim, playerSide, entityId);不可见 → false
        // 3) 满 MaxActive → 先移除 Items[0](最旧)
        // 4) 追加条目:Origin=e.PosF,Text=DamageText(amount),ColorKind=(e.Side==playerSide?0:1),Age=0
    public void Advance(float dt)  // 全体 Age+=dt;Age>Life 的移除
    public static string DamageText(float amount) // "-"+System.Math.Round(amount, System.MidpointRounding.AwayFromZero)
    public static float RiseOffset(float age)     // age*1.2(上浮米数,渲染端用)

namespace IronCurtain.View

    /// <summary>
    /// M8.6 点选/框选己方(蓝方)主将 + 右键移动命令(手动级)+ T 键自动布阵 + 淡出提示 + 选中圈/点击标记。

        private const float PickRadius = 3f;      // 点选主将命中半径(米)
        private const float RingRadius = 1.3f;    // 选中圈半径(米)
        private const float HintDuration = 2.5f;  // 提示淡出时长(秒)

        private static readonly Plane GroundPlane = new Plane(Vector3.up, Vector3.zero);

        private SimCore _sim;
        private Camera _cam;
        private readonly List<int> _selected = new List<int>();

        private bool _dragging;
        private Vector2 _dragStart;
        private string _selKey = "";

        private readonly List<GameObject> _rings = new List<GameObject>();
        private GameObject _ringRoot;
        private float _hintT; // 提示剩余显示时间(秒)

        private GameObject _marker;
        private Material _markerMat;
        private float _markerT;
        private Rect _retreatBtnRect; // M8.7 撤退按钮占位(兼做点击穿透过滤)

        public void Init(SimCore sim, Camera cam)
        {
            _sim = sim;
            _cam = cam;

        private bool MouseOverPanel
        {
            get
            {
                Vector2 m = Input.mousePosition;
                // 面板 Rect 为 GUI 坐标(y 从顶起),鼠标 y 从底起,需翻转
                // 撤退按钮 Rect 仅在有选中时参与拦截(清空选择后残留旧值会让该区域左键永久失效——盲测挑出)
                return BarracksOverviewPanel.LastPanelRect.Contains(new Vector2(m.x, Screen.height - m.y))
                    // 小地图区域点击不穿透到世界
                    || MinimapPanel.LastPanelRect.Contains(new Vector2(m.x, Screen.height - m.y))
                    || (_selected.Count >= 1 && _retreatBtnRect.Contains(new Vector2(m.x, Screen.height - m.y)));
            }

        private void Update()
        {
            if (_sim == null || _cam == null) return;
            PruneSelection();
            _hintT = Mathf.Max(0f, _hintT - Time.deltaTime);

            HandleLeftButton();
            HandleRightButton();
            HandleFormationKey();
            UpdateRings();
            UpdateMarker();
        }

        private void PruneSelection()
        {
            for (int i = _selected.Count - 1; i >= 0; i--)
            {
                var e = _sim.GetEntity(_selected[i]);
                if (e == null || e.State == EntityState.Dead || e.IsSoldier)
                    _selected.RemoveAt(i);
            }
        }

        private void HandleLeftButton()
        {
            if (Input.GetMouseButtonDown(0) && !MouseOverPanel)
            {
                _dragging = true;
                _dragStart = Input.mousePosition;
            }
            if (!Input.GetMouseButtonUp(0) || !_dragging) return;
            _dragging = false;
            Vector2 m = Input.mousePosition;
            float dx = m.x - _dragStart.x, dy = m.y - _dragStart.y;
            bool click = dx * dx + dy * dy < 64f; // 屏幕位移平方和 <64px² = 点选
            if (click)
            {
                int id = PickGeneral();
                _selected.Clear();
                if (id != 0) _selected.Add(id);
            }
            else
            {
                var rect = Rect.MinMaxRect(
                    Mathf.Min(_dragStart.x, m.x), Mathf.Min(_dragStart.y, m.y),
                    Mathf.Max(_dragStart.x, m.x), Mathf.Max(_dragStart.y, m.y));
                _selected.Clear();
                foreach (var kv in _sim.Entities)
                {
                    var e = kv.Value;
                    if (e == null || e.Kind != EntityKind.Unit || e.IsSoldier || e.Side != Side.Blue
                        || e.State == EntityState.Dead) continue;
                    var sp = _cam.WorldToScreenPoint(BoardView.WorldPos(e.PosF));
                    if (rect.Contains(new Vector2(sp.x, sp.y), true)) _selected.Add(e.Id);
                }
            }
        }

        private int PickGeneral()
        {
            Vector3 wp;
            if (!RayGround(out wp)) return 0;
            var p = new SimVec2(wp.x, 3f - wp.z);
            int best = 0;
            float bestD = PickRadius * PickRadius;
            int soldierGeneral = 0;
            float soldierD = PickRadius * PickRadius;
            foreach (var kv in _sim.Entities)
            {
                var e = kv.Value;
                if (e == null || e.Kind != EntityKind.Unit || e.Side != Side.Blue || e.State == EntityState.Dead) continue;
                float dx = e.PosF.X - p.X, dy = e.PosF.Y - p.Y;
                float d = dx * dx + dy * dy;
                if (e.IsSoldier)
                {
                    if (d < soldierD) { soldierD = d; soldierGeneral = e.GeneralId; }
                }
                else if (d < bestD) { bestD = d; best = e.Id; }
            }
            if (best != 0) return best;      // 主将优先
            return soldierGeneral;           // 命中士兵 → 其主将
        }

        private void HandleRightButton()
        {
            if (!Input.GetMouseButtonDown(1) || _selected.Count == 0 || MouseOverPanel) return;
            Vector3 wp;
            if (!RayGround(out wp)) return;
            var grid = new SimVec2(wp.x, 3f - wp.z).ToGrid();
            foreach (var id in _selected)
                _sim.SetMoveTarget(id, grid, true);
            SpawnMarker(wp);
        }

        {

        private bool RayGround(out Vector3 wp)

            if (!GroundPlane.Raycast(ray, out enter)) return false;
            wp = ray.GetPoint(enter);
            return true;

            }

        {

            _rings.Clear();

            if (_selected.Count == 0) return;

                if (e == null) continue;
                ViewAssets.DecoBox(ringMat, new Vector3(0f, 0.03f, 0f),
                    new Vector3(RingRadius * 2f, 0.03f, RingRadius * 2f), _ringRoot.transform, "Ring" + id);
                _rings.Add(_ringRoot.transform.GetChild(_ringRoot.transform.childCount - 1).gameObject);
                _rings[_rings.Count - 1].transform.position = BoardView.WorldPos(e.PosF);
            }
        }

        private void SpawnMarker(Vector3 wp)
        {
            if (_marker != null) Destroy(_marker);
            Material baseMat = ViewAssets.Flat(new Color(1f, 1f, 1f, 0.45f));
            _markerMat = Instantiate(baseMat); // 独立实例,可调 alpha
            _marker = new GameObject("ClickMarker");
            ViewAssets.DecoBox(_markerMat, new Vector3(0f, 0.04f, 0f), new Vector3(2f, 0.02f, 2f), _marker.transform, "Disc");
            _marker.transform.position = wp;
            _markerT = 1f;
        }

        private void UpdateMarker()
        {
            if (_marker == null) return;
            _markerT -= Time.deltaTime;
            if (_markerT <= 0f) { Destroy(_marker); _marker = null; _markerMat = null; return; }
            float a = Mathf.Clamp01(_markerT); // 1s 线性淡出
            if (_markerMat != null)
            {
                if (_markerMat.HasProperty("_BaseColor"))
                    _markerMat.SetColor("_BaseColor", new Color(1f, 1f, 1f, 0.45f * a));
                else _markerMat.color = new Color(1f, 1f, 1f, 0.45f * a);
            }
            _marker.transform.localScale = Vector3.one * Mathf.Max(0.05f, a); // 不透明材质下的可见消退兜底
        }

        private void OnGUI()
        {
            if (_selected.Count >= 1)
            {
                _retreatBtnRect = new Rect(Screen.width * 0.5f - 60f, Screen.height - 60f, 120f, 40f);
                if (GUI.Button(_retreatBtnRect, "撤退"))
                    foreach (var id in new List<int>(_selected))
                        RetreatSystem.OrderRetreat(_sim, id);
            }
            if (_selected.Count < 2 || _hintT <= 0f) return;
            var style = new GUIStyle(GUI.skin.label) { fontSize = 24, alignment = TextAnchor.MiddleCenter };
            style.normal.textColor = Color.white;
            var prev = GUI.color;
            GUI.color = new Color(1f, 1f, 1f, Mathf.Clamp01(_hintT / HintDuration)); // alpha 从 1 线性降到 0
            GUI.Label(new Rect(Screen.width * 0.5f - 160f, 24f, 320f, 36f), "按 T 键自动布阵", style);
            GUI.color = prev;
        }
    }

using UnityEngine;

namespace IronCurtain.View
{
    /// <summary>D18 镜头边缘滑动:鼠标贴屏幕边缘 → 匀速平移焦点;速度为公开可配置参数。</summary>
    public sealed class CameraRig : MonoBehaviour
    {
        public float EdgeScrollSpeed = 28f;  // 平移速度(米/秒),公开可配置
        public float EdgeMarginPx = 12f;     // 边缘带宽(像素)

        private Camera _cam;
        private Vector3 _offset;             // 相机位置 - 焦点(保持既有构图)
        private float _focusCol, _focusRow;

        public void Init(Camera cam, float focusColM, float focusRowM)
        {
            _cam = cam;
            _offset = cam.transform.position - WorldOf(focusColM, focusRowM);
            SetFocus(focusColM, focusRowM);
        }

        public void SetFocus(float colM, float rowM)
        {
            _focusCol = MinimapLayout.ClampFocusX(colM);
            _focusRow = MinimapLayout.ClampFocusY(rowM);
            if (_cam != null)
                _cam.transform.position = WorldOf(_focusCol, _focusRow) + _offset; // 旋转不动,偏移保构图
        }

        private static Vector3 WorldOf(float colM, float rowM)
        {
            return new Vector3(colM, 0f, 3f - rowM);
        }

        private static bool OverRect(Rect r, Vector2 mouse)
        {
            // GUI 坐标 y 从顶起,鼠标 y 从底起,需翻转
            return r.width > 0f && r.height > 0f && r.Contains(new Vector2(mouse.x, Screen.height - mouse.y));
        }

        private void Update()
        {
            if (_cam == null) return;
            Vector2 m = Input.mousePosition;
            // 悬停任何 IMGUI 面板时跳过(防误滚)
            if (OverRect(BarracksOverviewPanel.LastPanelRect, m) || OverRect(MinimapPanel.LastPanelRect, m)) return;

            float dx = 0f, dy = 0f; // dy 为行米增量(北=行增)
            if (m.x <= EdgeMarginPx) dx = -1f;
            else if (m.x >= Screen.width - EdgeMarginPx) dx = 1f;
            if (m.y >= Screen.height - EdgeMarginPx) dy = 1f;          // 屏幕上边缘 → 北
            else if (m.y <= EdgeMarginPx) dy = -1f;                     // 屏幕下边缘 → 南(rowY 减小)
            if (dx == 0f && dy == 0f) return;
            float step = EdgeScrollSpeed * Time.deltaTime;
            SetFocus(_focusCol + dx * step, _focusRow + dy * step);
        }
    }
}

using IronCurtain.Sim;
using UnityEngine;

            boardGo.transform.SetParent(transform, false);
            Board = boardGo.AddComponent<BoardView>();
            Board.Init(Sim, transform);
            Board.PlayerSide = Side.Blue;
            Sim.AddEventListener(Board.OnSimEvent);
            var overview = gameObject.AddComponent<BarracksOverviewPanel>();
            overview.Init(Sim, Side.Blue);
            gameObject.AddComponent<CaptureRunner>();
            var selection = gameObject.AddComponent<SelectionSystem>();
            selection.Init(Sim, Cam);

            var fogGo = new GameObject("FogView");
            fogGo.transform.SetParent(transform, false);
            _fog = fogGo.AddComponent<FogView>();
            _fog.Init(Sim, Side.Blue, transform);
            var minimap = gameObject.AddComponent<MinimapPanel>();
            minimap.Init(Sim, Side.Blue, _rig, _fog, Cam);

            Debug.Log("[GameBootstrap] Match started");
        }

        private void Update()
        {
            if (Paused || Sim == null) return;
            double interval = 1.0 / Sim.Config.TickRate;
            _accumulator += Time.deltaTime * Speed;
            while (_accumulator >= interval)
            {
                _accumulator -= interval;
                Sim.Step();
            }
            Board.TickViews();
        }

        {
            if (Sim != null) Sim.ClearEventListeners();
        }

            Cam.orthographicSize = 27f;

            go.transform.LookAt(new Vector3(143f, 0f, -12f));

            data.renderPostProcessing = true;

            go.transform.SetParent(transform, false);
            var vol = go.AddComponent<Volume>();
            vol.isGlobal = true;
            vol.profile = profile;
        }
    }
