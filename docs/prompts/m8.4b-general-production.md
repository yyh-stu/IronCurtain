# M8.4b 将兵化改造·主基地产将+名额制(Sim 第二批)

你是《铁幕:前线》Sim 层主程。将兵制拍板(权威=PROJECT_STATE 变更#9):**兵营只承担补兵;将一律从主基地产出,20s/将,出场自带满编 20 兵;人口=将名额制,测试版上限 4(开局赠 3:磐石/断峰/坠日);测试期六将均可产(不做选将)**。M8.4a 已落地补兵池与 SpawnSoldier。

## 二、需求与规范(权威)
### 2.1 SimCore 人口→名额制(签名不动,语义换)
- `PopulationOf(Side)` = 该方存活主将数:Kind==Unit && !IsSoldier && Side==side && State!=Dead 的实体数。
- `PopulationCapOf(Side)` = **GeneralCap=4**(公开常量 `public const int GeneralCap = 4;`,注释"测试版 4;正式版 8")。
- `PopCostOf(HeroId)` = 恒 1(每个将占 1 名额)。
- 其余人口相关调用点(若有)按新语义自洽;不改方法名,避免波及 BarracksOverviewModel/AiSystem。

### 2.2 ProductionSystem 改为"主基地产将"(兵营不再产任何东西)
- `public const float GeneralProductionSeconds = 20f;`(暂定 20s,注释注明)。
- `Tick(sim, dt)`:遍历 Building==**Base** 且 `Prod != null` 的建筑;每槽:
  - 空闲且 Pending 有货:**名额校验通过**(`sim.PopulationOf(side) < sim.PopulationCapOf(side)`)才 Dequeue 开工;名额满则等待(不报错)。
  - 生产进度按 GeneralProductionSeconds;完成后:**名额再校验**,满则 WaitingForSpawn 等待;否则在**基地朝敌侧前缘**(dir=蓝 +1/红 -1)找出生格(复用 M8.1a 前缘规则:`col = b.Pos.Col + dir*(ceil(HalfW)+1)`,候选行 {0,-1,1,-2,2},过滤越界/椭圆/水域/占地格/单位 0.5m 内),调用 `RegimentSystem.SpawnGeneral(sim, b.Side, slot.Current.Value, 出生格)`,然后清槽;出生格全占 → WaitingForSpawn=true 下 tick 重试。
- `Enqueue(sim, buildingId, hero, out error)`:目标必须是 Building==Base("目标不是主基地");其余入队逻辑照旧(最空槽)。
- `CancelLast`:照旧(槽语义不变)。`OnBarracksDestroyed` 改名为或保留(保留原名但作用于基地亦可;若保留,废墟化时基地 Prod 清空——把调用点改为 Base)。
- 出生点占用集合:`BuildingFootprint.CellsOccupied` 对全图非废墟建筑建 HashSet<long>(同 M8.1a)。

### 2.4 AiSystem 产将(FillProduction 或等价)
- 原"给兵营入队英雄"改为:找己方 Base 建筑,`ProductionSystem.Enqueue(sim, baseId, hero, out _)`,在 `PopulationOf < PopulationCapOf` 且队列空时入队(简单/普通都产,产将类型沿用原优先级逻辑;普通 AI 镇岳配比等逻辑能保留则保留,不能则简化为按原顺序产)。

### 2.5 ProductionTests.cs 重写(将生产口径;测试意图保持:节奏/名额/满员等待)
- 助手:建基地 `MakeBase(side, row, col)`(SpawnBuilding Base + Prod=new ProductionSet())。原兵营测试全部改为基地。
- 断言方向:
  1. `T1_SingleBase_Cadence`:1 基地 5 个将入队(名额 4)→ 前 4 个按 20s 节奏产出(带满编 20 兵,断言每个产出的主将实体的士兵数==20),第 5 个待名额(名额满,等将死)。

  5. `T5_Cancel`:取消进行中/待产返回成功,队列回退。
  6. `T6_Register`:产出的将=非士兵 Unit、Hero 正确、位于基地前缘(距基地中心 > HalfW+0.5)。

### 3.1 ProductionSystem.cs
```csharp
using System.Collections.Generic;

namespace IronCurtain.Sim
{
    /// <summary>
    /// 生产系统:兵营 3 队列、人口校验、取消返还、出生点占用等待、自动行军。
    /// 纯 tick 驱动,无 UnityEngine 依赖。M8.1a:出生点贴建筑占地前缘。
    /// </summary>
    public static class ProductionSystem
    {
        /// <summary>浮点比较容差(防 0.1f 累积漂移)。</summary>
        private const float Epsilon = 1e-4f;

        /// <summary>出生候选行序:同排/上/下/再上/再下(从近到远)。</summary>
        private static readonly int[] SpawnDRows = { 0, -1, 1, -2, 2 };

        public static void Tick(SimCore sim, float dt)
        {
            // 快照遍历:出生会向 Entities 字典添加实体,禁止边枚举边修改
            var snapshot = new List<Entity>(sim.Entities.Values);

            foreach (var e in snapshot)
            {
                if (e.Kind != EntityKind.Building) continue;
                if (e.Building != BuildingId.Barracks) continue;
                var prod = e.Prod;
                if (prod == null) continue;

                for (int i = 0; i < prod.Slots.Length; i++)
                {
                    var slot = prod.Slots[i];

                    // 空闲槽开工:人口校验通过才从待产队列取出
                    if (slot.Current == null && slot.Pending.Count > 0)
                    {
                        var hero = slot.Pending.Peek();
                        int pop = sim.PopCostOf(hero);
                        if (sim.PopulationOf(e.Side) + pop <= sim.PopulationCapOf(e.Side))
                        {
                            slot.Current = hero;
                            slot.Progress = 0f;
                            slot.WaitingForSpawn = false;
                            slot.Pending.Dequeue();
                        }
                    }

                    if (slot.Current != null)
                    {
                        if (!slot.WaitingForSpawn && slot.Progress + dt < ProductionSeconds(sim, e.Side, slot.Current.Value) - Epsilon)
                        {
                            slot.Progress += dt;
                            continue;
                        }

                        // 完成或等待出生中:尝试出生
                        if (TrySpawnUnit(sim, e, slot.Current.Value, out var unit))
                        {
                            slot.Current = null;
                            slot.Progress = 0f;
                            slot.WaitingForSpawn = false;
                            sim.SetMoveTarget(unit.Id, sim.RallyPointOf(e.Side));
                        }
                        else
                        {
                            slot.WaitingForSpawn = true; // 出生点全占,下 tick 重试
                        }
                    }
                }
            }
        }

        private static double ProductionSeconds(SimCore sim, Side side, HeroId hero)
        {
            if (hero == HeroId.ZhenYue && sim.IsUpgraded(side, HeroId.ZhenYue))
                return sim.Game.Production.ZhenYueUpgradedSeconds; // 迅捷:7.5s
            int pop = sim.PopCostOf(hero);
            return pop <= 1 ? sim.Game.Production.Pop1Seconds
                            : sim.Game.Production.ZhenYueSeconds;
        }

        private static bool TrySpawnUnit(SimCore sim, Entity barracks, HeroId heroId, out Entity unit)
        {
            unit = null;
            int dir = barracks.Side == Side.Blue ? 1 : -1;

            // 全图非废墟建筑占地格集合(多格占用,M8.1a)
            var occupied = new HashSet<long>();
            foreach (var e in sim.Entities.Values)
                if (e.Kind == EntityKind.Building && !e.Rubbled)
                    foreach (var cell in BuildingFootprint.CellsOccupied(e))
                        occupied.Add(((long)cell.Row << 32) | (uint)cell.Col);

            // 前缘列:占地半宽外再让 1 格
            int col = barracks.Pos.Col + dir * ((int)System.Math.Ceiling(BuildingFootprint.HalfW(barracks)) + 1);
            for (int i = 0; i < SpawnDRows.Length; i++)
            {
                int row = barracks.Pos.Row + SpawnDRows[i];
                if (row < 0 || row >= LaneGrid.Rows || col < 0 || col >= LaneGrid.Cols) continue;
                var cand = new GridPos(row, col);
                if (!TerrainMap.IsInside(cand)) continue;                    // 椭圆外不可出生
                if (TerrainMap.IsWater(SimVec2.FromGrid(cand))) continue;    // 水域不可出生
                long key = ((long)row << 32) | (uint)col;
                if (occupied.Contains(key)) continue;

                // 连续坐标下出生点还要避开既有单位(0.5 米内)
                var center = SimVec2.FromGrid(cand);
                bool tooClose = false;
                foreach (var e in sim.Entities.Values)
                    if (e.Kind == EntityKind.Unit && SimVec2.Distance(e.PosF, center) < 0.5f)
                    { tooClose = true; break; }
                if (tooClose) continue;

                var c = sim.Game.Heroes[(int)heroId];
                var u = sim.SpawnEntity(barracks.Side, EntityKind.Unit, cand);
                u.Hero = heroId; u.MaxHp = c.MaxHp; u.Hp = c.MaxHp;
                u.Attack = c.Attack; u.AttackIntervalSec = (float)c.AttackIntervalSec;
                u.Range = c.Range; u.MoveSpeed = (float)c.MoveSpeed;
                if (c.Shield > 0) { u.Shield = c.Shield; u.ShieldMax = c.Shield; }
                u.PosF = SimVec2.FromGrid(cand); // 格中心连续坐标
                unit = u;
                return true;
            }
            return false;
        }

        public static bool Enqueue(SimCore sim, int barracksId, HeroId hero, out string error)
        {
            error = null;
            if (!sim.Entities.TryGetValue(barracksId, out var b) ||
                b.Kind != EntityKind.Building || b.Building != BuildingId.Barracks)
            {
                error = "目标不是兵营";
                return false;
            }
            var prod = b.Prod;
            if (prod == null) { error = "兵营无生产集"; return false; }

            int best = 0;
            for (int i = 1; i < prod.Slots.Length; i++)
                if (prod.Slots[i].Pending.Count < prod.Slots[best].Pending.Count) best = i;
            prod.Slots[best].Pending.Enqueue(hero);
            return true;
        }

        public static bool CancelLast(SimCore sim, int barracksId, int slotIndex, out string error)
        {
            error = null;
            if (!sim.Entities.TryGetValue(barracksId, out var b) ||
                b.Kind != EntityKind.Building || b.Building != BuildingId.Barracks)
            {
                error = "目标不是兵营";
                return false;
            }
            var prod = b.Prod;
            if (prod == null) { error = "兵营无生产集"; return false; }
            if (slotIndex < 0 || slotIndex >= prod.Slots.Length) { error = "槽索引越界"; return false; }

            var slot = prod.Slots[slotIndex];
            if (slot.Pending.Count > 0)
            {
                var list = new List<HeroId>(slot.Pending);
                list.RemoveAt(list.Count - 1);
                slot.Pending.Clear();
                foreach (var h in list) slot.Pending.Enqueue(h);
                return true;
            }
            if (slot.Current != null)
            {
                slot.Current = null;
                slot.Progress = 0f;
                slot.WaitingForSpawn = false;
                return true;
            }
            error = "队列为空";
            return false;
        }

        public static void OnBarracksDestroyed(SimCore sim, Entity barracks)
        {
            if (barracks == null) return;
            if (barracks.Prod != null) barracks.Prod.Clear();
        }
    }
}
```

### 3.2 SimCore.cs
```csharp
using System;
using System.Collections.Generic;

namespace IronCurtain.Sim
{
    public sealed class SimCore
    {
        public SimConfig Config { get; }
        public SeededRng Rng { get; }
        public long Tick { get; private set; }
        public int EntityCount => _entities.Count;
        public IReadOnlyDictionary<int, Entity> Entities => _entities;

        /// <summary>游戏数值配置(由调用方注入;null 时战斗模块不生效)。</summary>
        public GameConfig Game { get; set; }

        /// <summary>胜负:大本营被摧毁时置位。</summary>
        public Side? Winner { get; private set; }
        public void SetWinner(Side side) { Winner = side; }

        // 补兵池(M8.4a):按阵营的全局池
        public readonly Dictionary<Side, float> ReinforceAccum = new Dictionary<Side, float> { { Side.Blue, 0f }, { Side.Red, 0f } };
        public readonly Dictionary<Side, int> ReinforcePool = new Dictionary<Side, int> { { Side.Blue, 0 }, { Side.Red, 0 } };

        // 升级(M4):每方已购买的英雄被动
        private readonly Dictionary<Side, HashSet<HeroId>> _upgrades = new Dictionary<Side, HashSet<HeroId>>();

        public bool IsUpgraded(Side side, HeroId hero)
        {
            return _upgrades.TryGetValue(side, out var s) && s.Contains(hero);
        }

        /// <summary>2EP 购买升级,单英雄每局一次。</summary>
        public bool BuyUpgrade(Side side, HeroId hero, out string error)
        {
            error = null;
            if (IsUpgraded(side, hero)) { error = "该英雄已升级"; return false; }
            if (!SpendEp(side, Game.Ep.UpgradeCost)) { error = "EP 不足"; return false; }
            if (!_upgrades.TryGetValue(side, out var s))
            {
                s = new HashSet<HeroId>();
                _upgrades[side] = s;
            }
            s.Add(hero);
            return true;
        }

        // 经济(M2)
        private readonly Dictionary<Side, float> _ep = new Dictionary<Side, float>();
        private readonly Dictionary<Side, float> _epAccum = new Dictionary<Side, float>();
        private readonly Dictionary<Side, GridPos> _rally = new Dictionary<Side, GridPos>
        {
            { Side.Blue, new GridPos(15, 270) },
            { Side.Red, new GridPos(15, 16) },
        };

        private readonly Dictionary<int, Entity> _entities = new Dictionary<int, Entity>();
        private readonly HashSet<int> _occupied = new HashSet<int>();
        private readonly Queue<ICommand> _commandQueue = new Queue<ICommand>();
        private readonly List<Action<SimEvent>> _listeners = new List<Action<SimEvent>>();
        private readonly LaneGrid _grid = new LaneGrid();
        private int _nextId = 1;

        public SimCore(SimConfig config, SeededRng rng)
        {
            Config = config;
            Rng = rng;
        }

        public Entity SpawnEntity(Side side, EntityKind kind, GridPos pos)
        {
            if (!_grid.InBounds(pos))
                throw new InvalidOperationException($"位置越界: {pos}");
            int idx = _grid.ToIndex(pos);
            if (_occupied.Contains(idx))
                throw new InvalidOperationException($"格子已占用: {pos}");
            // 连续坐标下单位不再占格;出生防重叠用距离校验(驻军/城内单位不占位)
            if (kind == EntityKind.Unit)
            {
                var center = SimVec2.FromGrid(pos);
                foreach (var o in _entities.Values)
                    if (o.Kind == EntityKind.Unit && o.GarrisonBuildingId == 0
                        && SimVec2.Distance(o.PosF, center) < 0.5f)
                        throw new InvalidOperationException($"出生点过近: {pos}");
            }

            var e = new Entity(_nextId++, side, kind, pos, 0f);
            e.PosF = SimVec2.FromGrid(pos);
            _entities[e.Id] = e;
            if (kind == EntityKind.Building)
                _occupied.Add(idx);
            Emit(new SimEvent(Tick, SimEventType.EntitySpawned, e.Id));
            return e;
        }

        public bool RemoveEntity(int id)
        {
            if (!_entities.TryGetValue(id, out var e)) return false;
            if (e.Kind == EntityKind.Building)
                _occupied.Remove(_grid.ToIndex(e.Pos));
            _entities.Remove(id);
            Emit(new SimEvent(Tick, SimEventType.EntityRemoved, id));
            return true;
        }

        public Entity GetEntity(int id) => _entities.TryGetValue(id, out var e) ? e : null;

        /// <summary>释放格位占用(驻军/入城交换等"离格但实体仍在"的场景)。</summary>
        public void FreeCell(GridPos pos)
        {
            _occupied.Remove(_grid.ToIndex(pos));
        }

        public void SubmitCommand(ICommand cmd) => _commandQueue.Enqueue(cmd);

        /// <summary>单位移动指令:目标越界/实体非单位时抛异常。</summary>
        public void SetMoveTarget(int id, GridPos target)
        {
            if (!_entities.TryGetValue(id, out var e))
                throw new InvalidOperationException($"实体不存在: {id}");
            if (e.Kind != EntityKind.Unit)
                throw new InvalidOperationException($"非线单位不可设置移动目标: {id}");
            if (!_grid.InBounds(target))
                throw new InvalidOperationException($"移动目标越界: {target}");

            e.MoveTarget = target;
            e.HasMoveTarget = true;
            e.State = EntityState.Moving;
        }

        public void Step()
        {
            Tick++;
            Emit(new SimEvent(Tick, SimEventType.StepBegin, 0));
            while (_commandQueue.Count > 0)
            {
                var cmd = _commandQueue.Dequeue();
                cmd.Execute(this);
                Emit(new SimEvent(Tick, SimEventType.CommandExecuted, 0));
            }
            MoveUnits();
            CombatUpdate();
            EconomyUpdate();
            SiegeSystem.Tick(this, 1f / Config.TickRate);
            TowerCombatSystem.Tick(this, 1f / Config.TickRate);
            GarrisonSystem.Tick(this, 1f / Config.TickRate);
            DestroySystem.Tick(this, 1f / Config.TickRate);
            HexSystem.Tick(this, 1f / Config.TickRate);
            VisionSystem.Tick(this, 1f / Config.TickRate);
            AiSystem.Tick(this, 1f / Config.TickRate);
            Emit(new SimEvent(Tick, SimEventType.StepEnd, 0));
        }

        public void AddEventListener(Action<SimEvent> listener) => _listeners.Add(listener);
        public void ClearEventListeners() => _listeners.Clear();

        // ===== 经济与生产挂钩(M2)=====

        /// <summary>生成建筑实体(归属方 owner)。中立建筑无血量;已方建筑按配置填数值。</summary>
        public Entity SpawnBuilding(Side owner, BuildingId type, GridPos pos)
        {
            var e = SpawnEntity(owner, EntityKind.Building, pos);
            e.Building = type;
            if (Game != null && owner != Side.Neutral)
            {
                var c = BuildingConfigOf(type);
                e.MaxHp = c.MaxHp;
                e.Hp = c.MaxHp;
                e.Attack = c.Attack;
                e.AttackIntervalSec = (float)c.AttackIntervalSec;
                e.Range = c.Range;
            }
            return e;
        }

        public BuildingCfg BuildingConfigOf(BuildingId id)
        {
            foreach (var b in Game.Buildings)
                if (b.Id == id) return b;
            throw new InvalidOperationException("建筑配置缺失: " + id);
        }

        /// <summary>某方拥有的资源点数量(按场上建筑实况计算)。</summary>
        public int OwnedResourcePointsOf(Side side)
        {
            int n = 0;
            foreach (var e in _entities.Values)
                if (e.Kind == EntityKind.Building && e.Building == BuildingId.ResourcePoint && e.Side == side)
                    n++;
            return n;
        }

        /// <summary>英雄人口占用(Game 配置)。</summary>
        public int PopCostOf(HeroId id)
        {
            foreach (var h in Game.Heroes)
                if (h.Id == id) return h.Population;
            return 0;
        }

        /// <summary>已用人口 = 场上单位人口 + 兵营进行中生产人口。</summary>
        public int PopulationOf(Side side)
        {
            int used = 0;
            foreach (var e in _entities.Values)
            {
                if (e.Side != side) continue;
                if (e.Kind == EntityKind.Unit && e.Hero != null)
                    used += PopCostOf(e.Hero.Value);
                else if (e.Kind == EntityKind.Building && e.Building == BuildingId.Barracks && e.Prod != null)
                    used += e.Prod.InProgressPop(id => PopCostOf(id));
            }
            return used;
        }

        public int PopulationCapOf(Side side)
        {
            int cap = (int)Game.Production.BasePopulation;
            cap += Game.Production.ResourcePointPopulation * OwnedResourcePointsOf(side);
            return cap + HexSystem.PopCapBonusOf(this, side);
        }

        public float EpOf(Side side) => _ep.TryGetValue(side, out var v) ? v : 0f;

        public bool SpendEp(Side side, float amount)
        {
            float cur = EpOf(side);
            if (cur < amount) return false;
            _ep[side] = cur - amount;
            return true;
        }

        public GridPos RallyPointOf(Side side) => _rally[side];
        public void SetRallyPoint(Side side, GridPos p) { _rally[side] = p; }

        private void EconomyUpdate()
        {
            if (Game == null) return;
            float dt = 1f / Config.TickRate;

            foreach (Side s in new[] { Side.Blue, Side.Red })
            {
                int pts = OwnedResourcePointsOf(s);
                if (pts <= 0) continue;
                float acc = _epAccum.TryGetValue(s, out var a) ? a : 0f;
                acc += pts * dt;
                float interval = (float)Game.Ep.IntervalSeconds;
                while (acc >= interval)
                {
                    acc -= interval;
                    float cur = EpOf(s);
                    if (cur < Game.Ep.Cap) _ep[s] = cur + 1f;
                }
                _epAccum[s] = acc;
            }

            ProductionSystem.Tick(this, dt);
            ReinforceSystem.Tick(this, dt);
        }

        private void MoveUnits()
        {
            float perTick = 1f / Config.TickRate;
            foreach (var e in _entities.Values)
            {
                if (e.IsSoldier) continue; // 士兵移动全归 RegimentSystem.Tick,不进命令移动分支
                if (e.Kind != EntityKind.Unit || (e.State != EntityState.Moving && e.State != EntityState.Routing) || e.MoveSpeed <= 0f)
                    continue;
                if (e.GarrisonBuildingId != 0) continue; // 驻军单位不占线移动
                if (!e.HasMoveTarget) continue;          // 无目标的 Routing 原地待命

                SimVec2 target = SimVec2.FromGrid(e.MoveTarget);
                SimVec2 delta = target - e.PosF;
                float dist = delta.Magnitude;
                if (dist <= 0.15f)
                {
                    e.HasMoveTarget = false;
                    e.State = EntityState.Idle;
                    e.MoveAccum = 0f;
                    continue;
                }

                float spd = MoveSpeedOf(e);
                float step = System.Math.Min(spd * perTick, dist);
                e.PosF = TerrainMap.ClampToEllipse(e.PosF + delta.Normalized() * step);
                PushOutOfBuildings(e);
                e.PosF = TerrainMap.ClampToEllipse(e.PosF);
                e.Pos = e.PosF.ToGrid();

                e.MoveAccum += step;
                if (e.MoveAccum >= 1f)
                {
                    e.MoveAccum = 0f;
                    Emit(new SimEvent(Tick, SimEventType.UnitAdvanced, e.Id)); // 节流:约每 1 米一次
                }

                if ((target - e.PosF).Magnitude <= 0.15f)
                {
                    e.HasMoveTarget = false;
                    e.State = EntityState.Idle;
                    e.MoveAccum = 0f;
                }
            }
            RegimentSystem.Tick(this, perTick); // 兵团底座(M8.2a):阵型跟随/无将续战,先于软碰撞
            SoftCollide();
        }

        private float MoveSpeedOf(Entity e)
        {
            float spd = e.MoveSpeed * e.Status.MoveMultiplier;
            if (e.IsCharging && e.Hero == HeroId.ZhuFeng && Game != null)
                spd = (float)ChargeSpeedOf(HeroId.ZhuFeng);
            if (e.Hero == HeroId.ZhenYue && Tick >= 1800) // 3 分钟后移速衰减(D11 默认 1.4)
                spd = IsUpgraded(e.Side, HeroId.ZhenYue) ? 2.2f : 1.4f;
            spd *= HexSystem.MoveMultiplierOf(this, e); // 海克斯移速层
            spd *= TerrainMap.MoveFactor(e.PosF);       // 涉水减半/桥上正常
            return spd;
        }

        /// <summary>建筑碰撞:把单位推出建筑占地矩形(连续坐标下的障碍语义,M8.1a 矩形化)。</summary>
        private void PushOutOfBuildings(Entity u)
        {
            foreach (var b in _entities.Values)
            {
                if (b.Kind != EntityKind.Building || b.Rubbled) continue;
                if (!BuildingFootprint.Contains(b, u.PosF)) continue;
                u.PosF = BuildingFootprint.ClosestPoint(b, u.PosF); // 点在矩形内→最小穿透轴推出,幂等无死循环
            }
        }

        /// <summary>单位间软碰撞:间距小于 0.9 米时沿连线对半推开。</summary>
        private void SoftCollide()
        {
            var units = new List<Entity>();
            foreach (var e in _entities.Values)
                if (e.Kind == EntityKind.Unit && e.GarrisonBuildingId == 0
                    && e.State != EntityState.Dead && e.State != EntityState.Exchanging)
                    units.Add(e);

            const float MinDist = 0.9f;
            var moved = new HashSet<int>();
            for (int i = 0; i < units.Count; i++)
            {
                for (int j = i + 1; j < units.Count; j++)
                {
                    var a = units[i];
                    var b = units[j];
                    // 兵团底座(M8.2a):同兵团士兵之间、士兵与己方主将之间不互推,避免阵型震荡
                    if ((a.IsSoldier || b.IsSoldier) && (a.GeneralId == b.Id || b.GeneralId == a.Id || (a.IsSoldier && b.IsSoldier && a.GeneralId != 0 && a.GeneralId == b.GeneralId))) continue;
                    float dx = a.PosF.X - b.PosF.X, dy = a.PosF.Y - b.PosF.Y;
                    float d2 = dx * dx + dy * dy;
                    if (d2 >= MinDist * MinDist) continue;
                    float d = (float)System.Math.Sqrt(d2);
                    float push = (MinDist - d) * 0.5f;
                    float nx = d > 1e-6f ? dx / d : 1f;
                    float ny = d > 1e-6f ? dy / d : 0f;
                    a.PosF = TerrainMap.ClampToEllipse(a.PosF + new SimVec2(nx * push, ny * push));
                    b.PosF = TerrainMap.ClampToEllipse(b.PosF - new SimVec2(nx * push, ny * push));
                    moved.Add(a.Id);
                    moved.Add(b.Id);
                }
            }
            // 仅被推挤的单位回写格(避免覆盖未移动单位的手动定位)
            foreach (var e in units)
                if (moved.Contains(e.Id))
                    e.Pos = e.PosF.ToGrid();
        }

        /// <summary>建筑外接半径(米)= Max(HalfW,HalfH)+0.4。兼容旧近战判定,下一批 M8.1b 换精确矩形距;新代码请用 BuildingFootprint。</summary>
        public static float BuildingCollisionRadius(Entity b)
        {
            return System.Math.Max(BuildingFootprint.HalfW(b), BuildingFootprint.HalfH(b)) + 0.4f;
        }

        private void CombatUpdate()
        {
            if (Game == null) return;
            float dt = 1f / Config.TickRate;
            var dead = new List<int>();

            foreach (var e in _entities.Values)
            {
                if (e.Kind != EntityKind.Unit || e.State == EntityState.Dead) continue;
                if (e.GarrisonBuildingId != 0) continue; // 驻军单位不参与场外逻辑

                e.Status.Tick(dt);
                e.SecondsSinceCombat += dt;
                HeroAbilitySystem.UpdatePanShiShield(e, dt);
                if (e.Hero == HeroId.ZhuFeng && e.IsCharging && !HeroAbilitySystem.CanCharge(e, this))
                    e.IsCharging = false;

                if (e.Attack <= 0f) continue;

                if (e.AttackCooldown > 0f)
                    e.AttackCooldown -= dt;

                Entity target = FindNearestEnemy(e, e.State == EntityState.Routing ? 1.4f : float.MaxValue);
                if (target == null)
                {
                    if (e.State == EntityState.Attacking) e.State = EntityState.Idle;
                    continue;
                }

                // 溃逃单位保持 Routing 状态(不可调遣语义),不转入 Attacking
                if (e.State != EntityState.Routing)
                {
                    e.HasMoveTarget = false;
                    e.State = EntityState.Attacking;
                }
                if (e.AttackCooldown <= 0f)
                {
                    e.AttackCooldown = e.AttackIntervalSec * e.Status.AttackIntervalMultiplier;
                    ApplyAttack(e, target, dead);
                }
            }

            foreach (int id in dead)
            {
                if (_entities.TryGetValue(id, out var e))
                {
                    if (e.Hero != null) HexSystem.OnUnitDied(this, e.Side); // 陷阵之志
                    Emit(new SimEvent(Tick, SimEventType.EntityDied, id));
                    if (e.Kind == EntityKind.Building)
                        _occupied.Remove(_grid.ToIndex(e.Pos));
                    _entities.Remove(id);
                    Emit(new SimEvent(Tick, SimEventType.EntityRemoved, id));
                }
            }
        }

        private Entity FindNearestEnemy(Entity e, float maxDist)
        {
            Entity best = null;
            float bestDist = float.MaxValue;
            foreach (var o in _entities.Values)
            {
                if (o.Kind != EntityKind.Unit || o.Side == e.Side || o.State == EntityState.Dead)
                    continue;
                if (o.GarrisonBuildingId != 0 || o.State == EntityState.Exchanging)
                    continue; // 城内/交换中的单位不可被城外锁定
                float d = SimVec2.Distance(e.PosF, o.PosF);
                float effRange = e.Range + 0.3f; // 近战贴身容差
                if (d > effRange || d > maxDist) continue;
                if (d < bestDist - 1e-4f || (System.Math.Abs(d - bestDist) <= 1e-4f && best != null && o.Id < best.Id))
                {
                    best = o;
                    bestDist = d;
                }
            }
            return best;
        }

        private void ApplyAttack(Entity attacker, Entity defender, List<int> dead)
        {
            attacker.SecondsSinceCombat = 0f;
            defender.SecondsSinceCombat = 0f;

            // 基础层(GDD 公式)
            float baseDmg = DamageCalculator.CalculateDamage(attacker, defender, Game, attacker.IsCharging);

            // 星落升级·诛灭:基础攻击 5 → 6(基础层内替换)
            bool zhuMie = attacker.Hero == HeroId.XingLuo && IsUpgraded(attacker.Side, HeroId.XingLuo);
            if (zhuMie && attacker.Attack > 0f) baseDmg = baseDmg / attacker.Attack * 6f;

            // 叠算层(D28):输出加算、承伤加算、免伤乘算;溃逃并入对应层
            float outSum = attacker.Status.AttackBonus + HexSystem.OutputBonusOf(this, attacker)
                         + (attacker.State == EntityState.Routing ? -0.3f : 0f);
            float takeSum = defender.Status.DamageTakenBonus + HexSystem.DamageTakenBonusOf(this, defender)
                          + (defender.State == EntityState.Routing ? 0.3f : 0f);
            float reduce = defender.Status.DamageReduction * HexSystem.DamageReductionOf(this, defender);
            float dmg = baseDmg * (1f + outSum) * (1f + takeSum) * reduce;

            // 附加项(层后加法)
            if (zhuMie) dmg += 0.01f * defender.MaxHp;

            // 护盾优先吸收,溢出扣生命
            float remaining = dmg;
            if (defender.Shield > 0f)
            {
                float absorbed = Math.Min(defender.Shield, remaining);
                defender.Shield -= absorbed;
                remaining -= absorbed;
            }
            defender.Hp -= remaining;

            Emit(new SimEvent(Tick, SimEventType.DamageDealt, defender.Id, dmg));

            HeroAbilitySystem.ApplyOnHitEffects(this, attacker, defender);

            // 断峰升级·破军:身前 90 度扇形(半径 2 米)内全部敌人受单次 75% 伤害
            if (attacker.Hero == HeroId.DuanFeng && IsUpgraded(attacker.Side, HeroId.DuanFeng))
                Cleave(attacker, defender, dead);

            if (defender.Hp <= 0f)
            {
                defender.State = EntityState.Dead;
                dead.Add(defender.Id);
            }
        }

        private void Cleave(Entity attacker, Entity mainTarget, List<int> dead)
        {
            float facing = mainTarget.PosF.X >= attacker.PosF.X ? 1f : -1f;
            foreach (var o in _entities.Values)
            {
                if (o.Id == mainTarget.Id) continue;
                if (o.Kind != EntityKind.Unit || o.Side == attacker.Side || o.State == EntityState.Dead) continue;
                if (o.GarrisonBuildingId != 0 || o.State == EntityState.Exchanging) continue;
                float dx = o.PosF.X - attacker.PosF.X;
                float dy = o.PosF.Y - attacker.PosF.Y;
                if (dx * facing < 0) continue;         // 身后不打
                if (dx * dx + dy * dy > 4f) continue;  // 半径 2 米扇形
                float baseDmg = 0.75f * DamageCalculator.CalculateDamage(attacker, o, Game, attacker.IsCharging);
                float outSum = attacker.Status.AttackBonus + (attacker.State == EntityState.Routing ? -0.3f : 0f);
                float takeSum = o.Status.DamageTakenBonus + (o.State == EntityState.Routing ? 0.3f : 0f);
                float dmg = baseDmg * (1f + outSum) * (1f + takeSum) * o.Status.DamageReduction;
                float remaining = dmg;
                if (o.Shield > 0f)
                {
                    float absorbed = System.Math.Min(o.Shield, remaining);
                    o.Shield -= absorbed;
                    remaining -= absorbed;
                }
                o.Hp -= remaining;
                Emit(new SimEvent(Tick, SimEventType.DamageDealt, o.Id, dmg));
                if (o.Hp <= 0f) { o.State = EntityState.Dead; dead.Add(o.Id); }
            }
        }

        private double ChargeSpeedOf(HeroId id)
        {
            foreach (var h in Game.Heroes)
                if (h.Id == id) return h.ChargeSpeed;
            return 0;
        }

        private void Emit(SimEvent ev)
        {
            for (int i = _listeners.Count - 1; i >= 0; i--)
                _listeners[i]?.Invoke(ev);
        }
    }
}
```

```csharp
using System;
using System.Collections.Generic;
using System.Linq;

namespace IronCurtain.Sim
{
    public enum AiTier { Simple, Normal }

    public static class AiSystem
    {
        const int DecisionInterval = 20;

        class AiState
        {
            public Side Side;
            public AiTier Tier;
            public int ProducedSinceZhenYue; // 已产 1 人口英雄计数
            public bool ZhenYueQueued;       // 下一只为镇岳
            public bool WaveActive;
            public int WaveStartCount;
            public List<int> WaveUnits = new List<int>();
        }

        static readonly Dictionary<Side, AiState> _ais = new Dictionary<Side, AiState>();

        public static void Configure(SimCore sim, Side side, AiTier tier)
        {
            _ais[side] = new AiState { Side = side, Tier = tier };
        }

        public static void Reset() => _ais.Clear();

        public static void Tick(SimCore sim, float dt)
        {
            if (sim.Tick % DecisionInterval != 0) return;
            foreach (var ai in _ais.Values.ToList())
            {
                if (ai.Tier == AiTier.Simple) TickSimple(sim, ai);
                else TickNormal(sim, ai);
            }
        }

        // ---------- 简单 AI ----------
        static void TickSimple(SimCore sim, AiState ai)
        {
            FillProduction(sim, ai, simple: true);
            PassiveDefend(sim, ai);
        }

        static void PassiveDefend(SimCore sim, AiState ai)
        {
            var myBuildings = sim.Entities.Values
                .Where(e => e.Side == ai.Side && e.Kind == EntityKind.Building && !e.Rubbled)
                .ToList();
            foreach (var b in myBuildings)
            {
                var threat = sim.Entities.Values
                    .Where(e => e.Kind == EntityKind.Unit && e.Side != ai.Side && e.State != EntityState.Dead
                                && VisionSystem.IsUnitVisible(sim, ai.Side, e.Id)
                                && Dist(sim, e, b) <= 8f)
                    .FirstOrDefault();
                if (threat == null) continue;
                var idle = sim.Entities.Values
                    .Where(e => e.Kind == EntityKind.Unit && e.Side == ai.Side && !e.IsSoldier
                                && e.State == EntityState.Idle && e.GarrisonBuildingId == 0)
                    .OrderBy(e => Dist(sim, e, b))
                    .Take(5).ToList();
                foreach (var u in idle)
                {
                    var t = FrontCell(sim, b);
                    if (t.HasValue) sim.SetMoveTarget(u.Id, t.Value);
                }
            }
        }

        // ---------- 普通 AI ----------
        static void TickNormal(SimCore sim, AiState ai)
        {
            // 1. 补产(配比)
            FillProduction(sim, ai, simple: false);
            // 2. 升级
            if (sim.EpOf(ai.Side) >= 2 && !sim.IsUpgraded(ai.Side, HeroId.ZhenYue))
                sim.BuyUpgrade(ai.Side, HeroId.ZhenYue, out _);
            // 3. 规律抽卡
            if (sim.Tick >= 1800 && HexSystem.CanDraw(sim, ai.Side))
                TryDrawHex(sim, ai);
            // 5. 受挫撤退(先于新波次判断)
            if (ai.WaveActive) CheckRetreat(sim, ai);
            // 4. 集结进攻
            if (!ai.WaveActive) TryLaunchWave(sim, ai);
            // 6. 主动防守
            ActiveDefend(sim, ai);
        }

        static void TryDrawHex(SimCore sim, AiState ai)
        {
            var prio = new[] { HexId.SteelTide, HexId.RapidDeploy, HexId.HoldFast, HexId.Vengeance,
                               HexId.LightMobility, HexId.IronWall, HexId.HiddenPath };
            var choices = HexSystem.DrawChoices(sim, ai.Side);
            foreach (var c in prio)
            {
                if (c == HexId.Sandbag) continue;
                if (choices.Contains(c) && HexSystem.UseHex(sim, ai.Side, c, 0, out _)) return;
            }
        }

        static void CheckRetreat(SimCore sim, AiState ai)
        {
            ai.WaveUnits = ai.WaveUnits
                .Where(id => sim.Entities.TryGetValue(id, out var e) && e.State != EntityState.Dead && !e.IsSoldier)
                .ToList();
            if (ai.WaveUnits.Count < ai.WaveStartCount * 0.4f)
            {
                var home = sim.Entities.Values
                    .FirstOrDefault(e => e.Side == ai.Side && e.Kind == EntityKind.Building && e.Building == BuildingId.Base);
                var front = home != null ? FrontCell(sim, home) : (GridPos?)null;
                foreach (var id in ai.WaveUnits)
                    if (front.HasValue) sim.SetMoveTarget(id, front.Value);
                ai.WaveActive = false;
                ai.WaveUnits.Clear();
            }
        }

        static void TryLaunchWave(SimCore sim, AiState ai)
        {
            var field = sim.Entities.Values
                .Where(e => e.Kind == EntityKind.Unit && e.Side == ai.Side && !e.IsSoldier
                            && e.State != EntityState.Dead && e.GarrisonBuildingId == 0
                            && e.State != EntityState.Exchanging)
                .ToList();
            if (field.Count < 8) return;

            var home = Home(sim, ai.Side);
            // 目标选择
            var neutral = sim.Entities.Values
                .Where(e => e.Kind == EntityKind.Building && e.Building == BuildingId.ResourcePoint
                            && e.Side == Side.Neutral && !e.Rubbled
                            && VisionSystem.IsBuildingKnown(sim, ai.Side, e.Id))
                .OrderBy(e => home != null ? Dist(sim, e, home) : 0f)
                .FirstOrDefault();
            int targetId; SiegeIntent intent;
            if (neutral != null) { targetId = neutral.Id; intent = SiegeIntent.Capture; }
            else
            {
                var enemyBuildings = sim.Entities.Values
                    .Where(e => e.Kind == EntityKind.Building && e.Side != ai.Side && e.Side != Side.Neutral
                                && !e.Rubbled && e.Building != BuildingId.Base
                                && VisionSystem.IsBuildingKnown(sim, ai.Side, e.Id))
                    .Select(e => new { E = e, G = VisionSystem.LastKnownGarrison(sim, ai.Side, e.Id) ?? 999 })
                    .OrderBy(x => x.G).ThenBy(x => home != null ? Dist(sim, x.E, home) : 0f)
                    .FirstOrDefault();
                if (enemyBuildings == null) return;
                targetId = enemyBuildings.E.Id;
                intent = enemyBuildings.E.Building == BuildingId.Tower ? SiegeIntent.Destroy : SiegeIntent.Capture;
            }

            var targetB = sim.GetEntity(targetId);
            var wave = field.OrderBy(e => targetB != null ? Dist(sim, e, targetB) : 0f).Take(10).Select(e => e.Id).ToList();
            if (SiegeSystem.SetIntent(sim, targetId, ai.Side, intent, out _))
            {
                SiegeSystem.IssueOrder(sim, wave, targetId, out _);
                ai.WaveActive = true;
                ai.WaveUnits = wave;
                ai.WaveStartCount = wave.Count;
            }
        }

        static void ActiveDefend(SimCore sim, AiState ai)
        {
            var visibleEnemies = sim.Entities.Values
                .Where(e => e.Kind == EntityKind.Unit && e.Side != ai.Side && e.State != EntityState.Dead
                            && VisionSystem.IsUnitVisible(sim, ai.Side, e.Id))
                .ToList();
            if (visibleEnemies.Count < 6) return;
            foreach (var b in sim.Entities.Values.Where(e => e.Side == ai.Side && e.Kind == EntityKind.Building && !e.Rubbled))
            {
                var near = visibleEnemies.Where(e => Dist(sim, e, b) <= 10f).ToList();
                if (near.Count < 6) continue;
                var idle = sim.Entities.Values
                    .Where(e => e.Kind == EntityKind.Unit && e.Side == ai.Side && !e.IsSoldier
                                && e.State == EntityState.Idle && e.GarrisonBuildingId == 0)
                    .OrderBy(e => Dist(sim, e, b))
                    .Take(6).ToList();
                // 将兵制中间态:非士兵无法驻军(仅士兵可),防御=调到城门前排站位;士兵驻军防御在 M8.5 接上
                foreach (var u in idle)
                {
                    var t = FrontCell(sim, b);
                    if (t.HasValue) sim.SetMoveTarget(u.Id, t.Value);
                }
                return;
            }
        }

        // ---------- 公共 ----------
        static void FillProduction(SimCore sim, AiState ai, bool simple)
        {
            foreach (var b in sim.Entities.Values)
            {
                if (b.Kind != EntityKind.Building || b.Building != BuildingId.Barracks || b.Prod == null) continue;
                if (b.Side != ai.Side || b.Rubbled) continue;
                for (int s = 0; s < b.Prod.Slots.Length; s++)
                {
                    var slot = b.Prod.Slots[s];
                    if (slot.Current != null || slot.Pending.Count > 0) continue;
                    HeroId hero;
                    if (simple) hero = HeroId.PanShi;
                    else if (ai.ZhenYueQueued) hero = HeroId.ZhenYue;
                    else hero = ai.ProducedSinceZhenYue % 2 == 0 ? HeroId.PanShi : HeroId.DuanFeng;
                    if (hero == HeroId.ZhenYue && sim.PopulationOf(ai.Side) + 3 > sim.PopulationCapOf(ai.Side))
                        return; // 人口不足镇岳,等待
                    ProductionSystem.Enqueue(sim, b.Id, hero, out _);
                    if (simple) continue;
                    if (hero == HeroId.ZhenYue) { ai.ZhenYueQueued = false; ai.ProducedSinceZhenYue = 0; }
                    else
                    {
                        ai.ProducedSinceZhenYue++;
                        if (ai.ProducedSinceZhenYue >= 3) ai.ZhenYueQueued = true;
                    }
                }
            }
        }

        static Entity Home(SimCore sim, Side side)
        {
            return sim.Entities.Values.FirstOrDefault(e => e.Side == side && e.Kind == EntityKind.Building && e.Building == BuildingId.Base);
        }

        /// <summary>城门站位格:建筑朝敌侧(碰撞半径+0.6 米),行序 15 优先、椭圆内、非水域。</summary>
        static GridPos? FrontCell(SimCore sim, Entity b)
        {
            int dx = b.Side == Side.Blue ? 1 : -1;
            float offset = SimCore.BuildingCollisionRadius(b) + 0.6f;
            var gate = (SimVec2.FromGrid(b.Pos) + new SimVec2(dx * offset, 0f)).ToGrid();
            foreach (int r in new[] { 0, -1, 1, -2, 2, -3, 3 })
            {
                var p = new GridPos(gate.Row + r, gate.Col);
                if (p.Col < 0 || p.Col >= LaneGrid.Cols || p.Row < 0 || p.Row >= LaneGrid.Rows) continue;
                if (!TerrainMap.IsInside(p)) continue;
                if (TerrainMap.IsWater(SimVec2.FromGrid(p))) continue;
                return p;
            }
            return null;
        }

        /// <summary>实体间距离(米制欧氏):单位取 PosF,建筑取格中心。</summary>

            SimVec2 pb = b.Kind == EntityKind.Unit ? b.PosF : SimVec2.FromGrid(b.Pos);
            return SimVec2.Distance(pa, pb);
```

```csharp
using System.Collections.Generic;

namespace IronCurtain.Sim
{
    /// <summary>
    /// 兵团底座(M8.2a 第一批):1 主将 + 20 兵,兵随将走(4×5 阵型跟随);
    /// 主将死亡士兵永不溃逃、无将状态就近逐杀至死。纯 tick,无 UnityEngine 依赖。
    /// </summary>
    public static class RegimentSystem
    {
        public const int SoldiersPerGeneral = 20;
        public const int FormationRows = 4;   // 4 行 × 5 列,行列距 1.0 米
        public const int FormationCols = 5;

        public static SimVec2 FacingOf(Entity g)
        {
            var def = g.Side == Side.Blue ? new SimVec2(1f, 0f) : new SimVec2(-1f, 0f);
            if (g.HasMoveTarget)
            {
                var d = (SimVec2.FromGrid(g.MoveTarget) - g.PosF).Normalized();
                if (d.SqrMagnitude > 1e-8f) return d;
            }
            return def;
        }

        public static SimVec2 SlotOffset(Entity g, int slot)
        {
            var facing = FacingOf(g);
            var perp = new SimVec2(facing.Y, -facing.X);
            int row = slot / FormationCols;
            int col = slot % FormationCols;
            return facing * -(row + 1) + perp * (col - 2);
        }

        public static void Tick(SimCore sim, float dt)
        {
            // 快照遍历:士兵死亡结算会移除实体,禁止边枚举边修改
            var snapshot = new List<Entity>(sim.Entities.Values);
            foreach (var e in snapshot)
            {
                if (e.Kind != EntityKind.Unit || !e.IsSoldier || e.State == EntityState.Dead) continue;
                if (e.GarrisonBuildingId != 0) continue;

                var g = e.GeneralId != 0 ? sim.GetEntity(e.GeneralId) : null;
                if (g != null && g.Kind == EntityKind.Unit && g.State != EntityState.Dead)
                {
                    // a) 主将存活:阵型跟随(士兵不写 HasMoveTarget/State)
                    var target = g.PosF + SlotOffset(g, e.FormationSlot);
                    float dist = SimVec2.Distance(e.PosF, target);
                    if (dist > 0.2f) StepMove(e, target, dist, dt);
                }
                else
                {
                    // b) 无将续战:就近索敌逐杀至死(永不溃逃/撤退/响应指令)
                    var enemy = NearestEnemy(sim, e);
                    if (enemy == null) continue;
                    float dist = SimVec2.Distance(e.PosF, enemy.PosF);
                    if (dist <= e.Range + 0.3f) continue; // 交战交给 CombatUpdate
                    StepMove(e, enemy.PosF, dist, dt);
                }
            }
        }

        private static void StepMove(Entity e, SimVec2 target, float dist, float dt)
        {
            float spd = e.MoveSpeed * TerrainMap.MoveFactor(e.PosF);
            float step = System.Math.Min(spd * dt, dist);
            e.PosF = TerrainMap.ClampToEllipse(e.PosF + (target - e.PosF).Normalized() * step);
            e.Pos = e.PosF.ToGrid();
        }

        /// <summary>最近敌索敌:与 SimCore.FindNearestEnemy 同一欧氏语义(单位、非己方、非死亡、非城内/交换)。</summary>
        private static Entity NearestEnemy(SimCore sim, Entity e)
        {
            Entity best = null;
            float bestDist = float.MaxValue;
            foreach (var o in sim.Entities.Values)
            {
                if (o.Kind != EntityKind.Unit || o.Side == e.Side || o.State == EntityState.Dead) continue;
                if (o.GarrisonBuildingId != 0 || o.State == EntityState.Exchanging) continue;
                float d = SimVec2.Distance(e.PosF, o.PosF);
                if (d < bestDist) { best = o; bestDist = d; }
            }
            return best;
        }

        /// <summary>生成一名士兵(属性=主将/10,攻速/移速/射程=主将,变装=主将 Hero),落位槽位偏移点。SpawnGeneral 与补兵共用。</summary>
        public static Entity SpawnSoldier(SimCore sim, Entity general, int slot)
        {
            var target = general.PosF + SlotOffset(general, slot);
            var s = sim.SpawnEntity(general.Side, EntityKind.Unit, target.ToGrid());
            s.IsSoldier = true;
            s.FormationSlot = slot;
            s.Hero = general.Hero;
            s.MaxHp = general.MaxHp / 10f;
            s.Hp = s.MaxHp;
            s.Attack = general.Attack / 10f;
            s.AttackIntervalSec = general.AttackIntervalSec;
            s.MoveSpeed = general.MoveSpeed;
            s.Range = general.Range;
            s.Shield = general.Shield / 10f;
            s.ShieldMax = s.Shield;
            s.GeneralId = general.Id;
            s.PosF = target;
            s.Pos = target.ToGrid();
            return s;
        }

        /// <summary>生成主将 + 20 名士兵的完整兵团,返回主将实体。</summary>
        public static Entity SpawnGeneral(SimCore sim, Side side, HeroId hero, GridPos pos)
        {
            var c = sim.Game.Heroes[(int)hero];

            var g = sim.SpawnEntity(side, EntityKind.Unit, pos);
            g.Hero = hero;
            g.MaxHp = c.MaxHp;
            g.Hp = c.MaxHp;
            g.Attack = c.Attack;
            g.AttackIntervalSec = (float)c.AttackIntervalSec;
            g.Range = c.Range;
            g.MoveSpeed = (float)c.MoveSpeed;
            if (c.Shield > 0) { g.Shield = c.Shield; g.ShieldMax = c.Shield; }

            for (int slot = 0; slot < SoldiersPerGeneral; slot++)
                SpawnSoldier(sim, g, slot);
            return g;
        }
```

```csharp
namespace IronCurtain.Sim
```

```csharp
    /// <summary>建筑占地矩形(米制半宽半高,中心=建筑格中心 SimVec2.FromGrid(b.Pos))。M8.1a:占地类,纯 C#。</summary>
    public static class BuildingFootprint
    {
        public static float HalfW(Entity b)
        {
            if (b.IsFort) return 1.5f;
            switch (b.Building)
            {
                case BuildingId.Base: return 2.5f;
                case BuildingId.Tower: return 2.0f;
                case BuildingId.Barracks: return 2.5f;
                case BuildingId.ResourcePoint: return 2.0f;
                default: return 1.5f;
            }
        }

        public static float HalfH(Entity b)
        {
            if (b.IsFort) return 1.5f;
            switch (b.Building)
            {
                case BuildingId.Base: return 2.5f;
                case BuildingId.Tower: return 2.0f;
                case BuildingId.Barracks: return 2.0f;
                case BuildingId.ResourcePoint: return 2.0f;
                default: return 1.5f;
            }
        }

        /// <summary>点是否在中心矩形内(含边界)。</summary>
        public static bool Contains(Entity b, SimVec2 p)
        {
            var c = SimVec2.FromGrid(b.Pos);
            float hw = HalfW(b), hh = HalfH(b);
            return p.X >= c.X - hw && p.X <= c.X + hw && p.Y >= c.Y - hh && p.Y <= c.Y + hh;
        }

        /// <summary>点在矩形外→矩形上最近点;点在矩形内→沿最小穿透轴推出后的缘点(供 PushOutOfBuildings 一次完成推出)。</summary>
        public static SimVec2 ClosestPoint(Entity b, SimVec2 p)
        {
            var c = SimVec2.FromGrid(b.Pos);
            float hw = HalfW(b), hh = HalfH(b);
            float minX = c.X - hw, maxX = c.X + hw, minY = c.Y - hh, maxY = c.Y + hh;
            if (p.X < minX || p.X > maxX || p.Y < minY || p.Y > maxY)
                return new SimVec2(Math.Min(Math.Max(p.X, minX), maxX),
                                   Math.Min(Math.Max(p.Y, minY), maxY));
            // 在矩形内:沿最小穿透轴推出
            float left = p.X - minX, right = maxX - p.X, down = p.Y - minY, up = maxY - p.Y;
            float m = Math.Min(Math.Min(left, right), Math.Min(down, up));
            if (m == left) return new SimVec2(minX, p.Y);
            if (m == right) return new SimVec2(maxX, p.Y);
            if (m == down) return new SimVec2(p.X, minY);
            return new SimVec2(p.X, maxY);
        }

        /// <summary>枚举矩形覆盖的所有格(格中心落在 [min,max) 半开区间内,保证基地25/箭塔16/兵营20/资源点16/堡垒9)。</summary>
        public static List<GridPos> CellsOccupied(Entity b)
        {
            var list = new List<GridPos>();
            var c = SimVec2.FromGrid(b.Pos);
            float hw = HalfW(b), hh = HalfH(b);
            int rc = (int)Math.Ceiling(hh), cc = (int)Math.Ceiling(hw);
            for (int dr = -rc; dr <= rc; dr++)
                for (int dc = -cc; dc <= cc; dc++)
                {
                    int row = b.Pos.Row + dr, col = b.Pos.Col + dc;
                    if (row < 0 || row >= LaneGrid.Rows || col < 0 || col >= LaneGrid.Cols) continue;
                    var ctr = SimVec2.FromGrid(new GridPos(row, col));
                    if (ctr.X >= c.X - hw && ctr.X < c.X + hw && ctr.Y >= c.Y - hh && ctr.Y < c.Y + hh)
                        list.Add(new GridPos(row, col));
                }
            return list;
        }
    }
```

```csharp
using System.Collections.Generic;

namespace IronCurtain.Sim
{
    /// <summary>
    /// 补兵池(M8.4a 第一批):全局 2s/兵、上限 60;玩家点补兵 → SJF(最缺人兵团先补、补齐为止);
    /// 交战兵团不可补、脱战 3s 重新入列。行军表现归 View(M8.6b),本类纯 Sim。
    /// </summary>
    public static class ReinforceSystem
    {
        public const int PoolCap = 60;              // 池上限(平拍)
        public const float SecondsPerSoldier = 2f;  // 2s/兵
        public const float OutOfCombatSeconds = 3f; // 脱战 3s 重新入列

        public static int PoolOf(SimCore sim, Side side) => sim.ReinforcePool[side];

        public static void Tick(SimCore sim, float dt)
        {
            sim.ReinforceAccum[Side.Blue] += dt / SecondsPerSoldier;
            sim.ReinforceAccum[Side.Red] += dt / SecondsPerSoldier;
            foreach (Side side in new[] { Side.Blue, Side.Red })
            {
                while (sim.ReinforceAccum[side] >= 1f && sim.ReinforcePool[side] < PoolCap)
                {
                    sim.ReinforcePool[side]++;
                    sim.ReinforceAccum[side] -= 1f;
                }
            }
        }

        public static bool RequestReinforce(SimCore sim, Side side, out string error)
        {
            error = null;

            // 1) 收集主将
            var generals = new List<Entity>();
            foreach (var e in sim.Entities.Values)
                if (e.Kind == EntityKind.Unit && !e.IsSoldier && e.Side == side && e.State != EntityState.Dead)
                    generals.Add(e);
            if (generals.Count == 0) { error = "无主将在场"; return false; }
            if (sim.ReinforcePool[side] <= 0) { error = "补兵池为空"; return false; }

            // 2) 交战判定 + 存活士兵计数
            var regiments = new List<System.Tuple<int, Entity>>(); // (cnt, general)
            bool anyEligible = false;
            foreach (var g in generals)
            {
                int cnt = 0;
                bool inCombat = g.SecondsSinceCombat < OutOfCombatSeconds;
                foreach (var e in sim.Entities.Values)
                {
                    if (!e.IsSoldier || e.GeneralId != g.Id || e.State == EntityState.Dead) continue;
                    cnt++;
                    if (e.SecondsSinceCombat < OutOfCombatSeconds) inCombat = true;
                }
                if (inCombat) continue;
                anyEligible = true;
                if (cnt < RegimentSystem.SoldiersPerGeneral)
                    regiments.Add(System.Tuple.Create(cnt, g));
            }
            if (!anyEligible) { error = "全部兵团交战中,不可补兵"; return false; }

            // 3) SJF:缺员数升序(同缺员按 Id 稳定序),逐团补齐为止
            regiments.Sort((a, b) => a.Item1 != b.Item1 ? a.Item1.CompareTo(b.Item1) : a.Item2.Id.CompareTo(b.Item2.Id));
            bool anyGiven = false;
            foreach (var reg in regiments)
            {
                if (sim.ReinforcePool[side] <= 0) break;
                var g = reg.Item2;
                var used = new HashSet<int>();
                foreach (var e in sim.Entities.Values)
                    if (e.IsSoldier && e.GeneralId == g.Id && e.State != EntityState.Dead)
                        used.Add(e.FormationSlot);
                int need = RegimentSystem.SoldiersPerGeneral - reg.Item1;
                int give = System.Math.Min(need, sim.ReinforcePool[side]);
                for (int slot = 0; slot < RegimentSystem.SoldiersPerGeneral && give > 0; slot++)
                {
                    if (used.Contains(slot)) continue;
                    RegimentSystem.SpawnSoldier(sim, g, slot);
                    sim.ReinforcePool[side]--;
                    give--;
                    anyGiven = true;
                }
            }
            if (!anyGiven) { error = "补兵池为空"; return false; }
            return true;
        }
    }
}
```
