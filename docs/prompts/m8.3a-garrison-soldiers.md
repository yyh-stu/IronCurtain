### 2.3 GarrisonSystem.cs 士兵口径(核心改造)
1. **仅士兵可驻军**:`EnterFormal`/`EnterTemp` 对每个单位校验 `u.IsSoldier`,非士兵直接跳过该单位并返回 error "仅士兵可驻军"(主将永不进城=将城外等候)。非士兵路过满员站定逻辑同样不再适用(只对士兵)。
2. **正式驻军**:现有入城语义不变(设 `GarrisonBuildingId`=城,进 `GarrisonFormal`),容量读取配置(已 30)。
3. **临时驻军=城后等候(不冻结移动)**:登记时不设 `GarrisonBuildingId`(士兵保持跟随主将的移动能力),改设 `u.TempGarrisonId=城`,并登记进建筑 `GarrisonTemp` 列表。士兵位置不强制(主将调走兵团=士兵跟随离开=截击)。
4. **离城自动解除(新增每 tick 检查,建议并入 GarrisonSystem.Tick 或对应 tick 处)**:对每个 TempGarrisonId!=0 的士兵:若其主将已死/不存在,或主将与城距离 > 15m,或士兵已死亡 → 从该城 `GarrisonTemp` 移除、`TempGarrisonId=0`。
5. **城守住补入正式(现有 intent 守卫逻辑保留)**:临时→正式转换时:士兵 `TempGarrisonId=0`、`GarrisonBuildingId=城`、进 `GarrisonFormal`(容量满则继续等候);敌方攻城意图存续期间仍不转换(现状)。
6. **易主/废墟的士兵处置(简化,用户拍板"士兵永不溃逃")**:`OnCityCaptured`/`OnBuildingRubbled` 中,对 `IsSoldier` 的临时/正式驻军:仅清列表、`TempGarrisonId=0`、`GarrisonBuildingId=0`(士兵留在兵团继续跟随主将;主将已死→无将续战),**不设溃逃 MoveTarget**;对非士兵(旧数据/兼容)保留现有溃逃逻辑(FindRoutDestination→城门)不变。
7. **出城 Exit(D24)**:对士兵出城:清 `GarrisonBuildingId`/`TempGarrisonId`,该士兵回到兵团阵型(跟随主将;主将死→无将续战);对非士兵保留现有语义(城门站定)。
8. 其余(容量/满员/堡垒 3 人/依托射击/回血)逻辑不动;`RallyPointOf` 等无关。

### 2.4 SiegeSystem.cs 交换只耗兵、主将免疫
- 交换消耗(KillDefenders 或等价路径):守城方消耗**正式驻军**中的单位(现有语义,现仅士兵可入正式,天然士兵口径)。
- 攻城方每次交换损失:**只从参与攻城的士兵中选 1 名阵亡;主将(非士兵)永不因交换阵亡**;若攻城单位中没有士兵在场,则攻方无损失(交换仍按节奏进行)。交换比例=1:1 兵/次(制作人拍板;现有 1:2 类分支删除或改 1:1,节奏/间隔参数不动)。
- 其余(意图/封城/拆塔/废墟化/易主判定)一律不动。

### 3.2 GarrisonSystem.cs
```csharp
using System.Collections.Generic;
using System.Linq;

namespace IronCurtain.Sim
{
    /// <summary>驻军系统:正式/临时驻军、依托射击、近战出击、撤出、建筑脱战回复(D27)。M8.1b:城门改占地矩形语义(D18)。</summary>
    public static class GarrisonSystem
    {
        // D27:建筑脱战 20 秒后每秒回复 4 点,至上限
        private const float RegenDelaySec = 20f;
        private const float RegenPerSec = 4f; // D27

        private static readonly HashSet<HeroId> RangedHeroes = new HashSet<HeroId> { HeroId.ZhuiRi, HeroId.XingLuo };

        public static bool EnterFormal(SimCore sim, IReadOnlyList<int> unitIds, int buildingId, out string error)
        {
            error = null;
            if (!sim.Entities.ContainsKey(buildingId)) { error = "建筑不存在"; return false; }
            var b = sim.GetEntity(buildingId);
            var cfg = sim.BuildingConfigOf(b.Building.Value);
            if (cfg.GarrisonFormal <= 0) { error = "该建筑不可驻军"; return false; }
            var accepted = new List<int>();
            foreach (var id in unitIds)
            {
                if (!sim.Entities.ContainsKey(id)) continue;
                var u = sim.GetEntity(id);
                if (u.Side != b.Side) { error = "建筑须为己方"; Rollback(sim, accepted, buildingId); return false; }
                if (sim.GetEntity(buildingId).GarrisonFormal.Count >= cfg.GarrisonFormal)
                {   // 驻军全满:路过或城门站定(用户新规,废除报错)
                    if (!u.HasMoveTarget && u.GarrisonBuildingId == 0)
                    {
                        var g = FindGateCell(sim, b);
                        if (g != null) sim.SetMoveTarget(id, g.Value);
                    }
                    continue;
                }
                DoEnter(sim, u, buildingId, sim.GetEntity(buildingId).GarrisonFormal);
                accepted.Add(id);
            }
            return true;
        }

        public static bool EnterTemp(SimCore sim, IReadOnlyList<int> unitIds, int buildingId, out string error)
        {
            error = null;
            if (!sim.Entities.ContainsKey(buildingId)) { error = "建筑不存在"; return false; }
            var b = sim.GetEntity(buildingId);
            int cap = b.IsFort ? 3 : (b.Building == null ? 0 : sim.BuildingConfigOf(b.Building.Value).GarrisonTemp); // 堡垒驻军上限 3(D32)
            if (cap <= 0) { error = "该建筑不可临时驻军"; return false; }
            var accepted = new List<int>();
            foreach (var id in unitIds)
            {
                if (!sim.Entities.ContainsKey(id)) continue;
                var u = sim.GetEntity(id);
                if (u.Side != b.Side) { error = "建筑须为己方"; RollbackTemp(sim, accepted, buildingId); return false; }
                if (sim.GetEntity(buildingId).GarrisonTemp.Count >= cap)
                {   // 临时驻军全满:路过或城门站定
                    if (!u.HasMoveTarget && u.GarrisonBuildingId == 0)
                    {
                        var g = FindGateCell(sim, b);
                        if (g != null) sim.SetMoveTarget(id, g.Value);
                    }
                    continue;
                }
                DoEnter(sim, u, buildingId, sim.GetEntity(buildingId).GarrisonTemp);
                accepted.Add(id);
            }
            return true;
        }

        public static bool Exit(SimCore sim, int buildingId, HeroId hero, out string error) // D24:每次放一只到城门站定
        {
            error = null;
            if (!sim.Entities.ContainsKey(buildingId)) { error = "建筑不存在"; return false; }
            var b = sim.GetEntity(buildingId);
            int id = b.GarrisonFormal.FirstOrDefault(x => sim.GetEntity(x).Hero == hero);
            bool inFormal = id != 0;
            if (id == 0) id = b.GarrisonTemp.FirstOrDefault(x => sim.GetEntity(x).Hero == hero);
            if (id == 0) { error = "城内无该兵种"; return false; }
            var gate = FindGateCell(sim, b);
            if (gate == null) { error = "城门无空位"; return false; }
            if (inFormal) b.GarrisonFormal.Remove(id); else { b.GarrisonTemp.Remove(id); }
            var u = sim.GetEntity(id);
            u.Pos = gate.Value; u.PosF = SimVec2.FromGrid(gate.Value); u.GarrisonBuildingId = 0; u.State = EntityState.Idle;
            u.HasMoveTarget = false;
            return true;
        }

        /// <summary>城池易主:旧主临时驻军向同一线后方最近己方据点溃逃(不可调遣,遇敌 -30% 交战)。</summary>
        public static void OnCityCaptured(SimCore sim, Entity b)
        {
            var fleeing = new List<int>(b.GarrisonTemp);
            b.GarrisonTemp.Clear();
            foreach (var id in fleeing)
            {
                var usedGates = new System.Collections.Generic.HashSet<long>(); // 溃逃单位分派不同城门格,避免堆叠无法抵达
                var u = sim.GetEntity(id);
                if (u == null) continue;
                var dest = FindRoutDestination(sim, u);
                if (dest == null) { u.GarrisonBuildingId = 0; u.State = EntityState.Idle; continue; }
                var gate = FindGateCell(sim, dest);
                if (gate == null)
                {   // 城门无位则就近找空位
                    for (int d = 1; d <= 8 && gate == null; d++)
                        for (int dr = 0; dr < LaneGrid.Rows && gate == null; dr++)
                        {
                            var p = new GridPos(dr, dest.Pos.Col + (u.Side == Side.Blue ? -d : d));
                            if (IsCellFree(sim, p) && TerrainMap.IsInside(p)) gate = p;
                        }
                }
                u.GarrisonBuildingId = 0;
                // 从城池后方出城,避免被已失守的城堵住退路
                int back = u.Side == Side.Blue ? -1 : 1;
                int behind = back * ((int)System.Math.Ceiling(BuildingFootprint.HalfW(b)) + 1); // 起点须落在失守建筑占地矩形之外
                var start = new GridPos(b.Pos.Row, b.Pos.Col + behind);
                if (!IsCellFree(sim, start))
                {
                    for (int dr = 0; dr < LaneGrid.Rows; dr++)
                    {
                        var alt = new GridPos(dr, b.Pos.Col + behind);
                        if (IsCellFree(sim, alt) && TerrainMap.IsInside(alt)) { start = alt; break; }
                    }
                }
                if (IsCellFree(sim, start)) { u.Pos = start; u.PosF = SimVec2.FromGrid(start); }
                u.State = EntityState.Routing;
                u.HasMoveTarget = true;
                u.MoveTarget = gate ?? dest.Pos;
            }
        }

        /// <summary>建筑化为废墟:全部驻军(正式+临时)强制后撤(途中遇敌交战 -30%)。</summary>
        public static void OnBuildingRubbled(SimCore sim, Entity b)
        {
            var fleeing = new List<int>(b.GarrisonFormal);
            fleeing.AddRange(b.GarrisonTemp);
            b.GarrisonFormal.Clear();
            b.GarrisonTemp.Clear();
            foreach (var id in fleeing)
            {
                var usedGates = new System.Collections.Generic.HashSet<long>(); // 溃逃单位分派不同城门格,避免堆叠无法抵达
                var u = sim.GetEntity(id);
                if (u == null) continue;
                var dest = FindRoutDestination(sim, u);
                if (dest == null) { u.GarrisonBuildingId = 0; u.State = EntityState.Idle; continue; }
                var gate = FindGateCell(sim, dest, usedGates);
                if (gate != null)
                    usedGates.Add(((long)gate.Value.Row << 32) | (uint)gate.Value.Col);
                if (gate == null)
                {
                    for (int d = 1; d <= 8 && gate == null; d++)
                        for (int dr = 0; dr < LaneGrid.Rows && gate == null; dr++)
                        {
                            var p = new GridPos(dr, dest.Pos.Col + (u.Side == Side.Blue ? d : -d));
                            if (IsCellFree(sim, p) && TerrainMap.IsInside(p)) gate = p;
                        }
                }
                u.GarrisonBuildingId = 0;
                int back = u.Side == Side.Blue ? -1 : 1;
                int behind = back * ((int)System.Math.Ceiling(BuildingFootprint.HalfW(b)) + 1); // 起点须落在失守建筑占地矩形之外
                var start = new GridPos(b.Pos.Row, b.Pos.Col + behind);
                if (!IsCellFree(sim, start))
                {
                    for (int dr = 0; dr < LaneGrid.Rows; dr++)
                    {
                        var alt = new GridPos(dr, b.Pos.Col + behind);
                        if (IsCellFree(sim, alt) && TerrainMap.IsInside(alt)) { start = alt; break; }
                    }
                }
                if (IsCellFree(sim, start)) { u.Pos = start; u.PosF = SimVec2.FromGrid(start); }
                u.State = EntityState.Routing;
                u.HasMoveTarget = true;
                u.MoveTarget = gate ?? dest.Pos;
            }
        }

        /// <summary>溃逃目的地:同一线后方(远离敌方向)最近的己方建筑;无则己方大本营。</summary>
        static Entity FindRoutDestination(SimCore sim, Entity u)
        {
            int dir = u.Side == Side.Blue ? -1 : 1; // 远离敌方向
            Entity best = null;
            int bestD = int.MaxValue;
            foreach (var e in sim.Entities.Values)
            {
                if (e.Kind != EntityKind.Building || e.Side != u.Side) continue;
                int d = (e.Pos.Col - u.Pos.Col) * dir;
                if (d <= 0) continue; // 只取后方
                if (d < bestD) { best = e; bestD = d; }
            }
            if (best != null) return best;
            foreach (var e in sim.Entities.Values)
                if (e.Kind == EntityKind.Building && e.Side == u.Side && e.Building == BuildingId.Base)
                    return e;
            return null;
        }

        public static void Tick(SimCore sim, float dt)
        {
            // 1) 临时驻军:正式有空缺且未封城 → 自动转正式(用户新规,废除 10 秒计时)
            foreach (var b in sim.Entities.Values.Where(e => e.Kind == EntityKind.Building).ToList())
            {
                if (b.Sealed) continue; // 封城期间不转
                // 敌方攻城意图存续期间不转:临时驻军受保护,防止"转正后被交换杀死"
                if (b.Intent != SiegeIntent.None && b.IntentIssuer != b.Side) continue;
                int cap = b.IsFort ? 3 : (b.Building == null ? 0 : sim.BuildingConfigOf(b.Building.Value).GarrisonFormal);
                while (b.GarrisonTemp.Count > 0 && b.GarrisonFormal.Count < cap)
                {
                    var id = b.GarrisonTemp[0];
                    b.GarrisonTemp.RemoveAt(0);
                    b.GarrisonFormal.Add(id);
                }
            }

            foreach (var b in sim.Entities.Values.Where(e => e.Kind == EntityKind.Building).ToList())
            {
                var garrison = b.GarrisonFormal.Concat(b.GarrisonTemp).ToList();
                // 2) 依托射击(远程)
                foreach (var id in garrison)
                {
                    var u = sim.GetEntity(id);
                    if (u == null || u.Hero == null || !RangedHeroes.Contains(u.Hero.Value)) continue;
                    u.AttackCooldown -= dt;
                    if (u.AttackCooldown > 0f) continue;
                    var tgt = FindRangedTarget(sim, b, (int)u.Range);
                    if (tgt == null) continue;
                    ApplyDamage(sim, u, tgt);
                    u.AttackCooldown = u.AttackIntervalSec;
                }
                // 3) 近战出击:敌单位欧氏≤2.5 米(建筑中心起算)
                var bc = SimVec2.FromGrid(b.Pos);
                bool threat = sim.Entities.Values.Any(e =>
                    e.Kind == EntityKind.Unit && e.Side != b.Side &&
                    e.GarrisonBuildingId == 0 && e.State != EntityState.Dead && e.State != EntityState.Exchanging &&
                    SimVec2.Distance(e.PosF, bc) <= 2.5f);
                if (threat)
                foreach (var id in garrison)
                {
                    var u = sim.GetEntity(id);
                    if (u == null || u.Hero == null || RangedHeroes.Contains(u.Hero.Value)) continue;
                    var gate = FindGateCell(sim, b);
                    if (gate == null) break;
                    b.GarrisonFormal.Remove(id); b.GarrisonTemp.Remove(id);
                    u.Pos = gate.Value; u.PosF = SimVec2.FromGrid(gate.Value); u.GarrisonBuildingId = 0; u.State = EntityState.Idle;
                }
            }

            // 4) 建筑脱战回复(D27):非大本营
            foreach (var b in sim.Entities.Values.Where(e => e.Kind == EntityKind.Building).ToList())
            {
                if (b.Building == BuildingId.Base) continue;
                b.SecondsSinceCombat += dt;
                if (b.SecondsSinceCombat >= RegenDelaySec && b.Hp < b.MaxHp)
                    b.Hp = System.Math.Min(b.MaxHp, b.Hp + RegenPerSec * dt);
            }
        }

        public static GridPos RallyPointOf(Side side) =>
            side == Side.Blue ? new GridPos(15, 270) : new GridPos(15, 16);

        private static void DoEnter(SimCore sim, Entity u, int buildingId, List<int> list)
        {
            sim.FreeCell(u.Pos);
            u.GarrisonBuildingId = buildingId;
            u.State = EntityState.Idle;
            u.HasMoveTarget = false;
            list.Add(u.Id);
        }

        private static void Rollback(SimCore sim, List<int> accepted, int buildingId)
        { foreach (var id in accepted) sim.GetEntity(buildingId).GarrisonFormal.Remove(id); }
        private static void RollbackTemp(SimCore sim, List<int> accepted, int buildingId)
        { foreach (var id in accepted) { sim.GetEntity(buildingId).GarrisonTemp.Remove(id); } }

        /// <summary>城门位置(D18,M8.1b):占地矩形敌侧缘 +0.6 米列起,行优先建筑行(±ceil(HalfH) 内扩散),椭圆内、非水域,列向外延展找最近空位;全占返回 null。</summary>
        private static GridPos? FindGateCell(SimCore sim, Entity b, System.Collections.Generic.HashSet<long> exclude = null)
        {
            int dir = b.Side == Side.Blue ? 1 : -1; // 朝敌侧:蓝方敌人在 +x、红方敌人在 -x(蓝基地列 15、红基地列 271)
            var gateF = SimVec2.FromGrid(b.Pos) + new SimVec2(dir * (BuildingFootprint.HalfW(b) + 0.6f), 0f);
            int gateCol = gateF.ToGrid().Col;
            int rh = (int)System.Math.Ceiling(BuildingFootprint.HalfH(b)); // 建筑占地行范围
            int[] rows = { 0, -1, 1, -2, 2, -3, 3 }; // 行优先建筑行,向外扩散
            for (int d = 0; d <= 4; d++)
            {
                foreach (int r in rows)
                {
                    if (r < -rh || r > rh) continue;
                    var p = new GridPos(b.Pos.Row + r, gateCol + dir * d);
                    if (p.Row < 0 || p.Row >= LaneGrid.Rows || p.Col < 0 || p.Col >= LaneGrid.Cols) continue;
                    if (!TerrainMap.IsInside(p)) continue;
                    if (TerrainMap.IsWater(SimVec2.FromGrid(p))) continue;
                    if (exclude != null && exclude.Contains(((long)p.Row << 32) | (uint)p.Col)) continue; // 溃逃单位分派不同城门格
                    if (IsCellFree(sim, p)) return p;
                }
            }
            return null; // 城门区全占 → 无空位
        }

        private static bool IsCellFree(SimCore sim, GridPos p)
        {
            var c = SimVec2.FromGrid(p);
            foreach (var e in sim.Entities.Values)
            {
                if (e.State == EntityState.Dead) continue;
                if (e.Kind == EntityKind.Building && !e.Rubbled && e.Pos == p) return false;
                if (e.Kind == EntityKind.Unit && e.GarrisonBuildingId == 0
                    && SimVec2.Distance(e.PosF, c) < 1.0f) return false;
            }
            return true;
        }

        private static Entity FindRangedTarget(SimCore sim, Entity b, int range)
        {
            Entity best = null; float bestD = float.MaxValue; int bestId = int.MaxValue;
            var bc = SimVec2.FromGrid(b.Pos);
            foreach (var e in sim.Entities.Values)
            {
                if (e.Kind != EntityKind.Unit || e.Side == b.Side) continue;
                if (e.GarrisonBuildingId != 0 || e.State == EntityState.Dead || e.State == EntityState.Exchanging) continue;
                float d = SimVec2.Distance(e.PosF, bc);
                if (d > range) continue;
                if (d < bestD || (System.Math.Abs(d - bestD) <= 1e-4f && e.Id < bestId)) { best = e; bestD = d; bestId = e.Id; }
            }
            return best;
        }

        private static void ApplyDamage(SimCore sim, Entity attacker, Entity defender)
        {
            float dmg = DamageCalculator.CalculateDamage(attacker, defender, sim.Game, false);
            if (defender.Shield > 0)
            {
                float abs = System.Math.Min(defender.Shield, dmg);
                defender.Shield -= abs; dmg -= abs;
            }
            if (dmg > 0f) defender.Hp -= dmg;
            if (defender.Hp <= 0f && defender.State != EntityState.Dead)
            { defender.State = EntityState.Dead; sim.RemoveEntity(defender.Id); }
        }
    }

    // 侧别辅助
    internal static class SideUtilities
    {
        public static Side SideOf(SimCore sim, IReadOnlyList<int> unitIds) =>
            unitIds.Count > 0 && sim.Entities.ContainsKey(unitIds[0]) ? sim.GetEntity(unitIds[0]).Side : Side.Blue;
    }
```

```csharp
using System.Collections.Generic;

namespace IronCurtain.Sim
{
    /// <summary>M3.1 建筑占领核心:意图/入城交换/封城/自动驻军。M8.1b:FrontCell 改占地矩形敌侧缘外 1 格,交换逻辑不动。</summary>
    public static class SiegeSystem
    {
        // 城内交换列表:建筑id → 攻方城内单位id
        static readonly Dictionary<int, List<int>> CityAttackers = new Dictionary<int, List<int>>();

        /// <summary>测试/重开局用:清空内部状态。</summary>
        public static void Reset() => CityAttackers.Clear();

        static bool TryGetBuilding(SimCore sim, int id, out Entity b)
        {
            b = null;
            return sim.Entities.TryGetValue(id, out b) && b.Kind == EntityKind.Building;
        }

        public static SiegeIntent GetIntent(SimCore sim, int buildingId)
        {
            return TryGetBuilding(sim, buildingId, out var b) ? b.Intent : SiegeIntent.None;
        }

        public static bool SetIntent(SimCore sim, int buildingId, Side issuer, SiegeIntent intent, out string error)
        {
            error = null;
            if (!TryGetBuilding(sim, buildingId, out var b)) { error = "目标不是建筑"; return false; }
            if (b.Rubbled) { error = "废墟需先重建"; return false; }
            if (b.Sealed) { error = "封城中不可切换意图"; return false; }
            if (b.Side == Side.Neutral && intent == SiegeIntent.Destroy) { error = "中立建筑不可摧毁"; return false; }
            if (b.Building == BuildingId.Base && intent == SiegeIntent.Capture) { error = "大本营不可占领"; return false; }
            b.Intent = intent;
            b.IntentIssuer = issuer;
            return true;
        }

        static bool Occupied(SimCore sim, GridPos p, int ignoreId)
        {
            var c = SimVec2.FromGrid(p);
            foreach (var e in sim.Entities.Values)
            {
                if (e.Id == ignoreId) continue;
                if (e.GarrisonBuildingId != 0) continue; // 驻军/城内单位不占位
                if (e.Kind == EntityKind.Building && !e.Rubbled && e.Pos == p) return true;
                if (e.Kind == EntityKind.Unit && SimVec2.Distance(e.PosF, c) < 1.0f) return true;
            }
            return false;
        }

        /// <summary>攻城前排(M8.1b):占地矩形敌侧缘外 1 格(ceil(HalfW)+1),行集合 {0,-1,1,-2,2} 相对建筑行;过滤椭圆/水域/其他建筑占地格。</summary>
        static GridPos? FrontCell(SimCore sim, Entity unit, Entity b)
        {
            int dir = unit.PosF.X <= SimVec2.FromGrid(b.Pos).X ? -1 : 1; // 攻方朝建筑方向,只在本侧找位
            int col = b.Pos.Col + dir * ((int)System.Math.Ceiling(BuildingFootprint.HalfW(b)) + 1);
            // 其他建筑占地格集合(同 M8.1a 出生点过滤)
            var blocked = new HashSet<long>();
            foreach (var o in sim.Entities.Values)
                if (o.Kind == EntityKind.Building && !o.Rubbled && o.Id != b.Id)
                    foreach (var c in BuildingFootprint.CellsOccupied(o)) blocked.Add(c.Row * 1000L + c.Col);
            int[] drs = { 0, -1, 1, -2, 2 };
            foreach (int dr in drs)
            {
                var p = new GridPos(b.Pos.Row + dr, col);
                if (p.Row < 0 || p.Row >= LaneGrid.Rows || p.Col < 0 || p.Col >= LaneGrid.Cols) continue;
                if (!TerrainMap.IsInside(p)) continue;
                if (TerrainMap.IsWater(SimVec2.FromGrid(p))) continue;
                if (blocked.Contains(p.Row * 1000L + p.Col)) continue;
                if (!Occupied(sim, p, unit.Id)) return p;
            }
            return null;
        }

        public static bool IssueOrder(SimCore sim, IReadOnlyList<int> unitIds, int buildingId, out string error)
        {
            error = null;
            if (!TryGetBuilding(sim, buildingId, out var b)) { error = "目标不是建筑"; return false; }
            foreach (var id in unitIds)
            {
                if (!sim.Entities.TryGetValue(id, out var u) || u.Kind != EntityKind.Unit || u.Hp <= 0) continue;
                if (u.State == EntityState.Routing) continue; // 溃逃期间不可调遣
                var cell = FrontCell(sim, u, b);
                if (cell == null) continue; // 无空位则跳过该单位
                u.TargetBuildingId = buildingId;
                sim.SetMoveTarget(id, cell.Value);
            }
            return true;
        }

        static List<int> AttackersOf(int buildingId)
        {
            if (!CityAttackers.TryGetValue(buildingId, out var l))
            {
                l = new List<int>();
                CityAttackers[buildingId] = l;
            }
            return l;
        }

        static void EnterCity(SimCore sim, Entity b, Entity u)
        {
            var list = AttackersOf(b.Id);
            if (list.Contains(u.Id)) return; // 防重复入城
            list.Add(u.Id);
            sim.FreeCell(u.Pos);           // 离格:释放格占用,不堵后续部队
            u.State = EntityState.Exchanging;
            b.Sealed = true;
        }

        static void Capture(SimCore sim, Entity b, List<int> survivors)
        {
            b.Side = b.IntentIssuer;
            if (b.Hp <= 0 && b.MaxHp <= 0)
            {
                var cfg = sim.BuildingConfigOf(b.Building.Value);
                b.MaxHp = cfg.MaxHp;
                b.Hp = cfg.MaxHp;
            }
            b.Sealed = false;
            CityAttackers.Remove(b.Id);
            GarrisonSystem.OnCityCaptured(sim, b); // 旧主临时驻军溃逃
            foreach (var id in survivors)
            {
                if (!sim.Entities.TryGetValue(id, out var e)) continue;
                e.TargetBuildingId = 0;
                e.State = EntityState.Idle;
                if (b.GarrisonFormal.Count < 9)
                {
                    b.GarrisonFormal.Add(id);
                    e.GarrisonBuildingId = b.Id;
                }
                // 超出 9 人者离城原地待命
            }
        }

        static void KillDefenders(SimCore sim, Entity b, int n)
        {
            // 交换只消耗正式驻军,临时驻军受保护(用户拍板,覆盖 GDD 2.4.1)
            for (int i = 0; i < n && b.GarrisonFormal.Count > 0; i++)
            {
                var id = b.GarrisonFormal[0];
                b.GarrisonFormal.RemoveAt(0);
                sim.RemoveEntity(id);
            }
        }

        public static void Tick(SimCore sim, float dt)
        {
            // 1) 到达判定
            var arrived = new List<Entity>();
            foreach (var e in sim.Entities.Values)
                if (e.Kind == EntityKind.Unit && e.TargetBuildingId != 0 && !e.HasMoveTarget
                    && e.GarrisonBuildingId == 0 && e.State != EntityState.Exchanging)
                    arrived.Add(e);
            foreach (var u in arrived)
            {
                if (!TryGetBuilding(sim, u.TargetBuildingId, out var b)) { u.TargetBuildingId = 0; continue; }
                if (b.Sealed)
                {
                    if (u.Side == b.IntentIssuer) EnterCity(sim, b, u); // 封城期间攻方增援进城
                    continue;
                }
                if (b.Intent == SiegeIntent.Capture)
                {
                    if (b.GarrisonFormal.Count + b.GarrisonTemp.Count > 0) EnterCity(sim, b, u);
                    else Capture(sim, b, new List<int> { u.Id });
                }
                else if (b.Intent == SiegeIntent.Destroy)
                {
                    u.State = EntityState.Attacking; // 拆塔结算在 M3.4
                }
            }

            // 2) 交换结算:每 10 tick 一轮
            if (sim.Tick % 10 != 0) return;
            var sealedBuildings = new List<Entity>();
            foreach (var e in sim.Entities.Values)
                if (e.Kind == EntityKind.Building && e.Sealed) sealedBuildings.Add(e);
            foreach (var b in sealedBuildings)
            {
                var atk = AttackersOf(b.Id);
                while (atk.Count > 0 && b.GarrisonFormal.Count > 0)
                {
                    var aid = atk[0];
                    atk.RemoveAt(0); // 该单位结束城内待命(无论生死都离开交换队列)
                    bool zhen = sim.Entities.TryGetValue(aid, out var a) && a.Hero == HeroId.ZhenYue;
                    KillDefenders(sim, b, zhen ? 2 : 1); // 镇岳 1:2,优先消耗守军
                    if (!zhen) sim.RemoveEntity(aid);    // 1:1 互换,攻方单位阵亡
                    else if (sim.Entities.ContainsKey(aid)) { a.State = EntityState.Idle; }
                }
                if (b.GarrisonFormal.Count == 0)
                {
                    // 正式守军被肃清 → 城池易主,临时驻军溃逃(临时受保护不参与交换)
                    var survivors = new List<int>(atk);
                    foreach (var id in survivors)
                        if (sim.Entities.TryGetValue(id, out var e2)) e2.State = EntityState.Idle;
                    Capture(sim, b, survivors);
                }
                else if (atk.Count == 0)
                {
                    // 攻方被肃清:解封,意图保留,剩余守军留城
                    b.Sealed = false;
                    CityAttackers.Remove(b.Id);
                }
            }
        }
    }
}
```

### 3.7 RegimentSystem.cs(M8.2 已落地,只读)
```csharp
using System.Collections.Generic;

namespace IronCurtain.Sim
{
    /// <summary>
    /// 兵团底座(M8.2a 第一批):1 主将 + 20 兵,兵随将走(4×5 阵型跟随);
    /// 主将死亡士兵永不溃逃、无将状态就近逐杀至死。纯 tick,无 UnityEngine 依赖。
    /// </summary>
    public static class RegimentSystem
    {
        public const int SoldiersPerGeneral = 20;
        public const int FormationRows = 4;   // 4 行 × 5 列,行列距 1.0 米
        public const int FormationCols = 5;

        public static SimVec2 FacingOf(Entity g)
        {
            var def = g.Side == Side.Blue ? new SimVec2(1f, 0f) : new SimVec2(-1f, 0f);
            if (g.HasMoveTarget)
            {
                var d = (SimVec2.FromGrid(g.MoveTarget) - g.PosF).Normalized();
                if (d.SqrMagnitude > 1e-8f) return d;
            }
            return def;
        }

        public static SimVec2 SlotOffset(Entity g, int slot)
        {
            var facing = FacingOf(g);
            var perp = new SimVec2(facing.Y, -facing.X);
            int row = slot / FormationCols;
            int col = slot % FormationCols;
            return facing * -(row + 1) + perp * (col - 2);
        }

        public static void Tick(SimCore sim, float dt)
        {
            // 快照遍历:士兵死亡结算会移除实体,禁止边枚举边修改
            var snapshot = new List<Entity>(sim.Entities.Values);
            foreach (var e in snapshot)
            {
                if (e.Kind != EntityKind.Unit || !e.IsSoldier || e.State == EntityState.Dead) continue;
                if (e.GarrisonBuildingId != 0) continue;

                var g = e.GeneralId != 0 ? sim.GetEntity(e.GeneralId) : null;
                if (g != null && g.Kind == EntityKind.Unit && g.State != EntityState.Dead)
                {
                    // a) 主将存活:阵型跟随(士兵不写 HasMoveTarget/State)
                    var target = g.PosF + SlotOffset(g, e.FormationSlot);
                    float dist = SimVec2.Distance(e.PosF, target);
                    if (dist > 0.2f) StepMove(e, target, dist, dt);
                }
                else
                {
                    // b) 无将续战:就近索敌逐杀至死(永不溃逃/撤退/响应指令)
                    var enemy = NearestEnemy(sim, e);
                    if (enemy == null) continue;
                    float dist = SimVec2.Distance(e.PosF, enemy.PosF);
                    if (dist <= e.Range + 0.3f) continue; // 交战交给 CombatUpdate
                    StepMove(e, enemy.PosF, dist, dt);
                }
            }
        }

        private static void StepMove(Entity e, SimVec2 target, float dist, float dt)
        {
            float spd = e.MoveSpeed * TerrainMap.MoveFactor(e.PosF);
            float step = System.Math.Min(spd * dt, dist);
            e.PosF = TerrainMap.ClampToEllipse(e.PosF + (target - e.PosF).Normalized() * step);
            e.Pos = e.PosF.ToGrid();
        }

        /// <summary>最近敌索敌:与 SimCore.FindNearestEnemy 同一欧氏语义(单位、非己方、非死亡、非城内/交换)。</summary>
        private static Entity NearestEnemy(SimCore sim, Entity e)
        {
            Entity best = null;
            float bestDist = float.MaxValue;
            foreach (var o in sim.Entities.Values)
            {
                if (o.Kind != EntityKind.Unit || o.Side == e.Side || o.State == EntityState.Dead) continue;
                if (o.GarrisonBuildingId != 0 || o.State == EntityState.Exchanging) continue;
                float d = SimVec2.Distance(e.PosF, o.PosF);
                if (d < bestDist) { best = o; bestDist = d; }
            }
            return best;
        }

        /// <summary>生成主将 + 20 名士兵的完整兵团,返回主将实体。</summary>
        public static Entity SpawnGeneral(SimCore sim, Side side, HeroId hero, GridPos pos)
        {
            var c = sim.Game.Heroes[(int)hero];
            var anchor = SimVec2.FromGrid(pos);
            // 朝向探针:仅用于出生时计算槽位偏移(默认朝向,与主将无移动目标时一致)
            var probe = new Entity(0, side, EntityKind.Unit, pos, 0f);

            // 先生成士兵(每生成一名立即摆到槽位,避免同点出生距离校验冲突),主将最后落位
            var soldiers = new List<Entity>(SoldiersPerGeneral);
            for (int slot = 0; slot < SoldiersPerGeneral; slot++)
            {
                var s = sim.SpawnEntity(side, EntityKind.Unit, pos);
                s.IsSoldier = true;
                s.FormationSlot = slot;
                s.Hero = hero; // 供类型与渲染(伤害英雄规则 M8.2b 收口,本轮已知)
                s.MaxHp = c.MaxHp / 10f;
                s.Hp = s.MaxHp;
                s.Attack = c.Attack / 10f;
                s.AttackIntervalSec = (float)c.AttackIntervalSec;
                s.MoveSpeed = (float)c.MoveSpeed;
                s.Range = c.Range;
                s.Shield = c.Shield / 10f;
                s.ShieldMax = s.Shield;
                s.PosF = anchor + SlotOffset(probe, slot);
                s.Pos = s.PosF.ToGrid();
                soldiers.Add(s);
            }

            var g = sim.SpawnEntity(side, EntityKind.Unit, pos);
            g.Hero = hero;
            g.MaxHp = c.MaxHp;
            g.Hp = c.MaxHp;
            g.Attack = c.Attack;
            g.AttackIntervalSec = (float)c.AttackIntervalSec;
            g.Range = c.Range;
            g.MoveSpeed = (float)c.MoveSpeed;
            if (c.Shield > 0) { g.Shield = c.Shield; g.ShieldMax = c.Shield; }

            foreach (var s in soldiers) s.GeneralId = g.Id;
            return g;
        }
    }
}
```

```csharp
using System;
using System.Collections.Generic;

namespace IronCurtain.Sim
{
    public sealed class SimCore
    {
        public SimConfig Config { get; }
        public SeededRng Rng { get; }
        public long Tick { get; private set; }
        public int EntityCount => _entities.Count;
        public IReadOnlyDictionary<int, Entity> Entities => _entities;

        /// <summary>游戏数值配置(由调用方注入;null 时战斗模块不生效)。</summary>
        public GameConfig Game { get; set; }

        /// <summary>胜负:大本营被摧毁时置位。</summary>
        public Side? Winner { get; private set; }
        public void SetWinner(Side side) { Winner = side; }

        // 升级(M4):每方已购买的英雄被动
        private readonly Dictionary<Side, HashSet<HeroId>> _upgrades = new Dictionary<Side, HashSet<HeroId>>();

        public bool IsUpgraded(Side side, HeroId hero)
        {
            return _upgrades.TryGetValue(side, out var s) && s.Contains(hero);
        }

        /// <summary>2EP 购买升级,单英雄每局一次。</summary>
        public bool BuyUpgrade(Side side, HeroId hero, out string error)
        {
            error = null;
            if (IsUpgraded(side, hero)) { error = "该英雄已升级"; return false; }
            if (!SpendEp(side, Game.Ep.UpgradeCost)) { error = "EP 不足"; return false; }
            if (!_upgrades.TryGetValue(side, out var s))
            {
                s = new HashSet<HeroId>();
                _upgrades[side] = s;
            }
            s.Add(hero);
            return true;
        }

        // 经济(M2)
        private readonly Dictionary<Side, float> _ep = new Dictionary<Side, float>();
        private readonly Dictionary<Side, float> _epAccum = new Dictionary<Side, float>();
        private readonly Dictionary<Side, GridPos> _rally = new Dictionary<Side, GridPos>
        {
            { Side.Blue, new GridPos(15, 270) },
            { Side.Red, new GridPos(15, 16) },
        };

        private readonly Dictionary<int, Entity> _entities = new Dictionary<int, Entity>();
        private readonly HashSet<int> _occupied = new HashSet<int>();
        private readonly Queue<ICommand> _commandQueue = new Queue<ICommand>();
        private readonly List<Action<SimEvent>> _listeners = new List<Action<SimEvent>>();
        private readonly LaneGrid _grid = new LaneGrid();
        private int _nextId = 1;

        public SimCore(SimConfig config, SeededRng rng)
        {
            Config = config;
            Rng = rng;
        }

        public Entity SpawnEntity(Side side, EntityKind kind, GridPos pos)
        {
            if (!_grid.InBounds(pos))
                throw new InvalidOperationException($"位置越界: {pos}");
            int idx = _grid.ToIndex(pos);
            if (_occupied.Contains(idx))
                throw new InvalidOperationException($"格子已占用: {pos}");
            // 连续坐标下单位不再占格;出生防重叠用距离校验(驻军/城内单位不占位)
            if (kind == EntityKind.Unit)
            {
                var center = SimVec2.FromGrid(pos);
                foreach (var o in _entities.Values)
                    if (o.Kind == EntityKind.Unit && o.GarrisonBuildingId == 0
                        && SimVec2.Distance(o.PosF, center) < 0.5f)
                        throw new InvalidOperationException($"出生点过近: {pos}");
            }

            var e = new Entity(_nextId++, side, kind, pos, 0f);
            e.PosF = SimVec2.FromGrid(pos);
            _entities[e.Id] = e;
            if (kind == EntityKind.Building)
                _occupied.Add(idx);
            Emit(new SimEvent(Tick, SimEventType.EntitySpawned, e.Id));
            return e;
        }

        public bool RemoveEntity(int id)
        {
            if (!_entities.TryGetValue(id, out var e)) return false;
            if (e.Kind == EntityKind.Building)
                _occupied.Remove(_grid.ToIndex(e.Pos));
            _entities.Remove(id);
            Emit(new SimEvent(Tick, SimEventType.EntityRemoved, id));
            return true;
        }

        public Entity GetEntity(int id) => _entities.TryGetValue(id, out var e) ? e : null;

        /// <summary>释放格位占用(驻军/入城交换等"离格但实体仍在"的场景)。</summary>
        public void FreeCell(GridPos pos)
        {
            _occupied.Remove(_grid.ToIndex(pos));
        }

        public void SubmitCommand(ICommand cmd) => _commandQueue.Enqueue(cmd);

        /// <summary>单位移动指令:目标越界/实体非单位时抛异常。</summary>
        public void SetMoveTarget(int id, GridPos target)
        {
            if (!_entities.TryGetValue(id, out var e))
                throw new InvalidOperationException($"实体不存在: {id}");
            if (e.Kind != EntityKind.Unit)
                throw new InvalidOperationException($"非线单位不可设置移动目标: {id}");
            if (!_grid.InBounds(target))
                throw new InvalidOperationException($"移动目标越界: {target}");

            e.MoveTarget = target;
            e.HasMoveTarget = true;
            e.State = EntityState.Moving;
        }

        public void Step()
        {
            Tick++;
            Emit(new SimEvent(Tick, SimEventType.StepBegin, 0));
            while (_commandQueue.Count > 0)
            {
                var cmd = _commandQueue.Dequeue();
                cmd.Execute(this);
                Emit(new SimEvent(Tick, SimEventType.CommandExecuted, 0));
            }
            MoveUnits();
            CombatUpdate();
            EconomyUpdate();
            SiegeSystem.Tick(this, 1f / Config.TickRate);
            TowerCombatSystem.Tick(this, 1f / Config.TickRate);
            GarrisonSystem.Tick(this, 1f / Config.TickRate);
            DestroySystem.Tick(this, 1f / Config.TickRate);
            HexSystem.Tick(this, 1f / Config.TickRate);
            VisionSystem.Tick(this, 1f / Config.TickRate);
            AiSystem.Tick(this, 1f / Config.TickRate);
            Emit(new SimEvent(Tick, SimEventType.StepEnd, 0));
        }

        public void AddEventListener(Action<SimEvent> listener) => _listeners.Add(listener);
        public void ClearEventListeners() => _listeners.Clear();

        // ===== 经济与生产挂钩(M2)=====

        /// <summary>生成建筑实体(归属方 owner)。中立建筑无血量;已方建筑按配置填数值。</summary>
        public Entity SpawnBuilding(Side owner, BuildingId type, GridPos pos)
        {
            var e = SpawnEntity(owner, EntityKind.Building, pos);
            e.Building = type;
            if (Game != null && owner != Side.Neutral)
            {
                var c = BuildingConfigOf(type);
                e.MaxHp = c.MaxHp;
                e.Hp = c.MaxHp;
                e.Attack = c.Attack;
                e.AttackIntervalSec = (float)c.AttackIntervalSec;
                e.Range = c.Range;
            }
            return e;
        }

        public BuildingCfg BuildingConfigOf(BuildingId id)
        {
            foreach (var b in Game.Buildings)
                if (b.Id == id) return b;
            throw new InvalidOperationException("建筑配置缺失: " + id);
        }

        /// <summary>某方拥有的资源点数量(按场上建筑实况计算)。</summary>
        public int OwnedResourcePointsOf(Side side)
        {
            int n = 0;
            foreach (var e in _entities.Values)
                if (e.Kind == EntityKind.Building && e.Building == BuildingId.ResourcePoint && e.Side == side)
                    n++;
            return n;
        }

        /// <summary>英雄人口占用(Game 配置)。</summary>
        public int PopCostOf(HeroId id)
        {
            foreach (var h in Game.Heroes)
                if (h.Id == id) return h.Population;
            return 0;
        }

        /// <summary>已用人口 = 场上单位人口 + 兵营进行中生产人口。</summary>
        public int PopulationOf(Side side)
        {
            int used = 0;
            foreach (var e in _entities.Values)
            {
                if (e.Side != side) continue;
                if (e.Kind == EntityKind.Unit && e.Hero != null)
                    used += PopCostOf(e.Hero.Value);
                else if (e.Kind == EntityKind.Building && e.Building == BuildingId.Barracks && e.Prod != null)
                    used += e.Prod.InProgressPop(id => PopCostOf(id));
            }
            return used;
        }

        public int PopulationCapOf(Side side)
        {
            int cap = (int)Game.Production.BasePopulation;
            cap += Game.Production.ResourcePointPopulation * OwnedResourcePointsOf(side);
            return cap + HexSystem.PopCapBonusOf(this, side);
        }

        public float EpOf(Side side) => _ep.TryGetValue(side, out var v) ? v : 0f;

        public bool SpendEp(Side side, float amount)
        {
            float cur = EpOf(side);
            if (cur < amount) return false;
            _ep[side] = cur - amount;
            return true;
        }

        public GridPos RallyPointOf(Side side) => _rally[side];
        public void SetRallyPoint(Side side, GridPos p) { _rally[side] = p; }

        private void EconomyUpdate()
        {
            if (Game == null) return;
            float dt = 1f / Config.TickRate;

            foreach (Side s in new[] { Side.Blue, Side.Red })
            {
                int pts = OwnedResourcePointsOf(s);
                if (pts <= 0) continue;
                float acc = _epAccum.TryGetValue(s, out var a) ? a : 0f;
                acc += pts * dt;
                float interval = (float)Game.Ep.IntervalSeconds;
                while (acc >= interval)
                {
                    acc -= interval;
                    float cur = EpOf(s);
                    if (cur < Game.Ep.Cap) _ep[s] = cur + 1f;
                }
                _epAccum[s] = acc;
            }

            ProductionSystem.Tick(this, dt);
        }

        private void MoveUnits()
        {
            float perTick = 1f / Config.TickRate;
            foreach (var e in _entities.Values)
            {
                if (e.IsSoldier) continue; // 士兵移动全归 RegimentSystem.Tick,不进命令移动分支
                if (e.Kind != EntityKind.Unit || (e.State != EntityState.Moving && e.State != EntityState.Routing) || e.MoveSpeed <= 0f)
                    continue;
                if (e.GarrisonBuildingId != 0) continue; // 驻军单位不占线移动
                if (!e.HasMoveTarget) continue;          // 无目标的 Routing 原地待命

                SimVec2 target = SimVec2.FromGrid(e.MoveTarget);
                SimVec2 delta = target - e.PosF;
                float dist = delta.Magnitude;
                if (dist <= 0.15f)
                {
                    e.HasMoveTarget = false;
                    e.State = EntityState.Idle;
                    e.MoveAccum = 0f;
                    continue;
                }

                float spd = MoveSpeedOf(e);
                float step = System.Math.Min(spd * perTick, dist);
                e.PosF = TerrainMap.ClampToEllipse(e.PosF + delta.Normalized() * step);
                PushOutOfBuildings(e);
                e.PosF = TerrainMap.ClampToEllipse(e.PosF);
                e.Pos = e.PosF.ToGrid();

                e.MoveAccum += step;
                if (e.MoveAccum >= 1f)
                {
                    e.MoveAccum = 0f;
                    Emit(new SimEvent(Tick, SimEventType.UnitAdvanced, e.Id)); // 节流:约每 1 米一次
                }

                if ((target - e.PosF).Magnitude <= 0.15f)
                {
                    e.HasMoveTarget = false;
                    e.State = EntityState.Idle;
                    e.MoveAccum = 0f;
                }
            }
            RegimentSystem.Tick(this, perTick); // 兵团底座(M8.2a):阵型跟随/无将续战,先于软碰撞
            SoftCollide();
        }

        private float MoveSpeedOf(Entity e)
        {
            float spd = e.MoveSpeed * e.Status.MoveMultiplier;
            if (e.IsCharging && e.Hero == HeroId.ZhuFeng && Game != null)
                spd = (float)ChargeSpeedOf(HeroId.ZhuFeng);
            if (e.Hero == HeroId.ZhenYue && Tick >= 1800) // 3 分钟后移速衰减(D11 默认 1.4)
                spd = IsUpgraded(e.Side, HeroId.ZhenYue) ? 2.2f : 1.4f;
            spd *= HexSystem.MoveMultiplierOf(this, e); // 海克斯移速层
            spd *= TerrainMap.MoveFactor(e.PosF);       // 涉水减半/桥上正常
            return spd;
        }

        /// <summary>建筑碰撞:把单位推出建筑占地矩形(连续坐标下的障碍语义,M8.1a 矩形化)。</summary>
        private void PushOutOfBuildings(Entity u)
        {
            foreach (var b in _entities.Values)
            {
                if (b.Kind != EntityKind.Building || b.Rubbled) continue;
                if (!BuildingFootprint.Contains(b, u.PosF)) continue;
                u.PosF = BuildingFootprint.ClosestPoint(b, u.PosF); // 点在矩形内→最小穿透轴推出,幂等无死循环
            }
        }

        /// <summary>单位间软碰撞:间距小于 0.9 米时沿连线对半推开。</summary>
        private void SoftCollide()
        {
            var units = new List<Entity>();
            foreach (var e in _entities.Values)
                if (e.Kind == EntityKind.Unit && e.GarrisonBuildingId == 0
                    && e.State != EntityState.Dead && e.State != EntityState.Exchanging)
                    units.Add(e);

            const float MinDist = 0.9f;
            var moved = new HashSet<int>();
            for (int i = 0; i < units.Count; i++)
            {
                for (int j = i + 1; j < units.Count; j++)
                {
                    var a = units[i];
                    var b = units[j];
                    // 兵团底座(M8.2a):同兵团士兵之间、士兵与己方主将之间不互推,避免阵型震荡
                    if ((a.IsSoldier || b.IsSoldier) && (a.GeneralId == b.Id || b.GeneralId == a.Id || (a.IsSoldier && b.IsSoldier && a.GeneralId != 0 && a.GeneralId == b.GeneralId))) continue;
                    float dx = a.PosF.X - b.PosF.X, dy = a.PosF.Y - b.PosF.Y;
                    float d2 = dx * dx + dy * dy;
                    if (d2 >= MinDist * MinDist) continue;
                    float d = (float)System.Math.Sqrt(d2);
                    float push = (MinDist - d) * 0.5f;
                    float nx = d > 1e-6f ? dx / d : 1f;
                    float ny = d > 1e-6f ? dy / d : 0f;
                    a.PosF = TerrainMap.ClampToEllipse(a.PosF + new SimVec2(nx * push, ny * push));
                    b.PosF = TerrainMap.ClampToEllipse(b.PosF - new SimVec2(nx * push, ny * push));
                    moved.Add(a.Id);
                    moved.Add(b.Id);
                }
            }
            // 仅被推挤的单位回写格(避免覆盖未移动单位的手动定位)
            foreach (var e in units)
                if (moved.Contains(e.Id))
                    e.Pos = e.PosF.ToGrid();
        }

        /// <summary>建筑外接半径(米)= Max(HalfW,HalfH)+0.4。兼容旧近战判定,下一批 M8.1b 换精确矩形距;新代码请用 BuildingFootprint。</summary>
        public static float BuildingCollisionRadius(Entity b)
        {
            return System.Math.Max(BuildingFootprint.HalfW(b), BuildingFootprint.HalfH(b)) + 0.4f;
        }

        private void CombatUpdate()
        {
            if (Game == null) return;
            float dt = 1f / Config.TickRate;
            var dead = new List<int>();

            foreach (var e in _entities.Values)
            {
                if (e.Kind != EntityKind.Unit || e.State == EntityState.Dead) continue;
                if (e.GarrisonBuildingId != 0) continue; // 驻军单位不参与场外逻辑

                e.Status.Tick(dt);
                e.SecondsSinceCombat += dt;
                HeroAbilitySystem.UpdatePanShiShield(e, dt);
                if (e.Hero == HeroId.ZhuFeng && e.IsCharging && !HeroAbilitySystem.CanCharge(e, this))
                    e.IsCharging = false;

                if (e.Attack <= 0f) continue;

                if (e.AttackCooldown > 0f)
                    e.AttackCooldown -= dt;

                Entity target = FindNearestEnemy(e, e.State == EntityState.Routing ? 1.4f : float.MaxValue);
                if (target == null)
                {
                    if (e.State == EntityState.Attacking) e.State = EntityState.Idle;
                    continue;
                }

                // 溃逃单位保持 Routing 状态(不可调遣语义),不转入 Attacking
                if (e.State != EntityState.Routing)
                {
                    e.HasMoveTarget = false;
                    e.State = EntityState.Attacking;
                }
                if (e.AttackCooldown <= 0f)
                {
                    e.AttackCooldown = e.AttackIntervalSec * e.Status.AttackIntervalMultiplier;
                    ApplyAttack(e, target, dead);
                }
            }

            foreach (int id in dead)
            {
                if (_entities.TryGetValue(id, out var e))
                {
                    if (e.Hero != null) HexSystem.OnUnitDied(this, e.Side); // 陷阵之志
                    Emit(new SimEvent(Tick, SimEventType.EntityDied, id));
                    if (e.Kind == EntityKind.Building)
                        _occupied.Remove(_grid.ToIndex(e.Pos));
                    _entities.Remove(id);
                    Emit(new SimEvent(Tick, SimEventType.EntityRemoved, id));
                }
            }
        }

        private Entity FindNearestEnemy(Entity e, float maxDist)
        {
            Entity best = null;
            float bestDist = float.MaxValue;
            foreach (var o in _entities.Values)
            {
                if (o.Kind != EntityKind.Unit || o.Side == e.Side || o.State == EntityState.Dead)
                    continue;
                if (o.GarrisonBuildingId != 0 || o.State == EntityState.Exchanging)
                    continue; // 城内/交换中的单位不可被城外锁定
                float d = SimVec2.Distance(e.PosF, o.PosF);
                float effRange = e.Range + 0.3f; // 近战贴身容差
                if (d > effRange || d > maxDist) continue;
                if (d < bestDist - 1e-4f || (System.Math.Abs(d - bestDist) <= 1e-4f && best != null && o.Id < best.Id))
                {
                    best = o;
                    bestDist = d;
                }
            }
            return best;
        }

        private void ApplyAttack(Entity attacker, Entity defender, List<int> dead)
        {
            attacker.SecondsSinceCombat = 0f;
            defender.SecondsSinceCombat = 0f;

            // 基础层(GDD 公式)
            float baseDmg = DamageCalculator.CalculateDamage(attacker, defender, Game, attacker.IsCharging);

            // 星落升级·诛灭:基础攻击 5 → 6(基础层内替换)
            bool zhuMie = attacker.Hero == HeroId.XingLuo && IsUpgraded(attacker.Side, HeroId.XingLuo);
            if (zhuMie && attacker.Attack > 0f) baseDmg = baseDmg / attacker.Attack * 6f;

            // 叠算层(D28):输出加算、承伤加算、免伤乘算;溃逃并入对应层
            float outSum = attacker.Status.AttackBonus + HexSystem.OutputBonusOf(this, attacker)
                         + (attacker.State == EntityState.Routing ? -0.3f : 0f);
            float takeSum = defender.Status.DamageTakenBonus + HexSystem.DamageTakenBonusOf(this, defender)
                          + (defender.State == EntityState.Routing ? 0.3f : 0f);
            float reduce = defender.Status.DamageReduction * HexSystem.DamageReductionOf(this, defender);
            float dmg = baseDmg * (1f + outSum) * (1f + takeSum) * reduce;

            // 附加项(层后加法)
            if (zhuMie) dmg += 0.01f * defender.MaxHp;

            // 护盾优先吸收,溢出扣生命
            float remaining = dmg;
            if (defender.Shield > 0f)
            {
                float absorbed = Math.Min(defender.Shield, remaining);
                defender.Shield -= absorbed;
                remaining -= absorbed;
            }
            defender.Hp -= remaining;

            Emit(new SimEvent(Tick, SimEventType.DamageDealt, defender.Id, dmg));

            HeroAbilitySystem.ApplyOnHitEffects(this, attacker, defender);

            // 断峰升级·破军:身前 90 度扇形(半径 2 米)内全部敌人受单次 75% 伤害
            if (attacker.Hero == HeroId.DuanFeng && IsUpgraded(attacker.Side, HeroId.DuanFeng))
                Cleave(attacker, defender, dead);

            if (defender.Hp <= 0f)
            {
                defender.State = EntityState.Dead;
                dead.Add(defender.Id);
            }
        }

        private void Cleave(Entity attacker, Entity mainTarget, List<int> dead)
        {
            float facing = mainTarget.PosF.X >= attacker.PosF.X ? 1f : -1f;
            foreach (var o in _entities.Values)
            {
                if (o.Id == mainTarget.Id) continue;
                if (o.Kind != EntityKind.Unit || o.Side == attacker.Side || o.State == EntityState.Dead) continue;
                if (o.GarrisonBuildingId != 0 || o.State == EntityState.Exchanging) continue;
                float dx = o.PosF.X - attacker.PosF.X;
                float dy = o.PosF.Y - attacker.PosF.Y;
                if (dx * facing < 0) continue;         // 身后不打
                if (dx * dx + dy * dy > 4f) continue;  // 半径 2 米扇形
                float baseDmg = 0.75f * DamageCalculator.CalculateDamage(attacker, o, Game, attacker.IsCharging);
                float outSum = attacker.Status.AttackBonus + (attacker.State == EntityState.Routing ? -0.3f : 0f);
                float takeSum = o.Status.DamageTakenBonus + (o.State == EntityState.Routing ? 0.3f : 0f);
                float dmg = baseDmg * (1f + outSum) * (1f + takeSum) * o.Status.DamageReduction;
                float remaining = dmg;
                if (o.Shield > 0f)
                {
                    float absorbed = System.Math.Min(o.Shield, remaining);
                    o.Shield -= absorbed;
                    remaining -= absorbed;
                }
                o.Hp -= remaining;
                Emit(new SimEvent(Tick, SimEventType.DamageDealt, o.Id, dmg));
                if (o.Hp <= 0f) { o.State = EntityState.Dead; dead.Add(o.Id); }
            }
        }

        private double ChargeSpeedOf(HeroId id)
        {
            foreach (var h in Game.Heroes)
                if (h.Id == id) return h.ChargeSpeed;
            return 0;
        }

        private void Emit(SimEvent ev)
        {
            for (int i = _listeners.Count - 1; i >= 0; i--)
                _listeners[i]?.Invoke(ev);
        }
    }
```
