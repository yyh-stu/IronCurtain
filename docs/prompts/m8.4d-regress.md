# M8.4d 回归修复:资源点驻军恢复 0 + 士兵不继承英雄被动(制作人拍板 2026-09-06)

## 一、角色与任务
前序模块有两处越权设计被制作人否决,本轮回归原设计。**只允许输出/修改以下 4 个文件,其余零改动:**
- `Assets/Resources/gameconfig.json`
- `Assets/Scripts/Sim/HeroAbilitySystem.cs`
- `Assets/Tests/EditMode/SiegeTests.cs`(仅 T1 适配)
- `Assets/Tests/EditMode/RegimentTests.cs`(仅追加 2 条断言)

## 二、规格(权威)
### 2.1 gameconfig.json
- `Buildings[2]`(资源点)的 `GarrisonFormal`/`GarrisonTemp` 由 30 改回 **0**(资源点不可驻军,原设计);其余任何数值不动。

### 2.2 HeroAbilitySystem.cs:士兵不继承英雄被动(与"士兵平伤"对齐)
1. **攻击附带被动**(断峰迟滞 / 坠日对镇岳重伤 / 坠日升级诛心脆弱):任一方为士兵(`attacker.IsSoldier || defender.IsSoldier`)则全部跳过(士兵的 Hero 字段仅用于渲染/类型标识,不触发被动;镇岳的士兵也不再被视为"镇岳"目标)。
2. **磐石脱战回盾** `UpdatePanShiShield`:`if (e.IsSoldier) return;` 加在最前。
3. **逐风冲锋阻断判定** `CanCharge`:aura 遍历里跳过士兵(`if (o.IsSoldier) continue;`)——敌方磐石/断峰的**士兵**不构成冲锋阻断源(它们不是真磐石/断峰)。
4. 其余逻辑不动。

### 2.3 SiegeTests.cs:T1 适配(仅此一条)
- 现状:T1 断言中立资源点占领后"进攻士兵自动正式驻军"——资源点驻军恢复 0 后不再成立。
- 改为断言:占领成功、补配置血量照旧;**幸存士兵不驻入**(GarrisonFormal.Count==0、士兵 GarrisonBuildingId==0 且仍在兵团内 GeneralId 不变、State!=Dead)。
- 其余测试零改动。

### 2.4 RegimentTests.cs 追加 2 条(既有 7 条不动)
- `T8_SoldierNoPassive_OnHit`:断峰**士兵**攻击敌方单位 N tick 后,目标无 Slow 状态;坠日士兵攻击镇岳士兵,目标无 Wound/Fragile。
- `T9_SoldierNoShieldRegen`:磐石士兵 Shield 置 10(ShieldMax 假设更高,不足额)、SecondsSinceCombat 置 99,步进 100 tick,断言 Shield 仍为 10(不回盾);对照:磐石**主将**同条件 Shield 回升。
- 风格照既有测试(静态 _cfg / NewSim / SpawnGeneral 取士兵)。

### 3.1 HeroAbilitySystem.cs
```csharp
using System;

namespace IronCurtain.Sim
{
    /// <summary>
    /// 英雄特殊机制(磐石回盾/星落沉默/坠日重伤/断峰迟滞/拒马/禁冲锋区)。
    /// 未定义参数采用决策记录默认值(见 docs 设计变更记录 D11-D16)。
    /// </summary>
    public static class HeroAbilitySystem
    {
        public const float SilenceSeconds = 2f;      // D15:沉默时长
        public const float PanShiRegenPerSec = 20f;  // D12:脱战回盾速率
        public const float PanShiRegenDelay = 5f;    // 脱战 5 秒开始回盾
        public const float PanShiChargeBlockRange = 1;  // D13:磐石禁冲锋范围(格)
        public const float DuanFengChargeBreakRange = 2; // 断峰拒马范围(格)

        /// <summary>命中后施加机制效果(在伤害结算之后调用)。</summary>
        public static void ApplyOnHitEffects(SimCore sim, Entity attacker, Entity defender)
        {
            // 磐石升级:完全免疫减速、沉默、重伤
            if (defender.Hero == HeroId.PanShi && sim.IsUpgraded(defender.Side, HeroId.PanShi))
                return;
            // 钢铁洪流:免疫负面(30s)
            if (HexSystem.DebuffImmune(sim, defender))
                return;

            // 星落:普攻沉默非镇岳单位
            if (attacker.Hero == HeroId.XingLuo && defender.Hero != HeroId.ZhenYue)
                defender.Status.Apply(DebuffType.Silence, SilenceSeconds);

            // 断峰:攻击自带迟滞
            if (attacker.Hero == HeroId.DuanFeng)
                defender.Status.Apply(DebuffType.Slow, 3f);

            // 坠日:攻击镇岳附加重伤
            if (attacker.Hero == HeroId.ZhuiRi && defender.Hero == HeroId.ZhenYue)
                defender.Status.Apply(DebuffType.Wound, 3f);

            // 坠日升级·诛心:攻击镇岳附加脆弱(承伤 +10%)
            if (attacker.Hero == HeroId.ZhuiRi && defender.Hero == HeroId.ZhenYue
                && sim.IsUpgraded(attacker.Side, HeroId.ZhuiRi))
                defender.Status.Apply(DebuffType.Fragile, 3f);
        }

        /// <summary>磐石脱战回盾:脱战满 5 秒后每秒回复 20 点。</summary>
        public static void UpdatePanShiShield(Entity e, float dt)
        {
            if (e.Hero != HeroId.PanShi) return;
            if (e.SecondsSinceCombat < PanShiRegenDelay) return;
            if (e.Shield >= e.ShieldMax) return;
            e.Shield = Math.Min(e.ShieldMax, e.Shield + PanShiRegenPerSec * dt);
        }

        /// <summary>逐风冲锋约束:沉默或附近有磐石/断峰时冲锋被打断。</summary>
        public static bool CanCharge(Entity zf, SimCore sim)
        {
            if (zf.Status.Has(DebuffType.Silence)) return false;
            foreach (var o in sim.Entities.Values)
            {
                if (o.Kind != EntityKind.Unit || o.Side == zf.Side || o.State == EntityState.Dead) continue;
                float d = SimVec2.Distance(zf.PosF, o.PosF);
                if (o.Hero == HeroId.PanShi && d <= PanShiChargeBlockRange) return false;
                if (o.Hero == HeroId.DuanFeng && d <= DuanFengChargeBreakRange) return false;
            }
            return true;
        }
    }
}
```

```csharp
using IronCurtain.Sim;

public class SiegeTests
{
    static GameConfig _cfg;
    SimCore sim;
    int blueTowerId, redTowerId, neutralId;
    int defCol = 200; // 守方兵团生成列(远离战场,登记后由驻军逻辑接管)

    [SetUp]
    public void SetUp()
    {
        if (_cfg == null)
            _cfg = GameConfigParser.FromJson(Resources.Load<TextAsset>("gameconfig").text);
        sim = new SimCore(new SimConfig(), new SeededRng(1));
        sim.Game = _cfg;
        SiegeSystem.Reset();
        blueTowerId = sim.SpawnBuilding(Side.Blue, BuildingId.Tower, new GridPos(15, 114)).Id;
        redTowerId = sim.SpawnBuilding(Side.Red, BuildingId.Tower, new GridPos(15, 130)).Id;
        neutralId = sim.SpawnBuilding(Side.Neutral, BuildingId.ResourcePoint, new GridPos(15, 122)).Id;
        defCol = 200;
    }

    Entity SpawnUnit(Side side, GridPos pos, HeroId? hero = null)
    {
        var u = sim.SpawnEntity(side, EntityKind.Unit, pos);
        if (hero != null)
        {
```

```csharp
            sim.RemoveEntity(probeA.Id);

            var probeB = sim.SpawnEntity(Side.Red, EntityKind.Unit, new GridPos(15, 114));
            probeB.Hero = HeroId.DuanFeng; probeB.Hp = probeB.MaxHp = 100f; probeB.Attack = 0f;
            sim.Step();
            // 把坠日士兵移远到 (15,122):距 (15,114)=8 > 6,不再覆盖;磐石士兵距其 4 > 3 亦不覆盖
            zrSoldier.PosF = SimVec2.FromGrid(new GridPos(15, 122));
            zrSoldier.Pos = new GridPos(15, 122);
            for (int i = 0; i < 35; i++) sim.Step(); // 视野信息 3s(30 tick)保鲜过期
            Assert.IsFalse(VisionSystem.IsUnitVisible(sim, Side.Blue, probeB.Id),
                "坠日士兵移远后,(15,114) 距磐石士兵 4 超出视野 3,应不可见");
            sim.RemoveEntity(probeB.Id);

            // 坠日士兵(远程 6):从 (15,112) 移回
            zrSoldier.PosF = SimVec2.FromGrid(new GridPos(15, 110));
            zrSoldier.Pos = new GridPos(15, 110);
            psSoldier.PosF = SimVec2.FromGrid(new GridPos(15, 108));
            psSoldier.Pos = new GridPos(15, 108); // 磐石士兵移开避免干扰
            for (int i = 0; i < 35; i++) sim.Step(); // 旧视野保鲜(此前坠日士兵在 112/122 处覆盖过远端格)过期

            var probeC = sim.SpawnEntity(Side.Red, EntityKind.Unit, new GridPos(15, 116));
            probeC.Hero = HeroId.DuanFeng; probeC.Hp = probeC.MaxHp = 100f; probeC.Attack = 0f;
            sim.Step();
            // 坠日士兵 (15,110) 视野 6:距 (15,116)=6 ≤ 6 → 可见;距磐石士兵(15,108)=8 不可见
            Assert.IsTrue(VisionSystem.IsUnitVisible(sim, Side.Blue, probeC.Id),
                "坠日士兵视野 6,应揭示 (15,116)");
            sim.RemoveEntity(probeC.Id);

            var probeD = sim.SpawnEntity(Side.Red, EntityKind.Unit, new GridPos(15, 117));
            probeD.Hero = HeroId.DuanFeng; probeD.Hp = probeD.MaxHp = 100f; probeD.Attack = 0f;
            sim.Step();
            // 距坠日士兵 7 > 6;距磐石士兵 9 > 3 → 不可见
            Assert.IsFalse(VisionSystem.IsUnitVisible(sim, Side.Blue, probeD.Id),
                "坠日士兵视野 6,(15,117) 距 7 不应被揭示");
        }
```
