# M8.3b 修复轮(第 2 轮)—— 5 条红收口

## 一、角色与任务
你是《铁幕:前线》Sim 层主程。M8.3a/b 落地后剩 5 条红(根因见 §2)。本轮收口。**允许输出/修改以下 6 个文件,其余零改动:**
- `Assets/Tests/EditMode/GarrisonTests.cs`
- `Assets/Tests/EditMode/SiegeTests.cs`
- `Assets/Tests/EditMode/DestroyTests.cs`
- `Assets/Tests/EditMode/AiTests.cs`
- `Assets/Scripts/Sim/SiegeSystem.cs`(仅当 §2.4 诊断确认是 Sim 缺口时最小修改,否则不动)
- `Assets/Scripts/Sim/GarrisonSystem.cs`(同上原则)

## 二、根因与修复规格
### 2.1 DestroyTests.T6_废墟化_守军后撤(3a 遗留,上轮漏排)
- 根因:守军原用英雄,3a 后仅士兵可驻军,场景失效。
- 修复:守军改为 `RegimentSystem.SpawnGeneral` 兵团的**士兵**(如 5 名士兵正式驻军);拆塔流程照旧;废墟后断言改为士兵口径:驻军列表清空、士兵 `GarrisonBuildingId==0` 且 `TempGarrisonId==0`、士兵 State!=Dead(解除登记回团,不强制溃逃)。测试意图不变(废墟化+守军处置)。
### 2.2 GarrisonTests.临时满员_新来者城门站定
- 根因:只摆了 20 名士兵(一个兵团),容量已改为 30。
- 修复:登记 30 名士兵临时驻军(可 SpawnGeneral 两个兵团取 30 名);第 31 名不登记。容量断言 30。
### 2.3 AiTests.T10_普通主动防守(3a 遗留)
- 根因:普通 AI 防守路径让英雄单位驻军,3a 后被拒导致防守失败。
- 中间态修复(士兵级 AI 是 M8.5):把断言调整为"AI 在威胁出现后调派单位前往被威胁建筑附近(该单位与建筑距离显著缩小/到达城门区域)";若 AI 的驻军调用报错会中断防守动作,则仅让 AI 对驻军失败**容错**(跳过驻军、继续把单位调到建筑旁),Sim 侧最小修改。测试意图=AI 主动防守,驻军深度在 M8.5 接上。
### 2.4 SiegeTests.T1_中立无人占领_直接占领并补配置血量并自动驻军
- 现象:占领成功、补血成功,但攻方士兵未自动正式驻军(Sim SiegeSystem.Capture 第 128 行已有"士兵幸存者自动驻军"分支)。
- 诊断:确认中性点无人占领路径传给 Capture 的 survivors 是否包含攻方士兵;若不含,Sim 最小修复(把参与占领的攻方士兵纳入 survivors);若是测试构造问题(如 AttackAndArrive 未让士兵进入交换/占领循环),改测试。修复后断言"进攻士兵自动正式驻军"成立。
### 2.5 SiegeTests.T4_交换1比1_守军与攻方士兵同归于尽
- 现象:守 1 攻 2 → 预期 1 次交换后守 0、攻剩 1 幸存者自动驻军;实际守军视角出现 1 名红方守军(或幸存者未驻军)。
- 诊断:交换循环多耗一轮/幸存者未入 Capture survivors/自动驻军分支未触发;按 M8.3a 语义(交换 1:1 兵、攻方主将免疫)最小修复 Sim 或测试,最终断言:守军 0、攻方士兵幸存 1、该幸存者正式驻军。

using System.Collections.Generic;
using System.Linq;
using NUnit.Framework;
using IronCurtain.Sim;
using UnityEngine;

public class SiegeTests
{
    static GameConfig _cfg;
    SimCore sim;
    int blueTowerId, redTowerId, neutralId;
    int defCol = 200; // 守方兵团生成列(远离战场,登记后由驻军逻辑接管)

    [SetUp]
    public void SetUp()
    {
        if (_cfg == null)
            _cfg = GameConfigParser.FromJson(Resources.Load<TextAsset>("gameconfig").text);
        sim = new SimCore(new SimConfig(), new SeededRng(1));
        sim.Game = _cfg;
        SiegeSystem.Reset();
        blueTowerId = sim.SpawnBuilding(Side.Blue, BuildingId.Tower, new GridPos(15, 114)).Id;
        redTowerId = sim.SpawnBuilding(Side.Red, BuildingId.Tower, new GridPos(15, 130)).Id;
        neutralId = sim.SpawnBuilding(Side.Neutral, BuildingId.ResourcePoint, new GridPos(15, 122)).Id;
        defCol = 200;
    }

    Entity SpawnUnit(Side side, GridPos pos, HeroId? hero = null)
    {
        var u = sim.SpawnEntity(side, EntityKind.Unit, pos);
        if (hero != null)
        {
            var c = sim.Game.Heroes[(int)hero.Value];
            u.Hero = hero; u.MaxHp = c.MaxHp; u.Hp = c.MaxHp; u.Attack = c.Attack;
            u.AttackIntervalSec = (float)c.AttackIntervalSec; u.Range = c.Range; u.MoveSpeed = (float)c.MoveSpeed;
            if (c.Shield > 0) { u.Shield = c.Shield; u.ShieldMax = c.Shield; }
        }
        else { u.MaxHp = 100; u.Hp = 100; u.MoveSpeed = 5f; }
        return u;
    }

    List<Entity> SoldiersOf(Entity g)
        => sim.Entities.Values
               .Where(x => x.Kind == EntityKind.Unit && x.IsSoldier && x.GeneralId == g.Id)
               .OrderBy(x => x.FormationSlot).ToList();

    /// <summary>守军:生成兵团并直接登记其士兵为正式驻军(驻军士兵不参与兵团跟随/索敌)。</summary>
    List<Entity> Garrison(int buildingId, int count, Side side)
    {
        var g = RegimentSystem.SpawnGeneral(sim, side, HeroId.PanShi, new GridPos(5, defCol));
        defCol += 25;
        var b = sim.Entities[buildingId];
        var defs = SoldiersOf(g).Take(count).ToList();
        foreach (var s in defs) { s.GarrisonBuildingId = buildingId; b.GarrisonFormal.Add(s.Id); }
        return defs;
    }

    bool RunUntil(System.Func<bool> cond, int maxTicks = 3000)
    {
        for (int i = 0; i < maxTicks; i++)
        {

        public static bool EnterFormal(SimCore sim, IReadOnlyList<int> unitIds, int buildingId, out string error)
        {
            error = null;
            if (!sim.Entities.ContainsKey(buildingId)) { error = "建筑不存在"; return false; }
            var b = sim.GetEntity(buildingId);
            var cfg = sim.BuildingConfigOf(b.Building.Value);
            if (cfg.GarrisonFormal <= 0) { error = "该建筑不可驻军"; return false; }
            var accepted = new List<int>();
            foreach (var id in unitIds)
            {
                if (!sim.Entities.ContainsKey(id)) continue;
                var u = sim.GetEntity(id);
                if (!u.IsSoldier) { error = "仅士兵可驻军"; continue; } // 主将永不进城
                if (u.Side != b.Side) { error = "建筑须为己方"; Rollback(sim, accepted, buildingId); return false; }
                if (b.GarrisonFormal.Count >= cfg.GarrisonFormal)
                {   // 驻军全满:路过或城门站定(仅士兵)
                    if (!u.HasMoveTarget && u.GarrisonBuildingId == 0)
                    {
                        var g = FindGateCell(sim, b);
                        if (g != null) sim.SetMoveTarget(id, g.Value);
                    }
                    continue;
                }
                DoEnterFormal(sim, u, buildingId, b.GarrisonFormal);
                accepted.Add(id);
            }
            return true;
        }

        public static bool EnterTemp(SimCore sim, IReadOnlyList<int> unitIds, int buildingId, out string error)
        {
            error = null;

            int cap = b.IsFort ? 3 : (b.Building == null ? 0 : sim.BuildingConfigOf(b.Building.Value).GarrisonTemp); // 堡垒驻军上限 3(D32)

        public static bool Exit(SimCore sim, int buildingId, HeroId hero, out string error) // D24:士兵回兵团;非士兵放城门站定
        {
            error = null;
            if (!sim.Entities.ContainsKey(buildingId)) { error = "建筑不存在"; return false; }
            var b = sim.GetEntity(buildingId);
            int id = b.GarrisonFormal.FirstOrDefault(x => sim.GetEntity(x).Hero == hero);
            bool inFormal = id != 0;
            if (id == 0) id = b.GarrisonTemp.FirstOrDefault(x => sim.GetEntity(x).Hero == hero);
            if (id == 0) { error = "城内无该兵种"; return false; }
            var u = sim.GetEntity(id);
            if (inFormal) b.GarrisonFormal.Remove(id); else b.GarrisonTemp.Remove(id);
            if (u.IsSoldier)
            {   // 士兵出城:回到兵团阵型(跟随主将;主将死→无将续战)
                u.GarrisonBuildingId = 0; u.TempGarrisonId = 0;
                u.State = EntityState.Idle; u.HasMoveTarget = false;
                return true;
            }
            var gate = FindGateCell(sim, b);
            if (gate == null) { error = "城门无空位"; return false; }
            u.Pos = gate.Value; u.PosF = SimVec2.FromGrid(gate.Value); u.GarrisonBuildingId = 0; u.State = EntityState.Idle;
            u.HasMoveTarget = false;
            return true;
        }

        /// <summary>城池易主:士兵只清登记(永不溃逃,留场续战);非士兵(旧数据)向同一线后方最近己方据点溃逃。</summary>
        public static void OnCityCaptured(SimCore sim, Entity b)
        {
            var fleeing = new List<int>(b.GarrisonTemp);
            b.GarrisonTemp.Clear();
            foreach (var id in fleeing)
            {
                var u = sim.GetEntity(id);
                if (u == null) continue;
                if (u.IsSoldier)
                {   // 士兵永不溃逃:清登记留兵团(主将已死→无将续战)
                    u.GarrisonBuildingId = 0; u.TempGarrisonId = 0; u.State = EntityState.Idle;
                    continue;
                }
                var dest = FindRoutDestination(sim, u);
                if (dest == null) { u.GarrisonBuildingId = 0; u.State = EntityState.Idle; continue; }
                var gate = FindGateCell(sim, dest);
                if (gate == null)
                {   // 城门无位则就近找空位
                    for (int d = 1; d <= 8 && gate == null; d++)
                        for (int dr = 0; dr < LaneGrid.Rows && gate == null; dr++)
                        {
                            var p = new GridPos(dr, dest.Pos.Col + (u.Side == Side.Blue ? -d : d));
                            if (IsCellFree(sim, p) && TerrainMap.IsInside(p)) gate = p;
                        }
                }
                u.GarrisonBuildingId = 0;
                // 从城池后方出城,避免被已失守的城堵住退路
                int back = u.Side == Side.Blue ? -1 : 1;
                int behind = back * ((int)System.Math.Ceiling(BuildingFootprint.HalfW(b)) + 1); // 起点须落在失守建筑占地矩形之外
                var start = new GridPos(b.Pos.Row, b.Pos.Col + behind);
                if (!IsCellFree(sim, start))
                {
                    for (int dr = 0; dr < LaneGrid.Rows; dr++)
                    {
                        var alt = new GridPos(dr, b.Pos.Col + behind);
                        if (IsCellFree(sim, alt) && TerrainMap.IsInside(alt)) { start = alt; break; }
                    }
                }
                if (IsCellFree(sim, start)) { u.Pos = start; u.PosF = SimVec2.FromGrid(start); }
                u.State = EntityState.Routing;
                u.HasMoveTarget = true;
                u.MoveTarget = gate ?? dest.Pos;
            }
        }

        /// <summary>建筑化为废墟:士兵只清登记(永不溃逃);非士兵(旧数据)强制后撤(途中遇敌交战 -30%)。</summary>
        public static void OnBuildingRubbled(SimCore sim, Entity b)
        {
            var fleeing = new List<int>(b.GarrisonFormal);
            fleeing.AddRange(b.GarrisonTemp);
            b.GarrisonFormal.Clear();
            b.GarrisonTemp.Clear();
            foreach (var id in fleeing)
            {
                var usedGates = new System.Collections.Generic.HashSet<long>(); // 溃逃单位分派不同城门格,避免堆叠无法抵达
                var u = sim.GetEntity(id);
                if (u == null) continue;
                if (u.IsSoldier)
                {   // 士兵永不溃逃:清登记留兵团
                    u.GarrisonBuildingId = 0; u.TempGarrisonId = 0; u.State = EntityState.Idle;
                    continue;

                var dest = FindRoutDestination(sim, u);
                if (dest == null) { u.GarrisonBuildingId = 0; u.State = EntityState.Idle; continue; }
                var gate = FindGateCell(sim, dest, usedGates);
                if (gate != null)
                    usedGates.Add(((long)gate.Value.Row << 32) | (uint)gate.Value.Col);
                if (gate == null)
                {
                    for (int d = 1; d <= 8 && gate == null; d++)
                        for (int dr = 0; dr < LaneGrid.Rows && gate == null; dr++)

                        }
                }
                u.GarrisonBuildingId = 0;
                int back = u.Side == Side.Blue ? -1 : 1;
                int behind = back * ((int)System.Math.Ceiling(BuildingFootprint.HalfW(b)) + 1); // 起点须落在失守建筑占地矩形之外

                if (!IsCellFree(sim, start))

                    }
                }
                if (IsCellFree(sim, start)) { u.Pos = start; u.PosF = SimVec2.FromGrid(start); }
                u.State = EntityState.Routing;
                u.HasMoveTarget = true;
                u.MoveTarget = gate ?? dest.Pos;
            }
        }

        /// <summary>溃逃目的地:同一线后方(远离敌方向)最近的己方建筑;无则己方大本营。</summary>
        static Entity FindRoutDestination(SimCore sim, Entity u)
        {
            int dir = u.Side == Side.Blue ? -1 : 1; // 远离敌方向
            Entity best = null;
            int bestD = int.MaxValue;
            foreach (var e in sim.Entities.Values)
            {
                if (e.Kind != EntityKind.Building || e.Side != u.Side) continue;
                int d = (e.Pos.Col - u.Pos.Col) * dir;
                if (d <= 0) continue; // 只取后方
                if (d < bestD) { best = e; bestD = d; }
            }
            if (best != null) return best;
            foreach (var e in sim.Entities.Values)
                if (e.Kind == EntityKind.Building && e.Side == u.Side && e.Building == BuildingId.Base)
                    return e;
            return null;
        }

        public static void Tick(SimCore sim, float dt)
        {
            // 0) 城后等候自动解除:主将已死/不存在、主将距城 > 15m、或城不存在 → 离城(士兵跟随主将=截击)
            foreach (var e in sim.Entities.Values.Where(x => x.Kind == EntityKind.Unit && x.TempGarrisonId != 0).ToList())
            {
                var g = e.GeneralId != 0 ? sim.GetEntity(e.GeneralId) : null;
                bool release = g == null || g.Kind != EntityKind.Unit || g.State == EntityState.Dead;
                if (!release)
                {
                    var tb = sim.GetEntity(e.TempGarrisonId);
                    release = tb == null || tb.Kind != EntityKind.Building ||
                              SimVec2.Distance(g.PosF, SimVec2.FromGrid(tb.Pos)) > TempFollowRange;
                }
                if (release)
                {
                    var tb2 = sim.GetEntity(e.TempGarrisonId);
                    if (tb2 != null) tb2.GarrisonTemp.Remove(e.Id);
                    e.TempGarrisonId = 0;
                }
            }

            // 1) 临时驻军:正式有空缺且未封城 → 自动转正式(城守住补入正式)
            foreach (var b in sim.Entities.Values.Where(e => e.Kind == EntityKind.Building).ToList())
            {
                if (b.Sealed) continue; // 封城期间不转
                // 敌方攻城意图存续期间不转:临时驻军受保护,防止"转正后被交换杀死"
                if (b.Intent != SiegeIntent.None && b.IntentIssuer != b.Side) continue;
                int cap = b.IsFort ? 3 : (b.Building == null ? 0 : sim.BuildingConfigOf(b.Building.Value).GarrisonFormal);
                while (b.GarrisonTemp.Count > 0 && b.GarrisonFormal.Count < cap)
                {
                    var id = b.GarrisonTemp[0];
                    b.GarrisonTemp.RemoveAt(0);
                    var u = sim.GetEntity(id);
                    if (u == null) continue;
                    u.TempGarrisonId = 0;
                    DoEnterFormal(sim, u, b.Id, b.GarrisonFormal);
                }
            }

            foreach (var b in sim.Entities.Values.Where(e => e.Kind == EntityKind.Building).ToList())
            {
                var garrison = b.GarrisonFormal.Concat(b.GarrisonTemp).ToList();
                // 2) 依托射击(远程)
                foreach (var id in garrison)
                {
                    var u = sim.GetEntity(id);
                    if (u == null || u.Hero == null || !RangedHeroes.Contains(u.Hero.Value)) continue;
                    u.AttackCooldown -= dt;
                    if (u.AttackCooldown > 0f) continue;
                    var tgt = FindRangedTarget(sim, b, (int)u.Range);
                    if (tgt == null) continue;
                    ApplyDamage(sim, u, tgt);
                    u.AttackCooldown = u.AttackIntervalSec;
                }
                // 3) 近战出击:敌单位欧氏≤2.5 米(建筑中心起算)
                var bc = SimVec2.FromGrid(b.Pos);
                bool threat = sim.Entities.Values.Any(e =>
                    e.Kind == EntityKind.Unit && e.Side != b.Side &&
                    e.GarrisonBuildingId == 0 && e.State != EntityState.Dead && e.State != EntityState.Exchanging &&
                    SimVec2.Distance(e.PosF, bc) <= 2.5f);
                if (threat)
                foreach (var id in garrison)
                {
                    var u = sim.GetEntity(id);
                    if (u == null || u.Hero == null || RangedHeroes.Contains(u.Hero.Value)) continue;
                    var gate = FindGateCell(sim, b);
                    if (gate == null) break;
                    b.GarrisonFormal.Remove(id); b.GarrisonTemp.Remove(id);
                    u.Pos = gate.Value; u.PosF = SimVec2.FromGrid(gate.Value);
                    u.GarrisonBuildingId = 0; u.TempGarrisonId = 0; u.State = EntityState.Idle;
                }
            }

            // 4) 建筑脱战回复(D27):非大本营
            foreach (var b in sim.Entities.Values.Where(e => e.Kind == EntityKind.Building).ToList())
            {
                if (b.Building == BuildingId.Base) continue;
                b.SecondsSinceCombat += dt;
                if (b.SecondsSinceCombat >= RegenDelaySec && b.Hp < b.MaxHp)
                    b.Hp = System.Math.Min(b.MaxHp, b.Hp + RegenPerSec * dt);
            }
        }

        private static void DoEnterFormal(SimCore sim, Entity u, int buildingId, List<int> list)

        private static void DoEnterTemp(Entity u, int buildingId, List<int> list)
        {   // 城后等候:只登记,不冻结移动(士兵保持跟随主将)
            u.TempGarrisonId = buildingId;

        private static void Rollback(SimCore sim, List<int> accepted, int buildingId)
        { foreach (var id in accepted) sim.GetEntity(buildingId).GarrisonFormal.Remove(id); }
        private static void RollbackTemp(SimCore sim, List<int> accepted, int buildingId)
        {
            var b = sim.GetEntity(buildingId);
            foreach (var id in accepted) { b.GarrisonTemp.Remove(id); var u = sim.GetEntity(id); if (u != null) u.TempGarrisonId = 0; }
        }

        /// <summary>城门位置(D18,M8.1b):占地矩形敌侧缘 +0.6 米列起,行优先建筑行(±ceil(HalfH) 内扩散),椭圆内、非水域,列向外延展找最近空位;全占返回 null。</summary>
        private static GridPos? FindGateCell(SimCore sim, Entity b, System.Collections.Generic.HashSet<long> exclude = null)
        {
            int dir = b.Side == Side.Blue ? 1 : -1; // 朝敌侧:蓝方敌人在 +x、红方敌人在 -x(蓝基地列 15、红基地列 271)
            var gateF = SimVec2.FromGrid(b.Pos) + new SimVec2(dir * (BuildingFootprint.HalfW(b) + 0.6f), 0f);
            int gateCol = gateF.ToGrid().Col;
            int rh = (int)System.Math.Ceiling(BuildingFootprint.HalfH(b)); // 建筑占地行范围
            int[] rows = { 0, -1, 1, -2, 2, -3, 3 }; // 行优先建筑行,向外扩散
            for (int d = 0; d <= 4; d++)
            {
                foreach (int r in rows)
                {
                    if (r < -rh || r > rh) continue;
                    var p = new GridPos(b.Pos.Row + r, gateCol + dir * d);
                    if (p.Row < 0 || p.Row >= LaneGrid.Rows || p.Col < 0 || p.Col >= LaneGrid.Cols) continue;
                    if (!TerrainMap.IsInside(p)) continue;
                    if (TerrainMap.IsWater(SimVec2.FromGrid(p))) continue;
                    if (exclude != null && exclude.Contains(((long)p.Row << 32) | (uint)p.Col)) continue; // 溃逃单位分派不同城门格
                    if (IsCellFree(sim, p)) return p;
                }
            }
            return null; // 城门区全占 → 无空位
        }

        private static bool IsCellFree(SimCore sim, GridPos p)
        {
            var c = SimVec2.FromGrid(p);
            foreach (var e in sim.Entities.Values)
            {
                if (e.State == EntityState.Dead) continue;
                if (e.Kind == EntityKind.Building && !e.Rubbled && e.Pos == p) return false;
                if (e.Kind == EntityKind.Unit && e.GarrisonBuildingId == 0
                    && SimVec2.Distance(e.PosF, c) < 1.0f) return false;
            }
            return true;
        }

        private static Entity FindRangedTarget(SimCore sim, Entity b, int range)
        {
            Entity best = null; float bestD = float.MaxValue; int bestId = int.MaxValue;
            var bc = SimVec2.FromGrid(b.Pos);
            foreach (var e in sim.Entities.Values)
            {
                if (e.Kind != EntityKind.Unit || e.Side == b.Side) continue;
                if (e.GarrisonBuildingId != 0 || e.State == EntityState.Dead || e.State == EntityState.Exchanging) continue;
                float d = SimVec2.Distance(e.PosF, bc);
                if (d > range) continue;
                if (d < bestD || (System.Math.Abs(d - bestD) <= 1e-4f && e.Id < bestId)) { best = e; bestD = d; bestId = e.Id; }
            }
            return best;
        }

        private static void ApplyDamage(SimCore sim, Entity attacker, Entity defender)
        {
            float dmg = DamageCalculator.CalculateDamage(attacker, defender, sim.Game, false);
            if (defender.Shield > 0)
            {
                float abs = System.Math.Min(defender.Shield, dmg);
                defender.Shield -= abs; dmg -= abs;
            }
            if (dmg > 0f) defender.Hp -= dmg;
            if (defender.Hp <= 0f && defender.State != EntityState.Dead)
            { defender.State = EntityState.Dead; sim.RemoveEntity(defender.Id); }
        }
    }

    // 侧别辅助
    internal static class SideUtilities
    {
        public static Side SideOf(SimCore sim, IReadOnlyList<int> unitIds) =>
            unitIds.Count > 0 && sim.Entities.ContainsKey(unitIds[0]) ? sim.GetEntity(unitIds[0]).Side : Side.Blue;
    }

using System.Collections.Generic;

namespace IronCurtain.Sim
{
    /// <summary>M3.1 建筑占领核心:意图/入城交换/封城/自动驻军。M8.3a:交换 1:1 只耗兵,主将免疫。M8.1b:FrontCell 占地矩形敌侧缘外 1 格。</summary>
    public static class SiegeSystem
    {
        // 城内交换列表:建筑id → 攻方城内单位id
        static readonly Dictionary<int, List<int>> CityAttackers = new Dictionary<int, List<int>>();

        /// <summary>测试/重开局用:清空内部状态。</summary>
        public static void Reset() => CityAttackers.Clear();

        static bool TryGetBuilding(SimCore sim, int id, out Entity b)
        {
            b = null;
            return sim.Entities.TryGetValue(id, out b) && b.Kind == EntityKind.Building;
        }

        public static SiegeIntent GetIntent(SimCore sim, int buildingId)
        {
            return TryGetBuilding(sim, buildingId, out var b) ? b.Intent : SiegeIntent.None;
        }

        public static bool SetIntent(SimCore sim, int buildingId, Side issuer, SiegeIntent intent, out string error)
        {
            error = null;
            if (!TryGetBuilding(sim, buildingId, out var b)) { error = "目标不是建筑"; return false; }
            if (b.Rubbled) { error = "废墟需先重建"; return false; }
            if (b.Sealed) { error = "封城中不可切换意图"; return false; }
            if (b.Side == Side.Neutral && intent == SiegeIntent.Destroy) { error = "中立建筑不可摧毁"; return false; }
            if (b.Building == BuildingId.Base && intent == SiegeIntent.Capture) { error = "大本营不可占领"; return false; }
            b.Intent = intent;
            b.IntentIssuer = issuer;
            return true;
        }

        static bool Occupied(SimCore sim, GridPos p, int ignoreId)
        {
            var c = SimVec2.FromGrid(p);
            foreach (var e in sim.Entities.Values)
            {
                if (e.Id == ignoreId) continue;
                if (e.GarrisonBuildingId != 0) continue; // 驻军/城内单位不占位
                if (e.Kind == EntityKind.Building && !e.Rubbled && e.Pos == p) return true;
                if (e.Kind == EntityKind.Unit && SimVec2.Distance(e.PosF, c) < 1.0f) return true;
            }
            return false;
        }

        /// <summary>攻城前排(M8.1b):占地矩形敌侧缘外 1 格(ceil(HalfW)+1),行集合 {0,-1,1,-2,2} 相对建筑行;过滤椭圆/水域/其他建筑占地格。</summary>
        static GridPos? FrontCell(SimCore sim, Entity unit, Entity b)
        {
            int dir = unit.PosF.X <= SimVec2.FromGrid(b.Pos).X ? -1 : 1; // 攻方朝建筑方向,只在本侧找位
            int col = b.Pos.Col + dir * ((int)System.Math.Ceiling(BuildingFootprint.HalfW(b)) + 1);
            // 其他建筑占地格集合(同 M8.1a 出生点过滤)
            var blocked = new HashSet<long>();
            foreach (var o in sim.Entities.Values)
                if (o.Kind == EntityKind.Building && !o.Rubbled && o.Id != b.Id)
                    foreach (var c in BuildingFootprint.CellsOccupied(o)) blocked.Add(c.Row * 1000L + c.Col);
            int[] drs = { 0, -1, 1, -2, 2 };
            foreach (int dr in drs)
            {
                var p = new GridPos(b.Pos.Row + dr, col);
                if (p.Row < 0 || p.Row >= LaneGrid.Rows || p.Col < 0 || p.Col >= LaneGrid.Cols) continue;
                if (!TerrainMap.IsInside(p)) continue;
                if (TerrainMap.IsWater(SimVec2.FromGrid(p))) continue;
                if (blocked.Contains(p.Row * 1000L + p.Col)) continue;
                if (!Occupied(sim, p, unit.Id)) return p;
            }
            return null;
        }

        public static bool IssueOrder(SimCore sim, IReadOnlyList<int> unitIds, int buildingId, out string error)
        {
            error = null;
            if (!TryGetBuilding(sim, buildingId, out var b)) { error = "目标不是建筑"; return false; }
            foreach (var id in unitIds)
            {
                if (!sim.Entities.TryGetValue(id, out var u) || u.Kind != EntityKind.Unit || u.Hp <= 0) continue;
                if (u.State == EntityState.Routing) continue; // 溃逃期间不可调遣
                var cell = FrontCell(sim, u, b);
                if (cell == null) continue; // 无空位则跳过该单位
                u.TargetBuildingId = buildingId;
                sim.SetMoveTarget(id, cell.Value);
            }
            return true;
        }

        static List<int> AttackersOf(int buildingId)
        {
            if (!CityAttackers.TryGetValue(buildingId, out var l))
            {
                l = new List<int>();
                CityAttackers[buildingId] = l;
            }
            return l;
        }

        static void EnterCity(SimCore sim, Entity b, Entity u)
        {
            var list = AttackersOf(b.Id);
            if (list.Contains(u.Id)) return; // 防重复入城
            list.Add(u.Id);
            sim.FreeCell(u.Pos);           // 离格:释放格占用,不堵后续部队
            u.State = EntityState.Exchanging;
            b.Sealed = true;
        }

        static void Capture(SimCore sim, Entity b, List<int> survivors)
        {
            b.Side = b.IntentIssuer;
            if (b.Hp <= 0 && b.MaxHp <= 0)
            {
                var cfg = sim.BuildingConfigOf(b.Building.Value);
                b.MaxHp = cfg.MaxHp;
                b.Hp = cfg.MaxHp;
            }
            b.Sealed = false;
            CityAttackers.Remove(b.Id);
            GarrisonSystem.OnCityCaptured(sim, b); // 旧主驻军处置(士兵只清登记不溃逃)
            int cap = b.IsFort ? 3 : (b.Building == null ? 0 : sim.BuildingConfigOf(b.Building.Value).GarrisonFormal);
            foreach (var id in survivors)
            {
                if (!sim.Entities.TryGetValue(id, out var e)) continue;
                e.TargetBuildingId = 0;
                e.State = EntityState.Idle;
                if (e.IsSoldier && b.GarrisonFormal.Count < cap)
                {
                    b.GarrisonFormal.Add(id);
                    e.GarrisonBuildingId = b.Id;
                }
                // 非士兵(主将)不入城,城外待命
            }
        }

        static void KillDefenders(SimCore sim, Entity b, int n)
        {
            // 交换只消耗正式驻军,临时驻军受保护(用户拍板,覆盖 GDD 2.4.1)
            for (int i = 0; i < n && b.GarrisonFormal.Count > 0; i++)
            {
                var id = b.GarrisonFormal[0];
                b.GarrisonFormal.RemoveAt(0);
                sim.RemoveEntity(id);
            }
        }

        /// <summary>攻方交换损失(M8.3a):只从城内士兵中选 1 名阵亡,主将(非士兵)永不因交换阵亡;无士兵在场则攻方无损失。</summary>
        static void KillAttacker(SimCore sim, List<int> atk, int aid)
        {
            var a = sim.GetEntity(aid);
            int victim = 0;
            if (a != null && a.IsSoldier) victim = aid;
            else
            {
                for (int i = 0; i < atk.Count; i++)
                {
                    var s = sim.GetEntity(atk[i]);
                    if (s != null && s.IsSoldier) { victim = atk[i]; atk.RemoveAt(i); break; }
                }
            }
            if (victim != 0) sim.RemoveEntity(victim); // 无士兵在场 → 攻方无损失
        }

        public static void Tick(SimCore sim, float dt)
        {
            // 1) 到达判定
            var arrived = new List<Entity>();
            foreach (var e in sim.Entities.Values)
                if (e.Kind == EntityKind.Unit && e.TargetBuildingId != 0 && !e.HasMoveTarget
                    && e.GarrisonBuildingId == 0 && e.State != EntityState.Exchanging)
                    arrived.Add(e);
            foreach (var u in arrived)
            {
                if (!TryGetBuilding(sim, u.TargetBuildingId, out var b)) { u.TargetBuildingId = 0; continue; }
                if (b.Sealed)
                {
                    if (u.Side == b.IntentIssuer) EnterCity(sim, b, u); // 封城期间攻方增援进城
                    continue;
                }
                if (b.Intent == SiegeIntent.Capture)
                {
                    if (b.GarrisonFormal.Count + b.GarrisonTemp.Count > 0) EnterCity(sim, b, u);
                    else Capture(sim, b, new List<int> { u.Id });
                }
                else if (b.Intent == SiegeIntent.Destroy)
                {
                    u.State = EntityState.Attacking; // 拆塔结算在 M3.4
                }
            }

            // 2) 交换结算:每 10 tick 一轮;1:1 兵/次,主将免疫
            if (sim.Tick % 10 != 0) return;
            var sealedBuildings = new List<Entity>();
            foreach (var e in sim.Entities.Values)
                if (e.Kind == EntityKind.Building && e.Sealed) sealedBuildings.Add(e);
            foreach (var b in sealedBuildings)
            {
                var atk = AttackersOf(b.Id);
                while (atk.Count > 0 && b.GarrisonFormal.Count > 0)
                {
                    var aid = atk[0];
                    atk.RemoveAt(0); // 该单位结束城内待命(无论生死都离开交换队列)
                    KillDefenders(sim, b, 1); // 1:1 兵/次(制作人拍板)
                    KillAttacker(sim, atk, aid); // 攻方只耗士兵,主将免疫
                }
                if (b.GarrisonFormal.Count == 0)
                {
                    // 正式守军被肃清 → 城池易主,临时驻军按新规处置(士兵不溃逃)
                    var survivors = new List<int>(atk);
                    foreach (var id in survivors)
                        if (sim.Entities.TryGetValue(id, out var e2)) e2.State = EntityState.Idle;
                    Capture(sim, b, survivors);
                }
                else if (atk.Count == 0)
                {
                    // 攻方被肃清:解封,意图保留,剩余守军留城
                    b.Sealed = false;
                    CityAttackers.Remove(b.Id);
                }
            }
        }
    }

        public static void Tick(SimCore sim, float dt)
        {
            var cleared = new List<int>();
            foreach (var kv in sim.Entities)
            {
                var e = kv.Value;
                if (e.TargetBuildingId == 0) continue;
                if (e.Kind != EntityKind.Unit) continue;

                // 逐风冲锋首击重置
                if (e.Hero == HeroId.ZhuFeng && !e.IsCharging)
                    e.ChargeHitUsed = false;

                var b = sim.GetEntity(e.TargetBuildingId);
                if (b == null || b.Rubbled || !sim.Entities.ContainsKey(e.TargetBuildingId))
                {
                    e.TargetBuildingId = 0;
                    e.State = EntityState.Idle;
                    continue;
                }

                // 意图校验
                if (b.Intent != SiegeIntent.Destroy || b.IntentIssuer != e.Side)
                    continue;

                // 中立非废墟建筑完全不受拆塔伤害
                if (b.Side == Side.Neutral && !b.Rubbled)
                    continue;

                // 距离校验(米制欧氏):近战需贴到建筑碰撞半径边缘,远程按射程
                float dist = SimVec2.Distance(e.PosF, SimVec2.FromGrid(b.Pos));
                if (DamageCalculator.IsRanged(e) ? dist > e.Range : dist > SimCore.BuildingCollisionRadius(b) + 0.4f)
                    continue;

                // 优先级 D26:射程内存在敌军单位 → 本 tick 不拆塔
                if (EnemyInRange(sim, e)) continue;

                // 攻击循环(冷却共享)
                while (e.AttackCooldown <= 0f)
                {
                    float dmg = (float)Rate(sim, e) * e.AttackIntervalSec;
                    if (e.Hero == HeroId.ZhuFeng && e.IsCharging && !e.ChargeHitUsed)
                    {
                        dmg = (float)sim.Game.Siege.ZhuFengChargeHit;
                        e.ChargeHitUsed = true;
                    }

                    b.Hp -= dmg;
                    e.AttackCooldown += (float)e.AttackIntervalSec;

                    if (b.Hp <= 0f)
                    {
                        if (b.Building == BuildingId.Base)
                        {
                            b.Hp = 0f;
                            sim.SetWinner(e.Side);
                        }
                        else
                        {
                            b.Rubbled = true;
                            b.Side = Side.Neutral;
                            b.Hp = 0f;
                            b.Intent = SiegeIntent.None;
                            b.IntentIssuer = Side.Neutral;
                            b.Sealed = false;
                            GarrisonSystem.OnBuildingRubbled(sim, b);
                            cleared.Add(b.Id);
                        }
                        break;
                    }
                }
            }

            if (cleared.Count > 0)
            {
                foreach (var kv in sim.Entities)
                {
                    var u = kv.Value;
                    if (u.Kind == EntityKind.Unit && u.TargetBuildingId != 0 && cleared.Contains(u.TargetBuildingId))
                    {
                        u.TargetBuildingId = 0;
                        u.State = EntityState.Idle;
                    }
                }
            }
        }

        private static bool EnemyInRange(SimCore sim, Entity e)
        {
            foreach (var kv in sim.Entities)
            {
                var o = kv.Value;
                if (o.Id == e.Id) continue;
                if (o.Kind != EntityKind.Unit) continue;
                if (o.Side == Side.Neutral || o.Side == e.Side) continue;
                if (o.GarrisonBuildingId != 0 || o.State == EntityState.Dead || o.State == EntityState.Exchanging) continue;
                float d = SimVec2.Distance(e.PosF, o.PosF);
                if (DamageCalculator.IsRanged(e) ? d <= e.Range : d <= 1.2f)
                    return true;
            }
            return false;
        }

        private static double Rate(SimCore sim, Entity e)
        {
            // 逐风升级·飞掠:拆塔 7/s
            if (e.Hero == HeroId.ZhuFeng && sim.IsUpgraded(e.Side, HeroId.ZhuFeng))
                return sim.Game.Siege.FlyoverPerSec;
            var s = sim.Game.Siege;
            switch (e.Hero)
            {
                case HeroId.ZhenYue: return s.ZhenYuePerSec;
                case HeroId.ZhuiRi: return s.RangedPerSec;
                case HeroId.XingLuo: return s.RangedPerSec;
                case HeroId.ZhuFeng: return s.ZhuFengPerSec;
                default: return s.MeleePerSec; // 磐石/断峰 及 Hero==null 单位
            }
        }

        /// <summary>对建筑结算伤害并走完整废墟/判负流程(攻城锥等直伤共用入口)。</summary>
        public static void ApplyBuildingDamage(SimCore sim, Entity b, float amount, Side attackerSide)
        {
            b.Hp -= amount;
            if (b.Hp > 0f) return;
            if (b.Building == BuildingId.Base)
            {
                b.Hp = 0f;
                sim.SetWinner(attackerSide);
                return;
            }
            b.Rubbled = true;
            b.Side = Side.Neutral;
            b.Hp = 0f;
            b.Intent = SiegeIntent.None;
            b.IntentIssuer = Side.Neutral;
            b.Sealed = false;
            GarrisonSystem.OnBuildingRubbled(sim, b);
            foreach (var e in sim.Entities.Values)
                if (e.TargetBuildingId == b.Id) { e.TargetBuildingId = 0; e.State = EntityState.Idle; }
        }

        public static bool Rebuild(SimCore sim, int buildingId, Side side, out string error)
        {
            error = null;
            if (!sim.Entities.TryGetValue(buildingId, out var b) || b.Kind != EntityKind.Building)

            if (!b.Rubbled)

                return false;

            if (!sim.SpendEp(side, (float)sim.Game.Ep.RebuildCost))

                error = "EP 不足";
                return false;

            b.Rubbled = false;
            b.Side = side;
            b.Hp = (float)sim.BuildingConfigOf(b.Building.Value).MaxHp;
            return true;
        }

using System;
using System.Collections.Generic;
using System.Linq;

namespace IronCurtain.Sim
{
    public enum AiTier { Simple, Normal }

    public static class AiSystem
    {
        const int DecisionInterval = 20;

        class AiState
        {
            public Side Side;
            public AiTier Tier;
            public int ProducedSinceZhenYue; // 已产 1 人口英雄计数
            public bool ZhenYueQueued;       // 下一只为镇岳
            public bool WaveActive;
            public int WaveStartCount;
            public List<int> WaveUnits = new List<int>();
        }

        static readonly Dictionary<Side, AiState> _ais = new Dictionary<Side, AiState>();

        public static void Configure(SimCore sim, Side side, AiTier tier)
        {
            _ais[side] = new AiState { Side = side, Tier = tier };
        }

        public static void Reset() => _ais.Clear();

        public static void Tick(SimCore sim, float dt)
        {
            if (sim.Tick % DecisionInterval != 0) return;
            foreach (var ai in _ais.Values.ToList())
            {
                if (ai.Tier == AiTier.Simple) TickSimple(sim, ai);
                else TickNormal(sim, ai);
            }
        }

        // ---------- 简单 AI ----------
        static void TickSimple(SimCore sim, AiState ai)
        {
            FillProduction(sim, ai, simple: true);
            PassiveDefend(sim, ai);
        }

        static void PassiveDefend(SimCore sim, AiState ai)
        {
            var myBuildings = sim.Entities.Values
                .Where(e => e.Side == ai.Side && e.Kind == EntityKind.Building && !e.Rubbled)
                .ToList();
            foreach (var b in myBuildings)
            {
                var threat = sim.Entities.Values
                    .Where(e => e.Kind == EntityKind.Unit && e.Side != ai.Side && e.State != EntityState.Dead
                                && VisionSystem.IsUnitVisible(sim, ai.Side, e.Id)
                                && Dist(sim, e, b) <= 8f)
                    .FirstOrDefault();
                if (threat == null) continue;
                var idle = sim.Entities.Values
                    .Where(e => e.Kind == EntityKind.Unit && e.Side == ai.Side && !e.IsSoldier
                                && e.State == EntityState.Idle && e.GarrisonBuildingId == 0)
                    .OrderBy(e => Dist(sim, e, b))
                    .Take(5).ToList();
                foreach (var u in idle)
                {
                    var t = FrontCell(sim, b);
                    if (t.HasValue) sim.SetMoveTarget(u.Id, t.Value);
                }
            }
        }

        // ---------- 普通 AI ----------
        static void TickNormal(SimCore sim, AiState ai)
        {
            // 1. 补产(配比)
            FillProduction(sim, ai, simple: false);
            // 2. 升级
            if (sim.EpOf(ai.Side) >= 2 && !sim.IsUpgraded(ai.Side, HeroId.ZhenYue))
                sim.BuyUpgrade(ai.Side, HeroId.ZhenYue, out _);
            // 3. 规律抽卡
            if (sim.Tick >= 1800 && HexSystem.CanDraw(sim, ai.Side))
                TryDrawHex(sim, ai);
            // 5. 受挫撤退(先于新波次判断)
            if (ai.WaveActive) CheckRetreat(sim, ai);
            // 4. 集结进攻
            if (!ai.WaveActive) TryLaunchWave(sim, ai);
            // 6. 主动防守
            ActiveDefend(sim, ai);
        }

        static void TryDrawHex(SimCore sim, AiState ai)
        {
            var prio = new[] { HexId.SteelTide, HexId.RapidDeploy, HexId.HoldFast, HexId.Vengeance,
                               HexId.LightMobility, HexId.IronWall, HexId.HiddenPath };
            var choices = HexSystem.DrawChoices(sim, ai.Side);
            foreach (var c in prio)
            {
                if (c == HexId.Sandbag) continue;
                if (choices.Contains(c) && HexSystem.UseHex(sim, ai.Side, c, 0, out _)) return;
            }
        }

        static void CheckRetreat(SimCore sim, AiState ai)
        {
            ai.WaveUnits = ai.WaveUnits
                .Where(id => sim.Entities.TryGetValue(id, out var e) && e.State != EntityState.Dead && !e.IsSoldier)
                .ToList();
            if (ai.WaveUnits.Count < ai.WaveStartCount * 0.4f)
            {
                var home = sim.Entities.Values
                    .FirstOrDefault(e => e.Side == ai.Side && e.Kind == EntityKind.Building && e.Building == BuildingId.Base);
                var front = home != null ? FrontCell(sim, home) : (GridPos?)null;
                foreach (var id in ai.WaveUnits)
                    if (front.HasValue) sim.SetMoveTarget(id, front.Value);
                ai.WaveActive = false;
                ai.WaveUnits.Clear();
            }
        }

        static void TryLaunchWave(SimCore sim, AiState ai)
        {
            var field = sim.Entities.Values
                .Where(e => e.Kind == EntityKind.Unit && e.Side == ai.Side && !e.IsSoldier
                            && e.State != EntityState.Dead && e.GarrisonBuildingId == 0
                            && e.State != EntityState.Exchanging)
                .ToList();
            if (field.Count < 8) return;

            var home = Home(sim, ai.Side);
            // 目标选择
            var neutral = sim.Entities.Values
                .Where(e => e.Kind == EntityKind.Building && e.Building == BuildingId.ResourcePoint
                            && e.Side == Side.Neutral && !e.Rubbled
                            && VisionSystem.IsBuildingKnown(sim, ai.Side, e.Id))
                .OrderBy(e => home != null ? Dist(sim, e, home) : 0f)
                .FirstOrDefault();
            int targetId; SiegeIntent intent;
            if (neutral != null) { targetId = neutral.Id; intent = SiegeIntent.Capture; }
            else
            {
                var enemyBuildings = sim.Entities.Values
                    .Where(e => e.Kind == EntityKind.Building && e.Side != ai.Side && e.Side != Side.Neutral
                                && !e.Rubbled && e.Building != BuildingId.Base
                                && VisionSystem.IsBuildingKnown(sim, ai.Side, e.Id))
                    .Select(e => new { E = e, G = VisionSystem.LastKnownGarrison(sim, ai.Side, e.Id) ?? 999 })
                    .OrderBy(x => x.G).ThenBy(x => home != null ? Dist(sim, x.E, home) : 0f)
                    .FirstOrDefault();
                if (enemyBuildings == null) return;
                targetId = enemyBuildings.E.Id;
                intent = enemyBuildings.E.Building == BuildingId.Tower ? SiegeIntent.Destroy : SiegeIntent.Capture;
            }

            var targetB = sim.GetEntity(targetId);
            var wave = field.OrderBy(e => targetB != null ? Dist(sim, e, targetB) : 0f).Take(10).Select(e => e.Id).ToList();
            if (SiegeSystem.SetIntent(sim, targetId, ai.Side, intent, out _))
            {
                SiegeSystem.IssueOrder(sim, wave, targetId, out _);
                ai.WaveActive = true;
                ai.WaveUnits = wave;
                ai.WaveStartCount = wave.Count;
            }
        }

        static void ActiveDefend(SimCore sim, AiState ai)
        {
            var visibleEnemies = sim.Entities.Values
                .Where(e => e.Kind == EntityKind.Unit && e.Side != ai.Side && e.State != EntityState.Dead
                            && VisionSystem.IsUnitVisible(sim, ai.Side, e.Id))
                .ToList();
            if (visibleEnemies.Count < 6) return;
            foreach (var b in sim.Entities.Values.Where(e => e.Side == ai.Side && e.Kind == EntityKind.Building && !e.Rubbled))
            {
                var near = visibleEnemies.Where(e => Dist(sim, e, b) <= 10f).ToList();
                if (near.Count < 6) continue;
                var idle = sim.Entities.Values
                    .Where(e => e.Kind == EntityKind.Unit && e.Side == ai.Side && !e.IsSoldier
                                && e.State == EntityState.Idle && e.GarrisonBuildingId == 0)
                    .OrderBy(e => Dist(sim, e, b))
                    .Take(6).ToList();
                if (b.Building != null && b.Building != BuildingId.Base && b.GarrisonFormal.Count < 5 && idle.Count > 0)
                    GarrisonSystem.EnterFormal(sim, idle.Select(e => e.Id).ToList(), b.Id, out _);
                else
                    foreach (var u in idle)
                    {
                        var t = FrontCell(sim, b);
                        if (t.HasValue) sim.SetMoveTarget(u.Id, t.Value);
                    }
                return;
            }
        }

        // ---------- 公共 ----------
        static void FillProduction(SimCore sim, AiState ai, bool simple)
        {
            foreach (var b in sim.Entities.Values)
            {
                if (b.Kind != EntityKind.Building || b.Building != BuildingId.Barracks || b.Prod == null) continue;
                if (b.Side != ai.Side || b.Rubbled) continue;
                for (int s = 0; s < b.Prod.Slots.Length; s++)
                {
                    var slot = b.Prod.Slots[s];
                    if (slot.Current != null || slot.Pending.Count > 0) continue;
                    HeroId hero;
                    if (simple) hero = HeroId.PanShi;
                    else if (ai.ZhenYueQueued) hero = HeroId.ZhenYue;
                    else hero = ai.ProducedSinceZhenYue % 2 == 0 ? HeroId.PanShi : HeroId.DuanFeng;
                    if (hero == HeroId.ZhenYue && sim.PopulationOf(ai.Side) + 3 > sim.PopulationCapOf(ai.Side))
                        return; // 人口不足镇岳,等待
                    ProductionSystem.Enqueue(sim, b.Id, hero, out _);
                    if (simple) continue;
                    if (hero == HeroId.ZhenYue) { ai.ZhenYueQueued = false; ai.ProducedSinceZhenYue = 0; }
                    else
                    {
                        ai.ProducedSinceZhenYue++;
                        if (ai.ProducedSinceZhenYue >= 3) ai.ZhenYueQueued = true;
                    }
                }
            }
        }

        static Entity Home(SimCore sim, Side side)
        {
            return sim.Entities.Values.FirstOrDefault(e => e.Side == side && e.Kind == EntityKind.Building && e.Building == BuildingId.Base);
        }

        /// <summary>城门站位格:建筑朝敌侧(碰撞半径+0.6 米),行序 15 优先、椭圆内、非水域。</summary>
        static GridPos? FrontCell(SimCore sim, Entity b)
        {
            int dx = b.Side == Side.Blue ? 1 : -1;
            float offset = SimCore.BuildingCollisionRadius(b) + 0.6f;
            var gate = (SimVec2.FromGrid(b.Pos) + new SimVec2(dx * offset, 0f)).ToGrid();
            foreach (int r in new[] { 0, -1, 1, -2, 2, -3, 3 })
            {
                var p = new GridPos(gate.Row + r, gate.Col);
                if (p.Col < 0 || p.Col >= LaneGrid.Cols || p.Row < 0 || p.Row >= LaneGrid.Rows) continue;
                if (!TerrainMap.IsInside(p)) continue;
                if (TerrainMap.IsWater(SimVec2.FromGrid(p))) continue;
                return p;
            }
            return null;
        }

        /// <summary>实体间距离(米制欧氏):单位取 PosF,建筑取格中心。</summary>
        static float Dist(SimCore sim, Entity a, Entity b)
        {
            SimVec2 pa = a.Kind == EntityKind.Unit ? a.PosF : SimVec2.FromGrid(a.Pos);
            SimVec2 pb = b.Kind == EntityKind.Unit ? b.PosF : SimVec2.FromGrid(b.Pos);
            return SimVec2.Distance(pa, pb);
        }
    }

using System.Collections.Generic;

namespace IronCurtain.Sim
{
    /// <summary>
    /// 兵团底座(M8.2a 第一批):1 主将 + 20 兵,兵随将走(4×5 阵型跟随);
    /// 主将死亡士兵永不溃逃、无将状态就近逐杀至死。纯 tick,无 UnityEngine 依赖。
    /// </summary>
    public static class RegimentSystem
    {
        public const int SoldiersPerGeneral = 20;
        public const int FormationRows = 4;   // 4 行 × 5 列,行列距 1.0 米
        public const int FormationCols = 5;

        public static SimVec2 FacingOf(Entity g)
        {
            var def = g.Side == Side.Blue ? new SimVec2(1f, 0f) : new SimVec2(-1f, 0f);
            if (g.HasMoveTarget)
            {
                var d = (SimVec2.FromGrid(g.MoveTarget) - g.PosF).Normalized();
                if (d.SqrMagnitude > 1e-8f) return d;
            }
            return def;
        }

        public static SimVec2 SlotOffset(Entity g, int slot)
        {
            var facing = FacingOf(g);
            var perp = new SimVec2(facing.Y, -facing.X);
            int row = slot / FormationCols;
            int col = slot % FormationCols;
            return facing * -(row + 1) + perp * (col - 2);
        }

        public static void Tick(SimCore sim, float dt)
        {
            // 快照遍历:士兵死亡结算会移除实体,禁止边枚举边修改
            var snapshot = new List<Entity>(sim.Entities.Values);
            foreach (var e in snapshot)
            {
                if (e.Kind != EntityKind.Unit || !e.IsSoldier || e.State == EntityState.Dead) continue;
                if (e.GarrisonBuildingId != 0) continue;

                var g = e.GeneralId != 0 ? sim.GetEntity(e.GeneralId) : null;
                if (g != null && g.Kind == EntityKind.Unit && g.State != EntityState.Dead)
                {
                    // a) 主将存活:阵型跟随(士兵不写 HasMoveTarget/State)
                    var target = g.PosF + SlotOffset(g, e.FormationSlot);
                    float dist = SimVec2.Distance(e.PosF, target);
                    if (dist > 0.2f) StepMove(e, target, dist, dt);
                }
                else
                {
                    // b) 无将续战:就近索敌逐杀至死(永不溃逃/撤退/响应指令)
                    var enemy = NearestEnemy(sim, e);
                    if (enemy == null) continue;
                    float dist = SimVec2.Distance(e.PosF, enemy.PosF);
                    if (dist <= e.Range + 0.3f) continue; // 交战交给 CombatUpdate
                    StepMove(e, enemy.PosF, dist, dt);
                }
            }
        }

        private static void StepMove(Entity e, SimVec2 target, float dist, float dt)
        {
            float spd = e.MoveSpeed * TerrainMap.MoveFactor(e.PosF);
            float step = System.Math.Min(spd * dt, dist);
            e.PosF = TerrainMap.ClampToEllipse(e.PosF + (target - e.PosF).Normalized() * step);
            e.Pos = e.PosF.ToGrid();
        }

        /// <summary>最近敌索敌:与 SimCore.FindNearestEnemy 同一欧氏语义(单位、非己方、非死亡、非城内/交换)。</summary>
        private static Entity NearestEnemy(SimCore sim, Entity e)
        {
            Entity best = null;
            float bestDist = float.MaxValue;
            foreach (var o in sim.Entities.Values)
            {
                if (o.Kind != EntityKind.Unit || o.Side == e.Side || o.State == EntityState.Dead) continue;
                if (o.GarrisonBuildingId != 0 || o.State == EntityState.Exchanging) continue;
                float d = SimVec2.Distance(e.PosF, o.PosF);
                if (d < bestDist) { best = o; bestDist = d; }
            }
            return best;
        }

        /// <summary>生成主将 + 20 名士兵的完整兵团,返回主将实体。</summary>
        public static Entity SpawnGeneral(SimCore sim, Side side, HeroId hero, GridPos pos)
        {
            var c = sim.Game.Heroes[(int)hero];
            var anchor = SimVec2.FromGrid(pos);
            // 朝向探针:仅用于出生时计算槽位偏移(默认朝向,与主将无移动目标时一致)
            var probe = new Entity(0, side, EntityKind.Unit, pos, 0f);

            // 先生成士兵(每生成一名立即摆到槽位,避免同点出生距离校验冲突),主将最后落位
            var soldiers = new List<Entity>(SoldiersPerGeneral);
            for (int slot = 0; slot < SoldiersPerGeneral; slot++)
            {
                var s = sim.SpawnEntity(side, EntityKind.Unit, pos);
                s.IsSoldier = true;
                s.FormationSlot = slot;
                s.Hero = hero; // 供类型与渲染(伤害英雄规则 M8.2b 收口,本轮已知)
                s.MaxHp = c.MaxHp / 10f;
                s.Hp = s.MaxHp;
                s.Attack = c.Attack / 10f;
                s.AttackIntervalSec = (float)c.AttackIntervalSec;
                s.MoveSpeed = (float)c.MoveSpeed;
                s.Range = c.Range;
                s.Shield = c.Shield / 10f;
                s.ShieldMax = s.Shield;
                s.PosF = anchor + SlotOffset(probe, slot);
                s.Pos = s.PosF.ToGrid();
                soldiers.Add(s);
            }

            var g = sim.SpawnEntity(side, EntityKind.Unit, pos);
            g.Hero = hero;
            g.MaxHp = c.MaxHp;
            g.Hp = c.MaxHp;
            g.Attack = c.Attack;
            g.AttackIntervalSec = (float)c.AttackIntervalSec;
            g.Range = c.Range;
            g.MoveSpeed = (float)c.MoveSpeed;
            if (c.Shield > 0) { g.Shield = c.Shield; g.ShieldMax = c.Shield; }

            foreach (var s in soldiers) s.GeneralId = g.Id;
            return g;
        }
    }
