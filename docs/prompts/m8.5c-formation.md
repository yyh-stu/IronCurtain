# M8.5c 自动布阵规划器(Sim 纯逻辑,View 的 T 键调用)

## 一、角色与任务
制作人拍板(变更#12b):**玩家框选多兵团→按 T 自动布阵**。布阵单元=**兵团原子**(将排阵、兵随将走)。规则:重装(磐石/镇岳)在前、战士(断峰/逐风)居中、射手(坠日/星落)在后;**朝向**:①框选内部队视野内有敌→朝敌(敌质心方向);②无敌→朝敌方大本营方向(蓝 +x/红 -x),或前进方向(仅当前进朝敌);③后撤且无敌→保持朝敌不转身。**多向受敌(敌人方位分布 >120°)→射手居中,近战在扇形外侧屏风**。本轮只做 Sim 规划器(View 接线在 M8.6)。

```csharp
public static SimVec2 FacingOf(Entity g)
{
    if (g.FacingOverride.HasValue) return g.FacingOverride.Value;
    SimVec2 enemyDir = g.Side == Side.Blue ? new SimVec2(1f, 0f) : new SimVec2(-1f, 0f);
    if (g.HasMoveTarget)
    {
        var dir = (SimVec2.FromGrid(g.MoveTarget) - g.PosF).Normalized();
        bool toward = (g.Side == Side.Blue && dir.X > 0.05f) || (g.Side == Side.Red && dir.X < -0.05f);
        if (toward) return dir;            // 前进方向(朝敌)→用之(斜向前进阵型随之斜)
        return enemyDir;                   // 后撤/横移 → 保持面朝敌不转身(规则③)
    }
    return enemyDir;
}
```

### 3.2 RegimentSystem.cs
```csharp
using System.Collections.Generic;

namespace IronCurtain.Sim
{
    /// <summary>
    /// 兵团底座(M8.2a 第一批):1 主将 + 20 兵,兵随将走(4×5 阵型跟随);
    /// 主将死亡士兵永不溃逃、无将状态就近逐杀至死。纯 tick,无 UnityEngine 依赖。
    /// </summary>
    public static class RegimentSystem
    {
        public const int SoldiersPerGeneral = 20;
        public const int FormationRows = 4;   // 4 行 × 5 列,行列距 1.0 米
        public const int FormationCols = 5;

        public static SimVec2 FacingOf(Entity g)
        {
            var def = g.Side == Side.Blue ? new SimVec2(1f, 0f) : new SimVec2(-1f, 0f);
            if (g.HasMoveTarget)
            {
                var d = (SimVec2.FromGrid(g.MoveTarget) - g.PosF).Normalized();
                if (d.SqrMagnitude > 1e-8f) return d;
            }
            return def;
        }

        public static SimVec2 SlotOffset(Entity g, int slot)
        {
            var facing = FacingOf(g);
            var perp = new SimVec2(facing.Y, -facing.X);
            int row = slot / FormationCols;
            int col = slot % FormationCols;
            return facing * -(row + 1) + perp * (col - 2);
        }

        public static void Tick(SimCore sim, float dt)
        {
            // 快照遍历:士兵死亡结算会移除实体,禁止边枚举边修改
            var snapshot = new List<Entity>(sim.Entities.Values);
            foreach (var e in snapshot)
            {
                if (e.Kind != EntityKind.Unit || !e.IsSoldier || e.State == EntityState.Dead) continue;
                if (e.GarrisonBuildingId != 0) continue;

                var g = e.GeneralId != 0 ? sim.GetEntity(e.GeneralId) : null;
                if (g != null && g.Kind == EntityKind.Unit && g.State != EntityState.Dead)
                {
                    // a) 主将存活:阵型跟随(士兵不写 HasMoveTarget/State)
                    var target = g.PosF + SlotOffset(g, e.FormationSlot);
                    float dist = SimVec2.Distance(e.PosF, target);
                    if (dist > 0.2f) StepMove(e, target, dist, dt);
                }
                else
                {
                    // b) 无将续战:就近索敌逐杀至死(永不溃逃/撤退/响应指令)
                    var enemy = NearestEnemy(sim, e);
                    if (enemy == null) continue;
                    float dist = SimVec2.Distance(e.PosF, enemy.PosF);
                    if (dist <= e.Range + 0.3f) continue; // 交战交给 CombatUpdate
                    StepMove(e, enemy.PosF, dist, dt);
                }
            }
        }

        private static void StepMove(Entity e, SimVec2 target, float dist, float dt)
        {
            float spd = e.MoveSpeed * TerrainMap.MoveFactor(e.PosF);
            float step = System.Math.Min(spd * dt, dist);
            e.PosF = TerrainMap.ClampToEllipse(e.PosF + (target - e.PosF).Normalized() * step);
            e.Pos = e.PosF.ToGrid();
        }

        /// <summary>最近敌索敌:与 SimCore.FindNearestEnemy 同一欧氏语义(单位、非己方、非死亡、非城内/交换)。</summary>
        private static Entity NearestEnemy(SimCore sim, Entity e)
        {
            Entity best = null;
            float bestDist = float.MaxValue;
            foreach (var o in sim.Entities.Values)
            {
                if (o.Kind != EntityKind.Unit || o.Side == e.Side || o.State == EntityState.Dead) continue;
                if (o.GarrisonBuildingId != 0 || o.State == EntityState.Exchanging) continue;
                float d = SimVec2.Distance(e.PosF, o.PosF);
                if (d < bestDist) { best = o; bestDist = d; }
            }
            return best;
        }

        /// <summary>生成一名士兵(属性=主将/10,攻速/移速/射程=主将,变装=主将 Hero),落位槽位偏移点。SpawnGeneral 与补兵共用。</summary>
        public static Entity SpawnSoldier(SimCore sim, Entity general, int slot, SimVec2? atPos = null)
        {
            var target = atPos ?? general.PosF + SlotOffset(general, slot);
            var s = sim.SpawnEntity(general.Side, EntityKind.Unit, target.ToGrid());
            s.IsSoldier = true;
            s.FormationSlot = slot;
            s.Hero = general.Hero;
            s.MaxHp = general.MaxHp / 10f;
            s.Hp = s.MaxHp;
            s.Attack = general.Attack / 10f;
            s.AttackIntervalSec = general.AttackIntervalSec;
            s.MoveSpeed = general.MoveSpeed;
            s.Range = general.Range;
            s.Shield = general.Shield / 10f;
            s.ShieldMax = s.Shield;
            s.GeneralId = general.Id;
            s.PosF = target;
            s.Pos = target.ToGrid();
            return s;
        }

        /// <summary>生成主将 + 20 名士兵的完整兵团,返回主将实体。</summary>
        public static Entity SpawnGeneral(SimCore sim, Side side, HeroId hero, GridPos pos)
        {
            var c = sim.Game.Heroes[(int)hero];

            var g = sim.SpawnEntity(side, EntityKind.Unit, pos);
            g.Hero = hero;
            g.MaxHp = c.MaxHp;
            g.Hp = c.MaxHp;
            g.Attack = c.Attack;
            g.AttackIntervalSec = (float)c.AttackIntervalSec;
            g.Range = c.Range;
            g.MoveSpeed = (float)c.MoveSpeed;
            if (c.Shield > 0) { g.Shield = c.Shield; g.ShieldMax = c.Shield; }

            for (int slot = 0; slot < SoldiersPerGeneral; slot++)
                SpawnSoldier(sim, g, slot);
            return g;
        }
    }
}
```

```csharp
using System;

namespace IronCurtain.Sim
{
    /// <summary>连续坐标(米制)。1 格 = 1 米;X=列方向,Y=行方向。</summary>
    public struct SimVec2
    {
        public float X;
        public float Y;

        public SimVec2(float x, float y)
        {
            X = x;
            Y = y;
        }

        public static SimVec2 FromGrid(GridPos p) => new SimVec2(p.Col + 0.5f, p.Row + 0.5f);

        public static SimVec2 operator +(SimVec2 a, SimVec2 b) => new SimVec2(a.X + b.X, a.Y + b.Y);
        public static SimVec2 operator -(SimVec2 a, SimVec2 b) => new SimVec2(a.X - b.X, a.Y - b.Y);
        public static SimVec2 operator *(SimVec2 a, float s) => new SimVec2(a.X * s, a.Y * s);

        public float SqrMagnitude => X * X + Y * Y;
        public float Magnitude => (float)Math.Sqrt(X * X + Y * Y);

        public SimVec2 Normalized()
        {
            float m = Magnitude;
            return m > 1e-6f ? new SimVec2(X / m, Y / m) : new SimVec2(0f, 0f);
        }

        public static float Distance(SimVec2 a, SimVec2 b)
        {
            float dx = a.X - b.X, dy = a.Y - b.Y;
            return (float)Math.Sqrt(dx * dx + dy * dy);
        }

        public static float SqrDistance(SimVec2 a, SimVec2 b)
        {
            float dx = a.X - b.X, dy = a.Y - b.Y;
            return dx * dx + dy * dy;
        }

        public static SimVec2 Lerp(SimVec2 a, SimVec2 b, float t) =>
            new SimVec2(a.X + (b.X - a.X) * t, a.Y + (b.Y - a.Y) * t);

        /// <summary>连续坐标 → 所在格(向下取整)。</summary>
        public GridPos ToGrid()
        {
            int col = X < 0 ? 0 : (int)X;
            int row = Y < 0 ? 0 : (int)Y;
            if (col > LaneGrid.Cols - 1) col = LaneGrid.Cols - 1;
            if (row > LaneGrid.Rows - 1) row = LaneGrid.Rows - 1;
            return new GridPos(row, col);
        }

        public override string ToString() => $"({X:F2},{Y:F2})";
    }
}
```

```csharp
        public override string ToString() => $"({Row},{Col})";
```

```csharp
using System;

namespace IronCurtain.Sim
{
    /// <summary>椭圆大地图地形规则(权威规格,详见 M6.5)。</summary>
    public static class TerrainMap
    {
        public const int Rows = 30;
        public const int Cols = 286;

        // 椭圆参数
        public const float CenterX = 143f;
        public const float CenterY = 15f;
        public const float HalfX = 143f;
        public const float HalfY = 15f;

        // 河(楚河汉界)
        public const float RiverMinX = 138f;
        public const float RiverMaxX = 148f;
        // 桥
        public const float BridgeTopMinY = 25f;
        public const float BridgeBotMaxY = 5f;
        // 中央岛
        public const float IsleMinX = 139f;
        public const float IsleMaxX = 147f;
        public const float IsleMinY = 11f;
        public const float IsleMaxY = 19f;

        public static bool IsInside(SimVec2 v)
        {
            float dx = (v.X - CenterX) / HalfX;
            float dy = (v.Y - CenterY) / HalfY;
            return dx * dx + dy * dy <= 1f;
        }

        public static bool IsInside(GridPos p)
        {
            return IsInside(SimVec2.FromGrid(p));
        }

        public static bool IsBridge(SimVec2 v)
        {
            if (v.X < RiverMinX || v.X >= RiverMaxX) return false;
            return v.Y >= BridgeTopMinY || v.Y < BridgeBotMaxY;
        }

        public static bool IsIsland(SimVec2 v)
        {
            return v.X >= IsleMinX && v.X < IsleMaxX && v.Y >= IsleMinY && v.Y < IsleMaxY;
        }

        public static bool IsWater(SimVec2 v)
        {
            if (v.X < RiverMinX || v.X >= RiverMaxX) return false;
            return !IsBridge(v) && !IsIsland(v);
        }

        public static float MoveFactor(SimVec2 v)
        {
            return IsWater(v) ? 0.5f : 1f;
        }

        /// <summary>径向投影:把 v 推回椭圆(含边界)。</summary>
        public static SimVec2 ClampToEllipse(SimVec2 v)
        {
            if (IsInside(v)) return v;
            float dx = v.X - CenterX;
            float dy = v.Y - CenterY;
            // 防退化:恰在中心时向右推
            if (dx == 0f && dy == 0f) return new SimVec2(CenterX + HalfX, CenterY);
            double a = HalfX, b = HalfY;
            // 参数角:取点方向,然后牛顿迭代求边界到方向射线的交点
            double theta = Math.Atan2(dy / b, dx / a);
            double ct = Math.Cos(theta), st = Math.Sin(theta);
            // 边界点: x = a*ct, y = b*st (椭圆参数化)
            float bx = (float)(CenterX + a * ct);
            float by = (float)(CenterY + b * st);
            // 参数化点保证在椭圆上;再向内收一点点避免浮点误差
            var r = new SimVec2(bx, by);
            if (!IsInside(r))
                r = new SimVec2((bx - CenterX) * 0.999f + CenterX, (by - CenterY) * 0.999f + CenterY);
            return r;
        }
    }
```

```csharp
        private void MoveUnits()
        {

            {

                if (e.GarrisonBuildingId != 0) continue; // 驻军单位不占线移动

                SimVec2 target = SimVec2.FromGrid(e.MoveTarget);
                SimVec2 delta = target - e.PosF;
                float dist = delta.Magnitude;
                if (dist <= 0.15f)
                {
                    e.HasMoveTarget = false;
                    e.State = EntityState.Idle;
                    e.MoveAccum = 0f;
                    continue;
                }

                float spd = MoveSpeedOf(e);
                float step = System.Math.Min(spd * perTick, dist);
                e.PosF = TerrainMap.ClampToEllipse(e.PosF + delta.Normalized() * step);
                PushOutOfBuildings(e);
                e.PosF = TerrainMap.ClampToEllipse(e.PosF);
                e.Pos = e.PosF.ToGrid();

                e.MoveAccum += step;
                if (e.MoveAccum >= 1f)
                {
                    e.MoveAccum = 0f;
                    Emit(new SimEvent(Tick, SimEventType.UnitAdvanced, e.Id)); // 节流:约每 1 米一次
                }

                if ((target - e.PosF).Magnitude <= 0.15f)
                {
                    e.HasMoveTarget = false;
                    e.State = EntityState.Idle;
                    e.MoveAccum = 0f;
                }
            }
            RegimentSystem.Tick(this, perTick); // 兵团底座(M8.2a):阵型跟随/无将续战,先于软碰撞
            SoftCollide();
        }

        /// <summary>单位间软碰撞:间距小于 0.9 米时沿连线对半推开。</summary>
        private void SoftCollide()
        {
            var units = new List<Entity>();
            foreach (var e in _entities.Values)
                if (e.Kind == EntityKind.Unit && e.GarrisonBuildingId == 0
                    && e.State != EntityState.Dead && e.State != EntityState.Exchanging)
                    units.Add(e);

            const float MinDist = 0.9f;
            var moved = new HashSet<int>();
            for (int i = 0; i < units.Count; i++)
            {
                for (int j = i + 1; j < units.Count; j++)
                {
                    var a = units[i];
                    var b = units[j];
                    // 兵团底座(M8.2a):同兵团士兵之间、士兵与己方主将之间不互推,避免阵型震荡
                    if ((a.IsSoldier || b.IsSoldier) && (a.GeneralId == b.Id || b.GeneralId == a.Id || (a.IsSoldier && b.IsSoldier && a.GeneralId != 0 && a.GeneralId == b.GeneralId))) continue;
                    float dx = a.PosF.X - b.PosF.X, dy = a.PosF.Y - b.PosF.Y;
                    float d2 = dx * dx + dy * dy;
                    if (d2 >= MinDist * MinDist) continue;
                    float d = (float)System.Math.Sqrt(d2);
                    float push = (MinDist - d) * 0.5f;
                    float nx = d > 1e-6f ? dx / d : 1f;
                    float ny = d > 1e-6f ? dy / d : 0f;
                    a.PosF = TerrainMap.ClampToEllipse(a.PosF + new SimVec2(nx * push, ny * push));
                    b.PosF = TerrainMap.ClampToEllipse(b.PosF - new SimVec2(nx * push, ny * push));
                    moved.Add(a.Id);
                    moved.Add(b.Id);
                }
            }
            // 仅被推挤的单位回写格(避免覆盖未移动单位的手动定位)
            foreach (var e in units)
                if (moved.Contains(e.Id))
                    e.Pos = e.PosF.ToGrid();
        }

        public static float BuildingCollisionRadius(Entity b)
        {
            return System.Math.Max(BuildingFootprint.HalfW(b), BuildingFootprint.HalfH(b)) + 0.4f;

        private void CombatUpdate()
        {
            if (Game == null) return;
            float dt = 1f / Config.TickRate;
            var dead = new List<int>();

            foreach (var e in _entities.Values)
            {
                if (e.Kind != EntityKind.Unit || e.State == EntityState.Dead) continue;
                if (e.GarrisonBuildingId != 0) continue; // 驻军单位不参与场外逻辑

                e.Status.Tick(dt);
                e.SecondsSinceCombat += dt;
                HeroAbilitySystem.UpdatePanShiShield(e, dt);
                if (e.Hero == HeroId.ZhuFeng && e.IsCharging && !HeroAbilitySystem.CanCharge(e, this))
                    e.IsCharging = false;

                if (e.Attack <= 0f) continue;

                if (e.AttackCooldown > 0f)
                    e.AttackCooldown -= dt;

                Entity target = FindNearestEnemy(e, e.State == EntityState.Routing ? 1.4f : float.MaxValue);
                if (target == null)
                {
                    if (e.State == EntityState.Attacking) e.State = EntityState.Idle;
                    continue;
                }

                // 溃逃单位保持 Routing 状态(不可调遣语义),不转入 Attacking
                if (e.State != EntityState.Routing)
                {
                    e.HasMoveTarget = false;
                    e.State = EntityState.Attacking;
                }
                if (e.AttackCooldown <= 0f)
                {
                    e.AttackCooldown = e.AttackIntervalSec * e.Status.AttackIntervalMultiplier;
                    ApplyAttack(e, target, dead);
                }
            }

            foreach (int id in dead)
            {
                if (_entities.TryGetValue(id, out var e))
                {
                    if (e.Hero != null) HexSystem.OnUnitDied(this, e.Side); // 陷阵之志
                    Emit(new SimEvent(Tick, SimEventType.EntityDied, id));
                    if (e.Kind == EntityKind.Building)
                        _occupied.Remove(_grid.ToIndex(e.Pos));
                    _entities.Remove(id);
                    Emit(new SimEvent(Tick, SimEventType.EntityRemoved, id));
                }
            }
        }

        private Entity FindNearestEnemy(Entity e, float maxDist)
        {
            Entity best = null;
            float bestDist = float.MaxValue;
            foreach (var o in _entities.Values)
            {
                if (o.Kind != EntityKind.Unit || o.Side == e.Side || o.State == EntityState.Dead)
                    continue;
                if (o.GarrisonBuildingId != 0 || o.State == EntityState.Exchanging)
                    continue; // 城内/交换中的单位不可被城外锁定
                float d = SimVec2.Distance(e.PosF, o.PosF);
                float effRange = e.Range + 0.3f; // 近战贴身容差
                if (d > effRange || d > maxDist) continue;
                if (d < bestDist - 1e-4f || (System.Math.Abs(d - bestDist) <= 1e-4f && best != null && o.Id < best.Id))
                {
                    best = o;
                    bestDist = d;
                }
            }
            return best;
        }

        private void ApplyAttack(Entity attacker, Entity defender, List<int> dead)
        {
            attacker.SecondsSinceCombat = 0f;
            defender.SecondsSinceCombat = 0f;

            // 基础层(GDD 公式)
            float baseDmg = DamageCalculator.CalculateDamage(attacker, defender, Game, attacker.IsCharging);

            // 星落升级·诛灭:基础攻击 5 → 6(基础层内替换)
            bool zhuMie = attacker.Hero == HeroId.XingLuo && IsUpgraded(attacker.Side, HeroId.XingLuo);
            if (zhuMie && attacker.Attack > 0f) baseDmg = baseDmg / attacker.Attack * 6f;

            // 叠算层(D28):输出加算、承伤加算、免伤乘算;溃逃并入对应层
            float outSum = attacker.Status.AttackBonus + HexSystem.OutputBonusOf(this, attacker)
                         + (attacker.State == EntityState.Routing ? -0.3f : 0f);
            float takeSum = defender.Status.DamageTakenBonus + HexSystem.DamageTakenBonusOf(this, defender)
                          + (defender.State == EntityState.Routing ? 0.3f : 0f);
            float reduce = defender.Status.DamageReduction * HexSystem.DamageReductionOf(this, defender);
            float dmg = baseDmg * (1f + outSum) * (1f + takeSum) * reduce;

            // 附加项(层后加法)
            if (zhuMie) dmg += 0.01f * defender.MaxHp;

            // 护盾优先吸收,溢出扣生命
            float remaining = dmg;
            if (defender.Shield > 0f)
            {
                float absorbed = Math.Min(defender.Shield, remaining);
                defender.Shield -= absorbed;
                remaining -= absorbed;
            }
            defender.Hp -= remaining;

            Emit(new SimEvent(Tick, SimEventType.DamageDealt, defender.Id, dmg));

            HeroAbilitySystem.ApplyOnHitEffects(this, attacker, defender);

            // 断峰升级·破军:身前 90 度扇形(半径 2 米)内全部敌人受单次 75% 伤害
            if (attacker.Hero == HeroId.DuanFeng && IsUpgraded(attacker.Side, HeroId.DuanFeng))
                Cleave(attacker, defender, dead);

            if (defender.Hp <= 0f)
            {
                defender.State = EntityState.Dead;
                dead.Add(defender.Id);
            }
        }

        private void Cleave(Entity attacker, Entity mainTarget, List<int> dead)
        {
            float facing = mainTarget.PosF.X >= attacker.PosF.X ? 1f : -1f;
            foreach (var o in _entities.Values)
            {
                if (o.Id == mainTarget.Id) continue;
                if (o.Kind != EntityKind.Unit || o.Side == attacker.Side || o.State == EntityState.Dead) continue;
                if (o.GarrisonBuildingId != 0 || o.State == EntityState.Exchanging) continue;
                float dx = o.PosF.X - attacker.PosF.X;
                float dy = o.PosF.Y - attacker.PosF.Y;
                if (dx * facing < 0) continue;         // 身后不打
                if (dx * dx + dy * dy > 4f) continue;  // 半径 2 米扇形
                float baseDmg = 0.75f * DamageCalculator.CalculateDamage(attacker, o, Game, attacker.IsCharging);
                float outSum = attacker.Status.AttackBonus + (attacker.State == EntityState.Routing ? -0.3f : 0f);
                float takeSum = o.Status.DamageTakenBonus + (o.State == EntityState.Routing ? 0.3f : 0f);
                float dmg = baseDmg * (1f + outSum) * (1f + takeSum) * o.Status.DamageReduction;
                float remaining = dmg;
                if (o.Shield > 0f)
                {
                    float absorbed = System.Math.Min(o.Shield, remaining);
                    o.Shield -= absorbed;
                    remaining -= absorbed;
                }
                o.Hp -= remaining;
                Emit(new SimEvent(Tick, SimEventType.DamageDealt, o.Id, dmg));
                if (o.Hp <= 0f) { o.State = EntityState.Dead; dead.Add(o.Id); }
            }
        }

        private double ChargeSpeedOf(HeroId id)
        {
            foreach (var h in Game.Heroes)
                if (h.Id == id) return h.ChargeSpeed;
            return 0;
        }

        private void Emit(SimEvent ev)
        {
            for (int i = _listeners.Count - 1; i >= 0; i--)
                _listeners[i]?.Invoke(ev);
        }
    }
```

```csharp
using NUnit.Framework;
using UnityEngine;

namespace IronCurtain.Tests
{
    public sealed class RegimentBehaviorTests
    {
        private static GameConfig _cfg;

        private static SimCore NewSim()
        {
            if (_cfg == null)
                _cfg = GameConfigParser.FromJson(Resources.Load<TextAsset>("gameconfig").text);
            var sim = new SimCore(new SimConfig(), new SeededRng(1));
            sim.Game = _cfg;
            return sim;
        }

        private static SimCore NewAutoSim()
        {
            var sim = NewSim();
            sim.Config.AutoBehavior = true;
            return sim;
        }

        private static bool RunUntil(SimCore sim, System.Func<bool> cond, int maxTicks)
        {
            for (int i = 0; i < maxTicks; i++)
            {
                if (cond()) return true;
                sim.Step();
            }
            return cond();
        }

        private static List<Entity> SoldiersOf(SimCore sim, int generalId)
        {
            var list = new List<Entity>();
            foreach (var e in sim.Entities.Values)
                if (e.Kind == EntityKind.Unit && e.IsSoldier && e.GeneralId == generalId) list.Add(e);
            return list;
        }

        [TearDown]
        public void TearDown()
        {
            AiSystem.Reset();
            VisionSystem.Reset();
            HexSystem.Reset();
            SiegeSystem.Reset();
        }

        [Test]
        public void T1_EngageOnVision()
        {
            var sim = NewAutoSim();
            var g = RegimentSystem.SpawnGeneral(sim, Side.Blue, HeroId.DuanFeng, new GridPos(15, 100));
            RegimentSystem.SpawnGeneral(sim, Side.Red, HeroId.PanShi, new GridPos(15, 108)); // 距 8,断峰主将视野 10 内

            bool ok = RunUntil(sim, () => g.AutoKind == 1, 100);
            Assert.IsTrue(ok, "视野内出现敌人应触发自动进攻");
            Assert.IsTrue(g.HasMoveTarget, "自动进攻应下发移动目标");
        }

        [Test]
        public void T2_SupportWithin15()
        {
            var sim = NewAutoSim();
            var a = RegimentSystem.SpawnGeneral(sim, Side.Blue, HeroId.PanShi, new GridPos(15, 100));
            RegimentSystem.SpawnGeneral(sim, Side.Red, HeroId.PanShi, new GridPos(15, 102)); // C:与 A 交战,距 B 8 > B 视野 6
            var b = RegimentSystem.SpawnGeneral(sim, Side.Blue, HeroId.PanShi, new GridPos(15, 114)); // B 距 A 14 ≤ 15;C(102) 距 B 12 > 视野 6

            bool ok = RunUntil(sim, () => b.AutoKind == 2 && b.AutoTargetId == a.Id, 300);
            Assert.IsTrue(ok, "15 格内友军交战应触发自动支援(A 交战中)");
        }

        [Test]
        public void T3_EngageBeatsSupport()
        {
            var sim = NewAutoSim();
            var a = RegimentSystem.SpawnGeneral(sim, Side.Blue, HeroId.PanShi, new GridPos(15, 100));
            RegimentSystem.SpawnGeneral(sim, Side.Red, HeroId.PanShi, new GridPos(15, 102)); // A 的交战对象
            var b = RegimentSystem.SpawnGeneral(sim, Side.Blue, HeroId.PanShi, new GridPos(15, 114)); // B 距 A 10

            var e = sim.SpawnEntity(Side.Red, EntityKind.Unit, new GridPos(15, 119)); // 距 B 5 ≤ 视野 6,B 可见
            e.Hero = HeroId.PanShi; e.Hp = e.MaxHp = 5000f; e.Attack = 0f;

            for (int i = 0; i < 20; i++) sim.Step();
            Assert.AreEqual(1, b.AutoKind, "索敌进攻优先于支援");
        }

        [Test]
        public void T4_PlayerInterrupt_NoPullback()
        {
            var sim = NewAutoSim();
            var a = RegimentSystem.SpawnGeneral(sim, Side.Blue, HeroId.PanShi, new GridPos(15, 100));
            RegimentSystem.SpawnGeneral(sim, Side.Red, HeroId.PanShi, new GridPos(15, 102));
            var b = RegimentSystem.SpawnGeneral(sim, Side.Blue, HeroId.PanShi, new GridPos(15, 114));

            Assert.IsTrue(RunUntil(sim, () => b.AutoKind == 2 && b.AutoTargetId == a.Id, 300), "前置:B 进入支援");

            // 玩家手动打断(反方向)
            sim.SetMoveTarget(b.Id, new GridPos(15, 125), true);
            Assert.AreEqual(0, b.AutoKind, "手动命令应清 AutoKind");
            Assert.AreEqual(2, b.SupKind, "手动命令应写 SupKind=2");
            Assert.AreEqual(a.Id, b.SupTargetId);

            float startDist = SimVec2.Distance(b.PosF, a.PosF);
            bool pulled = false;
            for (int i = 0; i < 100; i++)
            {
                sim.Step();
                if (b.AutoKind == 2 && b.AutoTargetId == a.Id) pulled = true;
            }
            Assert.IsFalse(pulled, "同事件支援复读应被禁");
            Assert.Greater(SimVec2.Distance(b.PosF, a.PosF), startDist, "手动目标期间应远离 A");

            // 新事件:敌 E 进入 B 行军路径视野 → 允许索敌进攻
            var e2 = sim.SpawnEntity(Side.Red, EntityKind.Unit, new GridPos(15, 128));
            e2.Hero = HeroId.PanShi; e2.Hp = e2.MaxHp = 5000f; e2.Attack = 0f;
            bool engaged = RunUntil(sim, () => b.AutoKind == 1, 200);
            Assert.IsTrue(engaged, "新事件(视野内新敌)可再次触发自动进攻");
        }

        [Test]
        public void T5_FlagOffNoAuto()
        {
            var sim = NewSim(); // 默认 Config,不开 AutoBehavior
            var a = RegimentSystem.SpawnGeneral(sim, Side.Blue, HeroId.PanShi, new GridPos(15, 100));
            RegimentSystem.SpawnGeneral(sim, Side.Red, HeroId.PanShi, new GridPos(15, 102));
            var b = RegimentSystem.SpawnGeneral(sim, Side.Blue, HeroId.PanShi, new GridPos(15, 114));

            for (int i = 0; i < 200; i++) sim.Step();
            Assert.AreEqual(0, b.AutoKind, "开关关闭时不应有自动决策");
            Assert.IsFalse(b.HasMoveTarget, "开关关闭时不应有自动移动");
        }

        [Test]
        public void T6_AiAutoReinforce()
        {
            var sim = NewAutoSim();
            AiSystem.Reset();
            VisionSystem.Reset();

            var home = sim.SpawnBuilding(Side.Blue, BuildingId.Base, new GridPos(15, 30));
            home.Prod = new ProductionSet();
            sim.SpawnBuilding(Side.Red, BuildingId.Base, new GridPos(15, 250));
            AiSystem.Configure(sim, Side.Blue, AiTier.Normal);

            var g = RegimentSystem.SpawnGeneral(sim, Side.Blue, HeroId.PanShi, new GridPos(15, 100));
            var soldiers = SoldiersOf(sim, g.Id);
            Assert.AreEqual(20, soldiers.Count);
            for (int i = 0; i < 15; i++) sim.RemoveEntity(soldiers[i].Id); // 杀 15 → 5/20
            sim.ReinforcePool[Side.Blue] = 10;

            int Count() => SoldiersOf(sim, g.Id).Count;
            bool done = RunUntil(sim, () => Count() > 5 || sim.ReinforcePool[Side.Blue] < 10, 600);
            Assert.IsTrue(done, "AI 应在池≥10 时自动补兵(≤600 tick)");
        }
    }
}
```
