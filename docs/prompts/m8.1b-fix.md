# M8.1b 修复轮(第 2 轮)—— 3 条红:视野规格纠偏 / 出生堵塞场景 / 溃逃目的地

## 一、角色与任务
你是《铁幕:前线》Sim 层主程。M8.1b 首轮有 3 条测试红,根因如下(其中第 1 条是编排者规格错误,按本节纠偏)。**只允许输出/修改以下 3 个文件,其余零改动:**

## 二、根因与修复规格(精确)

### 修复 1:VisionSystem 建筑视野(编排者规格纠偏)
- 错误:首轮把建筑视野从"半径圆盘"改成了"只标占地格",导致 `建筑视野_塔12格_大本营20格` 失败。
- 正确语义(与 M5.0 权威一致):建筑视野=**半径圆盘**(大本营 20、箭塔 12、其他建筑 10,半径取自现有配置/常量);**建筑视野圆盘不做椭圆过滤**(建筑自身格与视盘都要标,历史测试 (1,0)→覆盖 (1,20) 依赖此语义;单位/精灵的椭圆过滤保持不动)。
- 实现:建筑分支 = ①`MarkDisc(st, sim.Tick, e.Pos, BuildingVisionRadius(e))`(恢复半径盘,无椭圆过滤——给 MarkDisc 增加一个 `bool skipOutsideEllipse` 参数或为建筑单独写一个不带过滤的标记循环,单位调用传 true、建筑调用传 false,禁止复制粘贴两份循环代码);②额外把 `BuildingFootprint.CellsOccupied(e)` 每格标记(radius 0,无过滤)。
- `BuildingVisionRadius`:Base→20,Tower→12,其余(Barracks/ResourcePoint/堡垒)→10。

### 修复 2:T10 出生堵塞场景坐标(当前堵点方向写反)
- 现状:兵营 (15,110),蓝方 dir=+1,HalfW=2.5 → 前缘外候选列 = `110 + (int)(Ceil(2.5)+1) = 114`,候选行 = 15+{0,-1,1,-2,2} = 13..17。首轮堵的是 cols 104..108(错误方向)。
- 修复:堵塞区改为 **col 114 与 col 115,rows 12..18** 的敌方单位(确保全部 5 个候选格 (13..17,114) 及外扩余量被占/被 0.5 米近距规则挡住)。断言意图不变(Step 后 slot.WaitingForSpawn == true 且 Current 非空)。若堵两列导致出生点尝试到 (114,13..17) 全被占,即可成立;若 115 列不必要,可只堵 114 列 5 格,但必须覆盖全部候选行。

### 修复 3:溃逃测试目的地被占导致 Routing 永续
- 根因:`城池易主_临时驻军溃逃回后方据点` 里红方进攻单位 (15,118) 恰好在蓝方溃逃目的地(蓝兵营城门格 ~col 119)路径上,软碰撞+接战使溃逃单位永远无法"抵达"(状态恒为 Routing),断言超时。
- 修复:调整该测试场景使溃逃目的地不被占:把红方进攻单位移到不挡溃逃路径的位置(如 (12,117) 或 (17,117)),或调整蓝兵营位置使城门路径通畅;**保持测试意图**(易主成功/临时驻军离城/抵达后方据点待命=非 Routing)。其余断言不动。
- 禁止改 GarrisonSystem 任何逻辑(若调整后发现是 Sim 侧真问题——比如溃逃目的地选择被新占地挡住——只允许改测试场景绕过,并在输出中说明;Sim 侧寻路问题留待 M8.2 兵团化统一处理)。

```csharp
using System.Collections.Generic;

namespace IronCurtain.Sim
{
    /// <summary>据点信息快照(冻结显示用,纯信息层)。</summary>
    public sealed class BuildingSnapshot
    {
        public float Hp;
        public float MaxHp;
        public Side Side;
        public bool Rubbled;
        public int GarrisonCount;
    }

    /// <summary>双方对称的视野/战争迷雾信息层。纯信息,不影响战斗/移动/索敌。M8.1b:建筑视野改占地矩形全部格。</summary>
    public static class VisionSystem
    {
        public const int FreshTicks = 30;              // 信息新鲜 3 秒(30 tick)
        public const int SpriteIntervalTicks = 300;    // 精灵每 30 秒出发一次
        const float SpriteSpeedFriendly = 1.5f;  // 己方区段 15 米/s(每 tick 1.5 米)
        const float SpriteSpeedContested = 0.6f;  // 后续区段 6 米/s(每 tick 0.6 米)
        const int Rows = LaneGrid.Rows;
        const int Cols = LaneGrid.Cols;

        sealed class SideState
        {
            public readonly Dictionary<long, long> LastSeen = new Dictionary<long, long>();
            public readonly Dictionary<int, BuildingSnapshot> Buildings = new Dictionary<int, BuildingSnapshot>();
            public long NextSpriteDeparture;
            public float SpriteTravel;   // 精灵已行进米数(浮点)
            public bool SpriteActive;    // 精灵在途
            public int SpriteCol = -1;   // 精灵当前列(渲染用,淡蓝/淡红发光)
        }

        static readonly SideState Blue = new SideState();
        static readonly SideState Red = new SideState();

        static SideState S(Side side) { return side == Side.Blue ? Blue : Red; }
        static long Key(GridPos p) { return p.Row * 1000L + p.Col; }

        public static void Reset()
        {
            Blue.LastSeen.Clear(); Blue.Buildings.Clear(); Blue.NextSpriteDeparture = 300; Blue.SpriteActive = false; Blue.SpriteTravel = 0f; Blue.SpriteCol = -1; // 首发第 30 秒
            Red.LastSeen.Clear(); Red.Buildings.Clear(); Red.NextSpriteDeparture = 300; Red.SpriteActive = false; Red.SpriteTravel = 0f; Red.SpriteCol = -1; // 首发第 30 秒
        }

        public static void Tick(SimCore sim, float dt)
        {
            if (sim.Game == null) return; // 无配置的裸模拟(旧测试)跳过
            TickSide(sim, Side.Blue);
            TickSide(sim, Side.Red);
        }

        static void TickSide(SimCore sim, Side side)
        {
            var st = S(side);

            // ---- 1) 野战单位来源 ----
            // 先找己方最靠敌方向的野战单位列(蓝取最大 Col,红取最小 Col),供鹰眼判定
            int forwardCol = side == Side.Blue ? int.MinValue : int.MaxValue;
            foreach (var e in sim.Entities.Values)
            {
                if (e.Kind != EntityKind.Unit || e.Side != side) continue;
                if (e.State == EntityState.Dead || e.GarrisonBuildingId != 0) continue;
                if (side == Side.Blue) { if (e.Pos.Col > forwardCol) forwardCol = e.Pos.Col; }
                else { if (e.Pos.Col < forwardCol) forwardCol = e.Pos.Col; }
            }
            foreach (var e in sim.Entities.Values)
            {
                if (e.Kind != EntityKind.Unit || e.Side != side) continue;
                if (e.State == EntityState.Dead || e.GarrisonBuildingId != 0) continue;
                MarkDisc(st, sim.Tick, e.Pos, UnitRadius(sim, side, e, forwardCol));
            }

            // ---- 2) 建筑来源(非废墟,M8.1b):标记占地矩形全部格(椭圆外格仍跳过) ----
            foreach (var e in sim.Entities.Values)
            {
                if (e.Kind != EntityKind.Building || e.Side != side || e.Rubbled) continue;
                foreach (var cell in BuildingFootprint.CellsOccupied(e))
                    MarkDisc(st, sim.Tick, cell, 0);
            }

            // ---- 3) 视野精灵 ----
            TickSprite(sim, side, st);

            // ---- 4) 建筑快照:本 tick 建筑格被揭示则刷新 ----
            foreach (var e in sim.Entities.Values)
            {
                if (e.Kind != EntityKind.Building) continue;
                long t;
                if (st.LastSeen.TryGetValue(Key(e.Pos), out t) && t == sim.Tick)
                {
                    st.Buildings[e.Id] = new BuildingSnapshot
                    {
                        Hp = e.Hp,
                        MaxHp = e.MaxHp,
                        Side = e.Side,
                        Rubbled = e.Rubbled,
                        GarrisonCount = e.GarrisonFormal.Count + e.GarrisonTemp.Count
                    };
                }
            }
        }

        static int UnitRadius(SimCore sim, Side side, Entity e, int forwardCol)
        {
            // 鹰眼:断峰是己方最靠前的野战单位(不存在更靠敌方向者)→ 10
            if (e.Hero == HeroId.DuanFeng && e.Pos.Col == forwardCol) return 10;
            if (e.Hero == null) return 6; // 无英雄的裸单位按近战 6 格
            return (e.Hero == HeroId.ZhuiRi || e.Hero == HeroId.XingLuo) ? 9 : 6; // 射手 9 / 近战(含断峰)6
        }

        static void TickSprite(SimCore sim, Side side, SideState st)
        {
            // 追平出发时刻表(300/600…)
            while (sim.Tick - st.NextSpriteDeparture >= SpriteIntervalTicks)
                st.NextSpriteDeparture += SpriteIntervalTicks;
            if (sim.Tick < st.NextSpriteDeparture) { st.SpriteActive = false; return; }

            GridPos? basePos = null;
            foreach (var e in sim.Entities.Values)
            {
                if (e.Kind == EntityKind.Building && e.Side == side && e.Building == BuildingId.Base)
                {
                    basePos = e.Pos;
                    break;
                }
            }
            if (!basePos.HasValue) { st.SpriteActive = false; return; }

            if (!st.SpriteActive) // 出发瞬间:复位行程
            {
                st.SpriteActive = true;
                st.SpriteTravel = 0f;
            }
            int dir = side == Side.Blue ? 1 : -1;
            int baseCol = basePos.Value.Col;
            // 己方最靠前建筑列(无则=大本营自身)
            int frontCol = baseCol;
            foreach (var e in sim.Entities.Values)
            {
                if (e.Kind != EntityKind.Building || e.Side != side || e.Rubbled) continue;
                if (side == Side.Blue && e.Pos.Col > frontCol) frontCol = e.Pos.Col;
                if (side == Side.Red && e.Pos.Col < frontCol) frontCol = e.Pos.Col;
            }
            // 两段速度:己方区段(至最靠前己方建筑)15 米/s,之后 6 米/s
            float frontDist = System.Math.Abs(frontCol - baseCol);
            float speed = st.SpriteTravel < frontDist ? SpriteSpeedFriendly : SpriteSpeedContested;
            st.SpriteTravel += speed;

            int col = baseCol + dir * (int)st.SpriteTravel;
            if (col < 0 || col >= Cols) { st.SpriteActive = false; return; }
            st.SpriteCol = col;
            for (int row = 0; row < Rows; row++)
            {
                var p = new GridPos(row, col);
                if (!TerrainMap.IsInside(p)) continue; // 只揭示椭圆内
                st.LastSeen[row * 1000L + col] = sim.Tick;
            }
        }

        /// <summary>精灵当前位置(渲染发光用;-1=不在途)。</summary>
        public static int SpriteColOf(Side side) => S(side).SpriteCol;

        static void MarkDisc(SideState st, long tick, GridPos center, int radius)
        {
            // 欧氏圆盘(米制视野):sqrt(dr²+dc²) ≤ radius;跳过椭圆外。radius=0 即单格标记(M8.1b 建筑占地格)
            for (int dr = -radius; dr <= radius; dr++)
            {
                int row = center.Row + dr;
                if (row < 0 || row >= Rows) continue;
                int dcMax = (int)System.Math.Floor(System.Math.Sqrt((double)(radius * radius - dr * dr)));
                for (int dc = -dcMax; dc <= dcMax; dc++)
                {
                    int col = center.Col + dc;
                    if (col < 0 || col >= Cols) continue;
                    var p = new GridPos(row, col);
                    if (!TerrainMap.IsInside(p)) continue;
                    st.LastSeen[row * 1000L + col] = tick;
                }
            }
        }

        public static bool IsFresh(SimCore sim, Side side, GridPos cell)
        {
            long t;
            return S(side).LastSeen.TryGetValue(Key(cell), out t) && sim.Tick - t <= FreshTicks;
        }

        public static bool IsUnitVisible(SimCore sim, Side side, int unitId)
        {
            Entity e;
            if (!TryGet(sim, unitId, out e) || e.State == EntityState.Dead) return false;
            var pos = e.Pos;
            if (e.GarrisonBuildingId != 0)
            {
                Entity b;
                if (TryGet(sim, e.GarrisonBuildingId, out b)) pos = b.Pos; // 驻军以建筑格判断
            }
            return IsFresh(sim, side, pos);
        }

        public static bool IsBuildingKnown(SimCore sim, Side side, int buildingId)
        {
            return S(side).Buildings.ContainsKey(buildingId); // 冻结快照也算已知
        }

        public static int? LastKnownGarrison(SimCore sim, Side side, int buildingId)
        {
            BuildingSnapshot s;
            return S(side).Buildings.TryGetValue(buildingId, out s) ? (int?)s.GarrisonCount : null;
        }

        public static float? LastKnownHp(SimCore sim, Side side, int buildingId)
        {
            BuildingSnapshot s;
            return S(side).Buildings.TryGetValue(buildingId, out s) ? (float?)s.Hp : null;
        }

        static bool TryGet(SimCore sim, int id, out Entity e)
        {
            e = null;
            if (id == 0) return false;
            Entity found;
            if (!sim.Entities.TryGetValue(id, out found)) return false;
            e = found;
            return true;
        }
    }
}
```

### 3.4 BuildingFootprint.cs(只读)
```csharp
using System;
using System.Collections.Generic;

namespace IronCurtain.Sim
{
    /// <summary>建筑占地矩形(米制半宽半高,中心=建筑格中心 SimVec2.FromGrid(b.Pos))。M8.1a:占地类,纯 C#。</summary>
    public static class BuildingFootprint
    {
        public static float HalfW(Entity b)
        {
            if (b.IsFort) return 1.5f;
            switch (b.Building)
            {
                case BuildingId.Base: return 2.5f;
                case BuildingId.Tower: return 2.0f;
                case BuildingId.Barracks: return 2.5f;
                case BuildingId.ResourcePoint: return 2.0f;
                default: return 1.5f;
            }
        }

        public static float HalfH(Entity b)
        {
            if (b.IsFort) return 1.5f;
            switch (b.Building)
            {
                case BuildingId.Base: return 2.5f;
                case BuildingId.Tower: return 2.0f;
                case BuildingId.Barracks: return 2.0f;
                case BuildingId.ResourcePoint: return 2.0f;
                default: return 1.5f;
            }
        }

        /// <summary>点是否在中心矩形内(含边界)。</summary>
        public static bool Contains(Entity b, SimVec2 p)
        {
            var c = SimVec2.FromGrid(b.Pos);
            float hw = HalfW(b), hh = HalfH(b);
            return p.X >= c.X - hw && p.X <= c.X + hw && p.Y >= c.Y - hh && p.Y <= c.Y + hh;
        }

        /// <summary>点在矩形外→矩形上最近点;点在矩形内→沿最小穿透轴推出后的缘点(供 PushOutOfBuildings 一次完成推出)。</summary>
        public static SimVec2 ClosestPoint(Entity b, SimVec2 p)
        {
            var c = SimVec2.FromGrid(b.Pos);
            float hw = HalfW(b), hh = HalfH(b);
            float minX = c.X - hw, maxX = c.X + hw, minY = c.Y - hh, maxY = c.Y + hh;
            if (p.X < minX || p.X > maxX || p.Y < minY || p.Y > maxY)
                return new SimVec2(Math.Min(Math.Max(p.X, minX), maxX),
                                   Math.Min(Math.Max(p.Y, minY), maxY));
            // 在矩形内:沿最小穿透轴推出
            float left = p.X - minX, right = maxX - p.X, down = p.Y - minY, up = maxY - p.Y;
            float m = Math.Min(Math.Min(left, right), Math.Min(down, up));
            if (m == left) return new SimVec2(minX, p.Y);
            if (m == right) return new SimVec2(maxX, p.Y);
            if (m == down) return new SimVec2(p.X, minY);
            return new SimVec2(p.X, maxY);
        }

        /// <summary>枚举矩形覆盖的所有格(格中心落在 [min,max) 半开区间内,保证基地25/箭塔16/兵营20/资源点16/堡垒9)。</summary>
        public static List<GridPos> CellsOccupied(Entity b)
        {
            var list = new List<GridPos>();
            var c = SimVec2.FromGrid(b.Pos);
            float hw = HalfW(b), hh = HalfH(b);
            int rc = (int)Math.Ceiling(hh), cc = (int)Math.Ceiling(hw);
            for (int dr = -rc; dr <= rc; dr++)
                for (int dc = -cc; dc <= cc; dc++)
                {
                    int row = b.Pos.Row + dr, col = b.Pos.Col + dc;
                    if (row < 0 || row >= LaneGrid.Rows || col < 0 || col >= LaneGrid.Cols) continue;
                    var ctr = SimVec2.FromGrid(new GridPos(row, col));
                    if (ctr.X >= c.X - hw && ctr.X < c.X + hw && ctr.Y >= c.Y - hh && ctr.Y < c.Y + hh)
                        list.Add(new GridPos(row, col));
                }
            return list;
        }
    }
}
```

### 3.5 ProductionSystem.cs(只读,出生点逻辑参考)
```csharp
using System.Collections.Generic;

namespace IronCurtain.Sim
{
    /// <summary>
    /// 生产系统:兵营 3 队列、人口校验、取消返还、出生点占用等待、自动行军。
    /// 纯 tick 驱动,无 UnityEngine 依赖。M8.1a:出生点贴建筑占地前缘。
    /// </summary>
    public static class ProductionSystem
    {
        /// <summary>浮点比较容差(防 0.1f 累积漂移)。</summary>
        private const float Epsilon = 1e-4f;

        /// <summary>出生候选行序:同排/上/下/再上/再下(从近到远)。</summary>
        private static readonly int[] SpawnDRows = { 0, -1, 1, -2, 2 };

        public static void Tick(SimCore sim, float dt)
        {
            // 快照遍历:出生会向 Entities 字典添加实体,禁止边枚举边修改
            var snapshot = new List<Entity>(sim.Entities.Values);

            foreach (var e in snapshot)
            {
                if (e.Kind != EntityKind.Building) continue;
                if (e.Building != BuildingId.Barracks) continue;
                var prod = e.Prod;
                if (prod == null) continue;

                for (int i = 0; i < prod.Slots.Length; i++)
                {
                    var slot = prod.Slots[i];

                    // 空闲槽开工:人口校验通过才从待产队列取出
                    if (slot.Current == null && slot.Pending.Count > 0)
                    {
                        var hero = slot.Pending.Peek();
                        int pop = sim.PopCostOf(hero);
                        if (sim.PopulationOf(e.Side) + pop <= sim.PopulationCapOf(e.Side))
                        {
                            slot.Current = hero;
                            slot.Progress = 0f;
                            slot.WaitingForSpawn = false;
                            slot.Pending.Dequeue();
                        }
                    }

                    if (slot.Current != null)
                    {
                        if (!slot.WaitingForSpawn && slot.Progress + dt < ProductionSeconds(sim, e.Side, slot.Current.Value) - Epsilon)
                        {
                            slot.Progress += dt;
                            continue;
                        }

                        // 完成或等待出生中:尝试出生
                        if (TrySpawnUnit(sim, e, slot.Current.Value, out var unit))
                        {
                            slot.Current = null;
                            slot.Progress = 0f;
                            slot.WaitingForSpawn = false;
                            sim.SetMoveTarget(unit.Id, sim.RallyPointOf(e.Side));
                        }
                        else
                        {
                            slot.WaitingForSpawn = true; // 出生点全占,下 tick 重试
                        }
                    }
                }
            }
        }

        private static double ProductionSeconds(SimCore sim, Side side, HeroId hero)
        {
            if (hero == HeroId.ZhenYue && sim.IsUpgraded(side, HeroId.ZhenYue))
                return sim.Game.Production.ZhenYueUpgradedSeconds; // 迅捷:7.5s
            int pop = sim.PopCostOf(hero);
            return pop <= 1 ? sim.Game.Production.Pop1Seconds
                            : sim.Game.Production.ZhenYueSeconds;
        }

        private static bool TrySpawnUnit(SimCore sim, Entity barracks, HeroId heroId, out Entity unit)
        {
            unit = null;
            int dir = barracks.Side == Side.Blue ? 1 : -1;

            // 全图非废墟建筑占地格集合(多格占用,M8.1a)
            var occupied = new HashSet<long>();
            foreach (var e in sim.Entities.Values)
                if (e.Kind == EntityKind.Building && !e.Rubbled)
                    foreach (var cell in BuildingFootprint.CellsOccupied(e))
                        occupied.Add(((long)cell.Row << 32) | (uint)cell.Col);

            // 前缘列:占地半宽外再让 1 格
            int col = barracks.Pos.Col + dir * ((int)System.Math.Ceiling(BuildingFootprint.HalfW(barracks)) + 1);
            for (int i = 0; i < SpawnDRows.Length; i++)
            {
                int row = barracks.Pos.Row + SpawnDRows[i];
                if (row < 0 || row >= LaneGrid.Rows || col < 0 || col >= LaneGrid.Cols) continue;
                var cand = new GridPos(row, col);
                if (!TerrainMap.IsInside(cand)) continue;                    // 椭圆外不可出生
                if (TerrainMap.IsWater(SimVec2.FromGrid(cand))) continue;    // 水域不可出生
                long key = ((long)row << 32) | (uint)col;
                if (occupied.Contains(key)) continue;

                // 连续坐标下出生点还要避开既有单位(0.5 米内)
                var center = SimVec2.FromGrid(cand);
                bool tooClose = false;
                foreach (var e in sim.Entities.Values)
                    if (e.Kind == EntityKind.Unit && SimVec2.Distance(e.PosF, center) < 0.5f)
                    { tooClose = true; break; }
                if (tooClose) continue;

                var c = sim.Game.Heroes[(int)heroId];
                var u = sim.SpawnEntity(barracks.Side, EntityKind.Unit, cand);
                u.Hero = heroId; u.MaxHp = c.MaxHp; u.Hp = c.MaxHp;
                u.Attack = c.Attack; u.AttackIntervalSec = (float)c.AttackIntervalSec;
                u.Range = c.Range; u.MoveSpeed = (float)c.MoveSpeed;
                if (c.Shield > 0) { u.Shield = c.Shield; u.ShieldMax = c.Shield; }
                u.PosF = SimVec2.FromGrid(cand); // 格中心连续坐标
                unit = u;
                return true;
            }
            return false;
        }

        public static bool Enqueue(SimCore sim, int barracksId, HeroId hero, out string error)
        {
            error = null;
            if (!sim.Entities.TryGetValue(barracksId, out var b) ||
                b.Kind != EntityKind.Building || b.Building != BuildingId.Barracks)
            {
                error = "目标不是兵营";
                return false;
            }
            var prod = b.Prod;
            if (prod == null) { error = "兵营无生产集"; return false; }

            int best = 0;
            for (int i = 1; i < prod.Slots.Length; i++)
                if (prod.Slots[i].Pending.Count < prod.Slots[best].Pending.Count) best = i;
            prod.Slots[best].Pending.Enqueue(hero);
            return true;
        }

        public static bool CancelLast(SimCore sim, int barracksId, int slotIndex, out string error)
        {
            error = null;
            if (!sim.Entities.TryGetValue(barracksId, out var b) ||
                b.Kind != EntityKind.Building || b.Building != BuildingId.Barracks)
            {
                error = "目标不是兵营";
                return false;
            }
            var prod = b.Prod;
            if (prod == null) { error = "兵营无生产集"; return false; }
            if (slotIndex < 0 || slotIndex >= prod.Slots.Length) { error = "槽索引越界"; return false; }

            var slot = prod.Slots[slotIndex];
            if (slot.Pending.Count > 0)
            {
                var list = new List<HeroId>(slot.Pending);
                list.RemoveAt(list.Count - 1);
                slot.Pending.Clear();
                foreach (var h in list) slot.Pending.Enqueue(h);
                return true;
            }
            if (slot.Current != null)
            {
                slot.Current = null;
                slot.Progress = 0f;
                slot.WaitingForSpawn = false;
                return true;
            }
            error = "队列为空";
            return false;
        }

        public static void OnBarracksDestroyed(SimCore sim, Entity barracks)
        {
            if (barracks == null) return;
            if (barracks.Prod != null) barracks.Prod.Clear();
        }
    }
}
```

```csharp
using System.Linq;

namespace IronCurtain.Sim
{
    /// <summary>驻军系统:正式/临时驻军、依托射击、近战出击、撤出、建筑脱战回复(D27)。M8.1b:城门改占地矩形语义(D18)。</summary>

        // D27:建筑脱战 20 秒后每秒回复 4 点,至上限
        private const float RegenDelaySec = 20f;
        private const float RegenPerSec = 4f; // D27

        private static readonly HashSet<HeroId> RangedHeroes = new HashSet<HeroId> { HeroId.ZhuiRi, HeroId.XingLuo };

        public static bool EnterFormal(SimCore sim, IReadOnlyList<int> unitIds, int buildingId, out string error)
        {
            error = null;
            if (!sim.Entities.ContainsKey(buildingId)) { error = "建筑不存在"; return false; }
            var b = sim.GetEntity(buildingId);

            var accepted = new List<int>();
            foreach (var id in unitIds)
            {
                if (!sim.Entities.ContainsKey(id)) continue;
                var u = sim.GetEntity(id);
                if (u.Side != b.Side) { error = "建筑须为己方"; Rollback(sim, accepted, buildingId); return false; }
                if (sim.GetEntity(buildingId).GarrisonFormal.Count >= cfg.GarrisonFormal)
                {   // 驻军全满:路过或城门站定(用户新规,废除报错)
                    if (!u.HasMoveTarget && u.GarrisonBuildingId == 0)
                    {
                        var g = FindGateCell(sim, b);
                        if (g != null) sim.SetMoveTarget(id, g.Value);
                    }
                    continue;
                }
                DoEnter(sim, u, buildingId, sim.GetEntity(buildingId).GarrisonFormal);
                accepted.Add(id);
            }
            return true;
        }

        public static bool EnterTemp(SimCore sim, IReadOnlyList<int> unitIds, int buildingId, out string error)
        {
            error = null;
            if (!sim.Entities.ContainsKey(buildingId)) { error = "建筑不存在"; return false; }
            var b = sim.GetEntity(buildingId);
            int cap = b.IsFort ? 3 : (b.Building == null ? 0 : sim.BuildingConfigOf(b.Building.Value).GarrisonTemp); // 堡垒驻军上限 3(D32)
            if (cap <= 0) { error = "该建筑不可临时驻军"; return false; }
            var accepted = new List<int>();
            foreach (var id in unitIds)
            {
                if (!sim.Entities.ContainsKey(id)) continue;
                var u = sim.GetEntity(id);
                if (u.Side != b.Side) { error = "建筑须为己方"; RollbackTemp(sim, accepted, buildingId); return false; }
                if (sim.GetEntity(buildingId).GarrisonTemp.Count >= cap)
                {   // 临时驻军全满:路过或城门站定
                    if (!u.HasMoveTarget && u.GarrisonBuildingId == 0)
                    {
                        var g = FindGateCell(sim, b);
                        if (g != null) sim.SetMoveTarget(id, g.Value);
                    }
                    continue;
                }
                DoEnter(sim, u, buildingId, sim.GetEntity(buildingId).GarrisonTemp);
                accepted.Add(id);
            }
            return true;
        }

        public static bool Exit(SimCore sim, int buildingId, HeroId hero, out string error) // D24:每次放一只到城门站定
        {
            error = null;
            if (!sim.Entities.ContainsKey(buildingId)) { error = "建筑不存在"; return false; }
            var b = sim.GetEntity(buildingId);
            int id = b.GarrisonFormal.FirstOrDefault(x => sim.GetEntity(x).Hero == hero);
            bool inFormal = id != 0;
            if (id == 0) id = b.GarrisonTemp.FirstOrDefault(x => sim.GetEntity(x).Hero == hero);
            if (id == 0) { error = "城内无该兵种"; return false; }
            var gate = FindGateCell(sim, b);
            if (gate == null) { error = "城门无空位"; return false; }
            if (inFormal) b.GarrisonFormal.Remove(id); else { b.GarrisonTemp.Remove(id); }
            var u = sim.GetEntity(id);
            u.Pos = gate.Value; u.PosF = SimVec2.FromGrid(gate.Value); u.GarrisonBuildingId = 0; u.State = EntityState.Idle;
            u.HasMoveTarget = false;
            return true;
        }

        /// <summary>城池易主:旧主临时驻军向同一线后方最近己方据点溃逃(不可调遣,遇敌 -30% 交战)。</summary>
        public static void OnCityCaptured(SimCore sim, Entity b)
        {
            var fleeing = new List<int>(b.GarrisonTemp);
            b.GarrisonTemp.Clear();
            foreach (var id in fleeing)
            {
                var u = sim.GetEntity(id);
                if (u == null) continue;
                var dest = FindRoutDestination(sim, u);
                if (dest == null) { u.GarrisonBuildingId = 0; u.State = EntityState.Idle; continue; }
                var gate = FindGateCell(sim, dest);
                if (gate == null)
                {   // 城门无位则就近找空位
                    for (int d = 1; d <= 8 && gate == null; d++)
                        for (int dr = 0; dr < LaneGrid.Rows && gate == null; dr++)
                        {
                            var p = new GridPos(dr, dest.Pos.Col + (u.Side == Side.Blue ? -d : d));
                            if (IsCellFree(sim, p) && TerrainMap.IsInside(p)) gate = p;
                        }
                }
                u.GarrisonBuildingId = 0;
                // 从城池后方出城,避免被已失守的城堵住退路
                int back = u.Side == Side.Blue ? -1 : 1;
                var start = new GridPos(b.Pos.Row, b.Pos.Col + back);
                if (!IsCellFree(sim, start))
                {
                    for (int dr = 0; dr < LaneGrid.Rows; dr++)
                    {
                        var alt = new GridPos(dr, b.Pos.Col + back);
                        if (IsCellFree(sim, alt) && TerrainMap.IsInside(alt)) { start = alt; break; }
                    }
                }
                if (IsCellFree(sim, start)) { u.Pos = start; u.PosF = SimVec2.FromGrid(start); }
                u.State = EntityState.Routing;
                u.HasMoveTarget = true;
                u.MoveTarget = gate ?? dest.Pos;
            }
        }

        /// <summary>建筑化为废墟:全部驻军(正式+临时)强制后撤(途中遇敌交战 -30%)。</summary>
        public static void OnBuildingRubbled(SimCore sim, Entity b)
        {
            var fleeing = new List<int>(b.GarrisonFormal);
            fleeing.AddRange(b.GarrisonTemp);
            b.GarrisonFormal.Clear();
            b.GarrisonTemp.Clear();
            foreach (var id in fleeing)
            {
                var u = sim.GetEntity(id);
                if (u == null) continue;
                var dest = FindRoutDestination(sim, u);
                if (dest == null) { u.GarrisonBuildingId = 0; u.State = EntityState.Idle; continue; }
                var gate = FindGateCell(sim, dest);
                if (gate == null)
                {
                    for (int d = 1; d <= 8 && gate == null; d++)
                        for (int dr = 0; dr < LaneGrid.Rows && gate == null; dr++)
                        {
                            var p = new GridPos(dr, dest.Pos.Col + (u.Side == Side.Blue ? -d : d));
                            if (IsCellFree(sim, p) && TerrainMap.IsInside(p)) gate = p;
                        }
                }
                u.GarrisonBuildingId = 0;
                int back = u.Side == Side.Blue ? -1 : 1;
                var start = new GridPos(b.Pos.Row, b.Pos.Col + back);
                if (!IsCellFree(sim, start))
                {
                    for (int dr = 0; dr < LaneGrid.Rows; dr++)
                    {
                        var alt = new GridPos(dr, b.Pos.Col + back);
                        if (IsCellFree(sim, alt) && TerrainMap.IsInside(alt)) { start = alt; break; }
                    }
                }
                if (IsCellFree(sim, start)) { u.Pos = start; u.PosF = SimVec2.FromGrid(start); }
                u.State = EntityState.Routing;
                u.HasMoveTarget = true;
                u.MoveTarget = gate ?? dest.Pos;
            }
        }

        /// <summary>溃逃目的地:同一线后方(远离敌方向)最近的己方建筑;无则己方大本营。</summary>
        static Entity FindRoutDestination(SimCore sim, Entity u)
        {
            int dir = u.Side == Side.Blue ? -1 : 1; // 远离敌方向
            Entity best = null;
            int bestD = int.MaxValue;
            foreach (var e in sim.Entities.Values)
            {
                if (e.Kind != EntityKind.Building || e.Side != u.Side) continue;
                int d = (e.Pos.Col - u.Pos.Col) * dir;
                if (d <= 0) continue; // 只取后方
                if (d < bestD) { best = e; bestD = d; }
            }
            if (best != null) return best;
            foreach (var e in sim.Entities.Values)
                if (e.Kind == EntityKind.Building && e.Side == u.Side && e.Building == BuildingId.Base)
                    return e;
            return null;
        }

        public static void Tick(SimCore sim, float dt)
        {
            // 1) 临时驻军:正式有空缺且未封城 → 自动转正式(用户新规,废除 10 秒计时)
            foreach (var b in sim.Entities.Values.Where(e => e.Kind == EntityKind.Building).ToList())
            {
                if (b.Sealed) continue; // 封城期间不转
                // 敌方攻城意图存续期间不转:临时驻军受保护,防止"转正后被交换杀死"
                if (b.Intent != SiegeIntent.None && b.IntentIssuer != b.Side) continue;
                int cap = b.IsFort ? 3 : (b.Building == null ? 0 : sim.BuildingConfigOf(b.Building.Value).GarrisonFormal);
                while (b.GarrisonTemp.Count > 0 && b.GarrisonFormal.Count < cap)
                {
                    var id = b.GarrisonTemp[0];
                    b.GarrisonTemp.RemoveAt(0);
                    b.GarrisonFormal.Add(id);
                }
            }

            foreach (var b in sim.Entities.Values.Where(e => e.Kind == EntityKind.Building).ToList())
            {
                var garrison = b.GarrisonFormal.Concat(b.GarrisonTemp).ToList();
                // 2) 依托射击(远程)
                foreach (var id in garrison)
                {
                    var u = sim.GetEntity(id);
                    if (u == null || u.Hero == null || !RangedHeroes.Contains(u.Hero.Value)) continue;
                    u.AttackCooldown -= dt;
                    if (u.AttackCooldown > 0f) continue;
                    var tgt = FindRangedTarget(sim, b, (int)u.Range);
                    if (tgt == null) continue;
                    ApplyDamage(sim, u, tgt);
                    u.AttackCooldown = u.AttackIntervalSec;
                }
                // 3) 近战出击:敌单位欧氏≤2.5 米(建筑中心起算)
                var bc = SimVec2.FromGrid(b.Pos);
                bool threat = sim.Entities.Values.Any(e =>
                    e.Kind == EntityKind.Unit && e.Side != b.Side &&
                    e.GarrisonBuildingId == 0 && e.State != EntityState.Dead && e.State != EntityState.Exchanging &&
                    SimVec2.Distance(e.PosF, bc) <= 2.5f);
                if (threat)
                foreach (var id in garrison)
                {
                    var u = sim.GetEntity(id);
                    if (u == null || u.Hero == null || RangedHeroes.Contains(u.Hero.Value)) continue;
                    var gate = FindGateCell(sim, b);
                    if (gate == null) break;
                    b.GarrisonFormal.Remove(id); b.GarrisonTemp.Remove(id);
                    u.Pos = gate.Value; u.PosF = SimVec2.FromGrid(gate.Value); u.GarrisonBuildingId = 0; u.State = EntityState.Idle;
                }
            }

            // 4) 建筑脱战回复(D27):非大本营
            foreach (var b in sim.Entities.Values.Where(e => e.Kind == EntityKind.Building).ToList())
            {
                if (b.Building == BuildingId.Base) continue;
                b.SecondsSinceCombat += dt;
                if (b.SecondsSinceCombat >= RegenDelaySec && b.Hp < b.MaxHp)
                    b.Hp = System.Math.Min(b.MaxHp, b.Hp + RegenPerSec * dt);
            }
        }

        public static GridPos RallyPointOf(Side side) =>
            side == Side.Blue ? new GridPos(15, 270) : new GridPos(15, 16);

        private static void DoEnter(SimCore sim, Entity u, int buildingId, List<int> list)
        {
            sim.FreeCell(u.Pos);
            u.GarrisonBuildingId = buildingId;
            u.State = EntityState.Idle;
            u.HasMoveTarget = false;
            list.Add(u.Id);
        }

        private static void Rollback(SimCore sim, List<int> accepted, int buildingId)
        { foreach (var id in accepted) sim.GetEntity(buildingId).GarrisonFormal.Remove(id); }
        private static void RollbackTemp(SimCore sim, List<int> accepted, int buildingId)
        { foreach (var id in accepted) { sim.GetEntity(buildingId).GarrisonTemp.Remove(id); } }

        /// <summary>城门位置(D18,M8.1b):占地矩形敌侧缘 +0.6 米列起,行优先建筑行(±ceil(HalfH) 内扩散),椭圆内、非水域,列向外延展找最近空位;全占返回 null。</summary>
        private static GridPos? FindGateCell(SimCore sim, Entity b)
        {
            int dir = b.Side == Side.Blue ? -1 : 1; // 朝敌侧(与 M8.1a 出生点 dir 一致)
            var gateF = SimVec2.FromGrid(b.Pos) + new SimVec2(dir * (BuildingFootprint.HalfW(b) + 0.6f), 0f);
            int gateCol = gateF.ToGrid().Col;
            int rh = (int)System.Math.Ceiling(BuildingFootprint.HalfH(b)); // 建筑占地行范围
            int[] rows = { 0, -1, 1, -2, 2, -3, 3 }; // 行优先建筑行,向外扩散
            for (int d = 0; d <= 4; d++)
            {
                foreach (int r in rows)
                {
                    if (r < -rh || r > rh) continue;
                    var p = new GridPos(b.Pos.Row + r, gateCol + dir * d);
                    if (p.Row < 0 || p.Row >= LaneGrid.Rows || p.Col < 0 || p.Col >= LaneGrid.Cols) continue;
                    if (!TerrainMap.IsInside(p)) continue;
                    if (TerrainMap.IsWater(SimVec2.FromGrid(p))) continue;
                    if (IsCellFree(sim, p)) return p;
                }
            }
            return null; // 城门区全占 → 无空位
        }

        private static bool IsCellFree(SimCore sim, GridPos p)
        {
            var c = SimVec2.FromGrid(p);
            foreach (var e in sim.Entities.Values)
            {
                if (e.State == EntityState.Dead) continue;
                if (e.Kind == EntityKind.Building && !e.Rubbled && e.Pos == p) return false;
                if (e.Kind == EntityKind.Unit && e.GarrisonBuildingId == 0
                    && SimVec2.Distance(e.PosF, c) < 1.0f) return false;
            }
            return true;
        }

        private static Entity FindRangedTarget(SimCore sim, Entity b, int range)
        {
            Entity best = null; float bestD = float.MaxValue; int bestId = int.MaxValue;
            var bc = SimVec2.FromGrid(b.Pos);
            foreach (var e in sim.Entities.Values)
            {
                if (e.Kind != EntityKind.Unit || e.Side == b.Side) continue;
                if (e.GarrisonBuildingId != 0 || e.State == EntityState.Dead || e.State == EntityState.Exchanging) continue;
                float d = SimVec2.Distance(e.PosF, bc);
                if (d > range) continue;
                if (d < bestD || (System.Math.Abs(d - bestD) <= 1e-4f && e.Id < bestId)) { best = e; bestD = d; bestId = e.Id; }
            }
            return best;
        }

        private static void ApplyDamage(SimCore sim, Entity attacker, Entity defender)
        {
            float dmg = DamageCalculator.CalculateDamage(attacker, defender, sim.Game, false);
            if (defender.Shield > 0)
            {
                float abs = System.Math.Min(defender.Shield, dmg);
                defender.Shield -= abs; dmg -= abs;
            }
            if (dmg > 0f) defender.Hp -= dmg;
            if (defender.Hp <= 0f && defender.State != EntityState.Dead)
            { defender.State = EntityState.Dead; sim.RemoveEntity(defender.Id); }
        }
    }

    }
```
