### 3.2 ProductionSystem.cs(生产 API 权威,只读)
```csharp
using System.Collections.Generic;

namespace IronCurtain.Sim
{
    /// <summary>
    /// 生产系统:兵营 3 队列、人口校验、取消返还、出生点占用等待、自动行军。
    /// 纯 tick 驱动,无 UnityEngine 依赖。M6.5:出生点加椭圆/水域过滤。
    /// </summary>
    public static class ProductionSystem
    {
        /// <summary>浮点比较容差(防 0.1f 累积漂移)。</summary>
        private const float Epsilon = 1e-4f;

        /// <summary>出生点偏移表:兵营前方 4 格(避让建筑碰撞半径),行序 同排/上/下;列偏移按阵营方向翻倍。</summary>
        private static readonly GridPos[] SpawnOffsets =
        {
            new GridPos(0, 4),
            new GridPos(-1, 4),
            new GridPos(1, 4),
        };

        public static void Tick(SimCore sim, float dt)
        {
            // 快照遍历:出生会向 Entities 字典添加实体,禁止边枚举边修改
            var snapshot = new List<Entity>(sim.Entities.Values);

            foreach (var e in snapshot)
            {
                if (e.Kind != EntityKind.Building) continue;
                if (e.Building != BuildingId.Barracks) continue;
                var prod = e.Prod;
                if (prod == null) continue;

                for (int i = 0; i < prod.Slots.Length; i++)
                {
                    var slot = prod.Slots[i];

                    // 空闲槽开工:人口校验通过才从待产队列取出
                    if (slot.Current == null && slot.Pending.Count > 0)
                    {
                        var hero = slot.Pending.Peek();
                        int pop = sim.PopCostOf(hero);
                        if (sim.PopulationOf(e.Side) + pop <= sim.PopulationCapOf(e.Side))
                        {
                            slot.Current = hero;
                            slot.Progress = 0f;
                            slot.WaitingForSpawn = false;
                            slot.Pending.Dequeue();
                        }
                    }

                    if (slot.Current != null)
                    {
                        if (!slot.WaitingForSpawn && slot.Progress + dt < ProductionSeconds(sim, e.Side, slot.Current.Value) - Epsilon)
                        {
                            slot.Progress += dt;
                            continue;
                        }

                        // 完成或等待出生中:尝试出生
                        if (TrySpawnUnit(sim, e, slot.Current.Value, out var unit))
                        {
                            slot.Current = null;
                            slot.Progress = 0f;
                            slot.WaitingForSpawn = false;
                            sim.SetMoveTarget(unit.Id, sim.RallyPointOf(e.Side));
                        }
                        else
                        {
                            slot.WaitingForSpawn = true; // 出生点全占,下 tick 重试
                        }
                    }
                }
            }
        }

        private static double ProductionSeconds(SimCore sim, Side side, HeroId hero)
        {
            if (hero == HeroId.ZhenYue && sim.IsUpgraded(side, HeroId.ZhenYue))
                return sim.Game.Production.ZhenYueUpgradedSeconds; // 迅捷:7.5s
            int pop = sim.PopCostOf(hero);
            return pop <= 1 ? sim.Game.Production.Pop1Seconds
                            : sim.Game.Production.ZhenYueSeconds;
        }

        private static bool TrySpawnUnit(SimCore sim, Entity barracks, HeroId heroId, out Entity unit)
        {
            unit = null;
            int dir = barracks.Side == Side.Blue ? 1 : -1;

            var occupied = new HashSet<long>();
            foreach (var e in sim.Entities.Values)
                if (e.Kind == EntityKind.Building && !e.Rubbled)
                    occupied.Add(((long)e.Pos.Row << 32) | (uint)e.Pos.Col);

            for (int i = 0; i < SpawnOffsets.Length; i++)
            {
                int row = barracks.Pos.Row + SpawnOffsets[i].Row;
                int col = barracks.Pos.Col + dir * SpawnOffsets[i].Col;
                if (row < 0 || row >= LaneGrid.Rows || col < 0 || col >= LaneGrid.Cols) continue;
                var cand = new GridPos(row, col);
                if (!TerrainMap.IsInside(cand)) continue;                    // 椭圆外不可出生
                if (TerrainMap.IsWater(SimVec2.FromGrid(cand))) continue;    // 水域不可出生
                long key = ((long)row << 32) | (uint)col;
                if (occupied.Contains(key)) continue;

                // 连续坐标下出生点还要避开既有单位(0.5 米内)
                var center = SimVec2.FromGrid(cand);
                bool tooClose = false;
                foreach (var e in sim.Entities.Values)
                    if (e.Kind == EntityKind.Unit && SimVec2.Distance(e.PosF, center) < 0.5f)
                    { tooClose = true; break; }
                if (tooClose) continue;

                var c = sim.Game.Heroes[(int)heroId];
                var u = sim.SpawnEntity(barracks.Side, EntityKind.Unit, cand);
                u.Hero = heroId; u.MaxHp = c.MaxHp; u.Hp = c.MaxHp;
                u.Attack = c.Attack; u.AttackIntervalSec = (float)c.AttackIntervalSec;
                u.Range = c.Range; u.MoveSpeed = (float)c.MoveSpeed;
                if (c.Shield > 0) { u.Shield = c.Shield; u.ShieldMax = c.Shield; }
                unit = u;
                return true;
            }
            return false;
        }

        public static bool Enqueue(SimCore sim, int barracksId, HeroId hero, out string error)
        {
            error = null;
            if (!sim.Entities.TryGetValue(barracksId, out var b) ||
                b.Kind != EntityKind.Building || b.Building != BuildingId.Barracks)
            {
                error = "目标不是兵营";
                return false;
            }
            var prod = b.Prod;
            if (prod == null) { error = "兵营无生产集"; return false; }

            int best = 0;
            for (int i = 1; i < prod.Slots.Length; i++)
                if (prod.Slots[i].Pending.Count < prod.Slots[best].Pending.Count) best = i;
            prod.Slots[best].Pending.Enqueue(hero);
            return true;
        }

        public static bool CancelLast(SimCore sim, int barracksId, int slotIndex, out string error)
        {
            error = null;
            if (!sim.Entities.TryGetValue(barracksId, out var b) ||
                b.Kind != EntityKind.Building || b.Building != BuildingId.Barracks)
            {
                error = "目标不是兵营";
                return false;
            }
            var prod = b.Prod;
            if (prod == null) { error = "兵营无生产集"; return false; }
            if (slotIndex < 0 || slotIndex >= prod.Slots.Length) { error = "槽索引越界"; return false; }

            var slot = prod.Slots[slotIndex];
            if (slot.Pending.Count > 0)
            {
                var list = new List<HeroId>(slot.Pending);
                list.RemoveAt(list.Count - 1);
                slot.Pending.Clear();
                foreach (var h in list) slot.Pending.Enqueue(h);
                return true;
            }
            if (slot.Current != null)
            {
                slot.Current = null;
                slot.Progress = 0f;
                slot.WaitingForSpawn = false;
                return true;
            }
            error = "队列为空";
            return false;
        }

        public static void OnBarracksDestroyed(SimCore sim, Entity barracks)
        {
            if (barracks == null) return;
            if (barracks.Prod != null) barracks.Prod.Clear();
        }
    }
}
```

### 3.3 ProductionSet.cs(队列结构权威,只读)
```csharp
using System;
using System.Collections.Generic;

namespace IronCurtain.Sim
{
    public sealed class ProductionSet
    {
        public const int QueueCount = 3;

        public sealed class Slot
        {
            public HeroId? Current;
            public float Progress;

            /// <summary>生产已完成但出生点被占,等待腾位中(GL M 方案,核实后采纳)。</summary>
            public bool WaitingForSpawn;

            public readonly Queue<HeroId> Pending = new Queue<HeroId>();
        }

        public readonly Slot[] Slots;

        public ProductionSet()
        {
            Slots = new Slot[QueueCount];
            for (int i = 0; i < QueueCount; i++) Slots[i] = new Slot();
        }

        public bool HasWork
        {
            get
            {
                for (int i = 0; i < Slots.Length; i++)
                {
                    if (Slots[i].Current != null) return true;
                    if (Slots[i].Pending.Count > 0) return true;
                }
                return false;
            }
        }

        public int InProgressPop(Func<HeroId, int> popOf)
        {
            int sum = 0;
            for (int i = 0; i < Slots.Length; i++)
            {
                var cur = Slots[i].Current;
                if (cur.HasValue) sum += popOf(cur.Value);
            }
            return sum;
        }

        public void Clear()
        {
            for (int i = 0; i < Slots.Length; i++)
            {
                Slots[i].Current = null;
                Slots[i].Progress = 0f;
                Slots[i].WaitingForSpawn = false;
                Slots[i].Pending.Clear();
            }
        }
    }
}
```

```csharp
using IronCurtain.Sim;
using UnityEngine;
using UnityEngine.Rendering;
using UnityEngine.Rendering.Universal;

namespace IronCurtain.View
{
    public sealed class GameBootstrap : MonoBehaviour
    {
        public SimCore Sim;
        public BoardView Board;
        public Camera Cam;
        public bool Paused;
        public float Speed = 1f;

        private double _accumulator;

        private void Awake()
        {
            if (Sim != null) return; // 域重载防护:已初始化则跳过
            var cfg = GameConfigParser.FromJson(Resources.Load<TextAsset>("gameconfig").text);
            Sim = MatchSetup.Build(cfg, null, AiTier.Normal);

            CreateCamera();
            CreateLighting();
            CreatePostProcess();

            var boardGo = new GameObject("Board");
            boardGo.transform.SetParent(transform, false);
            Board = boardGo.AddComponent<BoardView>();
            Board.Init(Sim, transform);
            Sim.AddEventListener(Board.OnSimEvent);
            gameObject.AddComponent<CaptureRunner>();

            Debug.Log("[GameBootstrap] Match started");
        }

        private void Update()
        {
            if (Paused || Sim == null) return;
            double interval = 1.0 / Sim.Config.TickRate;
            _accumulator += Time.deltaTime * Speed;
            while (_accumulator >= interval)
            {
                _accumulator -= interval;
                Sim.Step();
            }
            Board.TickViews();
        }

        private void OnDestroy()
        {
            if (Sim != null) Sim.ClearEventListeners();
        }

        private void CreateCamera()
        {
            var go = new GameObject("MainCamera");
            go.transform.SetParent(transform, false);
            go.tag = "MainCamera";
            Cam = go.AddComponent<Camera>();
            Cam.clearFlags = CameraClearFlags.SolidColor;
            Cam.backgroundColor = new Color32(0x79, 0xb5, 0xea, 0xff); // 明亮天空蓝(部落冲突式明快)
            Cam.nearClipPlane = 0.3f;
            Cam.farClipPlane = 300f;
            Cam.orthographic = true;
            Cam.orthographicSize = 27f;
            go.transform.position = new Vector3(143f, 55f, -50f);
            go.transform.LookAt(new Vector3(143f, 0f, -12f));
            var data = go.AddComponent<UniversalAdditionalCameraData>();
            data.renderPostProcessing = true;
            go.AddComponent<AudioListener>();
        }

        private void CreateLighting()
        {
            var go = new GameObject("Directional Light");
            go.transform.SetParent(transform, false);
            var light = go.AddComponent<Light>();
            light.type = LightType.Directional;
            light.color = new Color(1f, 0.98f, 0.93f);
            light.intensity = 1.05f;
            go.transform.rotation = Quaternion.Euler(55f, -35f, 0f);
            RenderSettings.ambientMode = UnityEngine.Rendering.AmbientMode.Flat;
            RenderSettings.ambientLight = new Color(0.55f, 0.56f, 0.58f, 1f);
        }

        private void CreatePostProcess()
        {
            var profile = ScriptableObject.CreateInstance<VolumeProfile>();
            var bloom = profile.Add<Bloom>(true);
            bloom.threshold.Override(0.55f);
            bloom.intensity.Override(1.7f);
            bloom.scatter.Override(0.7f);
            var tone = profile.Add<Tonemapping>(true);
```

```csharp
using System.Collections.Generic;
using NUnit.Framework;
using IronCurtain.Sim;
using UnityEngine;

namespace IronCurtain.Tests
{
    public class ProductionTests
    {
        private static GameConfig _cfg;
        private SimCore sim;

        [SetUp]
        public void SetUp()
        {
            if (_cfg == null)
                _cfg = GameConfigParser.FromJson(Resources.Load<TextAsset>("gameconfig").text);
            sim = new SimCore(new SimConfig(), new SeededRng(1));
            sim.Game = _cfg;
        }

        private Entity MakeBarracks(Side side, int row, int col)
        {
            var b = sim.SpawnBuilding(side, BuildingId.Barracks, new GridPos(row, col));
            b.Prod = new ProductionSet();
            return b;
        }

        private void Step(int ticks)
        {
            for (int i = 0; i < ticks; i++) sim.Step();
        }

        private int HeroUnitsOf(Side side)
        {
            int n = 0;
            foreach (var e in sim.Entities.Values)
                if (e.Kind == EntityKind.Unit && e.Side == side && e.Hero != null) n++;
            return n;
        }

        [Test]
        public void T1_SingleBarracks_Cadence()
        {
            var b = MakeBarracks(Side.Blue, 15, 110);
            for (int i = 0; i < 5; i++)
                Assert.IsTrue(ProductionSystem.Enqueue(sim, b.Id, HeroId.PanShi, out _), "入队应成功");
            // 5 只按"最少待产项"分配:槽0×2、槽1×2、槽2×1 → 三队列并行
            Step(59);
            Assert.AreEqual(0, HeroUnitsOf(Side.Blue), "tick59 不应有单位产出");
            Step(1);
            Assert.AreEqual(3, HeroUnitsOf(Side.Blue), "tick60 三队列并行各出第 1 只");
            Step(60);
            Assert.AreEqual(5, HeroUnitsOf(Side.Blue), "tick120 前 5 只应全部产出");
        }

        [Test]
        public void T2_ZhenYue_Production()
        {
            var b = MakeBarracks(Side.Blue, 15, 110);
            Assert.IsTrue(ProductionSystem.Enqueue(sim, b.Id, HeroId.ZhenYue, out _), "镇岳入队应成功");
            Step(1);
            Assert.AreEqual(2, sim.PopulationOf(Side.Blue), "进行中镇岳应占用 2 人口");
            Step(88);
            Assert.AreEqual(0, HeroUnitsOf(Side.Blue), "tick89 前不应产出");
            Step(1);
            Assert.AreEqual(1, HeroUnitsOf(Side.Blue), "90 tick 后镇岳应产出");
        }

        [Test]
        public void T3_ThreeQueues_Parallel()

                Assert.IsTrue(ProductionSystem.Enqueue(sim, b.Id, HeroId.PanShi, out _), "入队应成功");
            Step(60);
            Assert.AreEqual(3, HeroUnitsOf(Side.Blue), "三队列应并行各出 1 只");
        }

        [Test]
        public void T4_CancelLast_QueueAndInProgress()
        {
            var b = MakeBarracks(Side.Blue, 15, 110);
            Assert.IsTrue(ProductionSystem.Enqueue(sim, b.Id, HeroId.PanShi, out _));
            Assert.IsTrue(ProductionSystem.Enqueue(sim, b.Id, HeroId.DuanFeng, out _));
            // 最少待产项分配:磐石→槽0,断峰→槽1
            Assert.AreEqual(1, b.Prod.Slots[0].Pending.Count, "磐石应入槽0");
            Assert.AreEqual(1, b.Prod.Slots[1].Pending.Count, "断峰应入槽1");

            Assert.IsTrue(ProductionSystem.CancelLast(sim, b.Id, 0, out _), "取消队尾应成功");
            Assert.AreEqual(0, b.Prod.Slots[0].Pending.Count, "取消后槽0应为空");
            Assert.IsNull(b.Prod.Slots[0].Current);

            Step(1); // 槽1 开工断峰(进行中)
            Assert.IsNotNull(b.Prod.Slots[1].Current, "断峰应已开工");
            Assert.IsTrue(ProductionSystem.CancelLast(sim, b.Id, 1, out _), "取消进行中应成功");
            Assert.IsNull(b.Prod.Slots[1].Current, "取消后 Current 应为空");

            Step(60);
            Assert.AreEqual(0, HeroUnitsOf(Side.Blue), "全部取消后不应有任何产出");
            Assert.AreEqual(0, sim.PopulationOf(Side.Blue), "人口应全部释放");

            string err;
            Assert.IsFalse(ProductionSystem.CancelLast(sim, b.Id, 0, out err), "空队列取消应失败");
            Assert.AreEqual("队列为空", err);
        }
```
