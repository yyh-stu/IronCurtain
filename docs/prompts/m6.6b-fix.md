# M6.6b 修复轮 —— 地形/建筑视觉修正(hy-vision 视觉验收 FAIL 修复)

## 一、角色与任务
你是《铁幕:前线》Unity 6(URP 17)表现层主程。M6.6 首轮视觉验收 FAIL(问题清单见 §2),本轮修复。**只允许输出以下 2 个文件,其余任何文件零改动:**
- `Assets/Scripts/View/TerrainView.cs`
- `Assets/Scripts/View/EntityView.cs`

## 二、问题与修复规格(权威,精确到数值,逐条落实)
1. **草地呈对角条纹**(线性 hash `(row*31+col*17)&0xFF<38` 必然形成斜纹,验收视觉读成"绿色锯齿路径")。把 TerrainView.cs `BuildGroundChunks` 中:
   `bool isGrass = TerrainMap.IsIsland(v) || ((row * 31 + col * 17) & 0xFF) < 38;`
   改为散点 hash(约 15% 散点):
   `bool isGrass = TerrainMap.IsIsland(v) || (unchecked((uint)(row * 73856093 ^ col * 19349663)) & 0xFFFF) < 9830;`
2. **河面太黑**(smoothness 0.92 镜面反射发黑,验收视觉读成"黑色通道")。`BuildWater` 材质从
   `ViewAssets.Flat(new Color32(0x2f, 0x6f, 0xb3, 0xff), 0.92f)`
   改为带弱自发光、降 smoothness:
   `ViewAssets.Mat(new Color32(0x3f, 0x8f, 0xd6, 0xff), 0f, 0.6f, new Color(0.05f, 0.14f, 0.28f), 0.5f)`
3. **虚空盒太小**(相机视野外露出天空蓝,验收视觉读成"椭圆外浅蓝背景")。`BuildVoid` 的 DecoBox 尺寸从 `(300f, 0.6f, 34f)` 改为 `(420f, 0.6f, 160f)`,中心保持 (143f, -0.4f, -12f)。
4. **建筑不够高大、单位偏小、光环盘过亮**(高相机下验收视觉读成"单层小屋/空白框")。EntityView.cs 数值修订表(其余数值不动):
   - Base:AddPiece 件高 `3.0f,0f` / `3.2f,3.0f` / `2.8f,6.2f` → **`3.6f,0f` / `4.2f,3.6f` / `3.6f,7.8f`**(总 11.4);`_topY = 9.0f`→**11.4f**;`AddGlowDisc(_model.transform, 6.0f)`→**7.2f**。
   - Barracks:`3.8f,0f` / `1.4f,3.8f` → **`4.4f,0f` / `1.8f,4.4f`**(总 6.2);`_topY = 5.2f`→**6.2f**;glow `4.4f`→**5.0f**;`AddFlag(..., new Vector3(3.2f, 0f, 3.2f), 3.2f)` → **`new Vector3(3.6f, 0f, 3.6f), 3.6f`**。
   - Tower:`2.0f,0f` / `3.2f,2.0f` / `1.6f,5.2f` → **`2.6f,0f` / `4.0f,2.6f` / `2.0f,6.6f`**(总 8.6);`_topY = 6.8f`→**8.6f**;glow `4.4f`→**5.6f**。
   - ResourcePoint 非堡垒:`5.6f, 0f`→**`7.0f, 0f`**;`_topY = 5.6f`→**7.0f**;glow `3.6f`→**4.4f**。
   - 堡垒 stall:`3.2f, 0f`→**`4.0f, 0f`**;`_topY = 3.6f`→**4.4f**;顶部光点 `1.4f, 3.9f`→**`1.7f, 4.7f`**、尺寸 `0.44f`→**`0.5f`**。
   - 通用旗帜(第 300 行附近 `if (!fort) AddFlag`):pos `(5.2f, 0f, ...)`→**`(6.0f, 0f, ...)`**、高 `4.4f`→**`5.0f`**。
   - SetRubbled:主件 `3.2f, 0f`→**`4.0f, 0f`**;rocks-small `0.6f`→**`0.7f`**;散件半径 `2.2f`→**`2.6f`**。
   - 单位:BuildUnit `AddPiece(_model.transform, HeroModelName(hero), 2.0f, 0f)`→**`2.4f, 0f`**;`AddGlowDisc(_model.transform, 1.0f)`→**`1.15f`**。
   - 光环盘减亮:AddGlowDisc 内 `ViewAssets.Mat(TeamGlowColor() * 0.32f, 0f, 0.7f, TeamGlowColor(), 1.0f)` 末参数 1.0f→**0.55f**。
5. 其余一律不动:驻军光点(2.3f/0.4f)保持、血条、桥/栏杆、相机、Sim、环境撒布、分块结构全不动。

### 3.1 TerrainView.cs 现状全文
```csharp
using System.Collections.Generic;
using IronCurtain.Sim;
using UnityEngine;

namespace IronCurtain.View
{
    /// <summary>M6.6 椭圆大地图地形渲染:石路+草地/河/桥/虚空 + 环境撒布。</summary>
    public static class TerrainView
    {
        private static readonly int[] BuildingCols = { 15, 46, 77, 108, 143, 178, 209, 240, 271 };

        public static void Build(Transform parent)
        {
            var terrainGo = new GameObject("Terrain");
            terrainGo.transform.SetParent(parent, false);
            Transform t = terrainGo.transform;

            BuildVoid(t);
            BuildGroundChunks(t);
            BuildWater(t);
            BuildBridges(t);
            BuildEnvironment(t);
        }

        private static void BuildVoid(Transform t)
        {
            Material voidMat = ViewAssets.Flat(new Color32(0x0a, 0x0c, 0x12, 0xff), 0.05f);
            ViewAssets.DecoBox(voidMat, new Vector3(143f, -0.4f, -12f), new Vector3(300f, 0.6f, 34f), t, "Void");
        }

        private static void BuildGroundChunks(Transform t)
        {
            Material stoneDark = ViewAssets.Flat(new Color32(0x78, 0x7a, 0x7e, 0xff), 0.35f);
            Material stoneLight = ViewAssets.Flat(new Color32(0x86, 0x88, 0x8d, 0xff), 0.35f);
            Material grass = ViewAssets.Flat(new Color32(0x5f, 0x81, 0x49, 0xff), 0.12f);
            var mats = new[] { stoneDark, stoneLight, grass };

            for (int c = 0; c < 5; c++)
            {
                int colMin = c * 60;
                int colMax = Mathf.Min(colMin + 60, TerrainMap.Cols) - 1;
                var dv = new List<Vector3>(); var dt = new List<int>();
                var lv = new List<Vector3>(); var lt = new List<int>();
                var gv = new List<Vector3>(); var gt = new List<int>();
                for (int row = 0; row < TerrainMap.Rows; row++)
                {
                    for (int col = colMin; col <= colMax; col++)
                    {
                        var v = new SimVec2(col + 0.5f, row + 0.5f);
                        if (!TerrainMap.IsInside(v) || TerrainMap.IsWater(v) || TerrainMap.IsBridge(v)) continue;
                        var center = new Vector3(col + 0.5f, -0.05f, 3f - row - 0.5f);
                        var size = new Vector3(1f, 0.1f, 1f);
                        bool isGrass = TerrainMap.IsIsland(v) || ((row * 31 + col * 17) & 0xFF) < 38;
                        if (isGrass) ViewAssets.AddBox(gv, gt, center, size);
                        else if (((row + col) & 1) == 0) ViewAssets.AddBox(dv, dt, center, size);
                        else ViewAssets.AddBox(lv, lt, center, size);
                    }
                }
                var mesh = ViewAssets.CreateMesh(new List<Vector3>[] { dv, lv, gv }, new List<int>[] { dt, lt, gt });
                ViewAssets.AttachMesh("Ground_" + c, t, mesh, mats);
            }
        }

        private static void BuildWater(Transform t)
        {
            var wv = new List<Vector3>(); var wt = new List<int>();
            for (int row = 0; row < TerrainMap.Rows; row++)
                for (int col = 0; col < TerrainMap.Cols; col++)
                {
                    var v = new SimVec2(col + 0.5f, row + 0.5f);
                    if (!TerrainMap.IsWater(v)) continue;
                    ViewAssets.AddBox(wv, wt, new Vector3(col + 0.5f, -0.17f, 3f - row - 0.5f), new Vector3(1f, 0.1f, 1f));
                }
            var mesh = ViewAssets.CreateMesh(new List<Vector3>[] { wv }, new List<int>[] { wt });
            ViewAssets.AttachMesh("River", t, mesh,
                new Material[] { ViewAssets.Flat(new Color32(0x2f, 0x6f, 0xb3, 0xff), 0.92f) });
        }

        private static void BuildBridges(Transform t)
        {
            var bv = new List<Vector3>(); var bt = new List<int>();
            for (int row = 0; row < TerrainMap.Rows; row++)
                for (int col = 0; col < TerrainMap.Cols; col++)
                {
                    var v = new SimVec2(col + 0.5f, row + 0.5f);
                    if (!TerrainMap.IsBridge(v)) continue;
                    ViewAssets.AddBox(bv, bt, new Vector3(col + 0.5f, 0.01f, 3f - row - 0.5f), new Vector3(1f, 0.1f, 1f));
                }
            var mesh = ViewAssets.CreateMesh(new List<Vector3>[] { bv }, new List<int>[] { bt });
            ViewAssets.AttachMesh("Bridges", t, mesh,
                new Material[] { ViewAssets.Flat(new Color32(0x9a, 0x7a, 0x52, 0xff), 0.3f) });

            Material railMat = ViewAssets.Flat(new Color32(0x6f, 0x56, 0x38, 0xff), 0.4f);
            // 两座桥:桥面 x ∈ [138,148),长 10;两侧栏杆截面 0.12×0.5,y 0.35
            AddBridgeRails(railMat, t, 0.5f);  // 下桥 Y∈[0,5),桥面中心 z=0.5
            AddBridgeRails(railMat, t, -24.5f); // 上桥 Y∈[25,30)
        }

        private static void AddBridgeRails(Material m, Transform t, float centerZ)
        {
            ViewAssets.DecoBox(m, new Vector3(143f, 0.35f, centerZ + 2.5f), new Vector3(10f, 0.5f, 0.12f), t, "BridgeRail_N");
            ViewAssets.DecoBox(m, new Vector3(143f, 0.35f, centerZ - 2.5f), new Vector3(10f, 0.5f, 0.12f), t, "BridgeRail_S");
        }

        private static void BuildEnvironment(Transform t)
        {
            var envGo = new GameObject("Environment");
            envGo.transform.SetParent(t, false);
            Transform env = envGo.transform;

            var items = new List<KeyValuePair<string, float>>(26);
            AddItems(items, "tree-large", 6, 4.4f);
            AddItems(items, "tree-small", 8, 2.2f);
            AddItems(items, "rocks-large", 4, 1.4f);
            AddItems(items, "rocks-small", 8, 0.7f);

            var rng = new System.Random(1234);
            bool warned = false;
            foreach (var it in items)
            {
                GameObject prefab = Resources.Load<GameObject>("Models/" + it.Key);
                if (prefab == null)
                {
                    if (!warned) { Debug.LogWarning("[TerrainView] Missing model: " + it.Key); warned = true; }
                    continue;
                }
                Vector3 pos;
                if (!FindFreeCell(rng, out pos)) continue;
                var go = Object.Instantiate(prefab, env);
                go.name = it.Key;
                var b = new Bounds(); bool found = false;
                foreach (var rr in go.GetComponentsInChildren<Renderer>())
                {
                    if (!found) { b = rr.bounds; found = true; }
                    else b.Encapsulate(rr.bounds);
                }
                if (found && b.size.y > 0.0001f) go.transform.localScale = Vector3.one * (it.Value / b.size.y);
                go.transform.position = pos;
                go.transform.rotation = Quaternion.Euler(0f, (float)rng.NextDouble() * 360f, 0f);
            }
        }

        private static void AddItems(List<KeyValuePair<string, float>> list, string name, int count, float h)
        {
            for (int i = 0; i < count; i++) list.Add(new KeyValuePair<string, float>(name, h));
        }

        private static bool FindFreeCell(System.Random rng, out Vector3 pos)
        {
            pos = Vector3.zero;
            for (int attempt = 0; attempt < 40; attempt++)
            {
                int row = rng.Next(TerrainMap.Rows);
                int col = rng.Next(TerrainMap.Cols);
                var v = new SimVec2(col + 0.5f, row + 0.5f);
                if (!TerrainMap.IsInside(v)) continue;
                if (TerrainMap.IsWater(v) || TerrainMap.IsBridge(v) || TerrainMap.IsIsland(v)) continue;
                bool near = false;
                foreach (int bc in BuildingCols)
                    if (Mathf.Abs(col - bc) < 5 && Mathf.Abs(row - 15) < 5) { near = true; break; }
                if (near) continue;
                pos = new Vector3(col + 0.5f, 0f, 3f - row - 0.5f);
                return true;
            }
            return false;
        }
    }
}
```

### 3.2 EntityView.cs 现状全文
```csharp
using System.Collections.Generic;
using IronCurtain.Sim;
using UnityEngine;

namespace IronCurtain.View
{
    public sealed class EntityView : MonoBehaviour
    {
        private Entity _e;
        private Side _side;
        private GameObject _model, _gr, _chunks;
        private Transform _hpRoot, _hpFill, _shFill;
        private float _w = 1.7f, _h = 0.26f, _deadT = -1f, _topY = 3.5f;
        private bool _dead, _rubbled;
        private int _gf = -1, _gt = -1;
        private Material _g, _y, _r, _bk, _sh, _rubG, _gA, _gB;

        public void Bind(Entity e, Side side, HeroId? hero, BuildingId? building)
        {
            _e = e; _side = side;
            InitMats();
            if (e.Kind == EntityKind.Unit) BuildUnit(hero.HasValue ? hero.Value : HeroId.PanShi);
            else BuildBuilding(building.HasValue ? building.Value : BuildingId.Tower);
            BuildHpBar();
            transform.position = e.Kind == EntityKind.Building ? BoardView.WorldPos(e.Pos) : BoardView.WorldPos(e.PosF);
            if (e.Kind == EntityKind.Unit)
                transform.rotation = Quaternion.Euler(0f, side == Side.Blue ? 90f : -90f, 0f);
        }

        private void InitMats()
        {
            if (_g != null) return;
            _bk = ViewAssets.Flat(new Color(0.18f, 0.18f, 0.20f));
            _g = ViewAssets.Mat(Color.green, 0f, 0.4f, Color.green, 0.8f);
            _y = ViewAssets.Mat(new Color(1f, 0.85f, 0f), 0f, 0.4f, new Color(1f, 0.85f, 0f), 0.8f);
            _r = ViewAssets.Mat(Color.red, 0f, 0.4f, Color.red, 0.8f);
            _sh = ViewAssets.Mat(new Color(0.4f, 0.8f, 1f), 0f, 0.4f, new Color(0.4f, 0.8f, 1f), 1.2f);
            _rubG = ViewAssets.Flat(new Color(0.15f, 0.14f, 0.13f));
            _gA = ViewAssets.Mat(Color.black, 0f, 0.9f, new Color(1f, 1f, 0.9f), 2.2f);
            _gB = ViewAssets.Mat(Color.black, 0f, 0.9f, new Color(0.4f, 0.4f, 0.4f), 0.7f);
        }

        public void FlashHit() { }

        public void SetDead()
        {
            if (_dead) return;
            _dead = true; _deadT = 0f;
        }

        public void SetHp(float hp, float maxHp)
        {
            if (_hpFill == null) return;
            float t = maxHp > 0f ? Mathf.Clamp01(hp / maxHp) : 0f;
            bool show = _e.Kind == EntityKind.Building || t < 0.999f;
            _hpRoot.gameObject.SetActive(show);
            _hpFill.localScale = new Vector3(_w * t, _h, 1f);
            _hpFill.localPosition = new Vector3(-_w * (1f - t) * 0.5f, 0f, 0f);
            _hpFill.GetComponent<MeshRenderer>().sharedMaterial = t > 0.5f ? _g : (t > 0.25f ? _y : _r);
        }

        public void TickBillboard(Camera cam)
        {
            if (_hpRoot != null && cam != null)
                _hpRoot.rotation = Quaternion.LookRotation(_hpRoot.position - cam.transform.position);
        }

        public void Tick(float dt, Camera cam)
        {
            if (_e == null) return;
            bool isB = _e.Kind == EntityKind.Building;
            bool hidden = !isB && _e.GarrisonBuildingId != 0;
            bool vis = !hidden && !_rubbled;
            if (_model != null && _model.activeSelf != vis) { _model.SetActive(vis); _hpRoot.gameObject.SetActive(!hidden); }
            if (!hidden)
            {
                Vector3 target = isB ? BoardView.WorldPos(_e.Pos) : BoardView.WorldPos(_e.PosF);
                if (isB) transform.position = target;
                else transform.position = Vector3.Lerp(transform.position, target, 1f - Mathf.Exp(-14f * dt));
                SetHp(_e.Hp, _e.MaxHp);
            }
            if (_e.Rubbled && !_rubbled) SetRubbled(true);
            if (!_e.Rubbled && _rubbled) SetRubbled(false);
            if (isB) UpdateGarrison();
            UpdateShield();
            TickBillboard(cam);
            if (_dead)
            {
                _deadT += dt;
                float s = Mathf.Max(0.01f, 1f - _deadT * 4f);
                transform.localScale = Vector3.one * s;
            }
            else transform.localScale = Vector3.one;
        }

        private void UpdateGarrison()
        {
            int f = _e.GarrisonFormal.Count, t = _e.GarrisonTemp.Count;
            if (f == t && f == _gf && t == _gt) return;
            _gf = f; _gt = t;
            if (_gr != null) Destroy(_gr);
            _gr = null;
            if (f + t == 0) return;
            _gr = new GameObject("Garrison");
            _gr.transform.SetParent(transform, false);
            int n = f + t;
            for (int i = 0; i < n; i++)
            {
                float a = i / (float)Mathf.Max(1, n) * Mathf.PI * 2f;
                ViewAssets.DecoBox(i < f ? _gA : _gB,
                    new Vector3(Mathf.Cos(a) * 2.3f, _topY, Mathf.Sin(a) * 2.3f),
                    Vector3.one * 0.4f, _gr.transform, "d" + i);
            }
        }

        private void UpdateShield()
        {
            if (_shFill == null) return;
            bool show = _e.Shield > 0.01f && _e.ShieldMax > 0.01f;
            _shFill.gameObject.SetActive(show);
            if (show)
            {
                float t = Mathf.Clamp01(_e.Shield / _e.ShieldMax);
                _shFill.localScale = new Vector3(_w * t, 0.05f, 1f);
                _shFill.localPosition = new Vector3(-_w * (1f - t) * 0.5f, 0.14f, 0f);
            }
        }

        private void BuildHpBar()
        {
            var root = new GameObject("HpBar");
            root.transform.SetParent(transform, false);
            root.transform.localPosition = new Vector3(0f, _e.Kind == EntityKind.Building ? _topY + 0.9f : 2.7f, 0f);
            _hpRoot = root.transform;
            ViewAssets.DecoQuad(_bk, Vector3.zero, new Vector2(_w, _h), root.transform, "Back");
            _hpFill = ViewAssets.DecoQuad(_g, Vector3.zero, new Vector2(_w, _h), root.transform, "Fill").transform;
            _shFill = ViewAssets.DecoQuad(_sh, new Vector3(0f, 0.14f, 0f), new Vector2(_w, 0.05f), root.transform, "Shield").transform;
            _shFill.gameObject.SetActive(false);
        }

        // ===== Kenney 模型加载与组合 =====

        private GameObject NewModelRoot()
        {
            var go = new GameObject("Model");
            go.transform.SetParent(transform, false);
            return go;
        }

        private float PieceHeight(GameObject go)
        {
            bool found = false; var b = new Bounds();
            foreach (var rr in go.GetComponentsInChildren<Renderer>())
            {
                if (!found) { b = rr.bounds; found = true; }
                else b.Encapsulate(rr.bounds);
            }
            return found ? b.size.y : 0f;
        }

        private GameObject AddPiece(Transform parent, string name, float targetH, float y)
        {
            var prefab = Resources.Load<GameObject>("Models/" + name);
            if (prefab == null) { Debug.LogWarning("[EntityView] Missing model: " + name); return null; }
            var go = Instantiate(prefab, parent);
            go.name = name;
            float ph = PieceHeight(go);
            float s = ph > 0.0001f ? targetH / ph : 1f;
            go.transform.localScale = Vector3.one * s;
            go.transform.localPosition = new Vector3(0f, y + targetH * 0.5f, 0f);
            go.transform.localRotation = Quaternion.identity;
            return go;
        }

        private Color TeamGlowColor()
        {
            if (_side == Side.Blue) return new Color(0.4f, 0.8f, 1f);
            if (_side == Side.Red) return new Color(1f, 0.4f, 0.35f);
            return new Color(0.35f, 1f, 0.55f);
        }

        private void AddGlowDisc(Transform parent, float radius)
        {
            Material glow = ViewAssets.Mat(TeamGlowColor() * 0.32f, 0f, 0.7f, TeamGlowColor(), 1.0f);
            ViewAssets.DecoBox(glow, new Vector3(0f, 0.02f, 0f), new Vector3(radius * 2f, 0.025f, radius * 2f), parent, "GlowDisc");
        }

        private void AddFlag(Transform parent, Vector3 pos, float height)
        {
            Color fc = _side == Side.Blue ? new Color(0.2f, 0.85f, 1f)
                : (_side == Side.Red ? new Color(1f, 0.3f, 0.25f) : new Color(0.5f, 0.75f, 0.5f));
            Material pm = ViewAssets.Flat(new Color(0.25f, 0.22f, 0.18f));
            Material fm = ViewAssets.Flat(fc);
            ViewAssets.DecoBox(pm, pos + new Vector3(0f, height * 0.5f, 0f), new Vector3(0.06f, height, 0.06f), parent, "FlagPole");
            ViewAssets.DecoBox(fm, pos + new Vector3(0.32f, height - 0.22f, 0f), new Vector3(0.58f, 0.36f, 0.04f), parent, "Flag");
        }

        private void TintRenderers(GameObject go, Color tint)
        {
            foreach (var rr in go.GetComponentsInChildren<Renderer>())
            {
                var mats = rr.sharedMaterials;
                for (int i = 0; i < mats.Length; i++)
                {
                    if (mats[i] == null) continue;
                    var m = Instantiate(mats[i]);
                    if (m.HasProperty("_BaseColor")) m.SetColor("_BaseColor", m.GetColor("_BaseColor") * tint);
                    else m.color = m.color * tint;
                    mats[i] = m;
                }
                rr.sharedMaterials = mats;
            }
        }

        private void GreyRenderers(GameObject go)
        {
            foreach (var rr in go.GetComponentsInChildren<Renderer>())
            {
                var mats = rr.sharedMaterials;
                for (int i = 0; i < mats.Length; i++)
                    if (mats[i] != null) mats[i] = Instantiate(ViewAssets.Flat(new Color(0.62f, 0.62f, 0.60f)));
                rr.sharedMaterials = mats;
            }
        }

        // ===== 单位:英雄 → Kenney 角色映射(后续替换机甲模型只改此处)=====

        private static string HeroModelName(HeroId hero)
        {
            switch (hero)
            {
                case HeroId.PanShi: return "character-a";
                case HeroId.ZhenYue: return "character-b";
                case HeroId.DuanFeng: return "character-c";
                case HeroId.ZhuFeng: return "character-d";
                case HeroId.ZhuiRi: return "character-e";
                case HeroId.XingLuo:
                default: return "character-f";
            }
        }

        private void BuildUnit(HeroId hero)
        {
            _model = NewModelRoot();
            var body = AddPiece(_model.transform, HeroModelName(hero), 2.0f, 0f);
            if (body != null)
                TintRenderers(body, _side == Side.Blue ? new Color(0.8f, 0.9f, 1f)
                    : (_side == Side.Red ? new Color(1f, 0.85f, 0.85f) : Color.white));
            AddGlowDisc(_model.transform, 1.0f);
        }

        private void BuildBuilding(BuildingId type)
        {
            _model = NewModelRoot();
            bool neutral = _side == Side.Neutral;
            bool fort = _e.IsFort && type == BuildingId.ResourcePoint;
            switch (type)
            {
                case BuildingId.Base:
                    AddPiece(_model.transform, "tower-square-base", 3.0f, 0f);
                    AddPiece(_model.transform, "tower-square-mid", 3.2f, 3.0f);
                    AddPiece(_model.transform, "tower-square-top-roof", 2.8f, 6.2f);
                    _topY = 9.0f; AddGlowDisc(_model.transform, 6.0f);
                    break;
                case BuildingId.Barracks:
                    AddPiece(_model.transform, "tower-square-mid-door", 3.8f, 0f);
                    AddPiece(_model.transform, "tower-square-roof", 1.4f, 3.8f);
                    _topY = 5.2f; AddGlowDisc(_model.transform, 4.4f);
                    AddFlag(_model.transform, new Vector3(3.2f, 0f, 3.2f), 3.2f);
                    break;
                case BuildingId.ResourcePoint:
                    if (fort)
                    {
                        var stall = AddPiece(_model.transform, "stall", 3.2f, 0f);
                        _topY = 3.6f;
                        Material gm = ViewAssets.Mat(Color.black, 0f, 0.9f, TeamGlowColor(), 2.2f);
                        for (int i = 0; i < 3; i++)
                        {
                            float a = i * Mathf.PI * 2f / 3f;
                            ViewAssets.DecoBox(gm, new Vector3(Mathf.Cos(a) * 1.4f, 3.9f, Mathf.Sin(a) * 1.4f),
                                Vector3.one * 0.44f, _model.transform, "fortglow" + i);
                        }
                        if (stall != null && neutral) GreyRenderers(stall);
                    }
                    else
                    {
                        AddPiece(_model.transform, neutral ? "watermill" : "windmill", 5.6f, 0f);
                        _topY = 5.6f;
                    }
                    AddGlowDisc(_model.transform, 3.6f);
                    break;
                case BuildingId.Tower:
                default:
                    AddPiece(_model.transform, "tower-hexagon-base", 2.0f, 0f);
                    AddPiece(_model.transform, "tower-hexagon-mid", 3.2f, 2.0f);
                    AddPiece(_model.transform, "tower-hexagon-roof", 1.6f, 5.2f);
                    _topY = 6.8f; AddGlowDisc(_model.transform, 4.4f);
                    break;
            }
            if (!fort) AddFlag(_model.transform, new Vector3(5.2f, 0f, type == BuildingId.Barracks ? -1.8f : 0f), 4.4f);
            if (neutral && type != BuildingId.ResourcePoint) GreyRenderers(_model);
            else if (!neutral)
                TintRenderers(_model, _side == Side.Blue ? new Color(0.82f, 0.88f, 1f) : new Color(1f, 0.78f, 0.72f)); // 建筑阵营 tint(红方暖、蓝方冷)
        }

        public void SetRubbled(bool on)
        {
            if (_rubbled == on || _model == null) return;
            _rubbled = on;
            if (_chunks != null) { Destroy(_chunks); _chunks = null; }
            _model.SetActive(!on);
            if (!on) return;
            _chunks = new GameObject("Rubble");
            _chunks.transform.SetParent(transform, false);
            string main = _e.Building == BuildingId.ResourcePoint ? "cart" : "siege-tower-demolished";
            var go = AddPiece(_chunks.transform, main, 3.2f, 0f);
            if (go != null) go.transform.localScale *= 0.9f;
            for (int i = 0; i < 3; i++)
            {
                float a = i * 2.1f + 0.7f;
                var rk = AddPiece(_chunks.transform, "rocks-small", 0.6f, 0f);
                if (rk != null) rk.transform.localPosition = new Vector3(Mathf.Cos(a) * 2.2f, 0.2f, Mathf.Sin(a) * 2.2f);
            }
            if (_chunks.transform.childCount == 0) { Destroy(_chunks); _chunks = null; }
        }
    }
```
