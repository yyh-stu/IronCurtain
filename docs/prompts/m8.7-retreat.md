# M8.7 撤退 13c 落地(Sim 撤退命令 + 溃逃系数统一 −25% + 防拉回 + View 撤退按钮)

## 一、需求规格(用户 2026-09-06 拍板 13c,权威)

1. **撤退触发**:玩家点选主将或框选多兵团后,屏幕出现撤退 UI 按钮,点击按钮 → 对每个选中主将执行撤退命令(Sim 侧提供命令入口)。
2. **撤退减益**:撤退期间血量与攻击力 −25%。**结算口径(已拍板,乘区近似)**:输出 −25%、承伤 +25%,与现有溃逃(Routing)伤害乘区同构。**现有溃逃系数 −0.3/+0.3 全部统一改为 −0.25/+0.25**(所有撤退/溃逃同理)。
3. **撤退目标**:最近己方非废墟据点(欧氏距离最近的 `Side==自己` 非 `Rubbled` 建筑),行至据点附近站定。
4. **兵团原子**:主将撤退 = 全兵团撤退——同兵团(GeneralId 指向该主将)所有非驻军活士兵同步进入撤退态(士兵 State=Routing,随阵型跟随主将移动)。
5. **防拉回(13b 边界)**:撤退期间与撤退到达后,自动行为层(RegimentBehaviorSystem)不得再把该兵团拉回战场(不下索敌/支援指令);**直到玩家下一次手动命令(manual SetMoveTarget)才恢复自动层接管**。被迫接战(敌人进入射程自动还击)保留,无需新逻辑。
6. **士兵"永不溃逃"**指城易主/废墟等自动溃逃场景;玩家授意撤退不受影响。
7. 撤退中的主将:走 Routing 状态机(只被迫接战——索敌半径 1.4m 贴身还击,不追击、不可被自动层调遣);玩家手动 SetMoveTarget 可随时打断撤退(State 回 Moving)。

### 5. RetreatSystem.cs(新增,全部逻辑在此)
```csharp
using System;

namespace IronCurtain.Sim
{
    /// <summary>玩家授意撤退(M8.7 13c):兵团原子撤退至最近己方非废墟据点附近站定;
    /// 撤退/溃逃期间输出 −25%、承伤 +25%(SimCore 乘区);到达后 RetreatHold 压制自动层,玩家手动命令解除。</summary>
    public static class RetreatSystem
    {
        public const float StandoffOffset = 1.2f; // 站定点=建筑边缘外 1.2 米

        /// <summary>下令撤退。id 不存在抛 InvalidOperationException(与 SetMoveTarget 同风格);
        /// 士兵/死亡/驻军中/交换中/无非废墟己方建筑 → return false(不动状态)。</summary>
        public static bool OrderRetreat(SimCore sim, int generalId)
        {
            if (!sim.Entities.TryGetValue(generalId, out var g))
                throw new InvalidOperationException($"实体不存在: {generalId}");
            if (g.Kind != EntityKind.Unit || g.IsSoldier || g.State == EntityState.Dead) return false;
            if (g.GarrisonBuildingId != 0 || g.State == EntityState.Exchanging) return false;

            Entity dest = null; float bestD = float.MaxValue;
            foreach (var b in sim.Entities.Values)
            {
                if (b.Kind != EntityKind.Building || b.Side != g.Side || b.Rubbled) continue;
                float d = SimVec2.Distance(g.PosF, b.PosF);
                if (d < bestD - 1e-4f) { dest = b; bestD = d; }
            }
            if (dest == null) return false;

            var dir = g.PosF - dest.PosF;
            if (dir.Magnitude < 1e-4f) dir = g.Side == Side.Blue ? new SimVec2(-1f, 0f) : new SimVec2(1f, 0f);
            dir = dir.Normalized();
            float half = System.Math.Max(BuildingFootprint.HalfW(dest), BuildingFootprint.HalfH(dest));
            SimVec2 stand = TerrainMap.ClampToEllipse(dest.PosF + dir * (half + StandoffOffset));

            g.State = EntityState.Routing;
            g.HasMoveTarget = true;
            g.MoveTarget = stand.ToGrid();
            g.Detour = null; g.DetourTicks = 0; g.StuckTicks = 0;
            g.AutoKind = 0; g.AutoTargetId = 0; // 撤退即玩家意志,清自动决策
            g.RetreatHold = true;

            foreach (var s in sim.Entities.Values)
                if (s.Kind == EntityKind.Unit && s.IsSoldier && s.GeneralId == g.Id
                    && s.State != EntityState.Dead && s.GarrisonBuildingId == 0)
                    s.State = EntityState.Routing; // 兵团原子:士兵同步撤退(减益+只被迫接战)
            return true;
        }
    }
}
```

```csharp
// ===== 现状全文(修改基准): Assets/Scripts/Sim/SimCore.cs =====
using System;
using System.Collections.Generic;

namespace IronCurtain.Sim
{
    public sealed class SimCore
    {
        public SimConfig Config { get; }
        public SeededRng Rng { get; }
        public long Tick { get; private set; }
        public int EntityCount => _entities.Count;
        public IReadOnlyDictionary<int, Entity> Entities => _entities;

        /// <summary>将名额上限(M8.4b 名额制)。测试版 4;正式版 8。</summary>
        public const int GeneralCap = 4;

        /// <summary>游戏数值配置(由调用方注入;null 时战斗模块不生效)。</summary>
        public GameConfig Game { get; set; }

        /// <summary>胜负:大本营被摧毁时置位。</summary>
        public Side? Winner { get; private set; }
        public void SetWinner(Side side) { Winner = side; }

        // 补兵池(M8.4a):按阵营的全局池
        public readonly Dictionary<Side, float> ReinforceAccum = new Dictionary<Side, float> { { Side.Blue, 0f }, { Side.Red, 0f } };
        public readonly Dictionary<Side, int> ReinforcePool = new Dictionary<Side, int> { { Side.Blue, 0 }, { Side.Red, 0 } };

        // 升级(M4):每方已购买的英雄被动
        private readonly Dictionary<Side, HashSet<HeroId>> _upgrades = new Dictionary<Side, HashSet<HeroId>>();

        public bool IsUpgraded(Side side, HeroId hero)
        {
            return _upgrades.TryGetValue(side, out var s) && s.Contains(hero);
        }

        /// <summary>2EP 购买升级,单英雄每局一次。</summary>
        public bool BuyUpgrade(Side side, HeroId hero, out string error)
        {
            error = null;
            if (IsUpgraded(side, hero)) { error = "该英雄已升级"; return false; }
            if (!SpendEp(side, Game.Ep.UpgradeCost)) { error = "EP 不足"; return false; }
            if (!_upgrades.TryGetValue(side, out var s))
            {
                s = new HashSet<HeroId>();
                _upgrades[side] = s;
            }
            s.Add(hero);
            return true;
        }

        // 经济(M2)
        private readonly Dictionary<Side, float> _ep = new Dictionary<Side, float>();
        private readonly Dictionary<Side, float> _epAccum = new Dictionary<Side, float>();
        private readonly Dictionary<Side, GridPos> _rally = new Dictionary<Side, GridPos>
        {
            { Side.Blue, new GridPos(15, 270) },
            { Side.Red, new GridPos(15, 16) },
        };

        private readonly Dictionary<int, Entity> _entities = new Dictionary<int, Entity>();
        private readonly HashSet<int> _occupied = new HashSet<int>();
        private readonly Queue<ICommand> _commandQueue = new Queue<ICommand>();
        private readonly List<Action<SimEvent>> _listeners = new List<Action<SimEvent>>();
        private readonly LaneGrid _grid = new LaneGrid();
        private int _nextId = 1;

        public SimCore(SimConfig config, SeededRng rng)
        {
            Config = config;
            Rng = rng;
        }

        public Entity SpawnEntity(Side side, EntityKind kind, GridPos pos)
        {
            if (!_grid.InBounds(pos))
                throw new InvalidOperationException($"位置越界: {pos}");
            int idx = _grid.ToIndex(pos);
            if (_occupied.Contains(idx))
                throw new InvalidOperationException($"格子已占用: {pos}");
            // 连续坐标下单位不再占格;出生防重叠用距离校验(驻军/城内单位不占位)
            if (kind == EntityKind.Unit)
            {
                var center = SimVec2.FromGrid(pos);
                foreach (var o in _entities.Values)
                    if (o.Kind == EntityKind.Unit && o.GarrisonBuildingId == 0
                        && o.State != EntityState.Dead
                        && SimVec2.Distance(o.PosF, center) < 0.5f)
                        throw new InvalidOperationException($"出生点过近: {pos}");
            }

            var e = new Entity(_nextId++, side, kind, pos, 0f);
            e.PosF = SimVec2.FromGrid(pos);
            _entities[e.Id] = e;
            if (kind == EntityKind.Building)
                _occupied.Add(idx);
            Emit(new SimEvent(Tick, SimEventType.EntitySpawned, e.Id));
            return e;
        }

        public bool RemoveEntity(int id)
        {
            if (!_entities.TryGetValue(id, out var e)) return false;
            if (e.Kind == EntityKind.Building)
                _occupied.Remove(_grid.ToIndex(e.Pos));
            _entities.Remove(id);
            Emit(new SimEvent(Tick, SimEventType.EntityRemoved, id));
            return true;

        private void EconomyUpdate()
        {
            if (Game == null) return;
            float dt = 1f / Config.TickRate;

            foreach (Side s in new[] { Side.Blue, Side.Red })
            {
                int pts = OwnedResourcePointsOf(s);
                if (pts <= 0) continue;
                float acc = _epAccum.TryGetValue(s, out var a) ? a : 0f;
                acc += pts * dt;
                float interval = (float)Game.Ep.IntervalSeconds;
                while (acc >= interval)
                {
                    acc -= interval;
                    float cur = EpOf(s);
                    if (cur < Game.Ep.Cap) _ep[s] = cur + 1f;

            ReinforceSystem.Tick(this, dt);

        private void MoveUnits()
        {
            float perTick = 1f / Config.TickRate;
            foreach (var e in _entities.Values)
            {
                if (e.IsSoldier) continue; // 士兵移动全归 RegimentSystem.Tick,不进命令移动分支
                if (e.Kind != EntityKind.Unit || (e.State != EntityState.Moving && e.State != EntityState.Routing) || e.MoveSpeed <= 0f)
                    continue;
                if (e.GarrisonBuildingId != 0) continue; // 驻军单位不占线移动
                if (!e.HasMoveTarget) continue;          // 无目标的 Routing 原地待命

                // M8.5d 旁路点状态机:抵达(<0.8m)或超时(>45 tick)→ 清除,转回原目标
                if (e.Detour.HasValue)
                {
                    e.DetourTicks++;
                    if (SimVec2.Distance(e.PosF, e.Detour.Value) < NavigationAid.DetourClearDist
                        || e.DetourTicks > NavigationAid.DetourTimeoutTicks)
                    {
                        e.Detour = null;
                        e.DetourTicks = 0;
                    }
                }

                SimVec2 origTarget = SimVec2.FromGrid(e.MoveTarget); // 原目标(唯一抵达判定基准)
                SimVec2 target = e.Detour ?? origTarget;             // 有效目标(移动朝向)

                // M8.5d 卡死检测:本 tick 实际位移 < 期望步长×StepRatio → 停滞计数
                float spd = MoveSpeedOf(e);
                float expect = spd * perTick;
                if (SimVec2.Distance(e.PosF, e.PrevPosF) < expect * NavigationAid.StepRatio) e.StuckTicks++;
                else e.StuckTicks = 0;
                e.PrevPosF = e.PosF; // 采样须在检查之后(与士兵分支同序);写在移动后会导致检查时距离恒 0
                if (e.StuckTicks >= NavigationAid.StuckThresholdTicks && !e.Detour.HasValue)
                {
                    e.Detour = NavigationAid.ComputeDetour(this, e, target);
                    e.StuckTicks = 0;
                }

                SimVec2 delta = origTarget - e.PosF;
                if (delta.Magnitude <= 0.15f)
                {
                    e.HasMoveTarget = false;
                    e.State = EntityState.Idle;
                    e.MoveAccum = 0f;
                    e.Detour = null;
                    e.DetourTicks = 0;
                    e.PrevPosF = e.PosF;
                    continue;
                }

                delta = target - e.PosF;
                float dist = delta.Magnitude;
                float step = System.Math.Min(spd * perTick, dist);
                e.PosF = TerrainMap.ClampToEllipse(e.PosF + delta.Normalized() * step);
                PushOutOfBuildings(e);
                e.PosF = TerrainMap.ClampToEllipse(e.PosF);
                e.Pos = e.PosF.ToGrid();

                e.MoveAccum += step;
                if (e.MoveAccum >= 1f)
                {
                    e.MoveAccum = 0f;
                    Emit(new SimEvent(Tick, SimEventType.UnitAdvanced, e.Id)); // 节流:约每 1 米一次
                }

                if ((origTarget - e.PosF).Magnitude <= 0.15f)
                {
                    e.HasMoveTarget = false;
                    e.State = EntityState.Idle;
                    e.MoveAccum = 0f;
                    e.Detour = null;
                    e.DetourTicks = 0;
                }
            }
            RegimentSystem.Tick(this, perTick); // 兵团底座(M8.2a):阵型跟随/无将续战,先于软碰撞
            SoftCollide();
        }

        private float MoveSpeedOf(Entity e)
        {
            float spd = e.MoveSpeed * e.Status.MoveMultiplier;
            if (e.IsCharging && e.Hero == HeroId.ZhuFeng && Game != null)
                spd = (float)ChargeSpeedOf(HeroId.ZhuFeng);
            if (e.Hero == HeroId.ZhenYue && Tick >= 1800) // 3 分钟后移速衰减(D11 默认 1.4)
                spd = IsUpgraded(e.Side, HeroId.ZhenYue) ? 2.2f : 1.4f;
            spd *= HexSystem.MoveMultiplierOf(this, e); // 海克斯移速层
            spd *= TerrainMap.MoveFactor(e.PosF);       // 涉水减半/桥上正常
            return spd;
        }

        /// <summary>建筑碰撞:把单位推出建筑占地矩形(连续坐标下的障碍语义,M8.1a 矩形化)。</summary>
        private void PushOutOfBuildings(Entity u)
        {
            foreach (var b in _entities.Values)
            {
                if (b.Kind != EntityKind.Building || b.Rubbled) continue;
                if (!BuildingFootprint.Contains(b, u.PosF)) continue;
                u.PosF = BuildingFootprint.ClosestPoint(b, u.PosF); // 点在矩形内→最小穿透轴推出,幂等无死锁
            }
        }

        /// <summary>单位间软碰撞:间距小于 0.9 米时沿连线对半推开。</summary>
        private void SoftCollide()
        {
            var units = new List<Entity>();
            foreach (var e in _entities.Values)
                if (e.Kind == EntityKind.Unit && e.GarrisonBuildingId == 0
                    && e.State != EntityState.Dead && e.State != EntityState.Exchanging)
                    units.Add(e);

            const float MinDist = 0.9f;
            var moved = new HashSet<int>();
            for (int i = 0; i < units.Count; i++)
            {
                for (int j = i + 1; j < units.Count; j++)
                {
                    var a = units[i];
                    var b = units[j];
                    // 兵团底座(M8.2a):同兵团士兵之间、士兵与己方主将之间不互推,避免阵型震荡
                    if ((a.IsSoldier || b.IsSoldier) && (a.GeneralId == b.Id || b.GeneralId == a.Id || (a.IsSoldier && b.IsSoldier && a.GeneralId != 0 && a.GeneralId == b.GeneralId))) continue;
                    float dx = a.PosF.X - b.PosF.X, dy = a.PosF.Y - b.PosF.Y;
                    float d2 = dx * dx + dy * dy;
                    if (d2 >= MinDist * MinDist) continue;
                    float d = (float)System.Math.Sqrt(d2);
                    float push = (MinDist - d) * 0.5f;
                    float nx = d > 1e-6f ? dx / d : 1f;
                    float ny = d > 1e-6f ? dy / d : 0f;
                    a.PosF = TerrainMap.ClampToEllipse(a.PosF + new SimVec2(nx * push, ny * push));
                    b.PosF = TerrainMap.ClampToEllipse(b.PosF - new SimVec2(nx * push, ny * push));
                    moved.Add(a.Id);
                    moved.Add(b.Id);
                }
            }
            // 仅被推挤的单位回写格(避免覆盖未移动单位的手动定位)
            foreach (var e in units)
                if (moved.Contains(e.Id))
                    e.Pos = e.PosF.ToGrid();
        }

        /// <summary>建筑外接半径(米)= Max(HalfW,HalfH)+0.4。兼容旧近战判定,下一批 M8.1b 换精确矩形距;新代码请用 BuildingFootprint。</summary>
        public static float BuildingCollisionRadius(Entity b)
        {
            return System.Math.Max(BuildingFootprint.HalfW(b), BuildingFootprint.HalfH(b)) + 0.4f;
        }

        private void CombatUpdate()
        {
            if (Game == null) return;
            float dt = 1f / Config.TickRate;
            var dead = new List<int>();

            foreach (var e in _entities.Values)
            {
                if (e.Kind != EntityKind.Unit || e.State == EntityState.Dead) continue;
                if (e.GarrisonBuildingId != 0) continue; // 驻军单位不参与场外逻辑

                e.Status.Tick(dt);
                e.SecondsSinceCombat += dt;
                HeroAbilitySystem.UpdatePanShiShield(e, dt);
                if (e.Hero == HeroId.ZhuFeng && e.IsCharging && !HeroAbilitySystem.CanCharge(e, this))
                    e.IsCharging = false;

                if (e.Attack <= 0f) continue;

                if (e.AttackCooldown > 0f)
                    e.AttackCooldown -= dt;

                Entity target = FindNearestEnemy(e, e.State == EntityState.Routing ? 1.4f : float.MaxValue);
                if (target == null)
                {
                    if (e.State == EntityState.Attacking) e.State = EntityState.Idle;
                    continue;
                }

                // 溃逃单位保持 Routing 状态(不可调遣语义),不转入 Attacking
                if (e.State != EntityState.Routing)
                {
                    e.HasMoveTarget = false;
                    e.State = EntityState.Attacking;
                }
                if (e.AttackCooldown <= 0f)
                {
                    e.AttackCooldown = e.AttackIntervalSec * e.Status.AttackIntervalMultiplier;
                    ApplyAttack(e, target, dead);
                }
            }

            foreach (int id in dead)
            {
                if (_entities.TryGetValue(id, out var e))
                {
                    if (e.Hero != null) HexSystem.OnUnitDied(this, e.Side); // 陷阵之志
                    Emit(new SimEvent(Tick, SimEventType.EntityDied, id));
                    if (e.Kind == EntityKind.Building)
                        _occupied.Remove(_grid.ToIndex(e.Pos));
                    _entities.Remove(id);
                    Emit(new SimEvent(Tick, SimEventType.EntityRemoved, id));
                }
            }
        }

        private Entity FindNearestEnemy(Entity e, float maxDist)
        {
            Entity best = null;
            float bestDist = float.MaxValue;
            foreach (var o in _entities.Values)
            {
                if (o.Kind != EntityKind.Unit || o.Side == e.Side || o.State == EntityState.Dead)
                    continue;
                if (o.GarrisonBuildingId != 0 || o.State == EntityState.Exchanging)
                    continue; // 城内/交换中的单位不可被城外锁定
                float d = SimVec2.Distance(e.PosF, o.PosF);
                float effRange = e.Range + 0.3f; // 近战贴身容差
                if (d > effRange || d > maxDist) continue;
                if (d < bestDist - 1e-4f || (System.Math.Abs(d - bestDist) <= 1e-4f && best != null && o.Id < best.Id))
                {
                    best = o;
                    bestDist = d;
                }
            }
            return best;
        }

        private void ApplyAttack(Entity attacker, Entity defender, List<int> dead)
        {
            attacker.SecondsSinceCombat = 0f;
            defender.SecondsSinceCombat = 0f;

            // 基础层(GDD 公式)
            float baseDmg = DamageCalculator.CalculateDamage(attacker, defender, Game, attacker.IsCharging);

            // 星落升级·诛灭:基础攻击 5 → 6(基础层内替换)
            bool zhuMie = attacker.Hero == HeroId.XingLuo && IsUpgraded(attacker.Side, HeroId.XingLuo);
            if (zhuMie && attacker.Attack > 0f) baseDmg = baseDmg / attacker.Attack * 6f;

            // 叠算层(D28):输出加算、承伤加算、免伤乘算;溃逃并入对应层
            float outSum = attacker.Status.AttackBonus + HexSystem.OutputBonusOf(this, attacker)
                         + (attacker.State == EntityState.Routing ? -0.3f : 0f);
            float takeSum = defender.Status.DamageTakenBonus + HexSystem.DamageTakenBonusOf(this, defender)
                          + (defender.State == EntityState.Routing ? 0.3f : 0f);
            float reduce = defender.Status.DamageReduction * HexSystem.DamageReductionOf(this, defender);
            float dmg = baseDmg * (1f + outSum) * (1f + takeSum) * reduce;

            // 附加项(层后加法)
            if (zhuMie) dmg += 0.01f * defender.MaxHp;

            // 护盾优先吸收,溢出扣生命
            float remaining = dmg;
            if (defender.Shield > 0f)
            {
                float absorbed = Math.Min(defender.Shield, remaining);
                defender.Shield -= absorbed;
                remaining -= absorbed;
            }
            defender.Hp -= remaining;

            Emit(new SimEvent(Tick, SimEventType.DamageDealt, defender.Id, dmg));

            HeroAbilitySystem.ApplyOnHitEffects(this, attacker, defender);

            // 断峰升级·破军:身前 90 度扇形(半径 2 米)内全部敌人受单次 75% 伤害
            if (attacker.Hero == HeroId.DuanFeng && IsUpgraded(attacker.Side, HeroId.DuanFeng))
                Cleave(attacker, defender, dead);

            if (defender.Hp <= 0f)
            {
                defender.State = EntityState.Dead;
                dead.Add(defender.Id);
            }
        }

        private void Cleave(Entity attacker, Entity mainTarget, List<int> dead)
        {
            float facing = mainTarget.PosF.X >= attacker.PosF.X ? 1f : -1f;
            foreach (var o in _entities.Values)
            {
                if (o.Id == mainTarget.Id) continue;
                if (o.Kind != EntityKind.Unit || o.Side == attacker.Side || o.State == EntityState.Dead) continue;
                if (o.GarrisonBuildingId != 0 || o.State == EntityState.Exchanging) continue;
                float dx = o.PosF.X - attacker.PosF.X;
                float dy = o.PosF.Y - attacker.PosF.Y;
                if (dx * facing < 0) continue;         // 身后不打
                if (dx * dx + dy * dy > 4f) continue;  // 半径 2 米扇形
                float baseDmg = 0.75f * DamageCalculator.CalculateDamage(attacker, o, Game, attacker.IsCharging);
                float outSum = attacker.Status.AttackBonus + (attacker.State == EntityState.Routing ? -0.3f : 0f);
                float takeSum = o.Status.DamageTakenBonus + (o.State == EntityState.Routing ? 0.3f : 0f);
                float dmg = baseDmg * (1f + outSum) * (1f + takeSum) * o.Status.DamageReduction;
                float remaining = dmg;
                if (o.Shield > 0f)
                {
                    float absorbed = System.Math.Min(o.Shield, remaining);
                    o.Shield -= absorbed;
                    remaining -= absorbed;
                }
                o.Hp -= remaining;
                Emit(new SimEvent(Tick, SimEventType.DamageDealt, o.Id, dmg));
                if (o.Hp <= 0f) { o.State = EntityState.Dead; dead.Add(o.Id); }
            }
        }

        private double ChargeSpeedOf(HeroId id)
        {
            foreach (var h in Game.Heroes)
                if (h.Id == id) return h.ChargeSpeed;
            return 0;
        }

        private void Emit(SimEvent ev)
        {
            for (int i = _listeners.Count - 1; i >= 0; i--)
                _listeners[i]?.Invoke(ev);
        }
```

```csharp
using System.Collections.Generic;

namespace IronCurtain.Sim
{
    /// <summary>
    /// 兵团底座(M8.2a 第一批):1 主将 + 20 兵,兵随将走(4×5 阵型跟随);
    /// 主将死亡士兵永不溃逃、无将状态就近逐杀至死。纯 tick,无 UnityEngine 依赖。
    /// </summary>
    public static class RegimentSystem
    {
        public const int SoldiersPerGeneral = 20;
        public const int FormationRows = 4;   // 4 行 × 5 列,行列距 1.0 米
        public const int FormationCols = 5;

        public static SimVec2 FacingOf(Entity g)
        {
            if (g.FacingOverride.HasValue) return g.FacingOverride.Value;
            SimVec2 enemyDir = g.Side == Side.Blue ? new SimVec2(1f, 0f) : new SimVec2(-1f, 0f);
            if (g.HasMoveTarget)
            {
                var dir = (SimVec2.FromGrid(g.MoveTarget) - g.PosF).Normalized();
                bool toward = (g.Side == Side.Blue && dir.X > 0.05f) || (g.Side == Side.Red && dir.X < -0.05f);
                if (toward) return dir;       // 前进方向(朝敌)→用之(斜向前进阵型随之斜)
                return enemyDir;              // 后撤/横移 → 保持面朝敌不转身(规则③)
            }
            return enemyDir;
        }

        public static SimVec2 SlotOffset(Entity g, int slot)
        {
            var facing = FacingOf(g);
            var perp = new SimVec2(facing.Y, -facing.X);
            int row = slot / FormationCols;
            int col = slot % FormationCols;
            return facing * -(row + 1) + perp * (col - 2);
        }

        public static void Tick(SimCore sim, float dt)
        {
            // 快照遍历:士兵死亡结算会移除实体,禁止边枚举边修改
            var snapshot = new List<Entity>(sim.Entities.Values);
            foreach (var e in snapshot)
            {
                if (e.Kind != EntityKind.Unit || !e.IsSoldier || e.State == EntityState.Dead) continue;
                if (e.GarrisonBuildingId != 0) continue;

                var g = e.GeneralId != 0 ? sim.GetEntity(e.GeneralId) : null;
                if (g != null && g.Kind == EntityKind.Unit && g.State != EntityState.Dead)
                {
                    // a) 主将存活:阵型跟随(士兵不写 HasMoveTarget/State)
                    var target = g.PosF + SlotOffset(g, e.FormationSlot);
                    float dist = SimVec2.Distance(e.PosF, target);
                    if (dist > 0.2f) StepMove(sim, e, target, dt);
                }
                else
                {
                    // b) 无将续战:就近索敌逐杀至死(永不溃逃/撤退/响应指令)
                    var enemy = NearestEnemy(sim, e);
                    if (enemy == null) continue;
                    float dist = SimVec2.Distance(e.PosF, enemy.PosF);
                    if (dist <= e.Range + 0.3f) continue; // 交战交给 CombatUpdate
                    StepMove(sim, e, enemy.PosF, dt);
                }
            }
        }

        /// <summary>士兵步进(M8.5d:旁路点状态机 + 卡死检测内联,与 SimCore.MoveUnits 同一套数值)。</summary>
        private static void StepMove(SimCore sim, Entity e, SimVec2 target, float dt)
        {
            // 旁路点:抵达(<0.8m)或超时(>45 tick)→ 清除,转回原目标
            if (e.Detour.HasValue)
            {
                e.DetourTicks++;
                if (SimVec2.Distance(e.PosF, e.Detour.Value) < NavigationAid.DetourClearDist
                    || e.DetourTicks > NavigationAid.DetourTimeoutTicks)
                {
                    e.Detour = null;
                    e.DetourTicks = 0;
                }
            }
            var goal = e.Detour ?? target;

            float spd = e.MoveSpeed * TerrainMap.MoveFactor(e.PosF);
            float step = System.Math.Min(spd * dt, SimVec2.Distance(e.PosF, goal));
            e.PosF = TerrainMap.ClampToEllipse(e.PosF + (goal - e.PosF).Normalized() * step);
            // 建筑推出(与主将路径同语义,士兵不得穿墙)
            foreach (var b in sim.Entities.Values)
                if (b.Kind == EntityKind.Building && !b.Rubbled && BuildingFootprint.Contains(b, e.PosF))
                    e.PosF = BuildingFootprint.ClosestPoint(b, e.PosF);
            e.Pos = e.PosF.ToGrid();

            // 卡死检测:位移 < 期望步长×0.25 → 停滞;满 15 tick 且无旁路 → 计算旁路点
            if (SimVec2.Distance(e.PosF, e.PrevPosF) < spd * dt * NavigationAid.StepRatio) e.StuckTicks++;
            else e.StuckTicks = 0;
            e.PrevPosF = e.PosF;
            if (e.StuckTicks >= NavigationAid.StuckThresholdTicks && !e.Detour.HasValue)
            {
                e.Detour = NavigationAid.ComputeDetour(sim, e, target);
                e.StuckTicks = 0;
            }
        }

        /// <summary>最近敌索敌:与 SimCore.FindNearestEnemy 同一欧氏语义(单位、非己方、非死亡、非城内/交换)。</summary>
        private static Entity NearestEnemy(SimCore sim, Entity e)
        {
            Entity best = null;
            float bestDist = float.MaxValue;
            foreach (var o in sim.Entities.Values)
            {
                if (o.Kind != EntityKind.Unit || o.Side == e.Side || o.State == EntityState.Dead) continue;
                if (o.GarrisonBuildingId != 0 || o.State == EntityState.Exchanging) continue;
                float d = SimVec2.Distance(e.PosF, o.PosF);
                if (d < bestDist) { best = o; bestDist = d; }
            }
            return best;
        }

        /// <summary>生成一名士兵(属性=主将/10,攻速/移速/射程=主将,变装=主将 Hero),落位槽位偏移点。SpawnGeneral 与补兵共用。</summary>
        public static Entity SpawnSoldier(SimCore sim, Entity general, int slot, SimVec2? atPos = null)
        {
            var target = atPos ?? general.PosF + SlotOffset(general, slot);
            var s = sim.SpawnEntity(general.Side, EntityKind.Unit, target.ToGrid());
            s.IsSoldier = true;
            s.FormationSlot = slot;
            s.Hero = general.Hero;
            s.MaxHp = general.MaxHp / 10f;
            s.Hp = s.MaxHp;
            s.Attack = general.Attack / 10f;
            s.AttackIntervalSec = general.AttackIntervalSec;
            s.MoveSpeed = general.MoveSpeed;
            s.Range = general.Range;
            s.Shield = general.Shield / 10f;
            s.ShieldMax = s.Shield;
            s.GeneralId = general.Id;
            s.PosF = target;
            s.PrevPosF = target;
            s.Pos = target.ToGrid();
            return s;
        }

        /// <summary>生成主将 + 20 名士兵的完整兵团,返回主将实体。</summary>
        public static Entity SpawnGeneral(SimCore sim, Side side, HeroId hero, GridPos pos)
        {
            var c = sim.Game.Heroes[(int)hero];

            var g = sim.SpawnEntity(side, EntityKind.Unit, pos);
            g.Hero = hero;
            g.MaxHp = c.MaxHp;
            g.Hp = c.MaxHp;
            g.Attack = c.Attack;
            g.AttackIntervalSec = (float)c.AttackIntervalSec;
            g.Range = c.Range;
            g.MoveSpeed = (float)c.MoveSpeed;
            if (c.Shield > 0) { g.Shield = c.Shield; g.ShieldMax = c.Shield; }

            for (int slot = 0; slot < SoldiersPerGeneral; slot++)
                SpawnSoldier(sim, g, slot);
            return g;
        }
    }
}
```

```csharp
// ===== 现状全文(修改基准): Assets/Scripts/Sim/RegimentBehaviorSystem.cs =====
using System.Collections.Generic;

namespace IronCurtain.Sim
{
    /// <summary>
    /// 兵团自主行为层(M8.5b):视野索敌自动进攻(最高优先)> 15 格内友军交战自动支援;
    /// 玩家手动命令打断后该次自动决策永久失效(Sup 字段防复读)。纯 tick,无 UnityEngine。
    /// </summary>
    public static class RegimentBehaviorSystem
    {
        public const float SupportRange = 15f;          // 支援判定半径(格/米)
        public const float SupportArriveOffset = 2f;    // 支援落点:友军前方约 2 格

        public static void Tick(SimCore sim, float dt)
        {
            var snapshot = new List<Entity>(sim.Entities.Values);
            foreach (var g in snapshot)
            {
                if (g.Kind != EntityKind.Unit || g.IsSoldier || g.State == EntityState.Dead) continue;
                if (g.GarrisonBuildingId != 0) continue;

                // 1) 手动/自动命令执行保护(防拉回):有目标在身一律不动,到位后自然重评估
                if (g.HasMoveTarget) continue;

                // 2) 到达即清:本次自动决策自然完成,后续按新事件重新评估
                if (g.AutoKind != 0) { g.AutoKind = 0; g.AutoTargetId = 0; }

                // 3) 索敌进攻(优先级最高)
                var enemy = NearestVisibleEnemy(sim, snapshot, g);
                if (enemy != null)
                {
                    if (g.SupKind == 1 && g.SupTargetId == enemy.Id) continue; // 该敌已被玩家打断过
                    sim.SetMoveTarget(g.Id, enemy.PosF.ToGrid(), false);
                    g.AutoKind = 1;
                    g.AutoTargetId = enemy.Id;
                    continue; // 本 tick 不再评估支援
                }

                // 4) 支援:15 格内交战友军
                var ally = NearestFightingAlly(sim, snapshot, g);
                if (ally != null)
                {
                    if (g.SupKind == 2 && g.SupTargetId == ally.Id) continue; // 该支援已被玩家打断过
                    var dir = (g.PosF - ally.PosF).Normalized();
                    var dest = TerrainMap.ClampToEllipse(ally.PosF + dir * SupportArriveOffset);
                    sim.SetMoveTarget(g.Id, dest.ToGrid(), false);
                    g.AutoKind = 2;
                    g.AutoTargetId = ally.Id;
                }
                // 5) "被迫接战"无需新逻辑:敌方进射程后 CombatUpdate 自动还击
            }
        }

        /// <summary>兵团视野内最近可见敌(成员视野口径:士兵 远程6/近战3;主将 近战6/射手9/断峰10)。</summary>
        static Entity NearestVisibleEnemy(SimCore sim, List<Entity> snapshot, Entity g)
        {
            var members = MembersOf(snapshot, g);
            Entity best = null;
            float bestDist = float.MaxValue;
            foreach (var o in snapshot)
            {
                if (o.Kind != EntityKind.Unit || o.Side == g.Side || o.State == EntityState.Dead) continue;
                if (o.GarrisonBuildingId != 0 || o.State == EntityState.Exchanging) continue;
                if (!VisibleByAny(members, o)) continue;
                float d = SimVec2.Distance(g.PosF, o.PosF);
                if (d < bestDist - 1e-4f || (System.Math.Abs(d - bestDist) <= 1e-4f && best != null && o.Id < best.Id))
                {
                    best = o;
                    bestDist = d;
                }
            }
            return best;
        }

        static Entity NearestFightingAlly(SimCore sim, List<Entity> snapshot, Entity g)
        {
            Entity best = null;
            float bestDist = float.MaxValue;
            foreach (var f in snapshot)
            {
                if (f.Id == g.Id || f.Kind != EntityKind.Unit || f.IsSoldier || f.Side != g.Side) continue;
                if (f.State == EntityState.Dead || f.GarrisonBuildingId != 0) continue;
                if (f.SecondsSinceCombat >= 3f) continue; // 非交战中
                float d = SimVec2.Distance(g.PosF, f.PosF);
                if (d > SupportRange) continue;
                if (d < bestDist - 1e-4f || (System.Math.Abs(d - bestDist) <= 1e-4f && best != null && f.Id < best.Id))
                {
                    best = f;
                    bestDist = d;
                }
            }
            return best;
        }

        static List<Entity> MembersOf(List<Entity> snapshot, Entity g)
        {
            var list = new List<Entity> { g };
            foreach (var e in snapshot)
            {
                if (!e.IsSoldier || e.GeneralId != g.Id || e.State == EntityState.Dead) continue;
                if (e.GarrisonBuildingId != 0) continue;
                list.Add(e);
            }
            return list;
        }

        static bool VisibleByAny(List<Entity> members, Entity enemy)
        {
            foreach (var m in members)
                if (SimVec2.Distance(m.PosF, enemy.PosF) <= VisionRadiusOf(m)) return true;
            return false;
        }

        /// <summary>视野半径:士兵 远程(坠日/星落)6/近战 3;主将 近战 6/坠日星落 9/断峰 10(与 VisionSystem 同口径)。</summary>
        static float VisionRadiusOf(Entity e)
        {
            bool ranged = e.Hero == HeroId.ZhuiRi || e.Hero == HeroId.XingLuo;
            if (e.IsSoldier) return ranged ? 6f : 3f;
            if (e.Hero == HeroId.DuanFeng) return 10f;
            return ranged ? 9f : 6f;
        }
    }
```

```csharp
// ===== 现状全文(修改基准): Assets/Scripts/View/SelectionSystem.cs =====
using System.Collections.Generic;
using System.Text;
using IronCurtain.Sim;
using UnityEngine;

namespace IronCurtain.View
{
    /// <summary>
    /// M8.6 点选/框选己方(蓝方)主将 + 右键移动命令(手动级)+ T 键自动布阵 + 淡出提示 + 选中圈/点击标记。
    /// </summary>
    public sealed class SelectionSystem : MonoBehaviour
    {
        private const float PickRadius = 3f;      // 点选主将命中半径(米)
        private const float RingRadius = 1.3f;    // 选中圈半径(米)
        private const float HintDuration = 2.5f;  // 提示淡出时长(秒)

        private static readonly Plane GroundPlane = new Plane(Vector3.up, Vector3.zero);

        private SimCore _sim;
        private Camera _cam;
        private readonly List<int> _selected = new List<int>();

        private bool _dragging;
        private Vector2 _dragStart;
        private string _selKey = "";

        private readonly List<GameObject> _rings = new List<GameObject>();
        private GameObject _ringRoot;
        private float _hintT; // 提示剩余显示时间(秒)

        private GameObject _marker;
        private Material _markerMat;
        private float _markerT;

        public void Init(SimCore sim, Camera cam)
        {
            _sim = sim;
            _cam = cam;
        }

        private bool MouseOverPanel
        {
            get
            {
                Vector2 m = Input.mousePosition;
                // 面板 Rect 为 GUI 坐标(y 从顶起),鼠标 y 从底起,需翻转
                return BarracksOverviewPanel.LastPanelRect.Contains(new Vector2(m.x, Screen.height - m.y));
            }
        }

        private void Update()
        {
            if (_sim == null || _cam == null) return;
            PruneSelection();
            _hintT = Mathf.Max(0f, _hintT - Time.deltaTime);

            HandleLeftButton();
            HandleRightButton();
            HandleFormationKey();
            UpdateRings();
            UpdateMarker();
        }

        private void PruneSelection()
        {
            for (int i = _selected.Count - 1; i >= 0; i--)
            {
                var e = _sim.GetEntity(_selected[i]);
                if (e == null || e.State == EntityState.Dead || e.IsSoldier)
                    _selected.RemoveAt(i);
            }
        }

        private void HandleLeftButton()
        {
            if (Input.GetMouseButtonDown(0) && !MouseOverPanel)
            {
                _dragging = true;
                _dragStart = Input.mousePosition;
            }
            if (!Input.GetMouseButtonUp(0) || !_dragging) return;
            _dragging = false;
            Vector2 m = Input.mousePosition;
            float dx = m.x - _dragStart.x, dy = m.y - _dragStart.y;
            bool click = dx * dx + dy * dy < 64f; // 屏幕位移平方和 <64px² = 点选
            if (click)
            {
                int id = PickGeneral();
                _selected.Clear();
                if (id != 0) _selected.Add(id);
            }
            else
            {
                var rect = Rect.MinMaxRect(
                    Mathf.Min(_dragStart.x, m.x), Mathf.Min(_dragStart.y, m.y),
                    Mathf.Max(_dragStart.x, m.x), Mathf.Max(_dragStart.y, m.y));
                _selected.Clear();
                foreach (var kv in _sim.Entities)
                {
                    var e = kv.Value;
                    if (e == null || e.Kind != EntityKind.Unit || e.IsSoldier || e.Side != Side.Blue
                        || e.State == EntityState.Dead) continue;
                    var sp = _cam.WorldToScreenPoint(BoardView.WorldPos(e.PosF));
                    if (rect.Contains(new Vector2(sp.x, sp.y), true)) _selected.Add(e.Id);
                }
            }
        }

        private int PickGeneral()
        {
            Vector3 wp;
            if (!RayGround(out wp)) return 0;
            var p = new SimVec2(wp.x, 3f - wp.z);
            int best = 0;
            float bestD = PickRadius * PickRadius;
            int soldierGeneral = 0;
            float soldierD = PickRadius * PickRadius;
            foreach (var kv in _sim.Entities)
            {
                var e = kv.Value;
                if (e == null || e.Kind != EntityKind.Unit || e.Side != Side.Blue || e.State == EntityState.Dead) continue;
                float dx = e.PosF.X - p.X, dy = e.PosF.Y - p.Y;
                float d = dx * dx + dy * dy;
                if (e.IsSoldier)
                {
                    if (d < soldierD) { soldierD = d; soldierGeneral = e.GeneralId; }
                }
                else if (d < bestD) { bestD = d; best = e.Id; }
            }
            if (best != 0) return best;      // 主将优先
            return soldierGeneral;           // 命中士兵 → 其主将
        }

        private void HandleRightButton()
        {
            if (!Input.GetMouseButtonDown(1) || _selected.Count == 0 || MouseOverPanel) return;
            Vector3 wp;
            if (!RayGround(out wp)) return;
            var grid = new SimVec2(wp.x, 3f - wp.z).ToGrid();
            foreach (var id in _selected)
                _sim.SetMoveTarget(id, grid, true);
            SpawnMarker(wp);
        }

        private void HandleFormationKey()
        {
            if (Input.GetKeyDown(KeyCode.T) && _selected.Count >= 2)
                RegimentFormationSystem.Apply(_sim, _selected);
        }

        private bool RayGround(out Vector3 wp)
        {
            wp = Vector3.zero;
            Ray ray = _cam.ScreenPointToRay(Input.mousePosition);
            float enter;
            if (!GroundPlane.Raycast(ray, out enter)) return false;
            wp = ray.GetPoint(enter);
            return true;
        }

        private void UpdateRings()
        {
            var key = BuildKey();
            if (key != _selKey)
            {
                bool grew = _selected.Count >= 2 && !_selKey.StartsWith(key);
                _selKey = key;
                if (grew) _hintT = HintDuration; // 仅选择变化瞬间重置,之后自然淡出
                RebuildRings();
            }
            for (int i = 0; i < _rings.Count && i < _selected.Count; i++)
            {
                var e = _sim.GetEntity(_selected[i]);
                if (e == null || _rings[i] == null) continue;
                _rings[i].transform.position = BoardView.WorldPos(e.PosF);
            }
        }

        private string BuildKey()
        {
            var sb = new StringBuilder();
            foreach (var id in _selected) sb.Append(id).Append(',');
            return sb.ToString();
        }

        private void RebuildRings()
        {
            foreach (var r in _rings) if (r != null) Destroy(r);
            _rings.Clear();
            if (_ringRoot != null) { Destroy(_ringRoot); _ringRoot = null; }
            if (_selected.Count == 0) return;
            Material ringMat = ViewAssets.Mat(new Color(0.2f, 1f, 0.35f) * 0.5f, 0f, 0.5f, new Color(0.2f, 1f, 0.35f), 0.45f);
            _ringRoot = new GameObject("SelectionRings");
            foreach (var id in _selected)
            {
                var e = _sim.GetEntity(id);
                if (e == null) continue;
                ViewAssets.DecoBox(ringMat, new Vector3(0f, 0.03f, 0f),
                    new Vector3(RingRadius * 2f, 0.03f, RingRadius * 2f), _ringRoot.transform, "Ring" + id);
                _rings.Add(_ringRoot.transform.GetChild(_ringRoot.transform.childCount - 1).gameObject);
                _rings[_rings.Count - 1].transform.position = BoardView.WorldPos(e.PosF);
            }
        }

        private void SpawnMarker(Vector3 wp)
        {
            if (_marker != null) Destroy(_marker);
            Material baseMat = ViewAssets.Flat(new Color(1f, 1f, 1f, 0.45f));
            _markerMat = Instantiate(baseMat); // 独立实例,可调 alpha
            _marker = new GameObject("ClickMarker");
            ViewAssets.DecoBox(_markerMat, new Vector3(0f, 0.04f, 0f), new Vector3(2f, 0.02f, 2f), _marker.transform, "Disc");
            _marker.transform.position = wp;
            _markerT = 1f;
        }

        private void UpdateMarker()
        {
            if (_marker == null) return;
            _markerT -= Time.deltaTime;
            if (_markerT <= 0f) { Destroy(_marker); _marker = null; _markerMat = null; return; }
            float a = Mathf.Clamp01(_markerT); // 1s 线性淡出
            if (_markerMat != null)
            {
                if (_markerMat.HasProperty("_BaseColor"))
                    _markerMat.SetColor("_BaseColor", new Color(1f, 1f, 1f, 0.45f * a));
                else _markerMat.color = new Color(1f, 1f, 1f, 0.45f * a);
            }
            _marker.transform.localScale = Vector3.one * Mathf.Max(0.05f, a); // 不透明材质下的可见消退兜底
        }

        private void OnGUI()
        {
            if (_selected.Count < 2 || _hintT <= 0f) return;
            var style = new GUIStyle(GUI.skin.label) { fontSize = 24, alignment = TextAnchor.MiddleCenter };
            style.normal.textColor = Color.white;
            var prev = GUI.color;
            GUI.color = new Color(1f, 1f, 1f, Mathf.Clamp01(_hintT / HintDuration)); // alpha 从 1 线性降到 0
            GUI.Label(new Rect(Screen.width * 0.5f - 160f, 24f, 320f, 36f), "按 T 键自动布阵", style);
            GUI.color = prev;
        }
    }
}
```
