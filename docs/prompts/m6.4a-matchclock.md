## 二、GDD 规则摘录(权威)

> 9.1 平局与加时补充规则:10 分钟时限到若双方大本营均未被摧毁,进入 2 分钟加时:双方人口上限各 +10、兵营产速 +10%、EP 改为每 60 秒产 1。加时内率先摧毁对方大本营获胜;加时结束仍未分胜负,按总分判定:5×大本营剩余血量 + 3×剩余英雄数 + 2×剩余 EP + 剩余建筑数(每座 1 分),总分高者胜;总分相同比总人口,人口相同比全局阵亡兵团数(少者胜)。

将兵制映射(项目拍板口径,已定稿,不要另发明):
- 「人口上限各 +10」→ 名额制下映射为**将名额 +1**(PopulationCapOf 在加时返回 GeneralCap+1);配置 OvertimePopBonus=10 不按字面直加。
- 「兵营产速 +10%」→ 兵营只承担补兵,映射为**全局补兵速度 ×(1+10%)**(2s/兵 → 2/1.1 s/兵);主基地产将 20s 不变。
- 「EP 改为每 60 秒产 1」→ EconomyUpdate 的 interval 在加时取 Match.OvertimeEpInterval(=60)。
- 「剩余英雄数」→ 存活主将数(PopulationOf);「剩余 EP」→ (int)EpOf;「剩余建筑数」→ Side==己方、非废墟、非 Dead 的全部建筑实体(含被己方占领的中立点与堡垒,每座 1 分,基地本身也计 1 座)。
- 「全局阵亡兵团数」→ 该方累计阵亡**主将**数(新计数器 LostGenerals,士兵不计)。
- 加时结束仍完全平(分/人口/阵亡全同)→ 平局:Winner 保持 null,Phase=Ended,MatchEnded 事件 SubjectId=-1。

### 4. MatchClockSystem.cs(新增,全部逻辑在此;≤300 行;禁 UnityEngine)
```csharp
namespace IronCurtain.Sim
{
    /// <summary>对局阶段(M6.4a)。</summary>
    public enum MatchPhase { Normal, Overtime, Ended }

    /// <summary>
    /// 对局时钟(M6.4a):10 分钟时限 → 2 分钟加时 → 积分判定。
    /// 摧毁大本营胜利(DestroySystem.SetWinner)由本系统统一收口为 Ended + MatchEnded 事件。
    /// 加时 buff 落点:EconomyUpdate(EP 60s)/PopulationCapOf(名额+1)/ReinforceSystem(补兵 ×1.1)。
    /// </summary>
    public static class MatchClockSystem
    {
        public static float ElapsedSeconds(SimCore sim) => sim.Tick / (float)sim.Config.TickRate;

        /// <summary>常规期剩余 / 加时剩余 / 已结束 0。</summary>
        public static float RemainingSeconds(SimCore sim)
        {
            if (sim.Game == null) return 0f;
            var m = sim.Game.Match;
            float total = (m.DurationMinutes + (sim.Phase == MatchPhase.Overtime ? m.OvertimeMinutes : 0)) * 60f;
            if (sim.Phase == MatchPhase.Ended) return 0f;
            return System.Math.Max(0f, total - ElapsedSeconds(sim));
        }

        public static void Tick(SimCore sim, float dt)
        {
            // 留场死扫描(塔击/海克斯置 Dead 不移除):首次发现即计数,与 CombatUpdate 清理块双路径防重
            foreach (var e in sim.Entities.Values)
                if (e.Kind == EntityKind.Unit && !e.IsSoldier && e.State == EntityState.Dead && !e.DeadCounted)
                { e.DeadCounted = true; sim.LostGenerals[e.Side]++; }

            if (sim.Phase == MatchPhase.Ended) return;
            if (sim.Game == null) return; // 无数值配置(纯战斗测试)不启用时钟

            // 摧毁胜利已发生 → 收口
            if (sim.Winner != null)
            {
                sim.Phase = MatchPhase.Ended;
                sim.Emit(new SimEvent(sim.Tick, SimEventType.MatchEnded, (int)sim.Winner.Value));
                return;
            }

            var m = sim.Game.Match;
            float duration = m.DurationMinutes * 60f;
            float elapsed = ElapsedSeconds(sim);

            // 常规时限到(双方大本营均在,否则上方已收口)→ 加时
            if (sim.Phase == MatchPhase.Normal && elapsed >= duration)
            {
                sim.Phase = MatchPhase.Overtime;
                sim.Emit(new SimEvent(sim.Tick, SimEventType.OvertimeStarted, 0));
            }

            // 加时结束 → 积分判定
            if (sim.Phase == MatchPhase.Overtime && elapsed >= duration + m.OvertimeMinutes * 60f)
            {
                sim.Phase = MatchPhase.Ended;
                var winner = JudgeByScore(sim);
                if (winner.HasValue) sim.SetWinner(winner.Value);
                sim.Emit(new SimEvent(sim.Tick, SimEventType.MatchEnded, winner.HasValue ? (int)winner.Value : -1));
            }
        }

        /// <summary>积分明细(GDD 9.1):5×基地血 + 3×将 + 2×EP + 建筑×1。</summary>
        public static void ScoreOf(SimCore sim, Side side,
            out int basePt, out int heroPt, out int epPt, out int bldPt, out int total)
        {
            var m = sim.Game.Match;
            basePt = 0; bldPt = 0;
            foreach (var e in sim.Entities.Values)
            {
                if (e.Kind != EntityKind.Building || e.Side != side || e.Rubbled || e.State == EntityState.Dead) continue;
                if (e.Building == BuildingId.Base) basePt += (int)e.Hp * m.ScoreBaseHpWeight;
                bldPt += m.ScoreBuildingWeight;
            }
            heroPt = sim.PopulationOf(side) * m.ScoreHeroWeight;
            epPt = (int)sim.EpOf(side) * m.ScoreEpWeight;
            total = basePt + heroPt + epPt + bldPt;
        }

        /// <summary>平局链:总分 → 存活将数 → 阵亡兵团数(少者胜)→ 平局。</summary>
        private static Side? JudgeByScore(SimCore sim)
        {
            ScoreOf(sim, Side.Blue, out _, out _, out _, out _, out int b);
            ScoreOf(sim, Side.Red, out _, out _, out _, out _, out int r);
            if (b != r) return b > r ? Side.Blue : Side.Red;
            int bp = sim.PopulationOf(Side.Blue), rp = sim.PopulationOf(Side.Red);
            if (bp != rp) return bp > rp ? Side.Blue : Side.Red;
            int bl = sim.LostGenerals[Side.Blue], rl = sim.LostGenerals[Side.Red];
            if (bl != rl) return bl < rl ? Side.Blue : Side.Red;
            return null;
        }
    }
}
```

/// <summary>M6.4a 对局倒计时:常规期=剩余常规时长;加时=剩余加时;结束定格 00:00。无配置回退正计时。</summary>
public static string FormatCountdown(SimCore sim)
{
    if (sim.Config.TickRate <= 0) return "00:00";
    if (sim.Game == null) return FormatClock(sim.Tick, sim.Config.TickRate);
    float remain = MatchClockSystem.RemainingSeconds(sim);
    long sec = (long)System.Math.Ceiling(remain);
    return (sec / 60).ToString("00") + ":" + (sec % 60).ToString("00");
}

}

    public enum SimEventType
    {
        StepBegin,
        StepEnd,
        EntitySpawned,

        CommandExecuted,
        UnitAdvanced,

        EntityDied
    }

    public struct SimEvent
    {
        public long Tick;
        public SimEventType Type;
        public int SubjectId;
        public float Amount;

        public SimEvent(long tick, SimEventType type, int subjectId, float amount = 0f)
        {
            Tick = tick;
            Type = type;

        private void EconomyUpdate()
        {

            float dt = 1f / Config.TickRate;

            foreach (Side s in new[] { Side.Blue, Side.Red })
            {

                if (pts <= 0) continue;
                float acc = _epAccum.TryGetValue(s, out var a) ? a : 0f;
                acc += pts * dt;
                float interval = (float)Game.Ep.IntervalSeconds;
                while (acc >= interval)
                {
                    acc -= interval;
                    float cur = EpOf(s);
                    if (cur < Game.Ep.Cap) _ep[s] = cur + 1f;
                }
                _epAccum[s] = acc;

            ProductionSystem.Tick(this, dt);
            ReinforceSystem.Tick(this, dt);
        }

        private void MoveUnits()
        {
            float perTick = 1f / Config.TickRate;
            foreach (var e in _entities.Values)
            {
                if (e.IsSoldier) continue; // 士兵移动全归 RegimentSystem.Tick,不进命令移动分支
                if (e.Kind != EntityKind.Unit || (e.State != EntityState.Moving && e.State != EntityState.Routing) || e.MoveSpeed <= 0f)
                    continue;
                if (e.GarrisonBuildingId != 0) continue; // 驻军单位不占线移动
                if (!e.HasMoveTarget) continue;          // 无目标的 Routing 原地待命

                // M8.5d 旁路点状态机:抵达(<0.8m)或超时(>45 tick)→ 清除,转回原目标
                if (e.Detour.HasValue)
                {
                    e.DetourTicks++;
                    if (SimVec2.Distance(e.PosF, e.Detour.Value) < NavigationAid.DetourClearDist
                        || e.DetourTicks > NavigationAid.DetourTimeoutTicks)
                    {
                        e.Detour = null;
                        e.DetourTicks = 0;
                    }
                }

                SimVec2 origTarget = SimVec2.FromGrid(e.MoveTarget); // 原目标(唯一抵达判定基准)
                SimVec2 target = e.Detour ?? origTarget;             // 有效目标(移动朝向)

                // M8.5d 卡死检测:本 tick 实际位移 < 期望步长×StepRatio → 停滞计数
                float spd = MoveSpeedOf(e);
                float expect = spd * perTick;
                if (SimVec2.Distance(e.PosF, e.PrevPosF) < expect * NavigationAid.StepRatio) e.StuckTicks++;
                else e.StuckTicks = 0;
                e.PrevPosF = e.PosF; // 采样须在检查之后(与士兵分支同序);写在移动后会导致检查时距离恒 0
                if (e.StuckTicks >= NavigationAid.StuckThresholdTicks && !e.Detour.HasValue)
                {
                    e.Detour = NavigationAid.ComputeDetour(this, e, target);
                    e.StuckTicks = 0;
                }

                SimVec2 delta = origTarget - e.PosF;
                if (delta.Magnitude <= 0.15f)
                {
                    e.HasMoveTarget = false;
                    e.State = EntityState.Idle;
                    e.MoveAccum = 0f;
                    e.Detour = null;
                    e.DetourTicks = 0;
                    e.PrevPosF = e.PosF;
                    continue;
                }

                delta = target - e.PosF;
                float dist = delta.Magnitude;
                float step = System.Math.Min(spd * perTick, dist);
                e.PosF = TerrainMap.ClampToEllipse(e.PosF + delta.Normalized() * step);
                PushOutOfBuildings(e);
                e.PosF = TerrainMap.ClampToEllipse(e.PosF);
                e.Pos = e.PosF.ToGrid();

                e.MoveAccum += step;
                if (e.MoveAccum >= 1f)
                {
                    e.MoveAccum = 0f;
                    Emit(new SimEvent(Tick, SimEventType.UnitAdvanced, e.Id)); // 节流:约每 1 米一次
                }

                if ((origTarget - e.PosF).Magnitude <= 0.15f)
                {
                    e.HasMoveTarget = false;
                    e.State = EntityState.Idle;
                    e.MoveAccum = 0f;
                    e.Detour = null;
                    e.DetourTicks = 0;
                }
            }
            RegimentSystem.Tick(this, perTick); // 兵团底座(M8.2a):阵型跟随/无将续战,先于软碰撞
            SoftCollide();
        }

        private float MoveSpeedOf(Entity e)
        {
            float spd = e.MoveSpeed * e.Status.MoveMultiplier;
            if (e.IsCharging && e.Hero == HeroId.ZhuFeng && Game != null)
                spd = (float)ChargeSpeedOf(HeroId.ZhuFeng);
            if (e.Hero == HeroId.ZhenYue && Tick >= 1800) // 3 分钟后移速衰减(D11 默认 1.4)
                spd = IsUpgraded(e.Side, HeroId.ZhenYue) ? 2.2f : 1.4f;
            spd *= HexSystem.MoveMultiplierOf(this, e); // 海克斯移速层

        /// <summary>建筑碰撞:把单位推出建筑占地矩形(连续坐标下的障碍语义,M8.1a 矩形化)。</summary>
        private void PushOutOfBuildings(Entity u)
        {

            {
                if (b.Kind != EntityKind.Building || b.Rubbled) continue;
                if (!BuildingFootprint.Contains(b, u.PosF)) continue;
                u.PosF = BuildingFootprint.ClosestPoint(b, u.PosF); // 点在矩形内→最小穿透轴推出,幂等无死锁
            }
        }

        /// <summary>单位间软碰撞:间距小于 0.9 米时沿连线对半推开。</summary>
        private void SoftCollide()
        {
            var units = new List<Entity>();
            foreach (var e in _entities.Values)
                if (e.Kind == EntityKind.Unit && e.GarrisonBuildingId == 0
                    && e.State != EntityState.Dead && e.State != EntityState.Exchanging)
                    units.Add(e);

            const float MinDist = 0.9f;
            var moved = new HashSet<int>();
            for (int i = 0; i < units.Count; i++)
            {
                for (int j = i + 1; j < units.Count; j++)
                {
                    var a = units[i];
                    var b = units[j];
                    // 兵团底座(M8.2a):同兵团士兵之间、士兵与己方主将之间不互推,避免阵型震荡
                    if ((a.IsSoldier || b.IsSoldier) && (a.GeneralId == b.Id || b.GeneralId == a.Id || (a.IsSoldier && b.IsSoldier && a.GeneralId != 0 && a.GeneralId == b.GeneralId))) continue;
                    float dx = a.PosF.X - b.PosF.X, dy = a.PosF.Y - b.PosF.Y;
                    float d2 = dx * dx + dy * dy;
                    if (d2 >= MinDist * MinDist) continue;
                    float d = (float)System.Math.Sqrt(d2);
                    float push = (MinDist - d) * 0.5f;
                    float nx = d > 1e-6f ? dx / d : 1f;
                    float ny = d > 1e-6f ? dy / d : 0f;
                    a.PosF = TerrainMap.ClampToEllipse(a.PosF + new SimVec2(nx * push, ny * push));
                    b.PosF = TerrainMap.ClampToEllipse(b.PosF - new SimVec2(nx * push, ny * push));
                    moved.Add(a.Id);
                    moved.Add(b.Id);
                }
            }
            // 仅被推挤的单位回写格(避免覆盖未移动单位的手动定位)
            foreach (var e in units)
                if (moved.Contains(e.Id))
                    e.Pos = e.PosF.ToGrid();
        }

        /// <summary>建筑外接半径(米)= Max(HalfW,HalfH)+0.4。兼容旧近战判定,下一批 M8.1b 换精确矩形距;新代码请用 BuildingFootprint。</summary>
        public static float BuildingCollisionRadius(Entity b)
        {
            return System.Math.Max(BuildingFootprint.HalfW(b), BuildingFootprint.HalfH(b)) + 0.4f;
        }

        private void CombatUpdate()
        {
            if (Game == null) return;
            float dt = 1f / Config.TickRate;
            var dead = new List<int>();

            foreach (var e in _entities.Values)
            {
                if (e.Kind != EntityKind.Unit || e.State == EntityState.Dead) continue;
                if (e.GarrisonBuildingId != 0) continue; // 驻军单位不参与场外逻辑

                e.Status.Tick(dt);
                e.SecondsSinceCombat += dt;
                HeroAbilitySystem.UpdatePanShiShield(e, dt);
                if (e.Hero == HeroId.ZhuFeng && e.IsCharging && !HeroAbilitySystem.CanCharge(e, this))
                    e.IsCharging = false;

                if (e.Attack <= 0f) continue;

                if (e.AttackCooldown > 0f)
                    e.AttackCooldown -= dt;

                Entity target = FindNearestEnemy(e, e.State == EntityState.Routing ? 1.4f : float.MaxValue);
                if (target == null)
                {
                    if (e.State == EntityState.Attacking) e.State = EntityState.Idle;
                    continue;
                }

                // 溃逃单位保持 Routing 状态(不可调遣语义),不转入 Attacking
                if (e.State != EntityState.Routing)
                {
                    e.HasMoveTarget = false;
                    e.State = EntityState.Attacking;
                }
                if (e.AttackCooldown <= 0f)
                {
                    e.AttackCooldown = e.AttackIntervalSec * e.Status.AttackIntervalMultiplier;
                    ApplyAttack(e, target, dead);
                }
            }

            foreach (int id in dead)
            {
                if (_entities.TryGetValue(id, out var e))
                {
                    if (e.Hero != null) HexSystem.OnUnitDied(this, e.Side); // 陷阵之志
                    Emit(new SimEvent(Tick, SimEventType.EntityDied, id));
                    if (e.Kind == EntityKind.Building)
                        _occupied.Remove(_grid.ToIndex(e.Pos));
                    _entities.Remove(id);
                    Emit(new SimEvent(Tick, SimEventType.EntityRemoved, id));
                }
            }
        }

        private Entity FindNearestEnemy(Entity e, float maxDist)
        {
            Entity best = null;
            float bestDist = float.MaxValue;
            foreach (var o in _entities.Values)
            {
                if (o.Kind != EntityKind.Unit || o.Side == e.Side || o.State == EntityState.Dead)
                    continue;
                if (o.GarrisonBuildingId != 0 || o.State == EntityState.Exchanging)
                    continue; // 城内/交换中的单位不可被城外锁定
                float d = SimVec2.Distance(e.PosF, o.PosF);
                float effRange = e.Range + 0.3f; // 近战贴身容差
                if (d > effRange || d > maxDist) continue;
                if (d < bestDist - 1e-4f || (System.Math.Abs(d - bestDist) <= 1e-4f && best != null && o.Id < best.Id))
                {
                    best = o;
                    bestDist = d;
                }
            }
            return best;
        }

        private void ApplyAttack(Entity attacker, Entity defender, List<int> dead)
        {
            attacker.SecondsSinceCombat = 0f;
            defender.SecondsSinceCombat = 0f;

            // 基础层(GDD 公式)
            float baseDmg = DamageCalculator.CalculateDamage(attacker, defender, Game, attacker.IsCharging);

            // 星落升级·诛灭:基础攻击 5 → 6(基础层内替换)
            bool zhuMie = attacker.Hero == HeroId.XingLuo && IsUpgraded(attacker.Side, HeroId.XingLuo);
            if (zhuMie && attacker.Attack > 0f) baseDmg = baseDmg / attacker.Attack * 6f;

            // 叠算层(D28):输出加算、承伤加算、免伤乘算;撤退/溃逃 −25%/+25%(M8.7)
            float outSum = attacker.Status.AttackBonus + HexSystem.OutputBonusOf(this, attacker)
                         + (attacker.State == EntityState.Routing ? -0.25f : 0f);
            float takeSum = defender.Status.DamageTakenBonus + HexSystem.DamageTakenBonusOf(this, defender)
                          + (defender.State == EntityState.Routing ? 0.25f : 0f);
            float reduce = defender.Status.DamageReduction * HexSystem.DamageReductionOf(this, defender);
            float dmg = baseDmg * (1f + outSum) * (1f + takeSum) * reduce;

            // 附加项(层后加法)
            if (zhuMie) dmg += 0.01f * defender.MaxHp;

            // 护盾优先吸收,溢出扣生命
            float remaining = dmg;
            if (defender.Shield > 0f)
            {
                float absorbed = Math.Min(defender.Shield, remaining);
                defender.Shield -= absorbed;
                remaining -= absorbed;
            }
            defender.Hp -= remaining;

            Emit(new SimEvent(Tick, SimEventType.DamageDealt, defender.Id, dmg));

            HeroAbilitySystem.ApplyOnHitEffects(this, attacker, defender);

            // 断峰升级·破军:身前 90 度扇形(半径 2 米)内全部敌人受单次 75% 伤害
            if (attacker.Hero == HeroId.DuanFeng && IsUpgraded(attacker.Side, HeroId.DuanFeng))
                Cleave(attacker, defender, dead);

            if (defender.Hp <= 0f)
            {
                defender.State = EntityState.Dead;
                dead.Add(defender.Id);
            }
        }

        private void Cleave(Entity attacker, Entity mainTarget, List<int> dead)
        {
            float facing = mainTarget.PosF.X >= attacker.PosF.X ? 1f : -1f;
            foreach (var o in _entities.Values)
            {
                if (o.Id == mainTarget.Id) continue;
                if (o.Kind != EntityKind.Unit || o.Side == attacker.Side || o.State == EntityState.Dead) continue;
                if (o.GarrisonBuildingId != 0 || o.State == EntityState.Exchanging) continue;
                float dx = o.PosF.X - attacker.PosF.X;
                float dy = o.PosF.Y - attacker.PosF.Y;
                if (dx * facing < 0) continue;         // 身后不打
                if (dx * dx + dy * dy > 4f) continue;  // 半径 2 米扇形
                float baseDmg = 0.75f * DamageCalculator.CalculateDamage(attacker, o, Game, attacker.IsCharging);
                // 撤退/溃逃 −25%/+25%(M8.7)
                float outSum = attacker.Status.AttackBonus + (attacker.State == EntityState.Routing ? -0.25f : 0f);
                float takeSum = o.Status.DamageTakenBonus + (o.State == EntityState.Routing ? 0.25f : 0f);
                float dmg = baseDmg * (1f + outSum) * (1f + takeSum) * o.Status.DamageReduction;
                float remaining = dmg;
                if (o.Shield > 0f)
                {
                    float absorbed = System.Math.Min(o.Shield, remaining);
                    o.Shield -= absorbed;
                    remaining -= absorbed;
                }
                o.Hp -= remaining;
                Emit(new SimEvent(Tick, SimEventType.DamageDealt, o.Id, dmg));
                if (o.Hp <= 0f) { o.State = EntityState.Dead; dead.Add(o.Id); }
            }
        }

        private double ChargeSpeedOf(HeroId id)
        {
            foreach (var h in Game.Heroes)
                if (h.Id == id) return h.ChargeSpeed;
            return 0;
        }

        private void Emit(SimEvent ev)
        {
            for (int i = _listeners.Count - 1; i >= 0; i--)
                _listeners[i]?.Invoke(ev);
        }
    }

using System.Collections.Generic;

namespace IronCurtain.Sim
{
    /// <summary>
    /// 补兵池(M8.4a 第一批):全局 2s/兵、上限 60;玩家点补兵 → SJF(最缺人兵团先补、补齐为止);
    /// 交战兵团不可补、脱战 3s 重新入列。行军表现归 View(M8.6b),本类纯 Sim。
    /// </summary>
    public static class ReinforceSystem
    {
        public const int PoolCap = 60;              // 池上限(平拍)
        public const float SecondsPerSoldier = 2f;  // 2s/兵
        public const float OutOfCombatSeconds = 3f; // 脱战 3s 重新入列

        public static int PoolOf(SimCore sim, Side side) => sim.ReinforcePool[side];

        public static void Tick(SimCore sim, float dt)
        {
            sim.ReinforceAccum[Side.Blue] += dt / SecondsPerSoldier;
            sim.ReinforceAccum[Side.Red] += dt / SecondsPerSoldier;
            foreach (Side side in new[] { Side.Blue, Side.Red })
            {
                while (sim.ReinforceAccum[side] >= 1f && sim.ReinforcePool[side] < PoolCap)
                {
                    sim.ReinforcePool[side]++;
                    sim.ReinforceAccum[side] -= 1f;
                }
            }
        }

        public static bool RequestReinforce(SimCore sim, Side side, out string error)
        {
            error = null;

            // 1) 收集主将
            var generals = new List<Entity>();
            foreach (var e in sim.Entities.Values)
                if (e.Kind == EntityKind.Unit && !e.IsSoldier && e.Side == side && e.State != EntityState.Dead)
                    generals.Add(e);
            if (generals.Count == 0) { error = "无主将在场"; return false; }
            if (sim.ReinforcePool[side] <= 0) { error = "补兵池为空"; return false; }

            // 2) 交战判定 + 存活士兵计数
            var regiments = new List<System.Tuple<int, Entity>>(); // (cnt, general)
            bool anyEligible = false;
            foreach (var g in generals)
            {
                int cnt = 0;
                bool inCombat = g.SecondsSinceCombat < OutOfCombatSeconds;
                foreach (var e in sim.Entities.Values)
                {
                    if (!e.IsSoldier || e.GeneralId != g.Id || e.State == EntityState.Dead) continue;
                    cnt++;
                    if (e.SecondsSinceCombat < OutOfCombatSeconds) inCombat = true;
                }
                if (inCombat) continue;
                anyEligible = true;
                if (cnt < RegimentSystem.SoldiersPerGeneral)
                    regiments.Add(System.Tuple.Create(cnt, g));
            }
            if (!anyEligible) { error = "全部兵团交战中,不可补兵"; return false; }

            // 3) SJF:缺员数升序(同缺员按 Id 稳定序),逐团补齐为止
            regiments.Sort((a, b) => a.Item1 != b.Item1 ? a.Item1.CompareTo(b.Item1) : a.Item2.Id.CompareTo(b.Item2.Id));
            bool anyGiven = false;
            foreach (var reg in regiments)
            {
                if (sim.ReinforcePool[side] <= 0) break;
                var g = reg.Item2;
                var used = new HashSet<int>();
                foreach (var e in sim.Entities.Values)
                    if (e.IsSoldier && e.GeneralId == g.Id && e.State != EntityState.Dead)
                        used.Add(e.FormationSlot);
                int need = RegimentSystem.SoldiersPerGeneral - reg.Item1;
                int give = System.Math.Min(need, sim.ReinforcePool[side]);
                for (int slot = 0; slot < RegimentSystem.SoldiersPerGeneral && give > 0; slot++)
                {
                    if (used.Contains(slot)) continue;
                    // 防崩:槽位落格若被建筑占地/单位占据,沿槽向再推 1m(至多 3 次);仍堵则跳过该槽不耗池
                    var off = RegimentSystem.SlotOffset(g, slot);
                    var pos = g.PosF + off;
                    bool placed = false;
                    for (int t = 0; t < 3 && !placed; t++)
                    {
                        if (SlotCellFree(sim, pos.ToGrid()))
                        {
                            var s = RegimentSystem.SpawnSoldier(sim, g, slot, pos);
                            s.MarchOrigin = MarchSource(sim, side, g.PosF);
                            sim.ReinforcePool[side]--;
                            give--;
                            anyGiven = true;
                            placed = true;
                        }
                        else pos = TerrainMap.ClampToEllipse(pos + off.Normalized());
                    }
                }
            }
            if (!anyGiven) { error = "补兵池为空"; return false; }
            return true;
        }

        /// <summary>行军源=距主将最近的己方存活非废墟兵营;无兵营则己方主基地;再无则 null(即时落位)。</summary>
        private static SimVec2? MarchSource(SimCore sim, Side side, SimVec2 from)
        {
            Entity best = null;
            float bestD = 0f;
            foreach (var e in sim.Entities.Values)
            {
                if (e.Kind != EntityKind.Building || e.Side != side || e.State == EntityState.Dead || e.Rubbled) continue;
                if (e.Building != BuildingId.Barracks) continue;
                float dx = e.Pos.Col - from.X;
                float dy = e.Pos.Row - from.Y;
                float d = dx * dx + dy * dy;
                if (best == null || d < bestD) { best = e; bestD = d; }
            }
            if (best != null) return new SimVec2(best.Pos.Col, best.Pos.Row);
            foreach (var e in sim.Entities.Values)
            {
                if (e.Kind != EntityKind.Building || e.Side != side || e.State == EntityState.Dead || e.Rubbled) continue;
                if (e.Building != BuildingId.Base) continue;
                float dx = e.Pos.Col - from.X;
                float dy = e.Pos.Row - from.Y;
                float d = dx * dx + dy * dy;
                if (best == null || d < bestD) { best = e; bestD = d; }
            }
            if (best != null) return new SimVec2(best.Pos.Col, best.Pos.Row);
            return null;
        }

        /// <summary>落格是否可出生:不在任何非废墟建筑占地内、距存活露天猫单位 ≥0.5m(镜像 SpawnEntity 校验)。</summary>
        static bool SlotCellFree(SimCore sim, GridPos cell)
        {
            var c = SimVec2.FromGrid(cell);
            foreach (var b in sim.Entities.Values)
                if (b.Kind == EntityKind.Building && !b.Rubbled && BuildingFootprint.Contains(b, c)) return false;
            foreach (var u in sim.Entities.Values)
                if (u.Kind == EntityKind.Unit && u.State != EntityState.Dead && u.GarrisonBuildingId == 0
                    && SimVec2.Distance(u.PosF, c) < 0.5f) return false;
            return true;
        }
    }
